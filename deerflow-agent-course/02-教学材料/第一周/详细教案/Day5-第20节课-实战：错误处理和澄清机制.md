```python
from typing import Dict, Any, Optional
from dataclasses import dataclass
from enum import Enum

class ErrorCategory(Enum):
    """错误类别（复习）"""
    TIMEOUT = "timeout"
    NETWORK = "network"
    PERMISSION = "permission"
    PARAMETER = "parameter"
    RESOURCE = "resource"
    UNKNOWN = "unknown"

@dataclass
class ToolError:
    """工具错误（复习）"""
    tool_name: str
    error_message: str
    category: ErrorCategory
    original_error: Optional[Exception] = None

class ErrorResponseFormatter:
    """错误响应格式化器"""
    
    def __init__(self):
        self.error_templates = {
            ErrorCategory.TIMEOUT: {
                "title": "⏱️ 操作超时",
                "message": "执行时间过长，请稍后重试",
                "action": "如果问题持续存在，请减少请求的数据量",
                "icon": "⏱️"
            },
            ErrorCategory.NETWORK: {
                "title": "🌐 网络错误",
                "message": "网络连接失败，请检查网络后重试",
                "action": "请检查网络连接，或稍后重试",
                "icon": "🌐"
            },
            ErrorCategory.PERMISSION: {
                "title": "🔐 权限不足",
                "message": "您没有执行此操作的权限",
                "action": "请联系管理员获取相应权限",
                "icon": "🔐"
            },
            ErrorCategory.PARAMETER: {
                "title": "📝 参数错误",
                "message": "输入参数有问题，无法继续执行",
                "action": "请检查输入的参数格式和取值范围",
                "icon": "📝"
            },
            ErrorCategory.RESOURCE: {
                "title": "💾 资源不足",
                "message": "系统资源不足，无法完成操作",
                "action": "请稍后重试，或减少操作规模",
                "icon": "💾"
            },
            ErrorCategory.UNKNOWN: {
                "title": "❓ 未知错误",
                "message": "发生了意外错误，我们正在排查",
                "action": "请记录错误信息并联系技术支持",
                "icon": "❓"
            }
        }
    
    def format_error(self, error: ToolError) -> Dict[str, Any]:
        """格式化错误响应"""
        template = self.error_templates.get(
            error.category,
            self.error_templates[ErrorCategory.UNKNOWN]
        )
        
        return {
            "type": "error",
            "title": template["title"],
            "message": template["message"],
            "detail": error.error_message[:100],  # 限制详细信息的长度
            "action": template["action"],
            "icon": template["icon"],
            "category": error.category.value,
            "timestamp": datetime.now().isoformat()
        }
    
    def format_clarification(self, clarification: Clarification) -> Dict[str, Any]:
        """格式化澄清请求"""
        return {
            "type": "clarification",
            "question": clarification.question,
            "suggestions": clarification.suggestions,
            "required": clarification.required,
            "trigger": clarification.trigger.value,
            "timestamp": datetime.now().isoformat()
        }
```

```python
import asyncio
from datetime import datetime

class RobustErrorHandlingSystem:
    """健壮的错误处理系统"""
    
    def __init__(self):
        # 初始化三个核心中间件
        self.error_middleware = ToolErrorHandlingMiddleware(max_retries=2)
        self.clarification_middleware = ClarificationMiddleware()
        self.title_middleware = TitleMiddleware()
        
        # 初始化格式化器
        self.formatter = ErrorResponseFormatter()
        
        # 系统状态
        self.stats = {
            "total_requests": 0,
            "successful_requests": 0,
            "failed_requests": 0,
            "clarifications_requested": 0,
            "titles_generated": 0
        }
    
    async def process_request(self, request: AgentRequest) -> Dict[str, Any]:
        """处理请求的完整流程"""
        self.stats["total_requests"] += 1
        
        print(f"📥 [{self.__class__.__name__}] 处理请求: {request.message[:50]}...")
        
        # 阶段1: 澄清检查
        print("  ├─ 阶段1: 澄清检查...")
        clarification_request = await self.clarification_middleware.before_agent(request)
        
        if request.metadata and request.metadata.get("needs_clarification"):
            self.stats["clarifications_requested"] += 1
            print(f"  │  ⚠️ 需要澄清")
            return self.formatter.format_clarification(
                request.metadata["clarification"]
            )
        
        # 阶段2: 标题生成
        print("  ├─ 阶段2: 标题生成...")
        title_request = await self.title_middleware.before_agent(request)
        generated_title = request.metadata.get("generated_title", "无标题")
        self.stats["titles_generated"] += 1
        
        # 阶段3: 模拟工具执行（可能出错）
        print("  ├─ 阶段3: 工具执行...")
        try:
            # 这里模拟工具执行，可能成功也可能失败
            result = await self._simulate_tool_execution(request)
            
            # 阶段4: 后置处理（标题保存）
            print("  ├─ 阶段4: 后置处理...")
            response = AgentResponse(
                thread_id=request.thread_id,
                content=result,
                success=True
            )
            response = await self.title_middleware.after_agent(response)
            
            self.stats["successful_requests"] += 1
            
            return {
                "type": "success",
                "title": generated_title,
                "content": result,
                "timestamp": datetime.now().isoformat()
            }
            
        except Exception as e:
            # 错误处理流程
            print(f"  │  ❌ 发生错误: {type(e).__name__}: {str(e)[:50]}")
            
            # 错误分类
            error = self.error_middleware.classify_error(e)
            error.tool_name = "main_tool"
            
            # 错误处理
            handling_result = await self.error_middleware.handle_error(error, {})
            
            # 格式化错误响应
            error_response = self.formatter.format_error(error)
            error_response["action_taken"] = handling_result.get("action", "unknown")
            
            self.stats["failed_requests"] += 1
            
            return error_response
    
    async def _simulate_tool_execution(self, request: AgentRequest) -> str:
        """模拟工具执行，有一定概率失败"""
        import random
        
        # 模拟不同类型的错误
        error_types = [
            ("timeout", TimeoutError, "工具执行超时"),
            ("network", ConnectionError, "网络连接失败"),
            ("permission", PermissionError, "权限不足"),
            ("parameter", ValueError, "参数无效"),
            ("resource", MemoryError, "内存不足"),
            ("success", None, None)  # 成功情况
        ]
        
        # 随机选择一个结果
        error_name, error_class, error_msg = random.choice(error_types)
        
        if error_name == "success":
            # 模拟成功执行
            await asyncio.sleep(0.1)  # 模拟处理时间
            return f"成功处理请求: {request.message}"
        else:
            # 模拟错误
            await asyncio.sleep(0.05)
            raise error_class(error_msg)
    
    def get_stats(self) -> Dict[str, int]:
        """获取系统统计信息"""
        return self.stats.copy()
```

## 应急预案

### 技术问题
1. **环境问题**：异步测试框架不兼容
   - 备用方案：提供同步版本的测试代码
   - 快速解决：使用替代的测试框架

2. **代码运行错误**：随机错误模拟不稳定
   - 备用方案：提供确定性的错误模拟模式
   - 逐步调试：固定随机种子确保可重复性

### 时间问题
1. **进度滞后**：学生理解困难
   - 调整策略：重点讲解核心架构，简化扩展部分
   - 课后补充：录制微视频讲解复杂部分

2. **进度超前**：学生掌握迅速
   - 拓展内容：介绍高级错误处理模式（断路器、舱壁模式）
   - 深入讨论：生产环境错误处理的最佳实践

### 内容问题
1. **内容过难**：学生跟不上
   - 简化讲解：用实际案例比喻复杂概念
   - 增加示例：提供更多简单易懂的代码示例

2. **内容过简**：学生觉得无聊
   - 增加挑战：提出进阶问题
   - 实际案例：分享生产环境中的错误处理案例
