def _safe_eval(self, expression: str) -> float:
    """安全评估数学表达式"""
    # 字符白名单检查

```python
def _generate_cache_key(self, tool_name: str, method: str, *args, **kwargs) -> str:
    """生成缓存键"""
    # 规范化参数
    normalized_args = []
    for arg in args:
        if isinstance(arg, (int, float, str, bool, type(None))):
            normalized_args.append(str(arg))
        elif isinstance(arg, (list, tuple)):
            normalized_args.append(tuple(arg))
        elif isinstance(arg, dict):
            # 对字典按键排序，确保顺序不影响缓存键
            sorted_items = sorted(arg.items())
            normalized_args.append(tuple(sorted_items))
        else:
            # 复杂对象使用repr，但可能影响性能
            normalized_args.append(repr(arg))
    
    # 规范化关键字参数
    normalized_kwargs = []
    for key in sorted(kwargs.keys()):  # 按键排序，确保顺序一致
        value = kwargs[key]
        if isinstance(value, (int, float, str, bool, type(None))):
            normalized_kwargs.append(f"{key}:{value}")
        elif isinstance(value, (list, tuple)):
            normalized_kwargs.append(f"{key}:{tuple(value)}")
        elif isinstance(value, dict):
            sorted_items = sorted(value.items())
            normalized_kwargs.append(f"{key}:{tuple(sorted_items)}")
        else:
            normalized_kwargs.append(f"{key}:{repr(value)}")
    
    # 生成缓存键
    key_parts = [tool_name, method]
    key_parts.extend(normalized_args)
    key_parts.extend(normalized_kwargs)
    
    cache_key = "|".join(key_parts)
    return hashlib.md5(cache_key.encode()).hexdigest()
```
