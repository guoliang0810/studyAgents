### 第57节：AppConfig加载流程

#### 配置系统设计哲学
DeerFlow配置系统遵循**约定优于配置**和**分层覆盖**原则：
1. **默认配置**：框架提供的合理默认值
2. **环境配置**：根据部署环境调整（开发、测试、生产）
3. **用户配置**：用户自定义覆盖
4. **运行时配置**：程序运行时动态修改

#### 配置加载顺序
```python
def load_app_config() -> AppConfig:
    """加载应用程序配置"""
    
    # 1. 加载默认配置
    config = load_default_config()
    
    # 2. 加载环境配置（基于DEPLOYMENT_ENV）
    env_config = load_environment_config()
    config = merge_configs(config, env_config)
    
    # 3. 加载用户配置文件
    user_config = load_user_config()
    config = merge_configs(config, user_config)
    
    # 4. 应用环境变量覆盖
    config = apply_environment_overrides(config)
    
    # 5. 验证配置有效性
    validate_config(config)
    
    return config
```

#### 配置合并策略
```python
def merge_configs(base: Dict, override: Dict) -> Dict:
    """深度合并两个配置字典"""
    result = base.copy()
    
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            # 递归合并字典
            result[key] = merge_configs(result[key], value)
        else:
            # 直接覆盖
            result[key] = value
    
    return result
```

```python
class ToolLoader:
    def __init__(self, config: ToolConfig):
        self.config = config
        self.tools: Dict[str, BaseTool] = {}
    
    async def load_tools(self) -> Dict[str, BaseTool]:
        """加载所有配置的工具"""
        
        for tool_name, tool_config in self.config.tools.items():
            if not tool_config.get("enabled", True):
                continue
            
            tool = await self.load_tool(tool_name, tool_config)
            self.tools[tool_name] = tool
        
        return self.tools
    
    async def load_tool(self, tool_name: str, config: Dict) -> BaseTool:
        """加载单个工具"""
        tool_type = config.get("type", "builtin")
        
        if tool_type == "builtin":
            return self.load_builtin_tool(tool_name, config)
        elif tool_type == "custom":
            return await self.load_custom_tool(tool_name, config)
        elif tool_type == "mcp":
            return await self.load_mcp_tool(tool_name, config)
        else:
            raise ValueError(f"Unknown tool type: {tool_type}")
```

```python
class ExtensionLoader:
    def __init__(self, config_path: str):
        self.config_path = config_path
        self.extensions: Dict[str, Any] = {}
    
    async def load_all(self):
        """加载所有扩展"""
        config = self.load_config()
        
        for category, extensions in config.get("extensions", {}).items():
            for name, ext_config in extensions.items():
                if ext_config.get("enabled", False):
                    extension = await self.load_extension(category, name, ext_config)
                    self.extensions[f"{category}.{name}"] = extension
    
    async def load_extension(self, category: str, name: str, config: Dict) -> Any:
        """加载单个扩展"""
        # 根据类别和名称动态导入扩展模块
        module_path = f"deerflow.extensions.{category}.{name}"
        module = importlib.import_module(module_path)
        
        # 创建扩展实例
        extension_class = getattr(module, f"{name.title()}Extension")
        extension = extension_class(config)
        
        # 初始化扩展
        await extension.initialize()
        
        return extension
```

```python
class MCPServerManager:
    def __init__(self, config: MCPConfig):
        self.config = config
        self.servers: Dict[str, MCPServer] = {}
    
    async def start_server(self, server_name: str) -> MCPServer:
        """启动MCP服务器"""
        server_config = self.config.servers.get(server_name)
        if not server_config:
            raise ValueError(f"Server {server_name} not found")
        
        # 创建服务器进程
        process = await asyncio.create_subprocess_exec(
            server_config.command,
            *server_config.args,
            env={**os.environ, **server_config.env},
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        
        server = MCPServer(
            name=server_name,
            process=process,
            config=server_config,
        )
        
        self.servers[server_name] = server
        
        # 等待服务器就绪
        await self.wait_for_server_ready(server)
        
        return server
    
    async def wait_for_server_ready(self, server: MCPServer, timeout: float = 30):
        """等待服务器就绪"""
        start_time = time.time()
        
        while time.time() - start_time < timeout:
            if await self.check_server_health(server):
                return
            
            await asyncio.sleep(0.5)
        
        raise TimeoutError(f"Server {server.name} failed to start within {timeout} seconds")
```

```python
@dataclass
class SkillState:
    """技能状态"""
    skill_name: str
    version: str
    enabled: bool
    last_used: float
    usage_count: int
    success_count: int
    failure_count: int
    avg_execution_time: float
    current_status: str  # "idle", "loading", "ready", "error"
    error_message: Optional[str] = None
    
    @property
    def success_rate(self) -> float:
        """计算成功率"""
        total = self.success_count + self.failure_count
        return self.success_count / total if total > 0 else 0.0
    
    @property
    def health_score(self) -> float:
        """计算健康度得分"""
        if not self.enabled:
            return 0.0
        
        scores = []
        
        # 成功率得分（权重0.5）
        scores.append(self.success_rate * 0.5)
        
        # 使用频率得分（权重0.2）
        freq_score = min(self.usage_count / 100, 1.0)  # 最多100次
        scores.append(freq_score * 0.2)
        
        # 响应时间得分（权重0.3）
        time_score = max(0, 1 - self.avg_execution_time / 10)  # 10秒为上限
        scores.append(time_score * 0.3)
        
        return sum(scores)
```

```python
class SkillStateManager:
    def __init__(self, storage_path: str):
        self.storage_path = storage_path
        self.states: Dict[str, SkillState] = self.load_states()
    
    def load_states(self) -> Dict[str, SkillState]:
        """从存储加载技能状态"""
        try:
            with open(self.storage_path, 'r') as f:
                data = json.load(f)
            
            states = {}
            for skill_name, state_data in data.items():
                states[skill_name] = SkillState(**state_data)
            
            return states
        except (FileNotFoundError, json.JSONDecodeError):
            return {}
    
    def save_states(self):
        """保存技能状态到存储"""
        data = {
            skill_name: dataclasses.asdict(state)
            for skill_name, state in self.states.items()
        }
        
        with open(self.storage_path, 'w') as f:
            json.dump(data, f, indent=2)
    
    def update_state(self, skill_name: str, updates: Dict[str, Any]):
        """更新技能状态"""
        if skill_name not in self.states:
            self.states[skill_name] = SkillState(
                skill_name=skill_name,
                version="unknown",
                enabled=True,
                last_used=0,
                usage_count=0,
                success_count=0,
                failure_count=0,
                avg_execution_time=0,
                current_status="idle",
            )
        
        state = self.states[skill_name]
        
        for key, value in updates.items():
            if hasattr(state, key):
                setattr(state, key, value)
        
        # 自动保存
        self.save_states()
```

```python
class ConfigVersionManager:
    def __init__(self, repo_path: str):
        self.repo_path = repo_path
        self.git = git.Repo(repo_path) if os.path.exists(repo_path) else None
    
    def create_version(self, config: Dict, message: str) -> str:
        """创建配置版本"""
        if not self.git:
            self.initialize_repo()
        
        # 写入配置文件
        config_path = os.path.join(self.repo_path, "config.yaml")
        with open(config_path, 'w') as f:
            yaml.dump(config, f)
        
        # 提交到Git
        self.git.index.add([config_path])
        commit = self.git.index.commit(message)
        
        return commit.hexsha
    
    def get_version(self, version_hash: str) -> Dict:
        """获取特定版本的配置"""
        config_path = os.path.join(self.repo_path, "config.yaml")
        
        # 切换到指定版本
        self.git.git.checkout(version_hash)
        
        # 读取配置
        with open(config_path, 'r') as f:
            config = yaml.safe_load(f)
        
        # 切换回主分支
        self.git.git.checkout("main")
        
        return config
    
    def compare_versions(self, version1: str, version2: str) -> Dict[str, Any]:
        """比较两个版本的配置差异"""
        config1 = self.get_version(version1)
        config2 = self.get_version(version2)
        
        diff = {}
        
        def find_diff(path, d1, d2):
            if d1 == d2:
                return
            
            if isinstance(d1, dict) and isinstance(d2, dict):
                for key in set(d1.keys()) | set(d2.keys()):
                    new_path = f"{path}.{key}" if path else key
                    find_diff(new_path, d1.get(key), d2.get(key))
            else:
                diff[path] = {"old": d1, "new": d2}
        
        find_diff("", config1, config2)
        
        return diff
```

#### 配置迁移工具
```python
class ConfigMigrationTool:
    def __init__(self):
        self.migrations: List[Callable] = []
    
    def add_migration(self, version: str, migration_func: Callable):
        """添加迁移函数"""
        self.migrations.append((version, migration_func))
    
    def migrate(self, config: Dict, target_version: str) -> Dict:
        """迁移配置到目标版本"""
        current_version = config.get("version", "1.0.0")
        
        # 按版本排序迁移
        self.migrations.sort(key=lambda x: x[0])
        
        result = config.copy()
        
        for version, migration_func in self.migrations:
            if version > current_version and version <= target_version:
                result = migration_func(result)
                result["version"] = version
        
        return result
    
    # 示例迁移函数
    @staticmethod
    def migrate_v1_to_v2(config: Dict) -> Dict:
        """从v1迁移到v2"""
        result = config.copy()
        
        # 重命名字段
        if "models" in result:
            for model_name, model_config in result["models"].items():
                if "api_key" in model_config:
                    model_config["credentials"] = {
                        "api_key": model_config.pop("api_key")
                    }
        
        return result
```

```python
class ChatModelFactory:
    PROVIDERS = {
        "openai": OpenAIProvider,
        "anthropic": AnthropicProvider,
        "cohere": CohereProvider,
        "local": LocalProvider,
    }
    
    def __init__(self, config: ModelConfig):
        self.config = config
        self.cache: Dict[str, BaseChatModel] = {}
    
    def create(self, model_name: str, **kwargs) -> BaseChatModel:
        """创建聊天模型"""
        # 检查缓存
        cache_key = f"{model_name}:{hash(str(kwargs))}"
        if cache_key in self.cache:
            return self.cache[cache_key]
        
        # 获取模型配置
        model_config = self.get_model_config(model_name)
        
        # 合并运行时参数
        merged_config = {**model_config.config, **kwargs}
        
        # 创建模型实例
        provider_class = self.PROVIDERS.get(model_config.provider)
        if not provider_class:
            raise ValueError(f"Unsupported provider: {model_config.provider}")
        
        provider = provider_class(model_config)
        model = provider.create_chat_model(merged_config)
        
        # 缓存模型实例
        self.cache[cache_key] = model
        
        return model
    
    def get_model_config(self, model_name: str) -> ModelConfig:
        """获取模型配置"""
        config = self.config.models.get(model_name)
        if not config:
            # 尝试模糊匹配
            similar = self.find_similar_model(model_name)
            if similar:
                raise ValueError(
                    f"Model {model_name} not found. Did you mean {similar}?"
                )
            raise ValueError(f"Model {model_name} not found")
        
        return config
```

#### 模型配置解析

```python
@dataclass
class ModelConfig:
    """模型配置"""
    provider: str
    model_name: str
    config: Dict[str, Any]
    capabilities: List[str]
    cost_per_token: float = 0.0
    
    @classmethod
    def from_dict(cls, data: Dict) -> "ModelConfig":
        """从字典创建配置"""
        return cls(
            provider=data["provider"],
            model_name=data["model_name"],
            config=data.get("config", {}),
            capabilities=data.get("capabilities", []),
            cost_per_token=data.get("cost_per_token", 0.0),
        )
    
    def has_capability(self, capability: str) -> bool:
        """检查是否支持某个能力"""
        return capability in self.capabilities
    
    def get_cost_estimate(self, input_tokens: int, output_tokens: int) -> float:
        """估算成本"""
        total_tokens = input_tokens + output_tokens
        return total_tokens * self.cost_per_token
```

```python
            # 1. 性能得分（基于历史数据）
            perf_score = self.usage_tracker.get_performance_score(model_name)

            cost = self.config.models[model_name].cost_per_token

            fit_score = self.calculate_fit_score(model_name, prompt)
```

```python
    def __init__(self, command: str, args: List[str], env: Dict[str, str]):
        self.command = command
        self.args = args
        self.env = env
        self.process: Optional[asyncio.subprocess.Process] = None

    async def connect(self):
        """启动子进程并建立连接"""
        self.process = await asyncio.create_subprocess_exec(
            self.command,
            *self.args,
            env={**os.environ, **self.env},
            stdin=asyncio.subprocess.PIPE,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
    
    async def call(self, method: str, params: Dict) -> Dict:
        """发送JSON-RPC请求"""
        self.request_id += 1
        
        request = {
            "jsonrpc": "2.0",
            "id": self.request_id,
            "method": method,
            "params": params,
        }
        
        # 发送请求
        request_json = json.dumps(request) + "\n"
        self.process.stdin.write(request_json.encode())
        await self.process.stdin.drain()
        
        # 读取响应
        line = await self.process.stdout.readline()
        response = json.loads(line.decode())
        
        if "error" in response:
            raise MCPError(response["error"])
        
        return response.get("result", {})
```

```python
class LazyLoadingManager:
    def __init__(self, config: LazyLoadingConfig):
        self.config = config
        self.loaded_components: Dict[str, Any] = {}
        self.loading_queue = asyncio.Queue()
        self.loading_tasks: Dict[str, asyncio.Task] = {}
    
    async def get(self, component_name: str) -> Any:
        """获取组件（延迟加载）"""
        # 检查是否已加载
        if component_name in self.loaded_components:
            return self.loaded_components[component_name]
        
        # 检查是否正在加载
        if component_name in self.loading_tasks:
            # 等待加载完成
            await self.loading_tasks[component_name]
            return self.loaded_components[component_name]
        
        # 立即加载（如果配置允许）
        if self.config.get("eager_load", False):
            component = await self.load_component(component_name)
            self.loaded_components[component_name] = component
            return component
        
        # 触发后台加载
        task = asyncio.create_task(
            self.background_load(component_name)
        )
        self.loading_tasks[component_name] = task
        
        # 同步加载（阻塞直到完成）
        component = await self.load_component(component_name)
        self.loaded_components[component_name] = component
        
        return component
    
    async def background_load(self, component_name: str):
        """后台加载组件"""
        try:
            component = await self.load_component(component_name)
            self.loaded_components[component_name] = component
        finally:
            # 清理任务记录
            self.loading_tasks.pop(component_name, None)
    
    async def preload(self, component_names: List[str]):
        """预加载组件"""
        tasks = []
        for name in component_names:
            if name not in self.loaded_components and name not in self.loading_tasks:
                task = asyncio.create_task(self.load_component(name))
                tasks.append((name, task))
        
        # 等待所有预加载完成
        for name, task in tasks:
            try:
                component = await task
                self.loaded_components[name] = component
            except Exception as e:
                logger.warning(f"Failed to preload {name}: {e}")
```

```python
class SmartPreloader:
    def __init__(self, usage_history: UsageHistory):
        self.usage_history = usage_history
        self.prediction_model = self.train_prediction_model()
    
    def train_prediction_model(self):
        """训练组件使用预测模型"""
        # 收集使用历史数据
        sequences = self.usage_history.get_usage_sequences()
        
        # 训练简单的马尔可夫模型
        transition_counts = defaultdict(Counter)
        
        for seq in sequences:
            for i in range(len(seq) - 1):
                current = seq[i]
                next_comp = seq[i + 1]
                transition_counts[current][next_comp] += 1
        
        # 计算转移概率
        transition_probs = {}
        for current, counts in transition_counts.items():
            total = sum(counts.values())
            transition_probs[current] = {
                next_comp: count / total
                for next_comp, count in counts.items()
            }
        
        return transition_probs
    
    def predict_next_components(self, current_components: List[str], limit: int = 3) -> List[str]:
        """预测接下来可能使用的组件"""
        predictions = {}
        
        for current in current_components:
            if current in self.prediction_model:
                for next_comp, prob in self.prediction_model[current].items():
                    predictions[next_comp] = predictions.get(next_comp, 0) + prob
        
        # 按概率排序
        sorted_predictions = sorted(
            predictions.items(),
            key=lambda x: x[1],
            reverse=True
        )
        
        return [comp for comp, prob in sorted_predictions[:limit]]
    
    async def smart_preload(self, current_context: Context):
        """智能预加载"""
        # 分析当前上下文
        current_components = self.extract_components_from_context(current_context)
        
        # 预测接下来可能需要的组件
        predicted = self.predict_next_components(current_components)
        
        # 触发预加载
```

```python
class MCPToolRegistry:
    def __init__(self):

        self.categories: Dict[str, List[str]] = defaultdict(list)
        self.metadata: Dict[str, ToolMetadata] = {}
    
    def register_tool(self, tool: MCPTool, metadata: ToolMetadata = None):
        """注册MCP工具"""
        if tool.name in self.tools:
            raise ValueError(f"Tool {tool.name} already registered")
        
        self.tools[tool.name] = tool
        
        if metadata:
            self.metadata[tool.name] = metadata

            for category in metadata.categories:

    def get_tool(self, name: str) -> Optional[MCPTool]:
        """获取工具"""
        return self.tools.get(name)
    
    def search_tools(self, query: str, category: str = None) -> List[ToolSearchResult]:
        """搜索工具"""
        results = []
        
        for tool_name, tool in self.tools.items():
            metadata = self.metadata.get(tool_name, ToolMetadata())
            
            # 按类别过滤
            if category and category not in metadata.categories:
                continue
            
            # 计算匹配分数
            score = self.calculate_match_score(tool_name, tool.description, query, metadata)
            
            if score > 0:
                results.append(ToolSearchResult(
                    tool_name=tool_name,
                    description=tool.description,
                    score=score,
                    metadata=metadata,
                ))
        
        # 按分数排序
        results.sort(key=lambda x: x.score, reverse=True)
        
        return results
    
    def calculate_match_score(self, name: str, description: str, query: str, metadata: ToolMetadata) -> float:
        """计算匹配分数"""
        score = 0.0
        
        # 名称匹配
        if query.lower() in name.lower():
            score += 2.0
        
        # 描述匹配
        if query.lower() in description.lower():
            score += 1.0
        
        # 标签匹配
        for tag in metadata.tags:
            if query.lower() in tag.lower():
                score += 0.5
```

```python
class VariableResolver:
    def __init__(self, context: Dict[str, Any]):
        self.context = context
        self.cache: Dict[str, Any] = {}
    
    async def resolve(self, expression: str) -> Any:
        """解析变量表达式"""
        # 检查缓存
        if expression in self.cache:
            return self.cache[expression]
        
        # 解析表达式
        if expression.startswith("${") and expression.endswith("}"):
            # 变量引用：${variable.name}
            var_path = expression[2:-1]
            value = await self.resolve_variable(var_path)
        elif expression.startswith("$(") and expression.endswith(")"):
            # 函数调用：$(function arg1 arg2)
            func_call = expression[2:-1]
            value = await self.resolve_function(func_call)
        elif ":" in expression:
            # 类型转换：type:value
            type_name, value_str = expression.split(":", 1)
            value = await self.cast_value(type_name, value_str)
        else:
            # 字面值
            value = await self.parse_literal(expression)
        
        # 缓存结果
        self.cache[expression] = value
        
        return value
    
    async def resolve_variable(self, var_path: str) -> Any:
        """解析变量路径"""
        parts = var_path.split(".")
        current = self.context
        
        for part in parts:
            if isinstance(current, dict):
                current = current.get(part)
            elif hasattr(current, part):
                current = getattr(current, part)
            else:
                raise ValueError(f"Variable {part} not found in path {var_path}")
            
            if current is None:
                break
        
        return current
    
    async def resolve_function(self, func_call: str) -> Any:
        """解析函数调用"""
        # 解析函数名和参数
        if " " in func_call:
            func_name, args_str = func_call.split(" ", 1)
            args = args_str.split()
        else:
            func_name = func_call
            args = []
        
        # 获取函数
        func = await self.get_function(func_name)
        if not func:
            raise ValueError(f"Function {func_name} not found")
        
        # 解析参数
        resolved_args = []
        for arg in args:
            resolved_arg = await self.resolve(arg)
            resolved_args.append(resolved_arg)
        
        # 调用函数
        if asyncio.iscoroutinefunction(func):
            return await func(*resolved_args)
        else:
            return func(*resolved_args)
```

```python
class ClassResolver:
    def __init__(self, search_paths: List[str] = None):
        self.search_paths = search_paths or []
        self.class_cache: Dict[str, Type] = {}
        self.module_cache: Dict[str, ModuleType] = {}
    
    async def resolve(self, class_path: str) -> Type:
        """解析类路径"""
        # 检查缓存
        if class_path in self.class_cache:
            return self.class_cache[class_path]
        
        # 解析模块和类名
        if ":" in class_path:
            module_path, class_name = class_path.split(":", 1)
        elif "." in class_path:
            parts = class_path.split(".")
            module_path = ".".join(parts[:-1])
            class_name = parts[-1]
        else:
            raise ValueError(f"Invalid class path: {class_path}")
        
        # 动态导入模块
        module = await self.import_module(module_path)
        
        # 获取类
        if not hasattr(module, class_name):
            raise ValueError(f"Class {class_name} not found in module {module_path}")
        
        cls = getattr(module, class_name)
        
        # 验证确实是类
        if not isinstance(cls, type):
            raise ValueError(f"{class_path} is not a class")
        
        # 缓存结果
        self.class_cache[class_path] = cls
        
        return cls
    
    async def import_module(self, module_path: str) -> ModuleType:
        """动态导入模块"""
        # 检查缓存
        if module_path in self.module_cache:
            return self.module_cache[module_path]
        
        # 尝试从搜索路径导入
        for search_path in self.search_paths:
            sys.path.insert(0, search_path)
        
        try:
            module = importlib.import_module(module_path)
            self.module_cache[module_path] = module
            return module
        except ImportError as e:
            # 尝试相对导入
            if module_path.startswith("."):
                # 需要知道调用者的模块
                raise ValueError(f"Relative imports require caller context: {module_path}")
            else:
                raise ValueError(f"Module {module_path} not found: {e}")
        finally:
            # 恢复搜索路径
            for _ in self.search_paths:
                sys.path.pop(0)
    
    async def create_instance(self, class_path: str, *args, **kwargs) -> Any:
        """创建类的实例"""
        cls = await self.resolve(class_path)
        return cls(*args, **kwargs)
```

```python
        await self.resolve_variables()

        components = await self.load_components()
        
        # 最后建立依赖关系
        await self.wire_dependencies(components)

    async def resolve_variables(self):
        """解析配置中的变量"""
        def resolve_dict(d: Dict) -> Dict:
            resolved = {}
            for key, value in d.items():
                if isinstance(value, dict):
                    resolved[key] = resolve_dict(value)
                elif isinstance(value, list):
                    resolved[key] = resolve_list(value)
                elif isinstance(value, str) and value.startswith("${") and value.endswith("}"):
                    # 变量引用
                    var_name = value[2:-1]
                    resolved[key] = self.variable_resolver.resolve(var_name)
                else:
                    resolved[key] = value
            return resolved
        
        def resolve_list(lst: List) -> List:
            resolved = []
            for item in lst:
                if isinstance(item, dict):
                    resolved.append(resolve_dict(item))
                elif isinstance(item, list):
                    resolved.append(resolve_list(item))
                elif isinstance(item, str) and item.startswith("${") and item.endswith("}"):
                    var_name = item[2:-1]
                    resolved.append(self.variable_resolver.resolve(var_name))
                else:
                    resolved.append(item)
            return resolved
        
        self.config = resolve_dict(self.config)
```

```python
class DependencyContainer:
    def __init__(self):
        self.services: Dict[str, Any] = {}
        self.factories: Dict[str, Callable] = {}
        self.singletons: Set[str] = set()
    
    def register(self, name: str, service: Any, singleton: bool = False):
        """注册服务"""
        self.services[name] = service
        if singleton:
            self.singletons.add(name)
    
    def register_factory(self, name: str, factory: Callable, singleton: bool = False):
        """注册工厂函数"""
        self.factories[name] = factory
        if singleton:
            self.singletons.add(name)
    
    async def get(self, name: str) -> Any:
        """获取服务"""
        # 检查是否已实例化（单例）
        if name in self.services and name in self.singletons:
            return self.services[name]
        
        # 检查是否有工厂函数
        if name in self.factories:
            factory = self.factories[name]
            
            # 解析工厂函数的依赖
            dependencies = self.get_dependencies(factory)
            dep_instances = await self.resolve_dependencies(dependencies)
            
            # 调用工厂函数
            instance = factory(*dep_instances)
            
            # 如果是单例，缓存实例
            if name in self.singletons:
                self.services[name] = instance
            
            return instance
        
        # 检查是否已注册
        if name in self.services:
            return self.services[name]
        
        raise ValueError(f"Service {name} not found")
    
    def get_dependencies(self, func: Callable) -> List[str]:
        """获取函数的依赖"""
        dependencies = []
        
        # 检查函数签名
        sig = inspect.signature(func)
        
        for param_name, param in sig.parameters.items():
            # 跳过self、cls等特殊参数
            if param_name in ["self", "cls"]:
                continue
            
            # 检查是否有类型注解
            if param.annotation != inspect.Parameter.empty:
                # 尝试从类型注解推断服务名称
                type_name = param.annotation.__name__.lower()
                dependencies.append(type_name)
            else:
                # 使用参数名作为服务名
                dependencies.append(param_name)
        
        return dependencies
    
    async def resolve_dependencies(self, dependencies: List[str]) -> List[Any]:
        """解析依赖列表"""
        instances = []
        
        for dep_name in dependencies:
            instance = await self.get(dep_name)
            instances.append(instance)
        
        return instances
```

```python
class UserFriendlyError(Exception):
    """用户友好的错误"""
    
    def __init__(self, message: str, details: Dict[str, Any] = None, suggestion: str = None):
        super().__init__(message)
        self.message = message
        self.details = details or {}
        self.suggestion = suggestion
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典格式"""
        return {
            "error": self.message,
            "details": self.details,
            "suggestion": self.suggestion,
            "type": self.__class__.__name__,
        }
    
    def __str__(self) -> str:
        base = f"{self.message}"
        if self.suggestion:
            base += f"\n\n建议：{self.suggestion}"
        if self.details:
            base += f"\n\n详情：{json.dumps(self.details, indent=2, ensure_ascii=False)}"
        return base
```

```python
class ErrorHandlingMiddleware:
    async def handle_error(self, error: Exception, context: Dict[str, Any]) -> Dict[str, Any]:
        """处理错误并生成友好的响应"""
        
        # 转换为用户友好的错误
        friendly_error = self.convert_to_friendly_error(error)
        
        # 添加上下文信息
        friendly_error.details["context"] = {
            "timestamp": time.time(),
            "request_id": context.get("request_id"),
            "component": context.get("component"),
        }
        
        # 记录错误日志
        self.log_error(friendly_error, context)

        response = friendly_error.to_dict()
        
        # 添加调试信息（仅开发环境）
        if context.get("environment") == "development":
            response["debug"] = {
                "original_error": str(error),
                "traceback": traceback.format_exc(),
            }
        
        return response

                dependency=str(error).split()[-1],
                reason=f"导入失败：{error}",

                message=f"值错误：{message}",
                suggestion="请检查输入值的格式和范围",

        elif isinstance(error, KeyError):
            return ConfigError(
                config_path="unknown",
                field=str(error),
                value=None,
                expected="必需的配置字段",
            )
        
        else:
            # 通用错误
            return UserFriendlyError(
                message=f"系统错误：{error_type}",
                details={"original_message": str(error)},
                suggestion="请联系技术支持或查看日志获取更多信息",
            )
```

```python
class CheckpointManager:

            except Exception as e:
```

```python
    async def load(self, checkpoint_id: str) -> Optional[Checkpoint]:
        """从文件系统加载检查点"""
        checkpoint_dir = os.path.join(self.base_dir, checkpoint_id)
        
        if not os.path.exists(checkpoint_dir):
            return None
        
        try:
            # 加载状态
            state_path = os.path.join(checkpoint_dir, "state.json")
            with open(state_path, 'r') as f:
                state_dict = json.load(f)
            
            # 加载元数据
            metadata_path = os.path.join(checkpoint_dir, "metadata.json")
            with open(metadata_path, 'r') as f:
                metadata = json.load(f)
            
            # 加载检查点信息
            info_path = os.path.join(checkpoint_dir, "checkpoint.json")
            with open(info_path, 'r') as f:
                info = json.load(f)
            
            # 重建检查点对象
            state = AgentState.from_dict(state_dict)
            
            checkpoint = Checkpoint(
                id=checkpoint_id,
                state=state,
                metadata=metadata,
                created_at=info.get("created_at", 0),
                tags=info.get("tags", []),
            )
            
            return checkpoint
        except Exception as e:
            raise CheckpointCorruptedError(f"Checkpoint {checkpoint_id} corrupted: {e}")
```

```python
class LongRunningTask:
    def __init__(self, task_id: str, checkpoint_manager: CheckpointManager):
        self.task_id = task_id
        self.checkpoint_manager = checkpoint_manager
        self.state = TaskState(id=task_id)
        self.checkpoint_interval = 300  # 每5分钟保存一次
        self.last_checkpoint_time = 0
    
    async def run(self):
        """运行长时间任务"""
        try:
            # 尝试恢复之前的检查点
            await self.restore_from_checkpoint()
            
            # 主循环
            while not self.state.is_completed:
                # 执行一个工作单元
                await self.process_work_unit()
                
                # 定期保存检查点
                current_time = time.time()
                if current_time - self.last_checkpoint_time > self.checkpoint_interval:
                    await self.save_checkpoint()
                    self.last_checkpoint_time = current_time
                
                # 检查是否应该暂停
                if await self.should_pause():
                    await self.save_checkpoint()
                    break
            
            # 任务完成
            if self.state.is_completed:
                await self.cleanup_checkpoints()
        
        except Exception as e:
            # 发生错误，保存检查点以便恢复
            await self.save_checkpoint()
            raise
    
    async def restore_from_checkpoint(self):
        """从检查点恢复"""
        # 查找最新的检查点
        checkpoints = await self.checkpoint_manager.list_checkpoints(
            tags=[f"task:{self.task_id}"],
            limit=1
        )
        
        if checkpoints:
            latest = checkpoints[0]
            self.state = await self.checkpoint_manager.restore_checkpoint(latest.id)
            logger.info(f"Task {self.task_id} restored from checkpoint {latest.id}")
    
    async def save_checkpoint(self):
        """保存检查点"""
        checkpoint = await self.checkpoint_manager.create_checkpoint(
            state=self.state,
            metadata={
                "task_id": self.task_id,
                "progress": self.state.progress,
                "step": self.state.current_step,
            },
        )
        
        logger.info(f"Checkpoint saved for task {self.task_id}: {checkpoint.id}")
    
    async def cleanup_checkpoints(self):
        """清理旧的检查点"""
        checkpoints = await self.checkpoint_manager.list_checkpoints(
            tags=[f"task:{self.task_id}"],
            limit=100
        )
        
        # 保留最近10个检查点
        if len(checkpoints) > 10:
            to_delete = checkpoints[10:]
            
            for checkpoint in to_delete:
                try:
                    await self.checkpoint_manager.delete_checkpoint(checkpoint.id)
                    logger.info(f"Deleted old checkpoint {checkpoint.id}")
                except Exception as e:
                    logger.warning(f"Failed to delete checkpoint {checkpoint.id}: {e}")
```

```python
class StateMigrationTool:
    def __init__(self, source_provider: CheckpointProvider, target_provider: CheckpointProvider):
        self.source = source_provider
        self.target = target_provider
    
    async def migrate(self, checkpoint_id: str) -> str:
        """迁移检查点到新提供者"""
        # 从源加载
        checkpoint = await self.source.load(checkpoint_id)
        if not checkpoint:
            raise ValueError(f"Checkpoint {checkpoint_id} not found in source")
        
        # 保存到目标
        new_id = await self.target.save(checkpoint)
        
        # 验证迁移
        verify_checkpoint = await self.target.load(new_id)
        if not verify_checkpoint:
            raise MigrationError("Migration verification failed")
        
        # 比较状态（简化验证）
        if checkpoint.state != verify_checkpoint.state:
            raise MigrationError("State mismatch after migration")
        
        logger.info(f"Checkpoint {checkpoint_id} migrated to {new_id}")
        
        return new_id
    
    async def batch_migrate(
        self, 
        tags: List[str] = None, 
        limit: int = 100,
        delete_source: bool = False
    ) -> MigrationReport:
        """批量迁移检查点"""
        # 列出源检查点
        checkpoints = await self.source.list(tags, limit)
        
        report = MigrationReport(total=len(checkpoints))
        
        for checkpoint_info in checkpoints:
            try:
                # 检查是否已存在
                existing = await self.target.load(checkpoint_info.id)
                if existing:
                    report.skipped += 1
                    logger.info(f"Checkpoint {checkpoint_info.id} already exists in target")
                    continue
                
                # 迁移
                new_id = await self.migrate(checkpoint_info.id)
                
                # 删除源（如果配置）
                if delete_source:
                    await self.source.delete(checkpoint_info.id)
                    report.deleted += 1
                
                report.migrated += 1
                report.mapping[checkpoint_info.id] = new_id
                
            except Exception as e:
                report.failed += 1
                report.errors[checkpoint_info.id] = str(e)
                logger.error(f"Failed to migrate {checkpoint_info.id}: {e}")
        
        return report
```

```python
class CheckpointCompressor:
    def __init__(self, compression_level: int = 6):
        self.compression_level = compression_level
    
    def compress_checkpoint(self, checkpoint: Checkpoint) -> CompressedCheckpoint:
        """压缩检查点"""
        # 序列化状态
        state_bytes = pickle.dumps(checkpoint.state)
        metadata_bytes = json.dumps(checkpoint.metadata).encode()
        
        # 压缩
        compressed_state = zlib.compress(state_bytes, self.compression_level)
        compressed_metadata = zlib.compress(metadata_bytes, self.compression_level)
        
        # 计算哈希（用于去重）
        state_hash = hashlib.sha256(state_bytes).hexdigest()
        
        return CompressedCheckpoint(
            id=checkpoint.id,
            state_hash=state_hash,
            compressed_state=compressed_state,
            compressed_metadata=compressed_metadata,
            original_size=len(state_bytes) + len(metadata_bytes),
            compressed_size=len(compressed_state) + len(compressed_metadata),
            created_at=checkpoint.created_at,
            tags=checkpoint.tags,
        )
    
    def decompress_checkpoint(self, compressed: CompressedCheckpoint) -> Checkpoint:
        """解压检查点"""
        # 解压
        state_bytes = zlib.decompress(compressed.compressed_state)
        metadata_bytes = zlib.decompress(compressed.compressed_metadata)
        
        # 反序列化
        state = pickle.loads(state_bytes)
        metadata = json.loads(metadata_bytes.decode())
        
        # 验证哈希
        current_hash = hashlib.sha256(state_bytes).hexdigest()
        if current_hash != compressed.state_hash:
            raise CheckpointCorruptedError("State hash mismatch")
        
        return Checkpoint(
            id=compressed.id,
            state=state,
            metadata=metadata,
            created_at=compressed.created_at,
            tags=compressed.tags,
        )
```

```python
class CheckpointCache:
    def __init__(self, max_size: int = 100):
        self.max_size = max_size
        self.cache: Dict[str, Checkpoint] = {}
        self.access_times: Dict[str, float] = {}
        self.hits = 0
        self.misses = 0
    
    async def get(self, checkpoint_id: str, loader: Callable) -> Checkpoint:
        """获取检查点（缓存优先）"""
        # 检查缓存
        if checkpoint_id in self.cache:
            self.hits += 1
            self.access_times[checkpoint_id] = time.time()
            return self.cache[checkpoint_id]
        
        # 缓存未命中，加载
        self.misses += 1
        checkpoint = await loader(checkpoint_id)
        
        # 添加到缓存
        self.put(checkpoint_id, checkpoint)
        
        return checkpoint
    
    def put(self, checkpoint_id: str, checkpoint: Checkpoint):
        """添加检查点到缓存"""
        # 如果缓存已满，移除最久未使用的
        if len(self.cache) >= self.max_size:
            self.evict_oldest()
        
        self.cache[checkpoint_id] = checkpoint
        self.access_times[checkpoint_id] = time.time()
    
    def evict_oldest(self):
        """移除最久未使用的检查点"""
        if not self.access_times:
            return
        
        oldest_id = min(self.access_times.items(), key=lambda x: x[1])[0]
        
        del self.cache[oldest_id]
        del self.access_times[oldest_id]
    
    def get_hit_rate(self) -> float:
        """获取缓存命中率"""
        total = self.hits + self.misses
        return self.hits / total if total > 0 else 0.0
    
    def clear(self):
        """清空缓存"""
        self.cache.clear()
        self.access_times.clear()
        self.hits = 0
        self.misses = 0
```
