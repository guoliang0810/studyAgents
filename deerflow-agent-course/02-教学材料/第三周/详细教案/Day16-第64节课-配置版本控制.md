## 核心代码示例

### ConfigVersionManager（来自学生讲义）
```python
class ConfigVersionManager:
    def __init__(self, repo_path: str):
        self.repo_path = repo_path
        self.git = git.Repo(repo_path) if os.path.exists(repo_path) else None
    
    def create_version(self, config: Dict, message: str) -> str:
        """创建配置版本"""
        if not self.git:
            self.initialize_repo()
        
        # 写入配置文件
        config_path = os.path.join(self.repo_path, "config.yaml")
        with open(config_path, 'w') as f:
            yaml.dump(config, f)
        
        # 提交到Git
        self.git.index.add([config_path])
        commit = self.git.index.commit(message)
        
        return commit.hexsha
    
    def get_version(self, version_hash: str) -> Dict:
        """获取特定版本的配置"""
        config_path = os.path.join(self.repo_path, "config.yaml")
        
        # 切换到指定版本
        self.git.git.checkout(version_hash)
        
        # 读取配置
        with open(config_path, 'r') as f:
            config = yaml.safe_load(f)
        
        # 切换回主分支
        self.git.git.checkout("main")
        
        return config
    
    def compare_versions(self, version1: str, version2: str) -> Dict[str, Any]:
        """比较两个版本的配置差异"""
        config1 = self.get_version(version1)
        config2 = self.get_version(version2)
        
        diff = {}
        
        def find_diff(path, d1, d2):
            if d1 == d2:
                return
            
            if isinstance(d1, dict) and isinstance(d2, dict):
                for key in set(d1.keys()) | set(d2.keys()):
                    new_path = f"{path}.{key}" if path else key
                    find_diff(new_path, d1.get(key), d2.get(key))
            else:
                diff[path] = {"old": d1, "new": d2}
        
        find_diff("", config1, config2)
        
        return diff
```

### ConfigMigrationTool（来自学生讲义）
```python
class ConfigMigrationTool:
    def __init__(self):
        self.migrations: List[Callable] = []
    
    def add_migration(self, version: str, migration_func: Callable):
        """添加迁移函数"""
        self.migrations.append((version, migration_func))
    
    def migrate(self, config: Dict, target_version: str) -> Dict:
        """迁移配置到目标版本"""
        current_version = config.get("version", "1.0.0")
        
        # 按版本排序迁移
        self.migrations.sort(key=lambda x: x[0])
        
        result = config.copy()
        
        for version, migration_func in self.migrations:
            if version > current_version and version <= target_version:
                result = migration_func(result)
                result["version"] = version
        
        return result
    
    # 示例迁移函数
    @staticmethod
    def migrate_v1_to_v2(config: Dict) -> Dict:
        """从v1迁移到v2"""
        result = config.copy()
        
        # 重命名字段
        if "models" in result:
            for model_name, model_config in result["models"].items():
                if "api_key" in model_config:
                    model_config["credentials"] = {
                        "api_key": model_config.pop("api_key")
                    }
        
        return result
```

### 配置版本管理器（来自互动教学）
```python
import json
from datetime import datetime

@dataclass
class ConfigVersion:
    """配置版本"""
    version: str
    config: Dict
    created_at: datetime
    created_by: str
    comment: str = ""

class ConfigVersionManager:
    """配置版本管理器"""
    
    def __init__(self, storage_path: str = "config_versions"):
        self.storage_path = storage_path
        self.versions: List[ConfigVersion] = []
    
    def save_version(
        self,
        version: str,
        config: Dict,
        created_by: str,
        comment: str = ""
    ):
        """保存版本"""
        v = ConfigVersion(
            version=version,
            config=config,
            created_at=datetime.now(),
            created_by=created_by,
            comment=comment
        )
        self.versions.append(v)
        self._persist(v)
        print(f"💾 保存配置版本: {version}")
    
    def get_version(self, version: str) -> Optional[Dict]:
        """获取指定版本"""
        for v in self.versions:
            if v.version == version:
                return v.config
        return None
    
    def list_versions(self) -> List[str]:
        """列出所有版本"""
        return [v.version for v in self.versions]
    
    def rollback(self, version: str) -> Dict:
        """回滚到指定版本"""
        config = self.get_version(version)
        if config:
            print(f"🔄 回滚到版本: {version}")
            return config
        raise ValueError(f"版本不存在: {version}")
    
    def _persist(self, version: ConfigVersion):
        """持久化版本"""
        os.makedirs(self.storage_path, exist_ok=True)
        path = f"{self.storage_path}/{version.version}.json"
        
        with open(path, 'w') as f:
            json.dump({
                "version": version.version,
                "config": version.config,
                "created_at": version.created_at.isoformat(),
                "created_by": version.created_by,
                "comment": version.comment
            }, f, indent=2)
```
