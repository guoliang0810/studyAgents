### 思考模式YAML配置（来自学生讲义）
```yaml
thinking:
  modes:
    fast:
      temperature: 0.3
      max_tokens: 500
      reasoning_depth: "shallow"
      use_cache: true
    
    balanced:
      temperature: 0.7
      max_tokens: 1000
      reasoning_depth: "medium"
      use_cache: true
    
    deep:
      temperature: 0.9
      max_tokens: 2000
      reasoning_depth: "deep"
      use_cache: false
      allow_reflection: true
      max_iterations: 3
    
    creative:
      temperature: 1.2
      max_tokens: 1500
      reasoning_depth: "medium"
      top_p: 0.95
      frequency_penalty: 0.5
```

### ThinkingModeSelector（来自学生讲义）
```python
class ThinkingModeSelector:
    def __init__(self, config: ThinkingConfig):
        self.config = config
    
    def select_mode(self, task_type: str, complexity: float, urgency: float) -> str:
        """根据任务特征选择思考模式"""
        
        # 规则1：基于任务类型
        mode_by_type = {
            "code_generation": "balanced",
            "debugging": "deep",
            "creative_writing": "creative",
            "data_analysis": "balanced",
            "quick_answer": "fast",
        }
        
        if task_type in mode_by_type:
            return mode_by_type[task_type]
        
        # 规则2：基于复杂度
        if complexity > 0.7:
            return "deep"
        elif complexity > 0.3:
            return "balanced"
        else:
            return "fast"
        
        # 规则3：基于紧急程度（覆盖规则）
        if urgency > 0.8:
            return "fast"
    
    def get_mode_config(self, mode_name: str) -> Dict[str, Any]:
        """获取思考模式配置"""
        mode_config = self.config.modes.get(mode_name)
        if not mode_config:
            raise ValueError(f"Thinking mode {mode_name} not found")
        
        return mode_config.copy()
```

### ThinkingConfig数据类（补充定义）
```python
from dataclasses import dataclass, field
from typing import Dict, Any

@dataclass
class ThinkingConfig:
    """思考模式配置"""
    modes: Dict[str, Dict[str, Any]] = field(default_factory=dict)
    
    @classmethod
    def from_yaml(cls, yaml_content: str) -> "ThinkingConfig":
        """从YAML内容创建配置"""
        import yaml
        data = yaml.safe_load(yaml_content)
        return cls(modes=data.get("thinking", {}).get("modes", {}))
    
    @classmethod
    def from_file(cls, file_path: str) -> "ThinkingConfig":
        """从YAML文件创建配置"""
        with open(file_path, 'r') as f:
            return cls.from_yaml(f.read())
    
    def validate(self) -> List[str]:
        """验证配置有效性，返回错误列表"""
        errors = []
        required_params = {"temperature", "max_tokens", "reasoning_depth"}
        
        for mode_name, mode_config in self.modes.items():
            for param in required_params:
                if param not in mode_config:
                    errors.append(f"模式 '{mode_name}' 缺少参数 '{param}'")
            
            # 验证temperature范围
            temp = mode_config.get("temperature")
            if temp is not None and not (0.0 <= temp <= 2.0):
                errors.append(f"模式 '{mode_name}' temperature参数超出范围: {temp}")
            
            # 验证max_tokens为正整数
            tokens = mode_config.get("max_tokens")
            if tokens is not None and (not isinstance(tokens, int) or tokens <= 0):
                errors.append(f"模式 '{mode_name}' max_tokens参数无效: {tokens}")
        
        return errors
```

    )
    
    # 处理特殊模式特性
    if mode_config.get("allow_reflection", False):
        result = apply_reflection(result, mode_config.get("max_iterations", 1))

        cache_result(prompt, result)
    
    return result
