## 📋 课程基本信息
- **课程名称**: 视觉能力支持
- **授课日期**: 2024年4月10日（周三）
- **上课时间**: 11:00-11:45（第67节课）
- **授课教师**: 张老师

## 🎯 教学目标

### 知识目标（学生将知道/理解）
1. 理解视觉AI模型的能力特点和应用场景
2. 掌握视觉模型配置的特殊参数和限制
3. 理解图像预处理管道的重要性和实现原理
4. 了解不同视觉模型（GPT-4 Vision, Claude-3）的特性差异

### 技能目标（学生将能够）
1. 能够配置YAML格式的视觉模型参数
2. 能够实现VisionPipeline进行图像预处理
3. 能够处理图像格式转换、大小调整、质量优化
4. 能够为不同视觉模型准备合适的输入图像

### 情感/态度目标
1. 培养学生对多模态AI系统的全面理解
2. 增强学生对图像数据处理的安全和隐私意识
3. 激发学生对计算机视觉与NLP结合的兴趣

## 📚 教学重点与难点
- **教学重点**: 视觉模型配置、图像预处理管道、格式和大小限制
- **教学难点**: 图像处理异步流水线、不同模型输入要求、性能优化
- **突破方法**: 通过实际图像处理演示、管道流程图、性能对比分析

### VisionPipeline（来自学生讲义）
```python
class VisionPipeline:
    def __init__(self, config: VisionConfig):
        self.config = config
        self.processors = self.initialize_processors()
    
    def initialize_processors(self) -> List[ImageProcessor]:
        """初始化图像处理器"""
        processors = []
        
        # 格式转换器
        if self.config.get("convert_to_supported", True):
            processors.append(FormatConverter(self.config.supported_formats))
        
        # 大小调整器
        max_size = parse_size(self.config.max_image_size)
        processors.append(SizeResizer(max_size))
        
        # 质量优化器
        if self.config.get("optimize_quality", True):
            processors.append(QualityOptimizer())
        
        # 元数据清理器
        if self.config.get("strip_metadata", True):
            processors.append(MetadataStripper())
        
        return processors
    
    async def process_image(self, image_data: bytes, format: str) -> bytes:
        """处理图像"""
        result = image_data
        
        for processor in self.processors:
            result = await processor.process(result, format)
            format = processor.output_format or format
        
        return result
    
    async def prepare_images_for_model(
        self, 
        images: List[ImageInfo], 
        model_name: str
    ) -> List[ProcessedImage]:
        """为模型准备图像"""
        model_config = self.config.vision_models.get(model_name)
        if not model_config:
            raise ValueError(f"Vision model {model_name} not found")
        
        processed_images = []
        
        for image in images:
            # 检查格式支持
            if image.format not in model_config.vision_config.supported_formats:
                image_data = await self.convert_format(
                    image.data, 
                    image.format,
                    model_config.vision_config.supported_formats[0]
                )
                image_format = model_config.vision_config.supported_formats[0]
            else:
                image_data = image.data
                image_format = image.format
            
            # 检查大小限制
            max_size = parse_size(model_config.vision_config.max_image_size)
            if len(image_data) > max_size:
                image_data = await self.resize_image(image_data, max_size)
            
            processed_images.append(ProcessedImage(
                data=image_data,
                format=image_format,
                original_size=len(image.data),
                processed_size=len(image_data),
            ))
        
        return processed_images
```

### 图像处理器基类和实现（补充定义）
```python
from abc import ABC, abstractmethod
from typing import Optional

class ImageProcessor(ABC):
    """图像处理器基类"""
    
    def __init__(self, output_format: Optional[str] = None):
        self.output_format = output_format
    
    @abstractmethod
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """处理图像数据"""
        pass

class FormatConverter(ImageProcessor):
    """格式转换器"""
    
    def __init__(self, supported_formats: List[str], target_format: str = "jpeg"):
        super().__init__(output_format=target_format)
        self.supported_formats = supported_formats
        self.target_format = target_format
    
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """转换图像格式"""
        if input_format.lower() in self.supported_formats:
            return image_data  # 已经是支持格式
        
        # 使用Pillow进行格式转换
        from PIL import Image
        import io
        
        image = Image.open(io.BytesIO(image_data))
        output_buffer = io.BytesIO()
        
        # 根据目标格式保存
        if self.target_format.lower() == "jpeg":
            if image.mode in ("RGBA", "LA"):
                # JPEG不支持透明度，需要转换为RGB
                background = Image.new("RGB", image.size, (255, 255, 255))
                background.paste(image, mask=image.split()[-1])
                image = background
            image.save(output_buffer, format="JPEG", quality=85)
        else:
            image.save(output_buffer, format=self.target_format.upper())
        
        return output_buffer.getvalue()

class SizeResizer(ImageProcessor):
    """大小调整器"""
    
    def __init__(self, max_size_bytes: int):
        super().__init__()
        self.max_size_bytes = max_size_bytes
    
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """调整图像大小以符合大小限制"""
        if len(image_data) <= self.max_size_bytes:
            return image_data
        
        # 逐步降低质量/尺寸直到符合要求
        from PIL import Image
        import io
        
        image = Image.open(io.BytesIO(image_data))
        current_data = image_data
        
        for quality in [85, 70, 50, 30]:
            output_buffer = io.BytesIO()
            image.save(output_buffer, format=input_format.upper(), quality=quality)
            current_data = output_buffer.getvalue()
            
            if len(current_data) <= self.max_size_bytes:
                break
        
        return current_data

class QualityOptimizer(ImageProcessor):
    """质量优化器"""
    
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """优化图像质量"""
        # 可以根据需要进行锐化、对比度调整等
        # 这里简单返回原数据
        return image_data

class MetadataStripper(ImageProcessor):
    """元数据清理器"""
    
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """清除图像元数据"""
        from PIL import Image
        import io
        
        image = Image.open(io.BytesIO(image_data))
        
        # 创建一个新图像，只复制像素数据
        clean_image = Image.new(image.mode, image.size)
        clean_image.putdata(list(image.getdata()))
        
        output_buffer = io.BytesIO()
        clean_image.save(output_buffer, format=input_format.upper())
        
        return output_buffer.getvalue()
```

```python
from dataclasses import dataclass, field
from typing import Dict, Any, List

@dataclass
class VisionModelConfig:
    """视觉模型配置"""
    provider: str
    model_name: str
    config: Dict[str, Any]
    capabilities: List[str]
    vision_config: Dict[str, Any]
    
    @classmethod
    def from_dict(cls, data: Dict) -> "VisionModelConfig":
        """从字典创建配置"""
        return cls(
            provider=data["provider"],
            model_name=data["model_name"],
            config=data.get("config", {}),
            capabilities=data.get("capabilities", []),
            vision_config=data.get("vision_config", {})
        )
    
    def can_handle_image(self, image_size: int, image_format: str) -> bool:
        """检查是否能处理指定图像"""
        max_size = parse_size(self.vision_config.get("max_image_size", "0MB"))
        supported_formats = self.vision_config.get("supported_formats", [])
        
        return (image_size <= max_size and 
                image_format.lower() in [f.lower() for f in supported_formats])

def parse_size(size_str: str) -> int:
    """解析大小字符串（如"20MB"）为字节数"""
    size_str = size_str.strip().upper()
    
    if size_str.endswith("MB"):
        return int(float(size_str[:-2]) * 1024 * 1024)
    elif size_str.endswith("KB"):
        return int(float(size_str[:-2]) * 1024)
    elif size_str.endswith("GB"):
        return int(float(size_str[:-2]) * 1024 * 1024 * 1024)
    else:
        return int(size_str)
```
