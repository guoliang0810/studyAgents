## 核心代码示例

### ModelRouter（来自学生讲义）
```python
class ModelRouter:
    def __init__(self, config: RouterConfig):
        self.config = config
        self.model_factory = ChatModelFactory(config.model_config)
        self.usage_tracker = UsageTracker()
    
    async def route(
        self, 
        prompt: str, 
        task_type: str = None,
        budget: float = None,
        latency_requirement: float = None
    ) -> BaseChatModel:
        """路由到合适的模型"""
        
        candidates = self.get_candidate_models(task_type)
        
        # 应用过滤规则
        filtered = self.filter_models(candidates, budget, latency_requirement)
        
        if not filtered:
            # 回退到默认模型
            return self.model_factory.create(self.config.default_model)
        
        # 选择最佳模型
        best_model = self.select_best_model(filtered, prompt)
        
        return self.model_factory.create(best_model)
    
    def get_candidate_models(self, task_type: str) -> List[str]:
        """获取候选模型列表"""
        if task_type:
            # 基于任务类型选择
            task_models = self.config.task_routing.get(task_type, [])
            if task_models:
                return task_models
        
        # 基于能力选择
        required_capabilities = self.analyze_required_capabilities(task_type)
        candidates = []
        
        for model_name, model_config in self.config.models.items():
            if all(model_config.has_capability(cap) for cap in required_capabilities):
                candidates.append(model_name)
        
        return candidates
    
    def select_best_model(self, candidates: List[str], prompt: str) -> str:
        """选择最佳模型"""
        scores = {}
        
        for model_name in candidates:
            score = 0.0
            
            # 1. 性能得分（基于历史数据）
            perf_score = self.usage_tracker.get_performance_score(model_name)
            score += perf_score * 0.4
            
            # 2. 成本得分（越便宜得分越高）
            cost = self.config.models[model_name].cost_per_token
            cost_score = max(0, 1 - cost * 1000)  # 归一化
            score += cost_score * 0.3
            
            # 3. 适合度得分（基于提示长度和复杂度）
            fit_score = self.calculate_fit_score(model_name, prompt)
            score += fit_score * 0.3
            
            scores[model_name] = score
        
        return max(scores.items(), key=lambda x: x[1])[0]
```

```python
        if strategy not in self.routing_strategies:
            raise ValueError(f"Unknown routing strategy: {strategy}")
        
        router = self.routing_strategies[strategy]
        return await router.route(prompt, context)
    
    async def adaptive_route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """自适应路由（自动选择最佳策略）"""
        
        # 分析请求特征
        features = self.extract_features(prompt, context)
        
        # 选择最适合的策略
        strategy = self.select_strategy_for_features(features)
        
        # 使用选定的策略进行路由
        return await self.route_with_strategy(strategy, prompt, context)
    
    def extract_features(self, prompt: str, context: Dict[str, Any]) -> Dict[str, float]:
        """提取路由特征"""
        features = {}
        
        # 提示长度特征
        features["prompt_length"] = len(prompt) / 1000  # 归一化
        
        # 复杂度特征（简单启发式）
        complexity_indicators = ["分析", "解释", "比较", "设计", "实现"]
        features["complexity"] = sum(1 for word in complexity_indicators if word in prompt) / 5
        
        # 紧急程度特征
        urgency_indicators = ["尽快", "紧急", "马上", "立即"]
        features["urgency"] = sum(1 for word in urgency_indicators if word in prompt) / 4
```

```python
class ModelRouter:
    """模型路由器"""
    
    def __init__(self):
        self.models: Dict[str, BaseModel] = {}
        self.usage_stats: Dict[str, int] = {}
    
    def register(self, name: str, model: BaseModel):
        """注册模型"""
        self.models[name] = model
        self.usage_stats[name] = 0
    
    async def route(
        self,
        task: str,
        required_capabilities: ModelCapabilities = None
    ) -> BaseModel:
        """路由到合适的模型"""
        
        for name, model in self.models.items():
            if required_capabilities:
                # 检查能力
                caps = model.capabilities
                if (required_capabilities.supports_vision and not caps.supports_vision):
                    continue
                if (required_capabilities.supports_function_calling and 
                    not caps.supports_function_calling):
                    continue
            
            # 选择
            self.usage_stats[name] += 1
            return model
        
        raise ValueError("没有合适的模型")
    
    def get_least_used(self) -> BaseModel:
        """获取使用最少的模型"""
        if not self.usage_stats:
            raise ValueError("没有可用的模型")
        
        least_used = min(self.usage_stats.items(), key=lambda x: x[1])
        return self.models[least_used[0]]
```

### 路由策略基类和实现（补充定义）

```python
from abc import ABC, abstractmethod
from typing import Dict, Any, List

class RoutingStrategy(ABC):
    """路由策略基类"""
    
    @abstractmethod
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """执行路由决策"""
        pass

class CostOptimizedStrategy(RoutingStrategy):
    """成本优化策略"""
    
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """选择成本最低的可用模型"""
        # 实现成本优先的逻辑
        models = context.get("available_models", [])
        if not models:
            raise ValueError("No available models")
        
        # 找到成本最低的模型
        cheapest = min(models, key=lambda m: m.cost_per_token)
        return RoutingDecision(
            model_name=cheapest.name,
            strategy="cost_optimized",
            confidence=0.8,
            reason="选择成本最低的模型"
        )

class PerformanceOptimizedStrategy(RoutingStrategy):
    """性能优化策略"""
    
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """选择性能最好的可用模型"""
        models = context.get("available_models", [])
        if not models:
            raise ValueError("No available models")
        
        # 找到性能最好的模型（基于历史数据）
        best_perf = max(models, key=lambda m: context.get("performance_scores", {}).get(m.name, 0))
        return RoutingDecision(
            model_name=best_perf.name,
            strategy="performance_optimized",
            confidence=0.7,
            reason="选择性能最好的模型"
        )

class BalancedStrategy(RoutingStrategy):
    """平衡策略"""
    
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """平衡成本和性能的选择"""
        models = context.get("available_models", [])
        if not models:
            raise ValueError("No available models")
        
        # 计算综合得分
        best_score = -1
        best_model = None
        
        for model in models:
            cost_score = 1 - min(model.cost_per_token * 1000, 1.0)
            perf_score = context.get("performance_scores", {}).get(model.name, 0.5)
            
            # 平衡权重：成本40%，性能60%
            total_score = cost_score * 0.4 + perf_score * 0.6
            
            if total_score > best_score:
                best_score = total_score
                best_model = model
        
        return RoutingDecision(
            model_name=best_model.name,
            strategy="balanced",
            confidence=best_score,
            reason=f"平衡选择：成本得分{best_score:.2f}"
        )

class LearningBasedStrategy(RoutingStrategy):
    """学习型策略"""
    
    def __init__(self):
        self.history: List[Dict] = []
    
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """基于历史学习的选择"""
        models = context.get("available_models", [])
        if not models:
            raise ValueError("No available models")
        
        # 简单学习：选择历史成功率最高的模型
        success_rates = {}
        for model in models:
            model_history = [h for h in self.history if h["model"] == model.name]
            if model_history:
                successes = sum(1 for h in model_history if h["success"])
                success_rates[model.name] = successes / len(model_history)
            else:
                success_rates[model.name] = 0.5  # 默认值
        
        best_model_name = max(success_rates.items(), key=lambda x: x[1])[0]
        best_model = next(m for m in models if m.name == best_model_name)
        
        return RoutingDecision(
            model_name=best_model.name,
            strategy="learning_based",
            confidence=success_rates[best_model_name],
            reason=f"历史成功率：{success_rates[best_model_name]:.1%}"
        )
```
