from dataclasses import dataclass, field
from enum import Enum
import re


class ModelCapability(Enum):
    """模型能力标签枚举"""
    TEXT_GENERATION = "text_generation"

@dataclass
class ModelCostConfig:
    """模型成本配置"""
    input_cost_per_token: float = 0.0
    output_cost_per_token: float = 0.0
    request_cost: float = 0.0
    monthly_fee: float = 0.0
    currency: str = "USD"
    
    def calculate_cost(self, input_tokens: int = 0, output_tokens: int = 0, 
                      requests: int = 0) -> float:

        token_cost = (input_tokens * self.input_cost_per_token +

        request_cost = requests * self.request_cost
        return token_cost + request_cost

            "input_cost_per_token": self.input_cost_per_token,
            "output_cost_per_token": self.output_cost_per_token,
            "request_cost": self.request_cost,
            "monthly_fee": self.monthly_fee,
            "currency": self.currency

@dataclass
class ModelConfig:
    """模型配置"""
    provider: str
    model_name: str
    api_key: Optional[str] = None
    base_url: Optional[str] = None
    timeout: float = 30.0
    max_retries: int = 3
    temperature: float = 0.7
    max_tokens: Optional[int] = None
    capabilities: Set[ModelCapability] = field(default_factory=set)
    cost_config: ModelCostConfig = field(default_factory=ModelCostConfig)
    metadata: Dict[str, Any] = field(default_factory=dict)
    
    def __post_init__(self):
        """初始化后验证"""
        self._validate()
    
    def _validate(self):
        """验证配置"""
        if not self.provider or not self.model_name:
            raise ValueError("provider和model_name是必需字段")
        
        if self.timeout <= 0:
            raise ValueError("timeout必须大于0")
        
        if self.max_retries < 0:
            raise ValueError("max_retries必须大于等于0")
        
        if not 0 <= self.temperature <= 2:
            raise ValueError("temperature必须在0到2之间")
        
        if self.max_tokens is not None and self.max_tokens <= 0:
            raise ValueError("max_tokens必须大于0")
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return {
            "provider": self.provider,
            "model_name": self.model_name,
            "api_key": self.api_key,
            "base_url": self.base_url,
            "timeout": self.timeout,
            "max_retries": self.max_retries,
            "temperature": self.temperature,
            "max_tokens": self.max_tokens,
            "capabilities": [cap.value for cap in self.capabilities],
            "cost_config": self.cost_config.to_dict(),
            "metadata": self.metadata
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "ModelConfig":
        """从字典创建"""
        capabilities = set()
        for cap_str in data.get("capabilities", []):
            capabilities.add(ModelCapability.from_string(cap_str))
        
        cost_config_data = data.get("cost_config", {})
        cost_config = ModelCostConfig.from_dict(cost_config_data)
        
        return cls(
            provider=data["provider"],
            model_name=data["model_name"],
            api_key=data.get("api_key"),
            base_url=data.get("base_url"),
            timeout=data.get("timeout", 30.0),
            max_retries=data.get("max_retries", 3),
            temperature=data.get("temperature", 0.7),
            max_tokens=data.get("max_tokens"),
            capabilities=capabilities,
            cost_config=cost_config,
            metadata=data.get("metadata", {})
        )
    
    def has_capability(self, capability: ModelCapability) -> bool:
        """检查是否支持特定能力"""
        return capability in self.capabilities
    
    def matches_capabilities(self, required_capabilities: Set[ModelCapability]) -> bool:
        """检查是否满足所有必需能力"""
        return required_capabilities.issubset(self.capabilities)
    
    def estimate_cost(self, input_tokens: int = 1000, output_tokens: int = 500) -> float:
        """估算成本"""
        return self.cost_config.calculate_cost(input_tokens, output_tokens)
```

from enum import Enum
from typing import Set, List, Dict


class ModelCapability(Enum):
    # 定义同上，此处省略重复定义
    
    @classmethod
    def from_string(cls, name: str) -> "ModelCapability":
        """从字符串转换（支持大小写不敏感）"""
        name_lower = name.lower().replace(" ", "_").replace("-", "_")
        for capability in cls:
            if capability.name.lower() == name_lower or capability.value == name_lower:
                return capability
        raise ValueError(f"未知的能力标签: {name}")


class CapabilitySystem:
    """能力标签系统工具类"""
    
    # 能力分组定义
    CAPABILITY_GROUPS: Dict[str, Set[ModelCapability]] = {
        "text": {
            ModelCapability.TEXT_GENERATION,
            ModelCapability.CODE_GENERATION,
            ModelCapability.TRANSLATION,
            ModelCapability.SUMMARIZATION,
            ModelCapability.SENTIMENT_ANALYSIS,
            ModelCapability.NAMED_ENTITY_RECOGNITION,
            ModelCapability.QUESTION_ANSWERING,
            ModelCapability.CHAT,
        },
        "vision": {
            ModelCapability.IMAGE_GENERATION,
            ModelCapability.IMAGE_UNDERSTANDING,
        },
        "audio": {
            ModelCapability.AUDIO_GENERATION,
            ModelCapability.AUDIO_TRANSCRIPTION,
        },
        "reasoning": {
            ModelCapability.REASONING,
            ModelCapability.MATH,
            ModelCapability.HIGH_ACCURACY,
        },
        "performance": {
            ModelCapability.LONG_CONTEXT,
            ModelCapability.LOW_LATENCY,
            ModelCapability.LOW_COST,
        },
        "advanced": {
            ModelCapability.FUNCTION_CALLING,
            ModelCapability.TEXT_EMBEDDING,
        }
    }
    
    # 相关能力映射
    RELATED_CAPABILITIES: Dict[ModelCapability, Set[ModelCapability]] = {
        ModelCapability.TEXT_GENERATION: {
            ModelCapability.CODE_GENERATION,
            ModelCapability.TRANSLATION,
            ModelCapability.SUMMARIZATION,
        },
        ModelCapability.CODE_GENERATION: {
            ModelCapability.TEXT_GENERATION,
            ModelCapability.REASONING,
        },
        ModelCapability.CHAT: {
            ModelCapability.TEXT_GENERATION,
            ModelCapability.QUESTION_ANSWERING,
        },
        # 其他映射...
    }
    
    @classmethod
    def get_capability_group(cls, capability: ModelCapability) -> str:
        """获取能力所属分组"""
        for group_name, capabilities in cls.CAPABILITY_GROUPS.items():
            if capability in capabilities:
                return group_name
        return "other"
    
    @classmethod
    def get_related_capabilities(cls, capability: ModelCapability) -> Set[ModelCapability]:
        """获取相关能力"""
        return cls.RELATED_CAPABILITIES.get(capability, set())
    
    @classmethod
    def capabilities_to_string(cls, capabilities: Set[ModelCapability]) -> str:
        """能力集合转字符串"""
        return ", ".join(sorted([cap.name for cap in capabilities]))
    
    @classmethod
    def matches_capabilities(cls, model_capabilities: Set[ModelCapability],
                           required_capabilities: Set[ModelCapability],
                           strict: bool = True) -> bool:
        """能力匹配
        
        Args:
            model_capabilities: 模型支持的能力
            required_capabilities: 需要的能力
            strict: 是否严格匹配（True: 所有必需能力都必须满足）
        
        Returns:
            是否匹配
        """
        if strict:
            # 严格匹配：所有必需能力都必须满足
            return required_capabilities.issubset(model_capabilities)
        else:
            # 宽松匹配：至少满足一个核心能力
            if not required_capabilities:
                return True
            
            # 定义核心能力
            CORE_CAPABILITIES = {
                ModelCapability.TEXT_GENERATION,
                ModelCapability.CODE_GENERATION,
                ModelCapability.IMAGE_GENERATION,
                ModelCapability.AUDIO_GENERATION,
            }
            
            # 检查是否有核心能力匹配
            core_required = required_capabilities.intersection(CORE_CAPABILITIES)
            if core_required:
                # 至少满足一个核心能力
                return any(cap in model_capabilities for cap in core_required)
            else:
                # 没有核心能力要求，检查是否有任何能力匹配
                return bool(required_capabilities.intersection(model_capabilities))
```

**解析**：
1. **能力分组**：将相似能力分组，便于管理和理解
2. **相关能力映射**：定义能力之间的关联关系，支持智能匹配
3. **灵活匹配策略**：支持严格匹配和宽松匹配两种模式
4. **工具方法**：提供能力转换、字符串化等实用方法
5. **扩展性**：易于添加新的能力标签和分组

# src/config/cost_config.py
from typing import Dict, Any, Optional
from dataclasses import dataclass, field
from datetime import datetime, timedelta
import json


@dataclass
class ModelCostConfig:
    """模型成本配置（完整版）"""
    input_cost_per_token: float = 0.0
    output_cost_per_token: float = 0.0
    request_cost: float = 0.0
    monthly_fee: float = 0.0
    currency: str = "USD"
    
    # 货币汇率（示例数据）
    EXCHANGE_RATES = {
        "USD": 1.0,
        "CNY": 7.2,
        "EUR": 0.92,
        "JPY": 150.0,
    }
    
    def calculate_cost(self, input_tokens: int = 0, output_tokens: int = 0, 
                      requests: int = 0, include_monthly: bool = True) -> float:
        """计算总成本"""
        token_cost = (input_tokens * self.input_cost_per_token + 
                     output_tokens * self.output_cost_per_token)
        request_cost = requests * self.request_cost
        monthly_cost = self.monthly_fee if include_monthly else 0.0
        
        return token_cost + request_cost + monthly_cost
    
    def estimate_cost(self, prompt: str, expected_response_length: int = 100) -> float:
        """基于文本估算成本
        
        Args:
            prompt: 输入文本
            expected_response_length: 预期响应长度（字符数）
        
        Returns:
            估算成本
        """
        # 简单估算：4个字符约1个token
        input_tokens = len(prompt) // 4
        output_tokens = expected_response_length // 4
        
        return self.calculate_cost(input_tokens, output_tokens)
    
    def convert_currency(self, amount: float, target_currency: str) -> float:
        """货币转换"""
        if self.currency == target_currency:
            return amount
        
        if self.currency not in self.EXCHANGE_RATES:
            raise ValueError(f"不支持源货币: {self.currency}")
        if target_currency not in self.EXCHANGE_RATES:
            raise ValueError(f"不支持目标货币: {target_currency}")
        
        # 转换：amount * (target_rate / source_rate)
        source_rate = self.EXCHANGE_RATES[self.currency]
        target_rate = self.EXCHANGE_RATES[target_currency]
        
        return amount * (target_rate / source_rate)
    
    def to_dict(self) -> Dict[str, Any]:
        return {
            "input_cost_per_token": self.input_cost_per_token,
            "output_cost_per_token": self.output_cost_per_token,
            "request_cost": self.request_cost,
            "monthly_fee": self.monthly_fee,
            "currency": self.currency
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "ModelCostConfig":
        return cls(**data)


class CostMonitor:
    """成本监控器"""
    
    def __init__(self, daily_budget: float = 10.0, monthly_budget: float = 300.0):
        self.daily_budget = daily_budget
        self.monthly_budget = monthly_budget
        self.daily_cost = 0.0
        self.monthly_cost = 0.0
        self.daily_reset_date = datetime.now().date()
        self.monthly_reset_date = datetime.now().replace(day=1).date()
        self.cost_history = []  # 记录成本历史
    
    def record_cost(self, cost: float, model_id: str, task_type: str = "") -> None:
        """记录成本"""
        self._check_reset()
        
        # 记录成本
        self.daily_cost += cost
        self.monthly_cost += cost
        
        # 添加到历史
        self.cost_history.append({
            "timestamp": datetime.now(),
            "cost": cost,
            "model_id": model_id,
            "task_type": task_type
        })
        
        # 检查预算
        self._check_budget()
    
    def _check_reset(self):
        """检查是否需要重置计数器"""
        today = datetime.now().date()
        
        # 检查日重置
        if today > self.daily_reset_date:
            self.daily_cost = 0.0
            self.daily_reset_date = today
        
        # 检查月重置
        current_month = datetime.now().replace(day=1).date()
        if current_month > self.monthly_reset_date:
            self.monthly_cost = 0.0
            self.monthly_reset_date = current_month
    
    def _check_budget(self):
        """检查预算"""
        warnings = []
        
        if self.daily_cost > self.daily_budget * 0.8:  # 80%警告
            warnings.append(f"日成本接近预算: {self.daily_cost:.2f}/{self.daily_budget:.2f}")
        
        if self.daily_cost > self.daily_budget:  # 超出预算
            raise ValueError(f"超出日预算: {self.daily_cost:.2f}/{self.daily_budget:.2f}")
        
        if self.monthly_cost > self.monthly_budget * 0.9:  # 90%警告
            warnings.append(f"月成本接近预算: {self.monthly_cost:.2f}/{self.monthly_budget:.2f}")
        
        if warnings:
            print("成本警告:", "; ".join(warnings))
    
    @property
    def remaining_daily_budget(self) -> float:
        """剩余日预算"""
        return max(0, self.daily_budget - self.daily_cost)
    
    @property
    def remaining_monthly_budget(self) -> float:
        """剩余月预算"""
        return max(0, self.monthly_budget - self.monthly_cost)
    
    def get_cost_report(self, period: str = "daily") -> Dict[str, Any]:
        """获取成本报告"""
        if period == "daily":
            return {
                "period": "daily",
                "total_cost": self.daily_cost,
                "budget": self.daily_budget,
                "remaining": self.remaining_daily_budget,
                "utilization": self.daily_cost / self.daily_budget if self.daily_budget > 0 else 0.0
            }
        else:  # monthly
            return {
                "period": "monthly",
                "total_cost": self.monthly_cost,
                "budget": self.monthly_budget,
                "remaining": self.remaining_monthly_budget,
                "utilization": self.monthly_cost / self.monthly_budget if self.monthly_budget > 0 else 0.0
            }

class ModelRegistry:

        """注册模型配置"""
        if model_id in self.models:
            raise ValueError(f"模型ID已存在: {model_id}")
        
        # 验证配置
        self._validate_config(config)
        
        # 存储配置
        self.models[model_id] = config
        
        # 更新能力索引
        for capability in config.capabilities:

                self.capability_index[capability] = set()
            self.capability_index[capability].add(model_id)

        self.provider_index[config.provider].add(model_id)

        cost_per_token = config.cost_config.input_cost_per_token + config.cost_config.output_cost_per_token
        self.cost_index.append((cost_per_token, model_id))
        self.cost_index.sort(key=lambda x: x[0])
    
    def _validate_config(self, config: ModelConfig) -> None:
        """验证配置"""
        # 这里可以添加更多业务规则验证
        pass
    
    def get(self, model_id: str) -> ModelConfig:
        """获取模型配置"""
        if model_id not in self.models:
            raise KeyError(f"模型ID不存在: {model_id}")
        return self.models[model_id]
    
    def unregister(self, model_id: str) -> None:
        """注销模型配置"""
        if model_id not in self.models:
            raise KeyError(f"模型ID不存在: {model_id}")
        
        config = self.models[model_id]
        
        # 从能力索引中移除
        for capability in config.capabilities:
            if capability in self.capability_index:
                self.capability_index[capability].discard(model_id)
                if not self.capability_index[capability]:
                    del self.capability_index[capability]
        
        # 从提供者索引中移除
        if config.provider in self.provider_index:
            self.provider_index[config.provider].discard(model_id)
            if not self.provider_index[config.provider]:
                del self.provider_index[config.provider]
        
        # 从成本索引中移除
        self.cost_index = [(cost, mid) for cost, mid in self.cost_index if mid != model_id]
        
        # 从主字典中移除
        del self.models[model_id]
    
    def find_by_capabilities(self, required_capabilities: Set[ModelCapability]) -> List[Tuple[str, ModelConfig]]:
        """根据能力要求查找模型"""
        if not required_capabilities:
            return list(self.models.items())
        
        # 使用倒排索引快速查找
        candidate_ids = None
        for capability in required_capabilities:
            if capability in self.capability_index:
                if candidate_ids is None:
                    candidate_ids = self.capability_index[capability].copy()
                else:
                    candidate_ids.intersection_update(self.capability_index[capability])
            else:
                # 如果某个能力没有模型支持，直接返回空
                return []
        
        if candidate_ids is None:
            return []
        
        # 验证候选模型是否真正满足所有能力要求
        results = []
        for model_id in candidate_ids:
            config = self.models[model_id]
            if config.matches_capabilities(required_capabilities):
                results.append((model_id, config))
        
        return results
    
    def find_by_provider(self, provider: str) -> List[Tuple[str, ModelConfig]]:
        """根据提供者查找模型"""
        if provider not in self.provider_index:
            return []
        
        results = []
        for model_id in self.provider_index[provider]:
            config = self.models[model_id]
            results.append((model_id, config))
        
        return results
    
    def find_lowest_cost(self, required_capabilities: Set[ModelCapability],

            candidates_with_cost.append((cost, model_id, config))

        # 返回最低成本模型

            _, model_id, config = candidates_with_cost[0]

           self.monthly_fee = monthly_fee

           total_tokens = usage.input_tokens + usage.output_tokens
           if total_tokens <= self.included_tokens:

## 📊 最佳实践总结

### 1. 配置设计最佳实践
- **环境分离**：开发、测试、生产环境使用不同配置
- **安全第一**：API密钥等敏感信息加密存储
- **版本控制**：配置和代码一起进行版本控制
- **文档化**：每个配置项都有详细说明
- **验证前置**：配置加载时立即验证

### 2. 性能优化最佳实践
- **索引优化**：为常用查询字段建立索引
- **缓存策略**：合理使用缓存，设置过期时间
- **连接池**：复用连接，减少建立连接开销
- **异步处理**：非实时任务使用异步处理
- **监控调优**：持续监控，根据数据调优

### 3. 可靠性最佳实践
- **健康检查**：定期检查服务健康状态
- **故障转移**：准备备用服务，快速故障转移
- **重试机制**：实现智能重试（指数退避）
- **熔断机制**：防止故障扩散
- **监控告警**：实时监控，及时告警

### 4. 成本管理最佳实践
- **预算设置**：设置明确的预算限制
- **成本监控**：实时监控成本，设置预警
- **优化分析**：定期分析成本，识别优化机会
- **资源优化**：使用合适规格的资源，避免浪费
- **成本分摊**：按项目、团队分摊成本
