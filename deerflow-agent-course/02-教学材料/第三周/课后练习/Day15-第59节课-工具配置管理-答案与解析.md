```python
import os
import re
import copy
from typing import Dict, Any, List, Optional, Union
from dataclasses import dataclass, field
from enum import Enum


class ToolType(Enum):
    """工具类型枚举"""
    BUILTIN = "builtin"      # 内置工具：系统自带的工具
    CUSTOM = "custom"        # 自定义工具：用户编写的工具
    MCP = "mcp"             # MCP工具：Model Context Protocol工具
    EXTERNAL = "external"    # 外部工具：通过API或CLI调用的外部程序
    SCRIPT = "script"        # 脚本工具：动态执行的脚本
    
    @classmethod
    def from_string(cls, tool_type_str: str) -> "ToolType":
        """从字符串转换为工具类型"""
        try:
            return cls(tool_type_str.lower())
        except ValueError:
            raise ValueError(f"未知的工具类型: {tool_type_str}")


class SecurityLevel(Enum):
    """安全级别枚举"""
    TRUSTED = "trusted"      # 信任级别：完全信任，无限制
    RESTRICTED = "restricted"  # 限制级别：有限制的操作
    SANDBOXED = "sandboxed"  # 沙箱级别：完全隔离，高度限制
    UNTRUSTED = "untrusted"  # 不信任级别：禁止执行
    
    @classmethod
    def from_string(cls, security_str: str) -> "SecurityLevel":
        """从字符串转换为安全级别"""
        try:
            return cls(security_str.lower())
        except ValueError:
            raise ValueError(f"未知的安全级别: {security_str}")


@dataclass
class ToolSecurityConfig:
    """工具安全配置"""
    security_level: SecurityLevel = SecurityLevel.RESTRICTED
    allowed_commands: List[str] = field(default_factory=list)      # 允许的命令列表
    blocked_commands: List[str] = field(default_factory=list)      # 禁止的命令列表
    max_execution_time: float = 30.0                               # 最大执行时间（秒）
    max_memory_mb: int = 100                                       # 最大内存使用（MB）
    network_access: bool = False                                   # 是否允许网络访问
    file_system_access: bool = True                                # 是否允许文件系统访问
    environment_variables: List[str] = field(default_factory=list)  # 允许访问的环境变量
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return {
            "security_level": self.security_level.value,
            "allowed_commands": self.allowed_commands,
            "blocked_commands": self.blocked_commands,
            "max_execution_time": self.max_execution_time,
            "max_memory_mb": self.max_memory_mb,
            "network_access": self.network_access,
            "file_system_access": self.file_system_access,
            "environment_variables": self.environment_variables
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "ToolSecurityConfig":
        """从字典创建"""
        return cls(
            security_level=SecurityLevel.from_string(data.get("security_level", "restricted")),
            allowed_commands=data.get("allowed_commands", []),
            blocked_commands=data.get("blocked_commands", []),
            max_execution_time=data.get("max_execution_time", 30.0),
            max_memory_mb=data.get("max_memory_mb", 100),
            network_access=data.get("network_access", False),
            file_system_access=data.get("file_system_access", True),
            environment_variables=data.get("environment_variables", [])
        )
    
    def is_command_allowed(self, command: str) -> bool:
        """检查命令是否允许执行"""
        # 如果命令在禁止列表中，不允许
        for blocked in self.blocked_commands:
            if re.match(blocked, command):
                return False
        
        # 如果允许列表为空，允许所有（除了禁止的）
        if not self.allowed_commands:
            return True
        
        # 检查是否在允许列表中
        for allowed in self.allowed_commands:
            if re.match(allowed, command):
                return True
        
        return False
    
    def validate(self) -> List[str]:
        """验证安全配置，返回错误列表"""
        errors = []
        
        if self.max_execution_time <= 0:
            errors.append("max_execution_time必须大于0")
        
        if self.max_memory_mb <= 0:
            errors.append("max_memory_mb必须大于0")
        
        # 检查命令模式的有效性
        for pattern in self.allowed_commands + self.blocked_commands:
            try:
                re.compile(pattern)
            except re.error:
                errors.append(f"无效的正则表达式模式: {pattern}")
        
        return errors

@dataclass
class ToolConfig:
    """工具配置"""
    tool_id: str                                     # 工具唯一标识
    tool_type: ToolType                              # 工具类型
    name: str                                       # 工具名称
    description: Optional[str] = None               # 工具描述
    version: str = "1.0.0"                         # 工具版本
    author: Optional[str] = None                    # 作者
    entry_point: Optional[str] = None               # 入口点（模块路径或函数名）
    module_path: Optional[str] = None               # 模块路径（对于自定义工具）
    class_name: Optional[str] = None                # 类名（对于面向对象的工具）
    function_name: Optional[str] = None             # 函数名（对于函数式工具）
    mcp_server_url: Optional[str] = None            # MCP服务器URL（对于MCP工具）
    command: Optional[str] = None                   # 命令（对于外部工具）
    script_content: Optional[str] = None            # 脚本内容（对于脚本工具）
    environment_variables: Dict[str, str] = field(default_factory=dict)  # 环境变量
    security_config: ToolSecurityConfig = field(default_factory=ToolSecurityConfig)  # 安全配置
    dependencies: List[ToolDependency] = field(default_factory=list)     # 依赖项
    metadata: Dict[str, Any] = field(default_factory=dict)               # 元数据
    
    def __post_init__(self):
        """初始化后验证"""
        errors = self.validate()
        if errors:
            raise ValueError(f"工具配置验证失败: {', '.join(errors)}")
    
    def validate(self) -> List[str]:
        """验证配置，返回错误列表"""
        errors = []
        
        # 基本验证
        if not self.tool_id:
            errors.append("tool_id不能为空")
        
        if not self.name:
            errors.append("name不能为空")
        
        # 类型特定验证
        if self.tool_type == ToolType.CUSTOM:
            if not self.module_path and not self.entry_point:
                errors.append("自定义工具需要module_path或entry_point")
        
        elif self.tool_type == ToolType.MCP:
            if not self.mcp_server_url:
                errors.append("MCP工具需要mcp_server_url")
        
        elif self.tool_type == ToolType.EXTERNAL:
            if not self.command:
                errors.append("外部工具需要command")
        
        elif self.tool_type == ToolType.SCRIPT:
            if not self.script_content:
                errors.append("脚本工具需要script_content")
        
        # 安全配置验证
        security_errors = self.security_config.validate()
        errors.extend(security_errors)
        
        return errors
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        result = {
            "tool_id": self.tool_id,
            "tool_type": self.tool_type.value,
            "name": self.name,
            "version": self.version,
            "description": self.description,
            "author": self.author,
            "entry_point": self.entry_point,
            "module_path": self.module_path,
            "class_name": self.class_name,
            "function_name": self.function_name,
            "mcp_server_url": self.mcp_server_url,
            "command": self.command,
            "script_content": self.script_content,
            "environment_variables": self.environment_variables,
            "security_config": self.security_config.to_dict(),
            "dependencies": [dep.to_dict() for dep in self.dependencies],
            "metadata": self.metadata
        }
        
        # 移除None值
        return {k: v for k, v in result.items() if v is not None}
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "ToolConfig":
        """从字典创建"""
        # 解析依赖项
        dependencies = []
        for dep_data in data.get("dependencies", []):
            dependencies.append(ToolDependency.from_dict(dep_data))
        
        # 创建安全配置
        security_data = data.get("security_config", {})
        security_config = ToolSecurityConfig.from_dict(security_data)
        
        return cls(
            tool_id=data["tool_id"],
            tool_type=ToolType.from_string(data["tool_type"]),
            name=data["name"],
            description=data.get("description"),
            version=data.get("version", "1.0.0"),
            author=data.get("author"),
            entry_point=data.get("entry_point"),
            module_path=data.get("module_path"),
            class_name=data.get("class_name"),
            function_name=data.get("function_name"),
            mcp_server_url=data.get("mcp_server_url"),
            command=data.get("command"),
            script_content=data.get("script_content"),
            environment_variables=data.get("environment_variables", {}),
            security_config=security_config,
            dependencies=dependencies,
            metadata=data.get("metadata", {})
        )
    
    def resolve_environment_variables(self, env_vars: Dict[str, str]) -> "ToolConfig":
        """解析环境变量引用
        
        支持格式：${VAR_NAME} 或 ${VAR_NAME:default_value}
        """
        resolved_config = copy.deepcopy(self)
        
        def resolve_value(value: Any) -> Any:
            if isinstance(value, str):
                # 替换环境变量引用
                def replace_var(match):
                    var_name = match.group(1)
                    default_value = match.group(3) if match.group(3) else None
                    
                    # 首先检查传入的环境变量
                    if var_name in env_vars:
                        return env_vars[var_name]
                    
                    # 然后检查系统环境变量
                    if var_name in os.environ:
                        return os.environ[var_name]
                    
                    # 最后使用默认值
                    if default_value is not None:
                        return default_value
                    
                    # 没有找到变量也没有默认值
                    raise ValueError(f"环境变量未定义且无默认值: {var_name}")
                
                # 匹配 ${VAR_NAME} 或 ${VAR_NAME:default}
                pattern = r'\$\{([a-zA-Z_][a-zA-Z0-9_]*)(:([^}]+))?\}'
                return re.sub(pattern, replace_var, value)
            
            elif isinstance(value, dict):
                return {k: resolve_value(v) for k, v in value.items()}
            elif isinstance(value, list):
                return [resolve_value(item) for item in value]
            else:
                return value
        
        # 解析环境变量字段
        resolved_config.environment_variables = resolve_value(self.environment_variables)
        resolved_config.command = resolve_value(self.command) if self.command else None
        resolved_config.mcp_server_url = resolve_value(self.mcp_server_url) if self.mcp_server_url else None
        resolved_config.script_content = resolve_value(self.script_content) if self.script_content else None
        
        return resolved_config
    
    def get_required_dependencies(self) -> List[str]:
        """获取必需的依赖工具ID列表"""
        return [dep.tool_id for dep in self.dependencies if dep.required]
```

**解析**：
1. **ToolConfig类**是工具配置的核心，包含了工具的所有配置信息。使用`@dataclass`装饰器简化代码，自动生成`__init__`等方法。
2. **环境变量解析**使用正则表达式匹配`${VAR_NAME}`格式，支持默认值`${VAR_NAME:default}`。解析时优先使用传入的环境变量，然后检查系统环境变量，最后使用默认值。
3. **配置验证**在`__post_init__`中自动执行，确保配置的完整性。验证包括基本字段非空检查和类型特定验证。
4. **序列化和反序列化**通过`to_dict()`和`from_dict()`方法实现，支持与YAML/JSON格式的互转换。

```python
# src/config/dependency.py
from typing import Dict, Any, List, Optional, Set, Tuple
from dataclasses import dataclass, field
import re


@dataclass
class EnhancedToolDependency(ToolDependency):
    """增强的工具依赖配置（支持版本约束）"""
    min_version: Optional[str] = None    # 最低版本
    max_version: Optional[str] = None    # 最高版本
    exact_version: Optional[str] = None  # 精确版本
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典（扩展版本）"""
        result = super().to_dict()
        if self.min_version:
            result["min_version"] = self.min_version
        if self.max_version:
            result["max_version"] = self.max_version
        if self.exact_version:
            result["exact_version"] = self.exact_version
        return result
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "EnhancedToolDependency":
        """从字典创建（扩展版本）"""
        base_dep = super().from_dict(data)
        return cls(
            tool_id=base_dep.tool_id,
            version=base_dep.version,
            required=base_dep.required,
            min_version=data.get("min_version"),
            max_version=data.get("max_version"),
            exact_version=data.get("exact_version")
        )
    
    def matches_version(self, version: str) -> bool:
        """检查版本是否匹配约束"""
        # 如果有精确版本要求，必须完全匹配
        if self.exact_version:
            return self._compare_versions(version, self.exact_version) == 0
        
        # 检查版本范围
        if self.min_version and self._compare_versions(version, self.min_version) < 0:
            return False
        
        if self.max_version and self._compare_versions(version, self.max_version) > 0:
            return False
        
        # 检查语义化版本约束
        if self.version:
            return self._matches_semver_constraint(version, self.version)
        
        return True
    
    def _compare_versions(self, v1: str, v2: str) -> int:
        """比较版本号，返回-1/0/1"""
        # 简化版本比较（实际应使用semver库）
        v1_parts = [int(part) for part in v1.split('.')]
        v2_parts = [int(part) for part in v2.split('.')]
        
        for i in range(max(len(v1_parts), len(v2_parts))):
            v1_part = v1_parts[i] if i < len(v1_parts) else 0
            v2_part = v2_parts[i] if i < len(v2_parts) else 0
            
            if v1_part < v2_part:
                return -1
            elif v1_part > v2_part:
                return 1
        
        return 0
    
    def _matches_semver_constraint(self, version: str, constraint: str) -> bool:
        """匹配语义化版本约束"""
        # 支持常见的语义化版本约束
        if constraint.startswith("^"):  # 兼容版本
            base_version = constraint[1:]
            # ^1.2.3 匹配 >=1.2.3 且 <2.0.0
            return (self._compare_versions(version, base_version) >= 0 and
                    self._compare_versions(version, "2.0.0") < 0)
        elif constraint.startswith("~"):  # 近似版本
            base_version = constraint[1:]
            # ~1.2.3 匹配 >=1.2.3 且 <1.3.0
            base_parts = base_version.split('.')
            if len(base_parts) >= 2:
                next_minor = f"{base_parts[0]}.{int(base_parts[1]) + 1}.0"
                return (self._compare_versions(version, base_version) >= 0 and
                        self._compare_versions(version, next_minor) < 0)
            return self._compare_versions(version, base_version) >= 0
        elif constraint.startswith(">="):
            base_version = constraint[2:]
            return self._compare_versions(version, base_version) >= 0
        elif constraint.startswith("<="):
            base_version = constraint[2:]
            return self._compare_versions(version, base_version) <= 0
        elif constraint.startswith(">"):
            base_version = constraint[1:]
            return self._compare_versions(version, base_version) > 0
        elif constraint.startswith("<"):
            base_version = constraint[1:]
            return self._compare_versions(version, base_version) < 0
        else:
            # 精确匹配
            return self._compare_versions(version, constraint) == 0


class DependencyResolver:
    """依赖解析器"""
    
    def __init__(self):
        self.dependency_graph: Dict[str, List[str]] = {}
        self.tool_versions: Dict[str, str] = {}
        self.dependency_details: Dict[str, List[EnhancedToolDependency]] = {}
    
    def add_tool(self, tool_id: str, version: str, 
                 dependencies: List[EnhancedToolDependency]):
        """添加工具及其依赖"""
        self.tool_versions[tool_id] = version
        self.dependency_details[tool_id] = dependencies
        
        # 只包含必需的依赖
        required_deps = [dep.tool_id for dep in dependencies if dep.required]
        self.dependency_graph[tool_id] = required_deps
    
    def build_dependency_graph(self) -> Dict[str, List[str]]:
        """构建依赖关系图"""
        return self.dependency_graph.copy()
    
    def topological_sort(self) -> List[str]:
        """拓扑排序工具ID"""
        graph = self.dependency_graph
        in_degree = {node: 0 for node in graph}
        
        # 计算入度
        for node, dependencies in graph.items():
            for dep in dependencies:
                if dep in in_degree:
                    in_degree[dep] += 1
                else:
                    in_degree[dep] = 1
        
        # 初始化队列
        queue = [node for node, degree in in_degree.items() if degree == 0]
        sorted_nodes = []
        
        # 拓扑排序
        while queue:
            node = queue.pop(0)
            sorted_nodes.append(node)
            
            for dependent in graph.get(node, []):
                if dependent in in_degree:
                    in_degree[dependent] -= 1
                    if in_degree[dependent] == 0:
                        queue.append(dependent)
        
        # 检查循环依赖
        if len(sorted_nodes) != len(graph):
            remaining = set(graph.keys()) - set(sorted_nodes)
            raise ValueError(f"检测到循环依赖: {remaining}")
        
        return sorted_nodes
    
    def detect_cycles(self) -> List[List[str]]:
        """检测循环依赖，返回所有循环路径"""
        graph = self.dependency_graph
        visited = set()
        on_stack = set()
        cycles = []
        
        def dfs(node, path):
            visited.add(node)
            on_stack.add(node)
            path.append(node)
            
            for neighbor in graph.get(node, []):
                if neighbor not in visited:
                    dfs(neighbor, path.copy())
                elif neighbor in on_stack:
                    # 找到循环
                    cycle_start = path.index(neighbor)
                    cycle = path[cycle_start:]
                    cycles.append(cycle)
            
            on_stack.remove(node)
            path.pop()
        
        for node in graph:
            if node not in visited:
                dfs(node, [])
        
        return cycles
    
    def resolve_dependencies(self, tool_id: str) -> Set[str]:
        """解析指定工具的所有依赖（包括间接依赖）"""
        if tool_id not in self.dependency_details:
            return set()
        
        dependencies = set()
        to_process = [tool_id]
        
        while to_process:
            current = to_process.pop()
            deps = self.dependency_details.get(current, [])
            
            for dep in deps:
                if dep.required:
                    dependencies.add(dep.tool_id)
                    to_process.append(dep.tool_id)
        
        return dependencies
    
    def validate_version_constraints(self) -> List[str]:
        """验证版本约束"""
        errors = []
        
        for tool_id, deps in self.dependency_details.items():
            tool_version = self.tool_versions.get(tool_id, "1.0.0")
            
            for dep in deps:
                if not dep.required:
                    continue
                
                dep_version = self.tool_versions.get(dep.tool_id)
                if not dep_version:
                    errors.append(f"{tool_id} 依赖的工具不存在: {dep.tool_id}")
                    continue
                
                if not dep.matches_version(dep_version):
                    errors.append(
                        f"{tool_id}@{tool_version} 依赖 {dep.tool_id}@{dep_version} "
                        f"不满足约束: {dep.version or dep.exact_version or f'{dep.min_version}-{dep.max_version}'}"
                    )
        
        return errors
```

**解析**：
1. **增强的依赖配置**支持完整的版本约束，包括语义化版本号（SemVer）。`matches_version()`方法实现了版本匹配逻辑。
2. **依赖解析器**提供了完整的依赖管理功能：
   - `topological_sort()`：实现拓扑排序算法，确定工具初始化顺序。
   - `detect_cycles()`：使用深度优先搜索检测循环依赖。
   - `resolve_dependencies()`：解析所有直接和间接依赖。
   - `validate_version_constraints()`：验证版本约束是否满足。
3. **循环依赖检测**是依赖管理的关键功能。本实现使用DFS检测所有循环路径，帮助开发者识别和解决循环依赖问题。
4. **版本约束解析**支持常见的语义化版本约束格式（^, ~, >=, <=, >, <），确保工具版本兼容性。

```python
import asyncio
import yaml
import json
import importlib
import importlib.util
from typing import Dict, Any, List, Optional
from pathlib import Path

from src.config.tool_config import ToolConfig, ToolType
from src.config.dependency import DependencyResolver

from src.tools.builtin_tool import BuiltinTool
from src.tools.custom_tool import CustomTool

from src.tools.external_tool import ExternalTool
from src.tools.script_tool import ScriptTool


class ToolLoader:
    """工具加载器"""
    
    def __init__(self):
        self.tools: Dict[str, BaseTool] = {}
        self.configs: Dict[str, ToolConfig] = {}
        self.dependency_resolver = DependencyResolver()
        self.initialized = False
    
    async def load_configs_from_yaml(self, yaml_path: str, 
                                   env_vars: Optional[Dict[str, str]] = None) -> List[ToolConfig]:
        """从YAML文件加载工具配置"""
        env_vars = env_vars or {}
        
        try:
            with open(yaml_path, 'r', encoding='utf-8') as f:
                yaml_data = yaml.safe_load(f)
            
            if not isinstance(yaml_data, list):
                raise ValueError("YAML文件应包含工具配置列表")
            
            configs = []
            for tool_data in yaml_data:
                # 创建配置
                config = ToolConfig.from_dict(tool_data)
                
                # 解析环境变量
                if env_vars:
                    config = config.resolve_environment_variables(env_vars)
                
                # 验证配置
                errors = config.validate()
                if errors:
                    raise ValueError(f"配置验证失败: {config.tool_id}, 错误: {', '.join(errors)}")
                
                configs.append(config)
                self.configs[config.tool_id] = config
            
            return configs
        except Exception as e:
            raise RuntimeError(f"加载YAML配置失败: {e}")
    
    async def load_configs_from_json(self, json_path: str,
                                   env_vars: Optional[Dict[str, str]] = None) -> List[ToolConfig]:
        """从JSON文件加载工具配置"""
        env_vars = env_vars or {}
        
        try:
            with open(json_path, 'r', encoding='utf-8') as f:
                json_data = json.load(f)
            
            if not isinstance(json_data, list):
                raise ValueError("JSON文件应包含工具配置列表")
            
            configs = []
            for tool_data in json_data:
                # 创建配置
                config = ToolConfig.from_dict(tool_data)
                
                # 解析环境变量
                if env_vars:
                    config = config.resolve_environment_variables(env_vars)
                
                # 验证配置
                errors = config.validate()
                if errors:
                    raise ValueError(f"配置验证失败: {config.tool_id}, 错误: {', '.join(errors)}")
                
                configs.append(config)
                self.configs[config.tool_id] = config
            
            return configs
        except Exception as e:
            raise RuntimeError(f"加载JSON配置失败: {e}")
    
    async def load_configs_from_directory(self, config_dir: str,
                                        env_vars: Optional[Dict[str, str]] = None) -> List[ToolConfig]:
        """从目录加载所有配置文件"""
        config_dir_path = Path(config_dir)
        configs = []
        
        # 加载YAML文件
        yaml_files = list(config_dir_path.glob("*.yaml")) + list(config_dir_path.glob("*.yml"))
        for yaml_file in yaml_files:
            yaml_configs = await self.load_configs_from_yaml(str(yaml_file), env_vars)
            configs.extend(yaml_configs)
        
        # 加载JSON文件
        json_files = list(config_dir_path.glob("*.json"))
        for json_file in json_files:
            json_configs = await self.load_configs_from_json(str(json_file), env_vars)
            configs.extend(json_configs)
        
        return configs
    
    async def load_tools(self, configs: List[ToolConfig]) -> Dict[str, BaseTool]:
        """加载工具实例"""
        loaded_tools = {}
        
        # 构建依赖关系
        for config in configs:
            self.dependency_resolver.add_tool(
                config.tool_id,
                config.version,
                config.dependencies
            )
        
        # 拓扑排序确定加载顺序
        try:
            sorted_tool_ids = self.dependency_resolver.topological_sort()
        except ValueError as e:
            raise RuntimeError(f"依赖解析失败: {e}")
        
        # 按依赖顺序加载工具
        for tool_id in sorted_tool_ids:
            config = self.configs.get(tool_id)
            if not config:
                continue
            
            try:
                tool = await self._create_tool_instance(config)
                self.tools[tool_id] = tool
                loaded_tools[tool_id] = tool
            except Exception as e:
                print(f"加载工具失败: {tool_id}, 错误: {e}")
                # 继续加载其他工具
                continue
        
        return loaded_tools
    
    async def _create_tool_instance(self, config: ToolConfig) -> BaseTool:
        """根据配置创建工具实例"""
        tool_type = config.tool_type
        
        if tool_type == ToolType.BUILTIN:
            return BuiltinTool(config)
        elif tool_type == ToolType.CUSTOM:
            return CustomTool(config)
        elif tool_type == ToolType.MCP:
            return MCPTool(config)
        elif tool_type == ToolType.EXTERNAL:
            return ExternalTool(config)
        elif tool_type == ToolType.SCRIPT:
            return ScriptTool(config)
        else:
            raise ValueError(f"未知的工具类型: {tool_type}")
    
    async def initialize_all(self) -> None:
        """初始化所有工具"""
        initialization_tasks = []
        
        for tool_id, tool in self.tools.items():
            if not tool.initialized:
                task = asyncio.create_task(tool.initialize())
                task.tool_id = tool_id  # 自定义属性
                initialization_tasks.append(task)
        
        # 并行初始化
        if initialization_tasks:
            results = await asyncio.gather(*initialization_tasks, return_exceptions=True)
            
            # 检查结果
            for task, result in zip(initialization_tasks, results):
                if isinstance(result, Exception):
                    print(f"工具初始化失败: {task.tool_id}, 错误: {result}")
        
        self.initialized = True
    
    async def shutdown_all(self) -> None:
        """关闭所有工具"""
        shutdown_tasks = []
        
        for tool in self.tools.values():
            if tool.initialized:
                task = asyncio.create_task(tool.shutdown())
                shutdown_tasks.append(task)
        
        # 并行关闭
        if shutdown_tasks:
            await asyncio.gather(*shutdown_tasks, return_exceptions=True)
        
        self.tools.clear()
        self.configs.clear()
        self.dependency_resolver = DependencyResolver()
        self.initialized = False
    
    async def execute_tool(self, tool_id: str, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """执行指定工具"""
        tool = self.tools.get(tool_id)
        if not tool:
            raise ValueError(f"工具未找到: {tool_id}")
        
        if not tool.initialized:
            raise ValueError(f"工具未初始化: {tool_id}")
        
        return await tool.execute(input_data)
    
    def get_tool(self, tool_id: str) -> Optional[BaseTool]:
        """获取工具实例"""
        return self.tools.get(tool_id)
    
    def list_tools(self) -> List[Dict[str, Any]]:
        """列出所有工具信息"""
        return [tool.to_dict() for tool in self.tools.values()]
    
    def validate_security_configs(self) -> List[str]:
        """验证所有工具的安全配置"""
        errors = []
        
        for tool_id, tool in self.tools.items():
            security_errors = tool.security_config.validate()
            for error in security_errors:
                errors.append(f"{tool_id}: {error}")
        
        return errors
    
    async def reload_tool(self, tool_id: str, new_config: ToolConfig) -> bool:
        """重新加载工具配置"""
        if tool_id not in self.tools:
            return False
        
        old_tool = self.tools[tool_id]
        
        try:
            # 关闭旧工具
            if old_tool.initialized:
                await old_tool.shutdown()
            
            # 创建新工具
            new_tool = await self._create_tool_instance(new_config)
            await new_tool.initialize()
            
            # 替换工具
            self.tools[tool_id] = new_tool
            self.configs[tool_id] = new_config
            
            return True
        except Exception as e:
            print(f"重新加载工具失败: {tool_id}, 错误: {e}")
            # 恢复旧工具
            if old_tool and not old_tool.initialized:
                await old_tool.initialize()
            self.tools[tool_id] = old_tool
            
            return False
```

5. **热重载支持**`reload_tool()`方法允许运行时更新工具配置，支持热升级和配置更新。
6. **安全配置验证**`validate_security_configs()`方法验证所有工具的安全配置，确保系统安全策略一致。

```python
class BuiltinTool(BaseTool):

    async def initialize(self) -> None:
        """初始化内置工具"""
        if self.config.function_name:
            # 动态导入函数
            if self.config.module_path:
                spec = importlib.util.spec_from_file_location(
                    self.config.tool_id,
                    self.config.module_path
                )
                module = importlib.util.module_from_spec(spec)
                spec.loader.exec_module(module)
                self.function = getattr(module, self.config.function_name)
            elif self.config.entry_point:
                # 从入口点导入
                module_path, func_name = self.config.entry_point.rsplit('.', 1)
                module = importlib.import_module(module_path)
                self.function = getattr(module, func_name)
        
        self.initialized = True
    
    async def execute(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """执行内置工具"""
        if not self.initialized:
            return {"error": f"工具未初始化: {self.tool_id}"}
        
        if not self.function:
            return {"error": f"工具函数未找到: {self.tool_id}"}
        
        # 验证输入
        errors = self.validate_input(input_data)
        if errors:
            return {"error": f"输入验证失败: {', '.join(errors)}"}
        
        try:
            # 执行函数
            if asyncio.iscoroutinefunction(self.function):
                result = await self.function(**input_data)
            else:
                result = self.function(**input_data)
            
            self.record_usage()
            return {"success": True, "result": result}
        except Exception as e:
            return {"error": f"执行失败: {str(e)}"}

        """关闭内置工具"""
        self.function = None
        self.initialized = False
```

```python
import asyncio
import importlib.util
from typing import Dict, Any

from src.tools.base_tool import BaseTool


class CustomTool(BaseTool):
    """自定义工具实现"""
    
    def __init__(self, config):
        super().__init__(config)
        self.instance = None
    
    async def initialize(self) -> None:
        """初始化自定义工具"""
        if not self.config.module_path:
            raise ValueError(f"自定义工具缺少module_path: {self.tool_id}")
        
        # 动态导入模块
        spec = importlib.util.spec_from_file_location(
            self.config.tool_id,
            self.config.module_path
        )
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
        
        # 创建实例
        if self.config.class_name:
            cls = getattr(module, self.config.class_name)
            self.instance = cls()
        elif self.config.function_name:
            self.instance = getattr(module, self.config.function_name)
        else:
            raise ValueError(f"自定义工具缺少class_name或function_name: {self.tool_id}")
        
        self.initialized = True
    
    async def execute(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """执行自定义工具"""
        if not self.initialized:
            return {"error": f"工具未初始化: {self.tool_id}"}
        
        # 验证输入
        errors = self.validate_input(input_data)
        if errors:
            return {"error": f"输入验证失败: {', '.join(errors)}"}
        
        try:
            # 执行实例方法或函数
            if hasattr(self.instance, 'execute'):
                if asyncio.iscoroutinefunction(self.instance.execute):
                    result = await self.instance.execute(**input_data)
                else:
                    result = self.instance.execute(**input_data)
            elif callable(self.instance):
                if asyncio.iscoroutinefunction(self.instance):
                    result = await self.instance(**input_data)
                else:
                    result = self.instance(**input_data)
            else:
                return {"error": f"自定义工具不可执行: {self.tool_id}"}
            
            self.record_usage()
            return {"success": True, "result": result}
        except Exception as e:
            return {"error": f"执行失败: {str(e)}"}
    
    async def shutdown(self) -> None:
        """关闭自定义工具"""
        if hasattr(self.instance, 'shutdown'):
            if asyncio.iscoroutinefunction(self.instance.shutdown):
                await self.instance.shutdown()
```

```python
import asyncio

from src.tools.base_tool import BaseTool


class ExternalTool(BaseTool):
    """外部工具实现"""
    
    def __init__(self, config):
        super().__init__(config)
        self.process = None
    
    async def initialize(self) -> None:
        """初始化外部工具"""
        if not self.config.command:
            raise ValueError(f"外部工具缺少command: {self.tool_id}")
        
        # 验证命令安全性
        command_parts = self.config.command.split()
        if command_parts:
            cmd = command_parts[0]
            if not self.is_command_allowed(cmd):
                raise ValueError(f"命令不被允许: {cmd}")
        
        self.initialized = True
    
    async def execute(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """执行外部工具"""
        if not self.initialized:
            return {"error": f"工具未初始化: {self.tool_id}"}
        
        if not self.config.command:
            return {"error": "未配置命令"}
        
        # 验证输入
        errors = self.validate_input(input_data)
        if errors:
            return {"error": f"输入验证失败: {', '.join(errors)}"}
        
        try:
            # 构建命令
            command = self.config.command
            args = input_data.get("args", [])
            
            # 安全检查
            cmd_parts = command.split()
            if cmd_parts and not self.is_command_allowed(cmd_parts[0]):
                return {"error": f"命令不被允许: {cmd_parts[0]}"}
            
            # 设置环境变量
            env = os.environ.copy()
            env.update(self.config.environment_variables)
            
            # 执行命令
            process = subprocess.Popen(
                [command] + args,
                shell=False,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                env=env,
                text=True
            )
            
            # 等待完成（带超时）
            timeout = self.security_config.max_execution_time
            try:
                stdout, stderr = process.communicate(timeout=timeout)
                return_code = process.returncode
            except subprocess.TimeoutExpired:
                process.kill()
                stdout, stderr = process.communicate()
                return {"error": f"命令执行超时: {timeout}秒"}
            
            self.record_usage()
            return {
                "success": return_code == 0,
                "return_code": return_code,
                "stdout": stdout,
                "stderr": stderr
            }
        except Exception as e:
            return {"error": f"执行失败: {str(e)}"}
    
    async def shutdown(self) -> None:
        """关闭外部工具"""
        if self.process and self.process.poll() is None:
            self.process.terminate()
            try:
                self.process.wait(timeout=5)
            except subprocess.TimeoutExpired:
                self.process.kill()
        
        self.process = None
        self.initialized = False
```

2. **内置工具**通过动态导入实现函数调用，支持同步和异步函数。这是性能最高、安全性最好的工具类型。
3. **自定义工具**支持从外部模块加载类或函数，提供了最大的灵活性。支持有状态工具（通过类实例）和无状态工具（通过函数）。

```python
                "max_processes": None, # 无限制
                "max_files": None,     # 无限制

                "network_access": False,
                "file_system_access": True
            }

                "memory_mb": 100,
                "max_processes": 5,
                "max_files": 50,
                "network_access": False,
                "file_system_access": False
            }
        else:  # UNTRUSTED

                "cpu_time": 0,         # 禁止执行
                "memory_mb": 0,
                "max_processes": 0,
                "max_files": 0,
                "network_access": False,
                "file_system_access": False
            }
    
    async def execute(self, command: str, args: List[str], 
                     env: Dict[str, str], working_dir: Optional[str] = None) -> Dict[str, Any]:
        """在沙箱中执行命令"""
        if self.security_level == SecurityLevel.UNTRUSTED:
            return {"error": "安全级别禁止执行"}
        
        # 创建临时目录
        self.temp_dir = tempfile.mkdtemp(prefix=f"{self.sandbox_id}_")
        
        try:
            # 设置资源限制
            self._set_resource_limits()
            
            # 准备执行环境
            execution_env = self._prepare_execution_env(env)
            execution_dir = working_dir or self.temp_dir
            
            # 执行命令
            result = await self._execute_command(
                command, args, execution_env, execution_dir
            )
            
            return result
        except Exception as e:
            return {"error": f"沙箱执行失败: {str(e)}"}
        finally:
            # 清理临时目录
            self._cleanup()
    
    def _set_resource_limits(self):
        """设置资源限制"""
        limits = self.resource_limits
        
        if limits["cpu_time"]:
            # 设置CPU时间限制（需要平台特定实现）
            pass
        
        if limits["memory_mb"]:
            # 设置内存限制
            memory_limit = limits["memory_mb"] * 1024 * 1024  # 转换为字节
            try:
                resource.setrlimit(resource.RLIMIT_AS, (memory_limit, memory_limit))
            except (ValueError, resource.error):
                pass  # 在某些平台上可能不支持
        
        if limits["max_processes"]:
            # 设置最大进程数
            try:
                resource.setrlimit(resource.RLIMIT_NPROC, 
                                 (limits["max_processes"], limits["max_processes"]))
            except (ValueError, resource.error):
                pass
        
        if limits["max_files"]:
            # 设置最大文件数
            try:
                resource.setrlimit(resource.RLIMIT_NOFILE,
                                 (limits["max_files"], limits["max_files"]))
            except (ValueError, resource.error):
                pass
    
    def _prepare_execution_env(self, env: Dict[str, str]) -> Dict[str, str]:
        """准备执行环境"""
        execution_env = os.environ.copy()
        execution_env.update(env)
        
        # 根据安全级别限制环境变量
        if self.security_level in [SecurityLevel.SANDBOXED, SecurityLevel.UNTRUSTED]:
            # 只保留基本的环境变量
            allowed_vars = {"PATH", "HOME", "TEMP", "TMP", "LANG", "LC_ALL"}
            execution_env = {k: v for k, v in execution_env.items() 
                           if k in allowed_vars}
        
        return execution_env
    
    async def _execute_command(self, command: str, args: List[str],
                             env: Dict[str, str], working_dir: str) -> Dict[str, Any]:
        """执行命令"""
        cmd_list = [command] + args
        
        try:
            # 使用subprocess执行
            process = subprocess.Popen(
                cmd_list,
                shell=False,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                env=env,
                cwd=working_dir,
                text=True
            )
            
            # 设置超时
            timeout = self.resource_limits.get("cpu_time")
            
            # 使用线程监控超时
            def kill_process():
                process.kill()
            
            timer = None
            if timeout:
                timer = threading.Timer(timeout, kill_process)
                timer.start()
            
            try:
                stdout, stderr = process.communicate()
                return_code = process.returncode
            finally:
                if timer:
                    timer.cancel()
            
            # 检查是否超时被杀
            if timer and timer.is_alive():
                timer.cancel()
            else:
                # 可能是超时被杀
                if process.returncode == -9:  # SIGKILL
                    return {"error": f"执行超时: {timeout}秒"}
            
            return {
                "success": return_code == 0,
                "return_code": return_code,
                "stdout": stdout,
                "stderr": stderr
            }
        except Exception as e:
            return {"error": f"命令执行失败: {str(e)}"}
    
    def _cleanup(self):
        """清理沙箱环境"""
        if self.temp_dir and os.path.exists(self.temp_dir):
            try:
                shutil.rmtree(self.temp_dir, ignore_errors=True)
            except:
                pass
            self.temp_dir = None
    
    @contextmanager
    def create_isolated_env(self):
        """创建隔离环境上下文管理器"""
        temp_dir = tempfile.mkdtemp(prefix=f"{self.sandbox_id}_isolated_")
        
        try:
            # 设置资源限制
            self._set_resource_limits()
            
            # 改变工作目录
            old_cwd = os.getcwd()
            os.chdir(temp_dir)
            
            yield temp_dir
        finally:
            # 恢复工作目录
            os.chdir(old_cwd)
            
            # 清理临时目录
            if os.path.exists(temp_dir):
                shutil.rmtree(temp_dir, ignore_errors=True)
```

2. **资源限制**使用Python的`resource`模块设置进程资源限制，包括内存、进程数、文件数等。这是操作系统级别的限制，比应用级限制更可靠。
3. **环境隔离**通过临时目录和改变工作目录实现文件系统隔离。这可以防止工具访问系统文件。
4. **超时控制**使用线程计时器监控执行时间，超时后终止进程。这可以防止工具无限运行。
5. **上下文管理器**`create_isolated_env()`方法提供了方便的上下文管理器接口，确保资源正确清理。

```python
# src/orchestrator/tool_orchestrator.py
import asyncio
import yaml
import json
from typing import Dict, Any, List, Optional, Set
from dataclasses import dataclass, field
from enum import Enum
import graphlib

from src.loader.tool_loader import ToolLoader


class NodeType(Enum):
    """节点类型"""
    TOOL = "tool"          # 工具节点
    CONDITION = "condition"  # 条件节点
    PARALLEL = "parallel"    # 并行节点
    SEQUENCE = "sequence"    # 序列节点


@dataclass
class WorkflowNode:
    """工作流节点"""
    node_id: str                          # 节点ID
    node_type: NodeType                   # 节点类型
    tool_id: Optional[str] = None         # 工具ID（对于工具节点）
    condition: Optional[str] = None       # 条件表达式（对于条件节点）
    inputs: Dict[str, Any] = field(default_factory=dict)  # 输入映射
    outputs: List[str] = field(default_factory=list)      # 输出名称
    depends_on: List[str] = field(default_factory=list)   # 依赖节点
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return {
            "node_id": self.node_id,
            "node_type": self.node_type.value,
            "tool_id": self.tool_id,
            "condition": self.condition,
            "inputs": self.inputs,
            "outputs": self.outputs,
            "depends_on": self.depends_on
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "WorkflowNode":
        """从字典创建"""
        return cls(
            node_id=data["node_id"],
            node_type=NodeType(data["node_type"]),
            tool_id=data.get("tool_id"),
            condition=data.get("condition"),
            inputs=data.get("inputs", {}),
            outputs=data.get("outputs", []),
            depends_on=data.get("depends_on", [])
        )


@dataclass
class WorkflowDefinition:
    """工作流定义"""
    name: str                               # 工作流名称
    version: str = "1.0"                    # 版本
    description: Optional[str] = None       # 描述
    nodes: List[WorkflowNode] = field(default_factory=list)  # 节点列表
    error_handling: Dict[str, Any] = field(default_factory=dict)  # 错误处理配置
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return {
            "name": self.name,
            "version": self.version,
            "description": self.description,
            "nodes": [node.to_dict() for node in self.nodes],
            "error_handling": self.error_handling
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "WorkflowDefinition":
        """从字典创建"""
        nodes = [WorkflowNode.from_dict(node_data) for node_data in data.get("nodes", [])]
        return cls(
            name=data["name"],
            version=data.get("version", "1.0"),
            description=data.get("description"),
            nodes=nodes,
            error_handling=data.get("error_handling", {})
        )


class ToolOrchestrator:
    """工具编排引擎"""
    
    def __init__(self, tool_loader: ToolLoader):
        self.tool_loader = tool_loader
        self.workflows: Dict[str, WorkflowDefinition] = {}
        self.execution_history: Dict[str, List[Dict[str, Any]]] = {}
    
    async def load_workflow_from_yaml(self, yaml_path: str) -> WorkflowDefinition:
        """从YAML加载工作流定义"""
        with open(yaml_path, 'r', encoding='utf-8') as f:
            yaml_data = yaml.safe_load(f)
        
        workflow = WorkflowDefinition.from_dict(yaml_data)
        self.workflows[workflow.name] = workflow
        
        return workflow
    
    async def load_workflow_from_json(self, json_path: str) -> WorkflowDefinition:
        """从JSON加载工作流定义"""
        with open(json_path, 'r', encoding='utf-8') as f:
            json_data = json.load(f)
        
        workflow = WorkflowDefinition.from_dict(json_data)
        self.workflows[workflow.name] = workflow
        
        return workflow
    
    def validate_workflow(self, workflow: WorkflowDefinition) -> List[str]:
        """验证工作流定义"""
        errors = []
        
        # 检查节点ID唯一性
        node_ids = set()
        for node in workflow.nodes:
            if node.node_id in node_ids:
                errors.append(f"节点ID重复: {node.node_id}")
            node_ids.add(node.node_id)
        
        # 检查依赖关系
        all_node_ids = {node.node_id for node in workflow.nodes}
        for node in workflow.nodes:
            for dep in node.depends_on:
                if dep not in all_node_ids:
                    errors.append(f"节点 {node.node_id} 依赖不存在的节点: {dep}")
        
        # 检查工具节点是否有tool_id
        for node in workflow.nodes:
            if node.node_type == NodeType.TOOL and not node.tool_id:
                errors.append(f"工具节点缺少tool_id: {node.node_id}")
        
        # 检查循环依赖
        try:
            self._topological_sort(workflow)
        except graphlib.CycleError as e:
            errors.append(f"工作流存在循环依赖: {e}")
        
        return errors
    
    def _topological_sort(self, workflow: WorkflowDefinition) -> List[str]:
        """拓扑排序工作流节点"""
        graph = {}
        for node in workflow.nodes:
            graph[node.node_id] = node.depends_on
        
        sorter = graphlib.TopologicalSorter(graph)
        return list(sorter.static_order())
    
    async def execute_workflow(self, workflow_name: str, 
                             initial_inputs: Dict[str, Any] = None) -> Dict[str, Any]:
        """执行工作流"""
        workflow = self.workflows.get(workflow_name)
        if not workflow:
            return {"error": f"工作流未找到: {workflow_name}"}
        
        # 验证工作流
        errors = self.validate_workflow(workflow)
        if errors:
            return {"error": f"工作流验证失败: {errors}"}
        
        # 初始化执行上下文
        execution_id = f"{workflow_name}_{int(asyncio.get_event_loop().time())}"
        context = {
            "execution_id": execution_id,
            "inputs": initial_inputs or {},
            "node_outputs": {},
            "errors": []
        }
        
        # 记录执行开始
        self._record_execution_start(execution_id, workflow_name, initial_inputs)
        
        try:
            # 拓扑排序确定执行顺序
            execution_order = self._topological_sort(workflow)
            
            # 按顺序执行节点
            for node_id in execution_order:
                node = next(n for n in workflow.nodes if n.node_id == node_id)
                
                # 执行节点
                result = await self._execute_node(node, context, workflow.error_handling)
                
                # 保存输出
                if result.get("success", False):
                    context["node_outputs"][node_id] = result.get("outputs", {})
                else:
                    context["errors"].append({
                        "node_id": node_id,
                        "error": result.get("error", "未知错误")
                    })
                    
                    # 错误处理
                    if not workflow.error_handling.get("continue_on_error", False):
                        break
            
            # 构建最终结果
            final_result = self._build_final_result(context, workflow)
            
            # 记录执行完成
            self._record_execution_complete(execution_id, final_result)
            
            return final_result
        except Exception as e:
            error_result = {"error": f"工作流执行失败: {str(e)}"}
            self._record_execution_error(execution_id, str(e))
            return error_result
    
    async def _execute_node(self, node: WorkflowNode, context: Dict[str, Any],
                          error_handling: Dict[str, Any]) -> Dict[str, Any]:
        """执行单个节点"""
        try:
            if node.node_type == NodeType.TOOL:
                return await self._execute_tool_node(node, context)
            elif node.node_type == NodeType.CONDITION:
                return await self._execute_condition_node(node, context)
            elif node.node_type == NodeType.PARALLEL:
                return await self._execute_parallel_node(node, context)
            elif node.node_type == NodeType.SEQUENCE:
                return await self._execute_sequence_node(node, context)
            else:
                return {"error": f"未知的节点类型: {node.node_type}"}
        except Exception as e:
            # 错误重试
            retry_count = error_handling.get("retry_count", 0)
            retry_delay = error_handling.get("retry_delay", 1.0)
            
            for attempt in range(retry_count):
                try:
                    await asyncio.sleep(retry_delay)
                    if node.node_type == NodeType.TOOL:
                        return await self._execute_tool_node(node, context)
                    else:
                        return {"error": f"重试失败: {str(e)}"}
                except Exception as retry_error:
                    e = retry_error
            
            return {"error": f"节点执行失败: {str(e)}"}
    
    async def _execute_tool_node(self, node: WorkflowNode, context: Dict[str, Any]) -> Dict[str, Any]:
        """执行工具节点"""
        # 解析输入（支持模板）
        inputs = self._resolve_inputs(node.inputs, context)
        
        # 执行工具
        result = await self.tool_loader.execute_tool(node.tool_id, inputs)
        
        if result.get("success", False):
            # 提取输出
            outputs = {}
            for output_name in node.outputs:
                # 支持输出映射
                if output_name in result:
                    outputs[output_name] = result[output_name]
                elif "result" in result:
                    outputs[output_name] = result["result"]
            
            return {"success": True, "outputs": outputs}
        else:
            return {"success": False, "error": result.get("error", "工具执行失败")}
    
    async def _execute_parallel_node(self, node: WorkflowNode, context: Dict[str, Any]) -> Dict[str, Any]:
        """执行并行节点"""
        # 获取子节点
        child_nodes = [n for n in context.get("workflow_nodes", []) 
                      if n.depends_on and node.node_id in n.depends_on]
        
        # 并行执行
        tasks = []
        for child_node in child_nodes:
            task = asyncio.create_task(self._execute_node(child_node, context, {}))
            tasks.append(task)
        
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 合并结果
        all_success = all(not isinstance(r, Exception) and r.get("success", False) 
                         for r in results if not isinstance(r, Exception))
        
        if all_success:
            outputs = {}
            for i, child_node in enumerate(child_nodes):
                if not isinstance(results[i], Exception):
                    outputs.update(results[i].get("outputs", {}))
            
            return {"success": True, "outputs": outputs}
        else:
            errors = [r if isinstance(r, Exception) else r.get("error") 
                     for r in results if isinstance(r, Exception) or not r.get("success", False)]
            return {"success": False, "error": f"并行节点执行失败: {errors}"}
    
    def _resolve_inputs(self, input_mapping: Dict[str, Any], context: Dict[str, Any]) -> Dict[str, Any]:
        """解析输入映射"""
        resolved = {}
        
        for key, value in input_mapping.items():
            if isinstance(value, str) and value.startswith("{{") and value.endswith("}}"):
                # 模板表达式
                expr = value[2:-2].strip()
                resolved_value = self._evaluate_expression(expr, context)
                resolved[key] = resolved_value
            else:
                resolved[key] = value
        
        return resolved
    
    def _evaluate_expression(self, expr: str, context: Dict[str, Any]) -> Any:
        """计算表达式"""
        # 简化实现：支持节点输出引用
        if expr.startswith("nodes."):
            parts = expr.split(".")
            if len(parts) >= 3 and parts[0] == "nodes":
                node_id = parts[1]
                output_key = ".".join(parts[2:])
                
                node_outputs = context.get("node_outputs", {})
                if node_id in node_outputs:
                    if output_key in node_outputs[node_id]:
                        return node_outputs[node_id][output_key]
                    elif output_key == "":
                        return node_outputs[node_id]
        
        # 支持简单Python表达式（安全环境）
        try:
            # 限制可用的内置函数
            safe_builtins = {
                "len": len,
                "str": str,
                "int": int,
                "float": float,
                "bool": bool,
                "list": list,
                "dict": dict,
                "tuple": tuple,
                "set": set
            }
            
            # 创建安全环境
            safe_env = {
                "__builtins__": safe_builtins,
                "context": context
            }
            
            return eval(expr, {"__builtins__": safe_builtins}, {"context": context})
        except:
            return expr
    
    def _build_final_result(self, context: Dict[str, Any], 
                          workflow: WorkflowDefinition) -> Dict[str, Any]:
        """构建最终结果"""
        # 收集所有输出
        final_outputs = {}
        for node_outputs in context["node_outputs"].values():
            final_outputs.update(node_outputs)
        
        return {
            "success": len(context["errors"]) == 0,
            "execution_id": context["execution_id"],
            "outputs": final_outputs,
            "errors": context["errors"],
            "nodes_executed": len(context["node_outputs"]),
            "total_nodes": len(workflow.nodes)
        }
    
    def _record_execution_start(self, execution_id: str, workflow_name: str,
                              inputs: Dict[str, Any]):
        """记录执行开始"""
        if execution_id not in self.execution_history:
            self.execution_history[execution_id] = []
        
        self.execution_history[execution_id].append({
            "timestamp": asyncio.get_event_loop().time(),
            "event": "start",
            "workflow": workflow_name,
            "inputs": inputs
        })
    
    def _record_execution_complete(self, execution_id: str, result: Dict[str, Any]):
        """记录执行完成"""
        if execution_id in self.execution_history:
            self.execution_history[execution_id].append({
                "timestamp": asyncio.get_event_loop().time(),
                "event": "complete",
                "result": result
            })
    
    def _record_execution_error(self, execution_id: str, error: str):
        """记录执行错误"""
        if execution_id in self.execution_history:
            self.execution_history[execution_id].append({
                "timestamp": asyncio.get_event_loop().time(),
                "event": "error",
                "error": error
            })
    
    def get_execution_history(self, execution_id: str) -> List[Dict[str, Any]]:
        """获取执行历史"""
        return self.execution_history.get(execution_id, [])
    
    def get_workflow_statistics(self, workflow_name: str) -> Dict[str, Any]:
        """获取工作流统计信息"""
        executions = [h for exec_id, history in self.execution_history.items()
                     for event in history if event.get("workflow") == workflow_name]
        
        if not executions:
            return {}
        
        success_count = sum(1 for e in executions if e.get("event") == "complete" 
                           and e.get("result", {}).get("success", False))
        error_count = sum(1 for e in executions if e.get("event") == "error")
        
        return {
            "total_executions": len([e for e in executions if e["event"] == "start"]),
            "success_count": success_count,
            "error_count": error_count,
            "success_rate": success_count / max(1, len(executions) / 2)  # 每个执行有开始和结束事件
        }
```

**解析**：
1. **工作流定义**使用YAML或JSON格式定义工具编排逻辑，支持节点依赖、条件分支和并行执行。
2. **拓扑排序**确保节点按依赖顺序执行，自动检测循环依赖。
3. **输入模板**支持表达式语言引用其他节点的输出，实现了节点间的数据传递。
4. **错误处理和重试**支持配置化的错误处理策略，包括重试次数和重试延迟。
5. **执行历史记录**详细记录每个工作流执行的过程和结果，便于调试和监控。
6. **并行执行**使用`asyncio.gather()`实现节点并行执行，提高工作流执行效率。

```python
            window = sorted_metrics[i:i+window_size]
            success_count = sum(1 for m in window if m.success)
            success_rates.append(success_count / len(window) if window else 0)
        
        if len(success_rates) >= 2:
            if success_rates[-1] > success_rates[0] + 0.1:
                success_trend = "improving"
            elif success_rates[-1] < success_rates[0] - 0.1:
                success_trend = "declining"
            else:
                success_trend = "stable"
        else:
            success_trend = "unknown"

            "success_trend": success_trend,

            [(tool_id, r["failure_count"]) for tool_id, r in tool_reports.items()],

            reverse=True

            "total_tools": len(tool_reports),
            "total_executions": total_executions,
            "total_successes": total_successes,
            "total_failures": total_failures,
            "overall_success_rate": total_successes / total_executions if total_executions else 0,
            "busiest_tools": busy_tools,
            "most_unreliable_tools": unreliable_tools,
            "period_covered": f"{min(r.get('min_timestamp', 0) for r in tool_reports.values())} to {max(r.get('max_timestamp', 0) for r in tool_reports.values())}"
        }
```

```python
        await self._update_local_model(aggregated_model)
    
    def _protect_privacy(self, update):
        # 差分隐私：添加噪声
        if self.privacy_method == "differential_privacy":
            noise = np.random.laplace(0, self.epsilon)
            return update + noise
        
        # 同态加密
        elif self.privacy_method == "homomorphic_encryption":
            return self.encryptor.encrypt(update)
        
        # 安全多方计算
        elif self.privacy_method == "secure_multiparty":
            return self.smpc.share(update)
```
