### 参考实现：skill_state.py

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
技能状态数据类
任务1：技能状态数据类设计
"""

import time
import dataclasses
from typing import Optional, List
from dataclasses import dataclass, field
from enum import Enum


class SkillStatus(Enum):
    """技能状态枚举"""
    IDLE = "idle"            # 空闲
    LOADING = "loading"      # 加载中
    READY = "ready"          # 准备就绪
    EXECUTING = "executing"  # 执行中
    ERROR = "error"          # 错误
    DISABLED = "disabled"    # 已禁用


class HealthLevel(Enum):
    """健康度等级"""
    CRITICAL = "critical"    # 危急：0-0.2
    POOR = "poor"            # 差：0.2-0.5
    FAIR = "fair"            # 一般：0.5-0.8
    GOOD = "good"            # 良好：0.8-0.95
    EXCELLENT = "excellent"  # 优秀：0.95-1.0


@dataclass
class SkillState:
    """技能状态数据类
    
    Attributes:
        skill_name: 技能名称（唯一标识）
        version: 技能版本
        enabled: 是否启用
        last_used: 最后使用时间（Unix时间戳）
        usage_count: 总使用次数
        success_count: 成功次数
        failure_count: 失败次数
        avg_execution_time: 平均执行时间（秒）
        current_status: 当前状态（SkillStatus枚举值）
        error_message: 错误信息（如果有）
        created_at: 状态创建时间
        last_updated: 最后更新时间
        tags: 技能标签（用于分类）
    """
    
    # 基本信息
    skill_name: str
    version: str = "1.0.0"
    enabled: bool = True
    
    # 使用统计
    last_used: float = 0.0
    usage_count: int = 0
    success_count: int = 0
    failure_count: int = 0
    avg_execution_time: float = 0.0
    
    # 状态信息
    current_status: str = SkillStatus.IDLE.value
    error_message: Optional[str] = None
    
    # 元数据
    created_at: float = field(default_factory=time.time)
    last_updated: float = field(default_factory=time.time)
    tags: List[str] = field(default_factory=list)
    
    @property
    def total_attempts(self) -> int:
        """总尝试次数"""
        return self.success_count + self.failure_count
    
    @property
    def success_rate(self) -> float:
        """计算成功率（0.0到1.0）"""
        if self.total_attempts == 0:
            return 0.0
        return self.success_count / self.total_attempts
    
    @property
    def availability(self) -> float:
        """可用性得分（基于启用状态和错误状态）"""
        if not self.enabled:
            return 0.0
        if self.current_status == SkillStatus.ERROR.value:
            return 0.3
        if self.current_status == SkillStatus.DISABLED.value:
            return 0.5
        return 1.0
    
    @property
    def performance_score(self) -> float:
        """性能得分（基于响应时间）"""
        if self.avg_execution_time <= 0:
            return 1.0
        
        # 响应时间评分：小于1秒为满分，超过10秒为0分
        if self.avg_execution_time < 1.0:
            return 1.0
        elif self.avg_execution_time > 10.0:
            return 0.0
        else:
            return 1.0 - (self.avg_execution_time - 1.0) / 9.0
    
    @property
    def usage_score(self) -> float:
        """使用频率得分（基于使用次数）"""
        if self.usage_count == 0:
            return 0.5  # 未使用的技能给中等分数
        
        # 使用次数评分：超过100次为满分
        return min(self.usage_count / 100.0, 1.0)
    
    @property
    def health_score(self) -> float:
        """综合健康度得分（0.0到1.0）
        
        健康度计算公式：
        健康度 = 成功率权重 × 成功率 + 可用性权重 × 可用性 + 
                性能权重 × 性能得分 + 使用权重 × 使用频率得分
        
        权重分配：
        - 成功率：0.4（最重要）
        - 可用性：0.3（次重要）
        - 性能：0.2（重要）
        - 使用频率：0.1（参考）
        """
        weights = {
            'success_rate': 0.4,
            'availability': 0.3,
            'performance': 0.2,
            'usage': 0.1
        }
        
        scores = {
            'success_rate': self.success_rate,
            'availability': self.availability,
            'performance': self.performance_score,
            'usage': self.usage_score
        }
        
        # 计算加权平均
        total_score = 0.0
        for key, weight in weights.items():
            total_score += scores[key] * weight
        
        return round(total_score, 3)
    
    @property
    def health_level(self) -> HealthLevel:
        """健康度等级（critical/poor/fair/good/excellent）"""
        score = self.health_score
        
        if score >= 0.95:
            return HealthLevel.EXCELLENT
        elif score >= 0.8:
            return HealthLevel.GOOD
        elif score >= 0.5:
            return HealthLevel.FAIR
        elif score >= 0.2:
            return HealthLevel.POOR
        else:
            return HealthLevel.CRITICAL
    
    def update_from_execution(self, success: bool, execution_time: float):
        """根据执行结果更新状态
        
        Args:
            success: 是否执行成功
            execution_time: 执行时间（秒）
        """
        # 更新使用统计
        self.last_used = time.time()
        self.usage_count += 1
        
        if success:
            self.success_count += 1
        else:
            self.failure_count += 1
        
        # 更新平均执行时间（指数移动平均）
        if self.total_attempts == 1:
            self.avg_execution_time = execution_time
        else:
            # EMA系数：0.3，给予新数据更高权重
            alpha = 0.3
            self.avg_execution_time = (
                alpha * execution_time + 
                (1 - alpha) * self.avg_execution_time
            )
        
        # 更新状态
        if success:
            self.current_status = SkillStatus.READY.value
            self.error_message = None
        else:
            self.current_status = SkillStatus.ERROR.value
            self.error_message = "Execution failed"
        
        # 更新最后修改时间
        self.last_updated = time.time()
    
    def to_dict(self) -> dict:
        """转换为字典（用于JSON序列化）"""
        data = dataclasses.asdict(self)
        # 添加计算属性
        data['success_rate'] = self.success_rate
        data['health_score'] = self.health_score
        data['health_level'] = self.health_level.value
        data['total_attempts'] = self.total_attempts
        return data
```

### 设计思路

1. **数据类设计**：使用Python的`@dataclass`装饰器简化数据类定义，自动生成`__init__`、`__repr__`等方法，减少样板代码。
2. **字段分类**：将字段分为基本信息、使用统计、状态信息和元数据四类，提高代码可读性和维护性。
3. **计算属性**：使用`@property`装饰器定义计算属性，将复杂的计算逻辑封装为属性访问，提供统一的接口。
4. **健康度算法**：采用加权平均算法计算综合健康度，权重分配反映不同指标的重要性（成功率最重要，使用频率最次要）。
5. **状态更新**：实现`update_from_execution`方法统一处理执行结果更新，确保状态一致性和数据完整性。

### 关键要点

1. **枚举类型**：使用`Enum`定义状态和健康度等级，避免魔法字符串，提高类型安全性和代码可读性。
2. **默认值设置**：使用`field(default_factory=...)`为时间戳字段设置动态默认值，确保每个实例都有唯一的时间戳。
3. **指数移动平均**：使用EMA算法计算平均执行时间，给予新数据更高权重（α=0.3），快速反映性能变化。
4. **健康度等级划分**：根据健康度得分划分为5个等级，提供直观的状态评估。
5. **序列化支持**：实现`to_dict`方法，将数据类转换为字典，方便JSON序列化和持久化存储。

### 常见错误

1. **JSON序列化错误**：
   ```python
   # 错误：TypeError: Object of type datetime is not JSON serializable
   # 解决方案：使用dataclasses.asdict()或自定义序列化方法
   json.dumps(state.to_dict(), indent=2)
   ```

2. **除以零错误**：
   ```python
   # 错误：ZeroDivisionError: division by zero
   # 解决方案：在计算成功率前检查total_attempts是否为0
   if self.total_attempts == 0:
       return 0.0
   ```

3. **状态更新不一致**：
   ```python
   # 错误：忘记更新last_updated时间戳
   # 解决方案：在update_from_execution方法中统一更新时间戳
   self.last_updated = time.time()
   ```

4. **性能得分计算错误**：
   ```python
   # 错误：线性插值计算错误
   # 解决方案：使用正确的线性插值公式
   return 1.0 - (self.avg_execution_time - 1.0) / 9.0
   ```

5. **健康度权重总和不为1**：
   ```python
   # 错误：权重总和不为1导致健康度得分超出预期范围
   # 解决方案：确保权重总和为1.0
   ```

## 🎯 任务2：技能状态管理器实现

### 参考实现：skill_state_manager.py

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
技能状态管理器
任务2：技能状态管理器实现
"""

import os
import json
import tempfile
import threading
import asyncio
from typing import Dict, Optional, Any, List
from pathlib import Path
from dataclasses import asdict
from skill_state import SkillState


class SkillStateManager:
    """技能状态管理器
    
    负责技能状态的加载、保存、更新和查询，支持并发安全和原子操作。
    
    Attributes:
        storage_path: 状态存储文件路径
        states: 技能状态字典（skill_name -> SkillState）
        _lock: 线程锁，确保并发安全
        _history: 状态变化历史记录
        _event_handlers: 事件处理器列表
    """
    
    def __init__(self, storage_path: str = "skill_states.json"):
        """初始化状态管理器
        
        Args:
            storage_path: 状态存储文件路径
        """
        self.storage_path = storage_path
        self.states: Dict[str, SkillState] = {}
        self._lock = threading.RLock()  # 可重入锁
        self._history: List[Dict[str, Any]] = []
        self._event_handlers: List[callable] = []
        
        # 自动加载现有状态
        self.load_states()
    
    def load_states(self) -> bool:
        """从存储文件加载技能状态
        
        Returns:
            bool: 加载是否成功
        """
        try:
            if not os.path.exists(self.storage_path):
                return True  # 文件不存在视为空状态
            
            with open(self.storage_path, 'r', encoding='utf-8') as f:
                data = json.load(f)
            
            with self._lock:
                self.states.clear()
                for skill_name, state_data in data.items():
                    # 将字典转换回SkillState对象
                    state = SkillState(**state_data)
                    self.states[skill_name] = state
                
                # 记录加载历史
                self._add_history("load", {
                    "skill_count": len(self.states),
                    "storage_path": self.storage_path
                })
            
            return True
            
        except (json.JSONDecodeError, IOError) as e:
            print(f"加载状态失败: {e}")
            return False
    
    def save_states(self) -> bool:
        """保存技能状态到存储文件（原子操作）
        
        原子保存步骤：
        1. 创建临时文件
        2. 写入状态数据
        3. 重命名临时文件为目标文件
        
        Returns:
            bool: 保存是否成功
        """
        try:
            # 准备序列化数据
            data = {}
            with self._lock:
                for skill_name, state in self.states.items():
                    data[skill_name] = state.to_dict()
            
            # 创建临时文件
            temp_dir = os.path.dirname(self.storage_path) or "."
            with tempfile.NamedTemporaryFile(
                mode='w', 
                encoding='utf-8',
                dir=temp_dir,
                delete=False
            ) as temp_file:
                temp_path = temp_file.name
                json.dump(data, temp_file, indent=2, ensure_ascii=False)
            
            # 原子重命名（在Unix上是原子的，在Windows上需要特殊处理）
            try:
                os.replace(temp_path, self.storage_path)
            except OSError:
                # Windows可能不支持原子替换，使用备份方案
                if os.path.exists(self.storage_path):
                    os.remove(self.storage_path)
                os.rename(temp_path, self.storage_path)
            
            # 记录保存历史
            self._add_history("save", {
                "skill_count": len(self.states),
                "storage_path": self.storage_path
            })
            
            return True
            
        except (IOError, OSError) as e:
            print(f"保存状态失败: {e}")
            # 清理临时文件
            if 'temp_path' in locals() and os.path.exists(temp_path):
                os.remove(temp_path)
            return False
    
    def get_state(self, skill_name: str) -> Optional[SkillState]:
        """获取技能状态
        
        Args:
            skill_name: 技能名称
            
        Returns:
            Optional[SkillState]: 技能状态，不存在则返回None
        """
        with self._lock:
            return self.states.get(skill_name)
    
    def get_all_states(self) -> Dict[str, SkillState]:
        """获取所有技能状态
        
        Returns:
            Dict[str, SkillState]: 技能状态字典的副本
        """
        with self._lock:
            return self.states.copy()
    
    async def update_state(self, skill_name: str, updates: Dict[str, Any], auto_save: bool = True) -> bool:
        """更新技能状态（支持异步）
        
        Args:
            skill_name: 技能名称
            updates: 更新字段字典
            auto_save: 是否自动保存
            
        Returns:
            bool: 更新是否成功
        """
        try:
            with self._lock:
                # 获取或创建状态
                if skill_name not in self.states:
                    state = SkillState(skill_name=skill_name)
                    self.states[skill_name] = state
                else:
                    state = self.states[skill_name]
                
                # 记录原始值（用于事件通知）
                old_values = {}
                for key in updates.keys():
                    if hasattr(state, key):
                        old_values[key] = getattr(state, key)
                
                # 应用更新
                for key, value in updates.items():
                    if hasattr(state, key):
                        setattr(state, key, value)
                
                # 更新最后修改时间
                state.last_updated = time.time()
                
                # 记录历史
                self._add_history("update", {
                    "skill_name": skill_name,
                    "updates": updates,
                    "old_values": old_values
                })
                
                # 触发事件
                self._trigger_event("state_updated", {
                    "skill_name": skill_name,
                    "state": state,
                    "updates": updates,
                    "old_values": old_values
                })
            
            # 自动保存
            if auto_save:
                # 在后台线程中执行保存，避免阻塞异步循环
                loop = asyncio.get_event_loop()
                await loop.run_in_executor(None, self.save_states)
            
            return True
            
        except Exception as e:
            print(f"更新状态失败: {e}")
            return False
    
    def get_health_summary(self) -> Dict[str, Any]:
        """获取健康度摘要
        
        Returns:
            Dict[str, Any]: 健康度摘要信息
        """
        with self._lock:
            if not self.states:
                return {
                    "total_skills": 0,
                    "avg_health_score": 0.0,
                    "health_distribution": {},
                    "unhealthy_skills": []
                }
            
            total_skills = len(self.states)
            health_scores = [state.health_score for state in self.states.values()]
            avg_health = sum(health_scores) / total_skills
            
            # 健康度分布
            distribution = {
                "excellent": 0,
                "good": 0,
                "fair": 0,
                "poor": 0,
                "critical": 0
            }
            
            unhealthy_skills = []
            for skill_name, state in self.states.items():
                level = state.health_level.value
                distribution[level] = distribution.get(level, 0) + 1
                
                # 记录不健康的技能
                if state.health_score < 0.5:
                    unhealthy_skills.append({
                        "skill_name": skill_name,
                        "health_score": state.health_score,
                        "health_level": level,
                        "issues": self._analyze_issues(state)
                    })
            
            return {
                "total_skills": total_skills,
                "avg_health_score": round(avg_health, 3),
                "health_distribution": distribution,
                "unhealthy_skills": unhealthy_skills,
                "timestamp": time.time()
            }
    
    def add_event_handler(self, handler: callable):
        """添加事件处理器
        
        Args:
            handler: 事件处理器函数，接收(event_type, event_data)参数
        """
        with self._lock:
            self._event_handlers.append(handler)
    
    def remove_event_handler(self, handler: callable):
        """移除事件处理器
        
        Args:
            handler: 要移除的事件处理器
        """
        with self._lock:
            if handler in self._event_handlers:
                self._event_handlers.remove(handler)
    
    def _add_history(self, action: str, data: Dict[str, Any]):
        """添加历史记录
        
        Args:
            action: 操作类型
            data: 操作数据
        """
        # 限制历史记录大小
        if len(self._history) >= 1000:
            self._history.pop(0)
        
        self._history.append({
            "timestamp": time.time(),
            "action": action,
            "data": data
        })
    
    def _trigger_event(self, event_type: str, event_data: Dict[str, Any]):
        """触发事件
        
        Args:
            event_type: 事件类型
            event_data: 事件数据
        """
        for handler in self._event_handlers:
            try:
                if asyncio.iscoroutinefunction(handler):
                    # 异步处理器
                    asyncio.create_task(handler(event_type, event_data))
                else:
                    # 同步处理器
                    handler(event_type, event_data)
            except Exception as e:
                print(f"事件处理器执行失败: {e}")
    
    def _analyze_issues(self, state: SkillState) -> List[str]:
        """分析技能问题
        
        Args:
            state: 技能状态
            
        Returns:
            List[str]: 问题描述列表
        """
        issues = []
        
        if not state.enabled:
            issues.append("技能被禁用")
        
        if state.current_status == SkillStatus.ERROR.value:
            issues.append("技能处于错误状态")
        
        if state.success_rate < 0.5 and state.total_attempts > 0:
            issues.append(f"成功率低 ({state.success_rate:.1%})")
        
        if state.avg_execution_time > 5.0:
            issues.append(f"响应时间慢 ({state.avg_execution_time:.1f}s)")
        
        if time.time() - state.last_used > 7 * 24 * 3600:  # 7天
            issues.append("长时间未使用")
        
        return issues
```

### 设计思路

1. **管理器架构**：采用集中式管理器模式，统一管理所有技能状态，提供一致的访问接口。
2. **并发安全**：使用可重入锁（`threading.RLock`）确保多线程/协程环境下的数据一致性。
3. **原子操作**：实现原子保存机制，通过临时文件+重命名避免数据损坏，确保状态持久化的可靠性。
4. **事件驱动**：支持事件订阅机制，允许外部组件监听状态变化，实现松耦合的系统集成。
5. **历史记录**：记录关键操作历史，支持调试和审计需求，同时限制记录大小避免内存泄漏。

### 关键要点

1. **原子保存实现**：
   ```python

   # 2. 写入数据
   # 3. 重命名为目标文件
   os.replace(temp_path, self.storage_path)  # Unix原子操作
   ```

2. **并发锁选择**：
   ```python
   # 可重入锁允许同一线程多次获取锁
   self._lock = threading.RLock()
   # 在异步环境中使用asyncio.Lock
   self._async_lock = asyncio.Lock()
   ```

3. **事件处理支持**：
   ```python
   # 同时支持同步和异步事件处理器
   if asyncio.iscoroutinefunction(handler):
       asyncio.create_task(handler(event_type, event_data))
   else:
       handler(event_type, event_data)
   ```

4. **历史记录管理**：


```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
技能健康监控器
任务3：技能健康监控器实现
"""

import asyncio
import time
import logging
from typing import Dict, List, Any, Callable, Optional
from skill_state_manager import SkillStateManager
from skill_state import SkillState, SkillStatus


logger = logging.getLogger(__name__)


class SkillHealthMonitor:
    """技能健康监控器
    
    定期检查技能健康状态，提供告警和自动恢复功能。
    
    Attributes:
        state_manager: 技能状态管理器
        check_interval: 检查间隔（秒）
        _monitoring: 是否正在监控
        _task: 监控任务
        _alert_handlers: 告警处理器列表
        _alert_history: 告警历史记录
    """
    
    def __init__(self, state_manager: SkillStateManager, check_interval: float = 60.0):
        """初始化健康监控器
        
        Args:
            state_manager: 技能状态管理器
            check_interval: 检查间隔（秒）
        """
        self.state_manager = state_manager
        self.check_interval = check_interval
        self._monitoring = False
        self._task: Optional[asyncio.Task] = None
        self._alert_handlers: List[Callable] = []
        self._alert_history: List[Dict[str, Any]] = []
    
    def add_alert_handler(self, handler: Callable):
        """添加告警处理器
        
        Args:
            handler: 告警处理器函数，接收(alert_data)参数
        """
        self._alert_handlers.append(handler)
    
    async def start_monitoring(self):
        """开始监控"""
        if self._monitoring:
            logger.warning("监控已在运行中")
            return
        
        self._monitoring = True
        self._task = asyncio.create_task(self._monitor_loop())
        logger.info(f"技能健康监控已启动，检查间隔: {self.check_interval}秒")
    
    async def stop_monitoring(self):
        """停止监控"""
        if not self._monitoring:
            logger.warning("监控未在运行中")
            return
        
        self._monitoring = False
        if self._task:
            self._task.cancel()
            try:
                await self._task
            except asyncio.CancelledError:
                pass
            self._task = None
        
        logger.info("技能健康监控已停止")
    
    async def _monitor_loop(self):
        """监控循环"""
        while self._monitoring:
            try:
                await self._check_health()
                await asyncio.sleep(self.check_interval)
            except asyncio.CancelledError:
                break
            except Exception as e:
                logger.error(f"监控循环异常: {e}")
                await asyncio.sleep(5)  # 异常后等待5秒再重试
    
    async def _check_health(self):
        """检查所有技能的健康状态"""
        states = self.state_manager.get_all_states()
        current_time = time.time()
        
        for skill_name, state in states.items():
            alerts = []
            
            # 检查低健康度
            if state.health_score < 0.5:
                alerts.append({
                    "type": "low_health",
                    "severity": "warning",
                    "message": f"技能健康度低: {state.health_score:.3f}",
                    "threshold": 0.5
                })
            
            # 检查错误状态
            if state.current_status == SkillStatus.ERROR.value:
                alerts.append({
                    "type": "error_state",
                    "severity": "error",
                    "message": f"技能处于错误状态: {state.error_message}",
                    "error_message": state.error_message
                })
            
            # 检查长时间闲置
            if state.last_used > 0 and (current_time - state.last_used) > 7 * 24 * 3600:
                idle_days = (current_time - state.last_used) / (24 * 3600)
                alerts.append({
                    "type": "idle_too_long",
                    "severity": "info",
                    "message": f"技能闲置时间过长: {idle_days:.1f}天",
                    "idle_days": idle_days
                })
            
            # 检查高失败率
            if state.total_attempts >= 5 and state.success_rate < 0.5:
                alerts.append({
                    "type": "high_failure_rate",
                    "severity": "warning",
                    "message": f"失败率高: {state.success_rate:.1%}",
                    "threshold": 0.5
                })
            
            # 处理告警
            for alert_data in alerts:
                alert_data.update({
                    "skill_name": skill_name,
                    "health_score": state.health_score,
                    "health_level": state.health_level.value,
                    "timestamp": current_time
                })
                
                # 记录告警历史
                self._record_alert(alert_data)
                
                # 触发告警处理器
                await self._trigger_alerts(alert_data)
        
        # 记录检查完成
        logger.debug(f"健康检查完成，检查了 {len(states)} 个技能")
    
    async def _trigger_alerts(self, alert_data: Dict[str, Any]):
        """触发告警处理器
        
        Args:
            alert_data: 告警数据
        """
        for handler in self._alert_handlers:
            try:
                if asyncio.iscoroutinefunction(handler):
                    await handler(alert_data)
                else:
                    # 在后台线程中执行同步处理器
                    loop = asyncio.get_event_loop()
                    await loop.run_in_executor(None, lambda: handler(alert_data))
            except Exception as e:
                logger.error(f"告警处理器执行失败: {e}")
    
    def _record_alert(self, alert_data: Dict[str, Any]):
        """记录告警历史
        
        Args:
            alert_data: 告警数据
        """
        # 限制历史记录大小
        if len(self._alert_history) >= 500:
            self._alert_history.pop(0)
        
        self._alert_history.append(alert_data)
    
    def get_alert_history(self, limit: int = 50) -> List[Dict[str, Any]]:
        """获取告警历史
        
        Args:
            limit: 返回记录数量限制
            
        Returns:
            List[Dict[str, Any]]: 告警历史记录
        """
        return self._alert_history[-limit:] if self._alert_history else []
    
    def get_alert_summary(self) -> Dict[str, Any]:
        """获取告警摘要
        
        Returns:
            Dict[str, Any]: 告警摘要信息
        """
        if not self._alert_history:
            return {
                "total_alerts": 0,
                "alert_types": {},
                "recent_alerts": []
            }
        
        # 统计告警类型
        alert_types = {}
        for alert in self._alert_history:
            alert_type = alert.get("type", "unknown")
            alert_types[alert_type] = alert_types.get(alert_type, 0) + 1
        
        # 最近告警（最近10条）
        recent_alerts = self._alert_history[-10:] if len(self._alert_history) >= 10 else self._alert_history
        
        return {
            "total_alerts": len(self._alert_history),
            "alert_types": alert_types,
            "recent_alerts": recent_alerts,
            "timestamp": time.time()
        }
```

### 设计思路

1. **异步监控**：基于异步循环实现定期检查，不阻塞主线程，适合集成到异步应用中。
2. **多维度检测**：实现四种告警检测：低健康度、错误状态、长时间闲置、高失败率，覆盖常见问题场景。
3. **可扩展处理器**：支持添加多个告警处理器，每个处理器可以是同步或异步函数，灵活应对不同告警处理需求。
4. **历史记录**：记录告警历史，支持查询和摘要生成，便于问题排查和趋势分析。
5. **容错设计**：监控循环包含异常处理，单个技能检查失败不影响其他技能，异常后自动恢复。

```python
    def cleanup_old_states(self, max_age_days: int = 30) -> List[str]:
        """清理超过指定天数的未使用技能状态

            max_age_days: 最大保留天数
            
        Returns:
            List[str]: 被清理的技能名称列表

        current_time = time.time()
        max_age_seconds = max_age_days * 24 * 3600
        
        cleaned_skills = []
        
        with self._lock:
            skills_to_remove = []
            
            for skill_name, state in self.states.items():
                # 计算未使用时间
                unused_time = current_time - state.last_used if state.last_used > 0 else float('inf')
                
                # 清理条件：超过最大天数且使用次数为0
                if unused_time > max_age_seconds and state.usage_count == 0:
                    skills_to_remove.append(skill_name)
            
            # 执行清理
            for skill_name in skills_to_remove:
                del self.states[skill_name]
                cleaned_skills.append(skill_name)
            
            if cleaned_skills:
                # 记录清理历史
                self._add_history("cleanup", {
                    "max_age_days": max_age_days,
                    "cleaned_skills": cleaned_skills,
                    "remaining_skills": len(self.states)
                })
                
                # 保存清理后的状态
                self.save_states()
        
        if cleaned_skills:
            logger.info(f"状态清理完成: 清理{len(cleaned_skills)}个技能，剩余{len(self.states)}个技能")
        
        return cleaned_skills

    def get_state_cached(self, skill_name: str) -> Optional[SkillState]:
        """获取技能状态（带缓存）
        
        Args:
            skill_name: 技能名称
            
        Returns:
            Optional[SkillState]: 技能状态，不存在则返回None
        """
        current_time = time.time()
        
        # 检查缓存是否有效
        if (current_time - self._cache_timestamp) < self._cache_ttl:
            # 从缓存返回
            return self._cache.get(skill_name)
        
        # 缓存失效，重新加载
        with self._lock:
            state = self.states.get(skill_name)
            if state:
                # 更新缓存
                self._cache[skill_name] = state
                self._cache_timestamp = current_time
        
        return state
    
    async def update_state_delayed(self, skill_name: str, updates: Dict[str, Any], 
                                   delay_save: bool = True) -> bool:
        """更新技能状态（延迟保存）
        
        Args:
            skill_name: 技能名称
            updates: 更新字段字典
            delay_save: 是否延迟保存
            
        Returns:
            bool: 更新是否成功
        """
        # 执行更新（不自动保存）
        success = await self.update_state(skill_name, updates, auto_save=False)
        
        if success and delay_save:
            # 记录待保存的更新
            self._pending_saves[skill_name] = updates
            
            # 启动延迟保存定时器（如果未运行）
            if not self._save_timer or self._save_timer.done():
                self._save_timer = asyncio.create_task(self._delayed_save())
        
        return success
    
    async def _delayed_save(self, delay_seconds: float = 2.0):
        """延迟保存
        
        Args:
            delay_seconds: 延迟秒数
        """
        await asyncio.sleep(delay_seconds)
        
        if self._pending_saves:
            # 执行批量保存
            with self._lock:
                for skill_name, updates in self._pending_saves.items():
                    if skill_name in self.states:
                        state = self.states[skill_name]
                        for key, value in updates.items():
                            if hasattr(state, key):
                                setattr(state, key, value)
                
                # 清空待保存队列
                self._pending_saves.clear()
            
            # 执行一次保存
            self.save_states()
    
    def save_incremental(self) -> bool:
        """增量保存（只保存变化的状态）
        
        Returns:
            bool: 保存是否成功
        """
        # 简化的增量保存实现
        # 实际生产环境中需要更复杂的变更跟踪机制
        return self.save_states()  # 暂时使用全量保存
```

### 设计思路

1. **继承扩展**：通过继承基础`SkillStateManager`类扩展高级功能，保持基础功能的兼容性。
2. **批量操作优化**：使用异步并发执行批量操作，提高处理效率，减少总体执行时间。
3. **智能清理**：基于使用时间和使用频率智能清理无用状态，避免数据膨胀。
4. **性能优化策略**：
   - **状态缓存**：减少频繁的文件读取，提高读取性能。
   - **延迟保存**：合并短时间内多次更新，减少IO操作频率。
   - **增量保存**：理论上只保存变化的状态，减少写入数据量。
5. **数据完整性**：导出导入支持校验和验证，确保数据完整性和一致性。

### 关键要点

1. **批量操作并发**：
   ```python
   # 使用asyncio.gather并发执行批量任务
   tasks = [self.update_state(skill_name, updates, auto_save=False) for skill_name in skill_names]
   results = await asyncio.gather(*tasks, return_exceptions=True)
   ```

2. **缓存策略**：
   ```python
   # 缓存有效期机制
   if (current_time - self._cache_timestamp) < self._cache_ttl:
       return self._cache.get(skill_name)
   ```

3. **延迟保存实现**：
   ```python
   # 收集待保存更新，定时批量保存
   self._pending_saves[skill_name] = updates
   self._save_timer = asyncio.create_task(self._delayed_save())
   ```

4. **数据校验**：
   ```python
   # 使用MD5校验和验证数据完整性
   checksum = hashlib.md5(data_str.encode('utf-8')).hexdigest()
   if import_data["checksum"] != calculated_checksum:
       logger.warning("数据校验失败")
   ```

5. **清理策略**：
   ```python
   # 基于使用时间和使用频率的清理条件
   if unused_time > max_age_seconds and state.usage_count == 0:
       skills_to_remove.append(skill_name)
   ```

### 常见错误

1. **并发批量操作冲突**：
   ```python
   # 错误：多个批量操作同时修改同一技能导致状态不一致
   # 解决方案：使用锁保护批量操作，或实现乐观锁机制
   ```

2. **缓存一致性问题**：
   ```python
   # 错误：缓存状态与磁盘状态不一致
   ```

   ```python
   # 错误：程序崩溃导致延迟保存的数据丢失
   ```
