#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
思考模式配置数据类实现
包含思考模式枚举、推理深度枚举和配置管理类
"""

from enum import Enum, StrEnum
from dataclasses import dataclass, field
from typing import Dict, List, Any, Optional
import yaml
import json


class ThinkingMode(StrEnum):
    """思考模式枚举"""
    FAST = "fast"
    BALANCED = "balanced"
    DEEP = "deep"
    CREATIVE = "creative"
    
    def description(self) -> str:
        """获取模式描述"""
        descriptions = {
            ThinkingMode.FAST: "快速响应模式，适用于简单查询和快速回答",
            ThinkingMode.BALANCED: "平衡模式，适用于大多数日常任务",
            ThinkingMode.DEEP: "深度思考模式，适用于复杂问题和需要推理的任务",
            ThinkingMode.CREATIVE: "创意生成模式，适用于写作、创意和发散性思考"
        }
        return descriptions.get(self, "未知模式")


class ReasoningDepth(StrEnum):
    """推理深度枚举"""
    SHALLOW = "shallow"
    MEDIUM = "medium"
    DEEP = "deep"
    
    def level_value(self) -> int:
        """获取深度数值（用于比较）"""
        values = {
            ReasoningDepth.SHALLOW: 1,
            ReasoningDepth.MEDIUM: 2,
            ReasoningDepth.DEEP: 3
        }
        return values.get(self, 0)


@dataclass
class ThinkingConfig:
    """思考模式配置数据类"""
    
    modes: Dict[str, Dict[str, Any]] = field(default_factory=dict)
    
    @classmethod
    def from_yaml(cls, yaml_content: str) -> "ThinkingConfig":
        """从YAML内容创建配置
        
        Args:
            yaml_content: YAML格式的配置内容
            
        Returns:
            ThinkingConfig实例
            
        Raises:
            ValueError: YAML解析失败
        """
        try:
            data = yaml.safe_load(yaml_content)
            if data is None:
                return cls()
            
            # 支持不同的YAML结构
            if "thinking" in data:
                modes = data.get("thinking", {}).get("modes", {})
            elif "modes" in data:
                modes = data.get("modes", {})
            else:
                modes = data
            
            return cls(modes=modes)
        except yaml.YAMLError as e:
            raise ValueError(f"YAML解析失败: {e}")
    
    @classmethod
    def from_file(cls, file_path: str) -> "ThinkingConfig":
        """从YAML文件创建配置
        
        Args:
            file_path: YAML文件路径
            
        Returns:
            ThinkingConfig实例
            
        Raises:
            ValueError: 文件读取失败
        """
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                return cls.from_yaml(f.read())
        except (IOError, OSError) as e:
            raise ValueError(f"文件读取失败: {e}")
    
    def validate(self) -> List[str]:
        """验证配置有效性
        
        Returns:
            错误消息列表，空列表表示配置有效
        """
        errors = []
        
        if not self.modes:
            errors.append("配置中未定义任何思考模式")
            return errors
        
        # 定义必需参数
        required_params = {"temperature", "max_tokens", "reasoning_depth"}
        
        for mode_name, mode_config in self.modes.items():
            # 检查必需参数
            for param in required_params:
                if param not in mode_config:
                    errors.append(f"模式 '{mode_name}' 缺少必需参数 '{param}'")
            
            # 验证temperature范围 (0.0-2.0)
            temp = mode_config.get("temperature")
            if temp is not None:
                if not isinstance(temp, (int, float)):
                    errors.append(f"模式 '{mode_name}' temperature参数类型错误: {type(temp)}")
                elif not 0.0 <= temp <= 2.0:
                    errors.append(f"模式 '{mode_name}' temperature参数超出范围 (0.0-2.0): {temp}")
            
            # 验证max_tokens为正整数
            tokens = mode_config.get("max_tokens")
            if tokens is not None:
                if not isinstance(tokens, int):
                    errors.append(f"模式 '{mode_name}' max_tokens参数类型错误: {type(tokens)}")
                elif tokens <= 0:
                    errors.append(f"模式 '{mode_name}' max_tokens参数必须为正整数: {tokens}")
            
            # 验证reasoning_depth有效值
            depth = mode_config.get("reasoning_depth")
            if depth is not None:
                valid_depths = ["shallow", "medium", "deep"]
                if isinstance(depth, str) and depth.lower() not in valid_depths:
                    errors.append(f"模式 '{mode_name}' reasoning_depth参数无效值: {depth}，有效值: {valid_depths}")
            
            # 验证可选参数
            optional_params = {
                "top_p": (0.0, 1.0),
                "frequency_penalty": (-2.0, 2.0),
                "presence_penalty": (-2.0, 2.0)
            }
            
            for param, (min_val, max_val) in optional_params.items():
                if param in mode_config:
                    value = mode_config[param]
                    if not isinstance(value, (int, float)):
                        errors.append(f"模式 '{mode_name}' {param}参数类型错误: {type(value)}")
                    elif not min_val <= value <= max_val:
                        errors.append(f"模式 '{mode_name}' {param}参数超出范围 ({min_val}-{max_val}): {value}")
        
        return errors
    
    def get_mode_config(self, mode_name: str) -> Dict[str, Any]:
        """获取指定模式的配置
        
        Args:
            mode_name: 模式名称
            
        Returns:
            模式配置字典
            
        Raises:
            ValueError: 模式不存在
        """
        if mode_name not in self.modes:
            raise ValueError(f"思考模式 '{mode_name}' 不存在")
        
        return self.modes[mode_name].copy()
    
    def add_mode(self, mode_name: str, config: Dict[str, Any]) -> None:
        """添加新的思考模式
        
        Args:
            mode_name: 模式名称
            config: 模式配置
        """
        self.modes[mode_name] = config.copy()
    
    def remove_mode(self, mode_name: str) -> bool:
        """移除思考模式
        
        Args:
            mode_name: 模式名称
            
        Returns:
            是否成功移除
        """
        if mode_name in self.modes:
            del self.modes[mode_name]
            return True
        return False
    
    def to_yaml(self) -> str:
        """将配置转换为YAML格式
        
        Returns:
            YAML格式字符串
        """
        data = {"thinking": {"modes": self.modes}}
        return yaml.dump(data, allow_unicode=True, sort_keys=False, default_flow_style=False)
    
    def to_dict(self) -> Dict[str, Any]:
        """将配置转换为字典
        
        Returns:
            配置字典
        """
        return {"modes": self.modes.copy()}
    
    def __str__(self) -> str:
        """字符串表示"""
        return f"ThinkingConfig(modes={list(self.modes.keys())})"


# ============================================================================

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
任务特征数据类实现
包含任务类型枚举和任务特征数据类
"""

from enum import Enum, StrEnum
from dataclasses import dataclass, field
from typing import Dict, List, Any, Optional


class TaskType(StrEnum):
    """任务类型枚举"""
    CODE_GENERATION = "code_generation"
    DEBUGGING = "debugging"
    CREATIVE_WRITING = "creative_writing"
    DATA_ANALYSIS = "data_analysis"
    QUICK_ANSWER = "quick_answer"
    RESEARCH = "research"
    PLANNING = "planning"
    TRANSLATION = "translation"
    SUMMARIZATION = "summarization"
    
    def description(self) -> str:
        """获取任务类型描述"""
        descriptions = {
            TaskType.CODE_GENERATION: "代码生成任务，编写程序代码",
            TaskType.DEBUGGING: "调试任务，查找和修复代码错误",
            TaskType.CREATIVE_WRITING: "创意写作任务，创作故事、文章等",
            TaskType.DATA_ANALYSIS: "数据分析任务，分析和处理数据",
            TaskType.QUICK_ANSWER: "快速回答任务，提供简洁直接的答案",
            TaskType.RESEARCH: "研究任务，深入研究和分析问题",
            TaskType.PLANNING: "规划任务，制定计划和方案",
            TaskType.TRANSLATION: "翻译任务，翻译文本内容",
            TaskType.SUMMARIZATION: "摘要任务，总结和提炼内容"
        }
        return descriptions.get(self, "未知任务类型")


@dataclass
class TaskFeatures:
    """任务特征数据类"""
    
    task_type: TaskType
    complexity: float = 0.5
    urgency: float = 0.5
    domain: Optional[str] = None
    constraints: Dict[str, Any] = field(default_factory=dict)
    
    def validate(self) -> List[str]:
        """验证任务特征有效性
        
        Returns:
            错误消息列表，空列表表示特征有效
        """
        errors = []
        
        # 验证复杂度范围
        if not isinstance(self.complexity, (int, float)):
            errors.append(f"复杂度必须是数值类型，当前类型: {type(self.complexity)}")
        elif not 0.0 <= self.complexity <= 1.0:
            errors.append(f"复杂度必须在0.0-1.0之间，当前值: {self.complexity}")
        
        # 验证紧急度范围
        if not isinstance(self.urgency, (int, float)):
            errors.append(f"紧急度必须是数值类型，当前类型: {type(self.urgency)}")
        elif not 0.0 <= self.urgency <= 1.0:
            errors.append(f"紧急度必须在0.0-1.0之间，当前值: {self.urgency}")
        
        # 验证任务类型
        try:
            TaskType(self.task_type)
        except ValueError:
            errors.append(f"无效的任务类型: {self.task_type}")
        
        return errors
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典
        
        Returns:
            任务特征字典
        """
        return {
            "task_type": self.task_type.value,
            "complexity": self.complexity,
            "urgency": self.urgency,
            "domain": self.domain,
            "constraints": self.constraints.copy()
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "TaskFeatures":
        """从字典创建任务特征
        
        Args:
            data: 任务特征字典
            
        Returns:
            TaskFeatures实例
        """
        task_type = TaskType(data.get("task_type", "quick_answer"))
        complexity = float(data.get("complexity", 0.5))
        urgency = float(data.get("urgency", 0.5))
        domain = data.get("domain")
        constraints = data.get("constraints", {})
        
        return cls(
            task_type=task_type,
            complexity=complexity,
            urgency=urgency,
            domain=domain,
            constraints=constraints
        )
    
    def __str__(self) -> str:
        """字符串表示"""
        return (f"TaskFeatures(type={self.task_type.value}, "
                f"complexity={self.complexity:.2f}, "
                f"urgency={self.urgency:.2f}, "
                f"domain={self.domain})")


# ============================================================================
# 工具函数
# ============================================================================

def analyze_task_from_text(text: str) -> TaskFeatures:
    """从文本分析任务特征
    
    Args:
        text: 任务描述文本
        
    Returns:
        任务特征
    """
    text_lower = text.lower()
    
    # 检测任务类型
    task_type = TaskType.QUICK_ANSWER  # 默认
    
    if any(word in text_lower for word in ["代码", "编程", "函数", "类", "def ", "class ", "import ", "python", "java"]):
        task_type = TaskType.CODE_GENERATION
    elif any(word in text_lower for word in ["错误", "bug", "调试", "修复", "问题", "异常", "崩溃"]):
        task_type = TaskType.DEBUGGING
    elif any(word in text_lower for word in ["写作", "故事", "创意", "诗歌", "文章", "小说", "散文"]):
        task_type = TaskType.CREATIVE_WRITING
    elif any(word in text_lower for word in ["数据", "分析", "统计", "图表", "报告", "excel", "数据库"]):
        task_type = TaskType.DATA_ANALYSIS
    elif any(word in text_lower for word in ["研究", "调查", "探索", "深入", "论文", "文献"]):
        task_type = TaskType.RESEARCH
    elif any(word in text_lower for word in ["计划", "规划", "安排", "日程", "方案", "策略"]):
        task_type = TaskType.PLANNING
    elif any(word in text_lower for word in ["翻译", "英文", "中文", "语言", "convert", "translate"]):
        task_type = TaskType.TRANSLATION
    elif any(word in text_lower for word in ["摘要", "总结", "概括", "要点", "summary", "summarize"]):
        task_type = TaskType.SUMMARIZATION
    
    # 估算复杂度（基于文本长度和关键词）
    complexity = 0.5
    if len(text) > 500:
        complexity = 0.8
    elif len(text) > 200:
        complexity = 0.6
    elif len(text) < 50:
        complexity = 0.3
    
    # 检测复杂关键词增加复杂度
    complex_keywords = ["复杂", "困难", "挑战", "高级", "深入", "多步骤", "系统"]
    if any(keyword in text_lower for keyword in complex_keywords):
        complexity = min(1.0, complexity + 0.2)
    
    # 检测紧急度关键词
    urgency = 0.5
    urgency_keywords = ["紧急", "尽快", "马上", "立刻", "急", "urgent", "asap", "deadline"]
    non_urgency_keywords = ["不急", "慢慢", "有空", "不着急", "whenever", "no rush"]
    
    if any(keyword in text_lower for keyword in urgency_keywords):
        urgency = 0.9
    elif any(keyword in text_lower for keyword in non_urgency_keywords):
        urgency = 0.2
    
    # 检测领域
    domain = None
    domain_keywords = {
        "scientific": ["科学", "研究", "实验", "论文", "学术"],
        "technical": ["技术", "编程", "代码", "系统", "软件"],
        "creative": ["创意", "艺术", "设计", "写作", "音乐"],
        "business": ["商业", "市场", "销售", "财务", "经济"],
        "educational": ["教育", "学习", "教学", "课程", "培训"]
    }
    
    for dom, keywords in domain_keywords.items():
        if any(keyword in text_lower for keyword in keywords):
            domain = dom
            break
    
    return TaskFeatures(
        task_type=task_type,
        complexity=complexity,
        urgency=urgency,
        domain=domain
    )


if __name__ == "__main__":
    # 测试代码

### 实现解析

#### 关键设计决策
1. **使用StrEnum而非普通Enum**：StrEnum提供了更好的字符串表示和序列化支持，适合配置系统。
2. **灵活的YAML解析**：支持多种YAML结构，提高了配置文件的兼容性。
3. **严格的参数验证**：不仅检查参数存在性，还检查类型、范围和有效性。
4. **描述性方法**：为枚举提供description()方法，提高用户体验。

#### 常见错误及避免
1. **枚举值比较错误**：使用StrEnum.value或直接比较枚举实例，避免字符串比较。
2. **YAML结构僵化**：支持多种YAML结构，避免强制用户使用特定格式。
3. **验证遗漏**：验证所有重要参数，包括可选参数的范围检查。
4. **类型提示不完整**：为所有方法和参数添加完整的类型提示。

#### 测试要点
1. **配置创建测试**：测试from_yaml()和from_file()方法。
2. **验证功能测试**：测试有效和无效配置的验证结果。
3. **模式管理测试**：测试add_mode()、remove_mode()、get_mode_config()。
4. **序列化测试**：测试to_yaml()和to_dict()的正确性。
5. **任务特征分析测试**：测试analyze_task_from_text()的准确性。

#### 扩展思考
1. **配置版本控制**：为配置添加版本字段，支持配置迁移。
2. **配置热重载**：监控配置文件变化，自动重新加载配置。
3. **配置模板**：提供配置模板生成功能。
4. **国际化支持**：为描述信息添加多语言支持。

# -*- coding: utf-8 -*-
"""
智能思考模式选择器实现
基于多规则决策逻辑选择最合适的思考模式
"""

import logging
from typing import Dict, List, Optional, Any, Tuple, Callable
from dataclasses import dataclass, asdict

class ThinkingModeSelector:
    """智能思考模式选择器"""
    
    def __init__(self, config: ThinkingConfig):
        """初始化选择器
        
        Args:
            config: 思考模式配置
        """
        self.config = config
        
        # 默认任务类型映射
        self._type_to_mode: Dict[TaskType, ThinkingMode] = {
            TaskType.CODE_GENERATION: ThinkingMode.BALANCED,
            TaskType.DEBUGGING: ThinkingMode.DEEP,
            TaskType.CREATIVE_WRITING: ThinkingMode.CREATIVE,
            TaskType.DATA_ANALYSIS: ThinkingMode.BALANCED,
            TaskType.QUICK_ANSWER: ThinkingMode.FAST,
            TaskType.RESEARCH: ThinkingMode.DEEP,
            TaskType.PLANNING: ThinkingMode.DEEP,
            TaskType.TRANSLATION: ThinkingMode.BALANCED,
            TaskType.SUMMARIZATION: ThinkingMode.FAST,
        }
        
        # 自定义规则列表
        self._custom_rules: List[Callable[[TaskFeatures], Optional[str]]] = []
        
        # 规则权重（用于置信度计算）
        self._rule_weights = {
            "custom": 0.3,
            "urgency": 0.4,
            "task_type": 0.2,
            "complexity": 0.1
        }
        
        # 初始化日志
        self.logger = logging.getLogger(__name__)
    
    def add_type_mapping(self, task_type: TaskType, mode: ThinkingMode) -> None:
        """添加或更新任务类型映射
        
        Args:
            task_type: 任务类型
            mode: 对应的思考模式
        """
        self._type_to_mode[task_type] = mode
        self.logger.info(f"添加任务类型映射: {task_type.value} -> {mode.value}")
    
    def add_custom_rule(self, rule_func: Callable[[TaskFeatures], Optional[str]]) -> None:
        """添加自定义规则
        
        Args:
            rule_func: 规则函数，接收TaskFeatures返回模式名称或None
        """
        self._custom_rules.append(rule_func)
        self.logger.info(f"添加自定义规则: {rule_func.__name__}")
    
    def select_mode(self, features: TaskFeatures) -> str:
        """选择思考模式
        
        Args:
            features: 任务特征
            
        Returns:
            选择的思考模式名称
            
        Raises:
            ValueError: 任务特征无效或无法选择模式
        """
        # 验证任务特征
        errors = features.validate()
        if errors:
            error_msg = f"任务特征无效: {', '.join(errors)}"
            self.logger.error(error_msg)
            raise ValueError(error_msg)
        
        selected_mode = self._select_mode_internal(features)
        
        # 验证选择的模式是否存在
        if selected_mode not in self.config.modes:
            error_msg = f"选择的模式 '{selected_mode}' 不存在于配置中"
            self.logger.error(error_msg)
            raise ValueError(error_msg)
        
        self.logger.info(f"为任务选择模式: {selected_mode} (类型: {features.task_type.value})")
        return selected_mode
    
    def _select_mode_internal(self, features: TaskFeatures) -> str:
        """内部模式选择逻辑"""
        
        # 1. 应用自定义规则（最高优先级）
        for i, rule in enumerate(self._custom_rules):
            try:
                result = rule(features)
                if result is not None:
                    self.logger.debug(f"自定义规则 #{i} 建议模式: {result}")
                    return result
            except Exception as e:
                self.logger.warning(f"自定义规则 #{i} 执行失败: {e}")
        
        # 2. 基于紧急程度（高优先级）
        if features.urgency > 0.8:
            self.logger.debug(f"高紧急度 ({features.urgency:.2f})，选择快速模式")
            return ThinkingMode.FAST.value
        
        # 3. 基于任务类型
        if features.task_type in self._type_to_mode:
            mode = self._type_to_mode[features.task_type]
            self.logger.debug(f"任务类型映射: {features.task_type.value} -> {mode.value}")
            return mode.value
        
        # 4. 基于复杂度
        if features.complexity > 0.7:
            self.logger.debug(f"高复杂度 ({features.complexity:.2f})，选择深度模式")
            return ThinkingMode.DEEP.value
        elif features.complexity > 0.3:
            self.logger.debug(f"中等复杂度 ({features.complexity:.2f})，选择平衡模式")
            return ThinkingMode.BALANCED.value
        else:
            self.logger.debug(f"低复杂度 ({features.complexity:.2f})，选择快速模式")
            return ThinkingMode.FAST.value
    
    def select_mode_with_confidence(self, features: TaskFeatures) -> Tuple[str, float]:
        """选择思考模式并返回置信度
        
        Args:
            features: 任务特征
            
        Returns:
            (模式名称, 置信度) 元组
        """
        try:
            # 获取基本模式选择
            mode_name = self.select_mode(features)
            
            # 计算置信度
            confidence = self._calculate_confidence(features, mode_name)
            
            self.logger.info(f"选择模式: {mode_name}，置信度: {confidence:.2f}")
            return mode_name, confidence
            
        except Exception as e:
            # 发生错误时返回默认模式和低置信度
            self.logger.error(f"模式选择失败: {e}")
            return ThinkingMode.BALANCED.value, 0.3
    
    def _calculate_confidence(self, features: TaskFeatures, selected_mode: str) -> float:
        """计算模式选择置信度
        
        Args:
            features: 任务特征
            selected_mode: 选择的模式
            
        Returns:
            置信度分数 (0.0-1.0)
        """
        confidence = 0.5  # 基础置信度
        
        # 基于紧急度的置信度调整
        if features.urgency > 0.8:
            if selected_mode == ThinkingMode.FAST.value:
                confidence += self._rule_weights["urgency"]
            else:
                confidence -= self._rule_weights["urgency"] * 0.5
        
        # 基于任务类型匹配的置信度调整
        if features.task_type in self._type_to_mode:
            expected_mode = self._type_to_mode[features.task_type].value
            if selected_mode == expected_mode:
                confidence += self._rule_weights["task_type"]
            else:
                confidence -= self._rule_weights["task_type"] * 0.5
        
        # 基于复杂度的置信度调整
        if features.complexity > 0.7:
            if selected_mode == ThinkingMode.DEEP.value:
                confidence += self._rule_weights["complexity"]
            else:
                confidence -= self._rule_weights["complexity"] * 0.5
        elif features.complexity < 0.3:
            if selected_mode == ThinkingMode.FAST.value:
                confidence += self._rule_weights["complexity"]
            else:
                confidence -= self._rule_weights["complexity"] * 0.5
        else:
            if selected_mode == ThinkingMode.BALANCED.value:
                confidence += self._rule_weights["complexity"] * 0.5
        
        # 基于自定义规则的置信度调整
        for rule in self._custom_rules:
            try:
                result = rule(features)
                if result == selected_mode:
                    confidence += self._rule_weights["custom"] * 0.5
            except:
                pass
        
        # 确保置信度在合理范围内
        confidence = max(0.1, min(1.0, confidence))
        
        return round(confidence, 2)
    
    def get_available_modes(self) -> List[str]:
        """获取所有可用思考模式
        
        Returns:
            模式名称列表
        """
        return list(self.config.modes.keys())
    
    def get_mode_description(self, mode_name: str) -> str:
        """获取模式描述
        
        Args:
            mode_name: 模式名称
            
        Returns:
            模式描述文本
        """
        try:
            mode_enum = ThinkingMode(mode_name)
            return mode_enum.description()
        except ValueError:
            mode_config = self.config.get_mode_config(mode_name)
            return mode_config.get("description", f"自定义模式: {mode_name}")
    
    def explain_selection(self, features: TaskFeatures) -> Dict[str, Any]:
        """解释模式选择过程
        
        Args:
            features: 任务特征
            
        Returns:
            选择解释字典
        """
        explanation = {
            "task_features": features.to_dict(),
            "applied_rules": [],
            "considered_modes": self.get_available_modes(),
            "final_selection": None,
            "confidence": 0.0,
            "selection_process": []
        }
        
        try:
            # 获取选择结果
            mode_name, confidence = self.select_mode_with_confidence(features)
            explanation["final_selection"] = mode_name
            explanation["confidence"] = confidence
            
            # 模拟选择过程（记录每个决策点）
            selection_process = []
            
            # 检查自定义规则
            for i, rule in enumerate(self._custom_rules):
                try:
                    result = rule(features)
                    if result is not None:
                        selection_process.append({
                            "step": "custom_rule",
                            "rule_index": i,
                            "rule_name": rule.__name__,
                            "suggested_mode": result,
                            "applied": result == mode_name
                        })
                        explanation["applied_rules"].append({
                            "rule_type": "custom",
                            "rule_index": i,
                            "rule_name": rule.__name__,
                            "suggested_mode": result
                        })
                except Exception as e:
                    selection_process.append({
                        "step": "custom_rule_error",
                        "rule_index": i,
                        "error": str(e)
                    })
            
            # 检查紧急度规则
            if features.urgency > 0.8:
                selection_process.append({
                    "step": "urgency_rule",
                    "urgency": features.urgency,
                    "threshold": 0.8,
                    "suggested_mode": ThinkingMode.FAST.value,
                    "applied": mode_name == ThinkingMode.FAST.value
                })
                explanation["applied_rules"].append({
                    "rule_type": "urgency_override",
                    "urgency": features.urgency,
                    "reason": "高紧急度强制使用快速模式"
                })
            
            # 检查任务类型映射
            if features.task_type in self._type_to_mode:
                expected_mode = self._type_to_mode[features.task_type].value
                selection_process.append({
                    "step": "task_type_mapping",
                    "task_type": features.task_type.value,
                    "mapped_mode": expected_mode,
                    "applied": mode_name == expected_mode
                })
                explanation["applied_rules"].append({
                    "rule_type": "task_type_mapping",
                    "task_type": features.task_type.value,
                    "mapped_mode": expected_mode
                })
            
            # 检查复杂度规则
            if features.complexity > 0.7:
                suggested_mode = ThinkingMode.DEEP.value
                reason = "高复杂度推荐深度模式"
            elif features.complexity > 0.3:
                suggested_mode = ThinkingMode.BALANCED.value
                reason = "中等复杂度推荐平衡模式"
            else:
                suggested_mode = ThinkingMode.FAST.value
                reason = "低复杂度推荐快速模式"
            
            selection_process.append({
                "step": "complexity_rule",
                "complexity": features.complexity,
                "suggested_mode": suggested_mode,
                "applied": mode_name == suggested_mode
            })
            explanation["applied_rules"].append({
                "rule_type": "complexity_based",
                "complexity": features.complexity,
                "reason": reason
            })
            
            explanation["selection_process"] = selection_process
            
        except Exception as e:
            explanation["error"] = str(e)
            self.logger.error(f"选择解释失败: {e}")
        
        return explanation
    
    def get_type_mappings(self) -> Dict[str, str]:
        """获取当前任务类型映射
        
        Returns:
            任务类型到思考模式的映射字典
        """
        return {
            task_type.value: mode.value
            for task_type, mode in self._type_to_mode.items()
        }

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
思考模式应用器实现
将思考模式配置应用到模型调用，实现缓存和性能统计
"""

import time
import json
import hashlib
import logging
import asyncio
from typing import Dict, List, Any, Optional, Callable, Tuple
from collections import defaultdict
from dataclasses import dataclass, field

@dataclass
class PerformanceStats:
    """性能统计数据类"""
    call_count: int = 0
    success_count: int = 0
    error_count: int = 0
    total_time: float = 0.0
    cache_hits: int = 0
    cache_misses: int = 0
    
    @property
    def average_time(self) -> float:
        """平均响应时间"""
        if self.call_count == 0:
            return 0.0
        return self.total_time / self.call_count
    
    @property
    def success_rate(self) -> float:
        """成功率"""
        if self.call_count == 0:
            return 0.0
        return self.success_count / self.call_count
    
    @property
    def cache_hit_rate(self) -> float:
        """缓存命中率"""
        total_cache_calls = self.cache_hits + self.cache_misses
        if total_cache_calls == 0:
            return 0.0
        return self.cache_hits / total_cache_calls


class ThinkingModeApplicator:
    """思考模式应用器"""
    
    def __init__(self, config: ThinkingConfig):
        """初始化应用器
        
        Args:
            config: 思考模式配置
        """
        self.config = config
        self._cache: Dict[str, Any] = {}
        self._performance_stats: Dict[str, PerformanceStats] = defaultdict(PerformanceStats)
        
        # 缓存配置
        self._max_cache_size = 1000
        self._cache_ttl = 3600  # 1小时
        
        # 初始化日志
        self.logger = logging.getLogger(__name__)
    
    def apply_to_model_call(self, mode_name: str, model_call_func: Callable,
                           prompt: str, **kwargs) -> Any:
        """将思考模式应用到模型调用
        
        Args:
            mode_name: 思考模式名称
            model_call_func: 模型调用函数
            prompt: 输入提示
            **kwargs: 额外参数
            
        Returns:
            模型调用结果
            
        Raises:
            ValueError: 模式不存在或配置无效
        """
        start_time = time.time()
        cache_hit = False
        
        try:
            # 获取模式配置
            mode_config = self.config.get_mode_config(mode_name)
            
            # 提取调用参数
            call_kwargs = self._extract_call_parameters(mode_config, kwargs)
            
            # 检查缓存
            cache_key = self._generate_cache_key(mode_name, prompt, call_kwargs)
            use_cache = mode_config.get("use_cache", True)
            
            if use_cache and cache_key in self._cache:
                self.logger.debug(f"缓存命中: {mode_name}")
                result = self._cache[cache_key]
                cache_hit = True
                self._performance_stats[mode_name].cache_hits += 1
            else:
                self.logger.debug(f"缓存未命中: {mode_name}")
                if use_cache:
                    self._performance_stats[mode_name].cache_misses += 1
                
                # 应用特殊模式特性
                if mode_config.get("allow_reflection", False):
                    max_iterations = mode_config.get("max_iterations", 1)
                    result = self._apply_with_reflection(
                        model_call_func, prompt, call_kwargs, max_iterations
                    )
                else:
                    # 普通调用
                    result = model_call_func(prompt, **call_kwargs)
                
                # 缓存结果
                if use_cache:
                    self._cache[cache_key] = result
                    self._cleanup_cache_if_needed()
            
            # 更新性能统计
            duration = time.time() - start_time
            self._update_stats(mode_name, True, duration, cache_hit)
            
            return result
            
        except Exception as e:
            # 更新错误统计
            duration = time.time() - start_time
            self._update_stats(mode_name, False, duration, cache_hit)
            self.logger.error(f"模式应用失败: {e}")
            raise e
    
    async def apply_to_model_call_async(self, mode_name: str, model_call_func: Callable,
                                       prompt: str, **kwargs) -> Any:
        """异步应用思考模式到模型调用
        
        Args:
            mode_name: 思考模式名称
            model_call_func: 异步模型调用函数
            prompt: 输入提示
            **kwargs: 额外参数
            
        Returns:
            模型调用结果
        """
        start_time = time.time()
        cache_hit = False
        
        try:
            # 获取模式配置
            mode_config = self.config.get_mode_config(mode_name)
            
            # 提取调用参数
            call_kwargs = self._extract_call_parameters(mode_config, kwargs)
            
            # 检查缓存
            cache_key = self._generate_cache_key(mode_name, prompt, call_kwargs)
            use_cache = mode_config.get("use_cache", True)
            
            if use_cache and cache_key in self._cache:
                self.logger.debug(f"缓存命中: {mode_name}")
                result = self._cache[cache_key]
                cache_hit = True
                self._performance_stats[mode_name].cache_hits += 1
            else:
                self.logger.debug(f"缓存未命中: {mode_name}")
                if use_cache:
                    self._performance_stats[mode_name].cache_misses += 1
                
                # 应用特殊模式特性
                if mode_config.get("allow_reflection", False):
                    max_iterations = mode_config.get("max_iterations", 1)
                    result = await self._apply_with_reflection_async(
                        model_call_func, prompt, call_kwargs, max_iterations
                    )
                else:
                    # 普通异步调用
                    result = await model_call_func(prompt, **call_kwargs)
                
                # 缓存结果
                if use_cache:
                    self._cache[cache_key] = result
                    self._cleanup_cache_if_needed()
            
            # 更新性能统计
            duration = time.time() - start_time
            self._update_stats(mode_name, True, duration, cache_hit)
            
            return result
            
        except Exception as e:
            # 更新错误统计
            duration = time.time() - start_time
            self._update_stats(mode_name, False, duration, cache_hit)
            self.logger.error(f"异步模式应用失败: {e}")
            raise e
    
    def _extract_call_parameters(self, mode_config: Dict[str, Any],
                                user_kwargs: Dict[str, Any]) -> Dict[str, Any]:
        """提取模型调用参数
        
        Args:
            mode_config: 模式配置
            user_kwargs: 用户提供的参数
            
        Returns:
            合并后的调用参数
        """
        # 参数映射：模式配置键 -> 模型调用参数键
        param_mapping = {
            "temperature": "temperature",
            "max_tokens": "max_tokens",
            "top_p": "top_p",
            "frequency_penalty": "frequency_penalty",
            "presence_penalty": "presence_penalty",
            "stop": "stop",
            "best_of": "best_of",
            "n": "n",
            "logprobs": "logprobs",
            "echo": "echo"
        }
        
        call_kwargs = {}
        
        # 从模式配置中提取参数
        for config_key, call_key in param_mapping.items():
            if config_key in mode_config:
                call_kwargs[call_key] = mode_config[config_key]
        
        # 用户提供的参数覆盖模式配置
        call_kwargs.update(user_kwargs)
        
        return call_kwargs
    
    def _generate_cache_key(self, mode_name: str, prompt: str,
                           call_kwargs: Dict[str, Any]) -> str:
        """生成缓存键
        
        Args:
            mode_name: 模式名称
            prompt: 输入提示
            call_kwargs: 调用参数
            
        Returns:
            缓存键字符串
        """
        # 创建可哈希的字典表示
        cache_dict = {
            "mode": mode_name,
            "prompt": prompt,
            "kwargs": self._normalize_kwargs(call_kwargs)
        }
        
        # 使用SHA256生成哈希
        cache_str = json.dumps(cache_dict, sort_keys=True, ensure_ascii=False)
        return hashlib.sha256(cache_str.encode('utf-8')).hexdigest()
    
    def _normalize_kwargs(self, kwargs: Dict[str, Any]) -> Dict[str, Any]:
        """标准化调用参数，确保可哈希
        
        Args:
            kwargs: 调用参数
            
        Returns:
            标准化后的参数
        """
        normalized = {}
        for key, value in kwargs.items():
            # 处理特殊类型
            if isinstance(value, (list, tuple)):
                normalized[key] = tuple(value)
            elif isinstance(value, dict):
                normalized[key] = tuple(sorted(value.items()))
            elif isinstance(value, set):
                normalized[key] = tuple(sorted(value))
            else:
                normalized[key] = value
        return normalized
    
    def _apply_with_reflection(self, model_call_func: Callable, prompt: str,
                              call_kwargs: Dict[str, Any], iterations: int) -> Any:
        """应用反思迭代
        
        Args:
            model_call_func: 模型调用函数
            prompt: 输入提示
            call_kwargs: 调用参数
            iterations: 迭代次数
            
        Returns:
            反思后的结果
        """
        if iterations <= 1:
            return model_call_func(prompt, **call_kwargs)
        
        current_result = model_call_func(prompt, **call_kwargs)
        
        for i in range(1, iterations):
            reflection_prompt = (
                f"请对以下内容进行第{i}次反思和改进:\n\n"
                f"原始提示: {prompt}\n\n"
                f"当前结果: {current_result}\n\n"
                f"请提供改进后的版本:"
            )
            
            current_result = model_call_func(reflection_prompt, **call_kwargs)
            self.logger.debug(f"反思迭代 {i} 完成")
        
        return current_result
    
    async def _apply_with_reflection_async(self, model_call_func: Callable, prompt: str,
                                          call_kwargs: Dict[str, Any], iterations: int) -> Any:
        """异步应用反思迭代
        
        Args:
            model_call_func: 异步模型调用函数
            prompt: 输入提示
            call_kwargs: 调用参数
            iterations: 迭代次数
            
        Returns:
            反思后的结果
        """
        if iterations <= 1:
            return await model_call_func(prompt, **call_kwargs)
        
        current_result = await model_call_func(prompt, **call_kwargs)
        
        for i in range(1, iterations):
            reflection_prompt = (
                f"请对以下内容进行第{i}次反思和改进:\n\n"
                f"原始提示: {prompt}\n\n"
                f"当前结果: {current_result}\n\n"
                f"请提供改进后的版本:"
            )
            
            current_result = await model_call_func(reflection_prompt, **call_kwargs)
            self.logger.debug(f"异步反思迭代 {i} 完成")
        
        return current_result
    
    def _update_stats(self, mode_name: str, success: bool,
                     duration: float, cache_hit: bool) -> None:
        """更新性能统计
        
        Args:
            mode_name: 模式名称
            success: 是否成功
            duration: 调用时长
            cache_hit: 是否缓存命中
        """
        stats = self._performance_stats[mode_name]
        stats.call_count += 1
        stats.total_time += duration
        
        if success:
            stats.success_count += 1
        else:
            stats.error_count += 1
        
        # 缓存命中统计已在调用时更新
    
    def get_performance_stats(self) -> Dict[str, Dict[str, Any]]:
        """获取性能统计
        
        Returns:
            性能统计字典
        """
        result = {}
        for mode_name, stats in self._performance_stats.items():
            result[mode_name] = {
                "call_count": stats.call_count,
                "success_count": stats.success_count,
                "error_count": stats.error_count,
                "total_time": stats.total_time,
                "average_time": stats.average_time,
                "success_rate": stats.success_rate,
                "cache_hits": stats.cache_hits,
                "cache_misses": stats.cache_misses,
                "cache_hit_rate": stats.cache_hit_rate
            }
        return result
    
    def get_cache_info(self) -> Dict[str, Any]:
        """获取缓存信息
        
        Returns:
            缓存信息字典
        """
        return {
            "size": len(self._cache),
            "max_size": self._max_cache_size,
            "ttl": self._cache_ttl
        }
    
    def clear_cache(self) -> None:
        """清空缓存"""
        self._cache.clear()
        self.logger.info("思考模式缓存已清空")
    
    def reset_stats(self) -> None:
        """重置性能统计"""
        self._performance_stats.clear()
        self.logger.info("性能统计已重置")
    
    def _cleanup_cache_if_needed(self) -> None:
        """清理缓存（如果需要）"""
        if len(self._cache) > self._max_cache_size:
            # 简单的LRU清理：删除前20%的条目
            keys_to_remove = list(self._cache.keys())[:self._max_cache_size // 5]
            for key in keys_to_remove:
                del self._cache[key]
            self.logger.info(f"清理缓存，移除了 {len(keys_to_remove)} 个条目")

5. **缓存清理机制**：实现LRU缓存清理，防止内存泄漏。

#### 常见错误及避免
1. **缓存键冲突**：使用哈希算法生成唯一键，确保不同输入产生不同键。

class AdaptiveThinkingModeTuner:
    """自适应思考模式调优器"""
    
    def __init__(self, config: ThinkingConfig):
        self.config = config
        self.feedback_history = []
        self.learning_rate = 0.1
        
    def record_feedback(self, task_features: TaskFeatures, 
                       mode_name: str, quality_score: float):
        """记录反馈"""
        self.feedback_history.append({
            'features': task_features,
            'mode': mode_name,
            'quality': quality_score,
            'timestamp': time.time()
        })
        
    def tune_parameters(self, mode_name: str) -> Dict[str, Any]:
        """调优参数"""
        # 获取该模式的历史反馈
        mode_feedback = [
            f for f in self.feedback_history 
            if f['mode'] == mode_name
        ]
        
        if not mode_feedback:
            return self.config.get_mode_config(mode_name)
        
        # 分析反馈，调整参数
        avg_quality = np.mean([f['quality'] for f in mode_feedback])
        
        # 根据质量调整temperature
        config = self.config.get_mode_config(mode_name).copy()
        current_temp = config.get('temperature', 0.7)
        
        if avg_quality < 0.5:
            # 质量低，降低随机性
            new_temp = max(0.1, current_temp - self.learning_rate)
        else:
            # 质量高，保持或轻微增加随机性
            new_temp = min(2.0, current_temp + self.learning_rate * 0.5)
        
        config['temperature'] = new_temp
        return config

### 5.3 分布式缓存

```python
import redis
import pickle

class RedisThinkingModeApplicator(ThinkingModeApplicator):
    """支持Redis的思考模式应用器"""
    
    def __init__(self, config: ThinkingConfig, redis_url: str = "redis://localhost:6379"):
        super().__init__(config)
        self.redis_client = redis.from_url(redis_url)
        self.local_cache = {}  # 本地缓存作为一级缓存
        
    def _get_from_cache(self, cache_key: str):
        """从缓存获取（先本地，后Redis）"""
        # 先检查本地缓存
        if cache_key in self.local_cache:
            return self.local_cache[cache_key]
        
        # 检查Redis缓存
        redis_data = self.redis_client.get(cache_key)
        if redis_data:
            result = pickle.loads(redis_data)
            # 存入本地缓存
            self.local_cache[cache_key] = result
            return result
        
        return None
    
    def _set_to_cache(self, cache_key: str, result: Any, ttl: int = 3600):
        """存入缓存（本地和Redis）"""
        # 存入本地缓存
        self.local_cache[cache_key] = result
        
        # 存入Redis
        redis_data = pickle.dumps(result)
        self.redis_client.setex(cache_key, ttl, redis_data)
```
