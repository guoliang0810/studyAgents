```python
class ImageProcessor(ABC):
    """图像处理器抽象基类"""
    
    def __init__(self, name: str, config: Optional[Dict[str, Any]] = None):
        self.name = name
        self.config = config or {}
        self.processed_count = 0
        self.success_count = 0
        self.failure_count = 0
        self.total_processing_time = 0.0
    
    @abstractmethod
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """处理图像数据"""
        pass
    
    def update_stats(self, success: bool, processing_time: float) -> None:
        """更新统计信息"""
        self.processed_count += 1
        if success:
            self.success_count += 1
        else:
            self.failure_count += 1
        self.total_processing_time += processing_time
```

```python
class FormatConverter(ImageProcessor):
    """图像格式转换器"""
    
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """转换图像格式"""
        from PIL import Image
        import io
        
        # 打开图像
        image = Image.open(io.BytesIO(image_data))
        
        # 转换为RGB模式（如果必要）
        if image.mode not in ("RGB", "L"):
            if image.mode == "RGBA":
                # 创建白色背景
                background = Image.new("RGB", image.size, (255, 255, 255))
                background.paste(image, mask=image.split()[-1])
                image = background
            else:
                image = image.convert("RGB")
        
        # 保存为指定格式
        output_buffer = io.BytesIO()
        save_kwargs = {}
        
        if self.target_format in ("jpeg", "jpg"):
            save_kwargs["quality"] = self.quality
            save_kwargs["optimize"] = self.optimize
        
        image.save(output_buffer, format=self.target_format, **save_kwargs)
        return output_buffer.getvalue()


class SizeResizer(ImageProcessor):
    """图像大小调整器"""
    
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """调整图像大小以满足限制"""
        # 如果已经在限制内，直接返回
        if len(image_data) <= self.max_size_bytes:
            return image_data
        
        # 渐进式优化策略
        return await self._progressive_optimize(image, input_format, len(image_data))


class QualityOptimizer(ImageProcessor):
    """图像质量优化器"""
    
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """优化图像质量"""
        from PIL import Image, ImageFilter, ImageOps
        import io
        
        image = Image.open(io.BytesIO(image_data))
        
        # 应用优化：锐化、对比度、亮度
        if self.sharpen_strength > 0:
            image = image.filter(ImageFilter.UnsharpMask(
                radius=2, percent=int(self.sharpen_strength * 100), threshold=0
            ))
        
        # 保存优化后的图像
        buffer = io.BytesIO()
        image.save(buffer, format=input_format, **save_kwargs)
        return buffer.getvalue()


class MetadataStripper(ImageProcessor):
    """图像元数据清理器"""
    
    async def process(self, image_data: bytes, input_format: str) -> bytes:
        """清理图像元数据"""
        from PIL import Image
        import io
        
        image = Image.open(io.BytesIO(image_data))
        
        # 创建新图像（复制像素数据，不复制元数据）
        new_image = Image.new(image.mode, image.size)
        new_image.putdata(list(image.getdata()))
        
        # 保存图像（不包含元数据）
        buffer = io.BytesIO()
        new_image.save(buffer, format=input_format)
        return buffer.getvalue()
```

```python
class VisionPipeline:
    """视觉处理管道"""
    
    def __init__(self, vision_config: VisionConfig, 
                 processor_chain: Optional[List[ImageProcessor]] = None):
        self.vision_config = vision_config
        self.processors = processor_chain or self._create_default_chain()
        self.stats = ProcessingStats()
        self.cache: Dict[str, ProcessedImage] = {}
    
    async def process_image(self, image_data: bytes, 
                           input_format: str) -> ProcessedImage:
        """处理单个图像"""
        start_time = time.time()
        original_size = len(image_data)
        
        current_data = image_data
        current_format = input_format
        processing_steps = []
        
        # 依次应用处理器（责任链模式）
        for processor in self.processors:
            try:
                processed_data = await processor.process(current_data, current_format)
                
                # 记录处理步骤
                step_name = f"{processor.name}"
                processing_steps.append(step_name)
                
                # 更新当前数据和格式
                current_data = processed_data
                
            except Exception as e:
                # 处理器失败，记录但继续（错误隔离）
                processing_steps.append(f"{processor.name}(失败: {str(e)})")
                continue
        
        processed_size = len(current_data)
        processing_time = time.time() - start_time
        
        # 创建处理结果
        result = ProcessedImage(
            data=current_data,
            format=current_format,
            original_size=original_size,
            processed_size=processed_size,
            processing_steps=processing_steps
        )
        
        # 更新统计
        self.stats.update(True, processing_time, original_size, processed_size)
        return result
    
    async def process_batch(self, images: List[Tuple[bytes, str]], 
                           max_concurrent: int = 10) -> List[ProcessedImage]:
        """批量处理多个图像"""
        semaphore = asyncio.Semaphore(max_concurrent)
        
        async def process_with_semaphore(image_data, image_format):
            async with semaphore:
                return await self.process_image(image_data, image_format)
        
        tasks = [
            process_with_semaphore(image_data, image_format)
            for image_data, image_format in images
        ]
        
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 处理异常结果
        processed_images = []
        for i, result in enumerate(results):
            if isinstance(result, Exception):
                # 创建失败的结果
                processed_images.append(ProcessedImage(
                    data=b"",
                    format="",
                    original_size=len(images[i][0]),
                    processed_size=0,
                    processing_steps=[f"处理失败: {str(result)}"]
                ))
            else:
                processed_images.append(result)
        
        return processed_images
    
    async def prepare_images_for_model(self, images: List[ImageInfo], 
                                      model_name: str) -> List[ProcessedImage]:
        """为特定模型准备图像"""
        model_config = self.vision_config.get_model_config(model_name)
        if not model_config:
            raise ValueError(f"模型 '{model_name}' 未找到")
        
        # 过滤模型不支持的图像
        supported_images = []
        for image_info in images:
            if model_config.can_handle_image(image_info):
                supported_images.append((image_info.data, image_info.format))
        
        # 批量处理支持的图像
        return await self.process_batch(supported_images)
```

```python
class VideoFrameExtractor(ImageProcessor):
    """视频帧提取器"""
    
    async def process(self, video_data: bytes, input_format: str) -> List[bytes]:
        """提取视频帧"""
        import cv2
        import tempfile
        
        # 将字节数据保存到临时文件
        with tempfile.NamedTemporaryFile(suffix=f".{input_format}", delete=False) as f:
            f.write(video_data)
            temp_path = f.name
        
        # 使用OpenCV提取视频帧
        cap = cv2.VideoCapture(temp_path)
        frames = []
        
        while len(frames) < self.max_frames:
            ret, frame = cap.read()
            if not ret:
                break
            
            # 调整帧大小
            if frame.shape[:2] != self.frame_size:
                frame = cv2.resize(frame, self.frame_size)
            
            # 编码为图像格式
            _, encoded = cv2.imencode(f".{self.output_format}", frame)
            frames.append(encoded.tobytes())
        
        cap.release()
        return frames
```

### 5.2 分布式处理扩展

```python
class DistributedVisionPipeline:
    """分布式视觉处理管道"""
    
    async def submit_task(self, image_data: bytes, image_format: str) -> str:
        """提交处理任务到分布式队列"""
        import redis
        import uuid
        
        task_id = str(uuid.uuid4())
        task_data = {
            "task_id": task_id,
            "image_data": image_data.hex(),
            "image_format": image_format
        }
        
        # 提交到Redis队列
        redis_client = redis.Redis.from_url(self.redis_url)
        redis_client.lpush(self.task_queue, json.dumps(task_data))
        
        return task_id
    
    async def get_result(self, task_id: str, timeout: float = 30.0) -> Optional[ProcessedImage]:
        """从分布式结果队列获取处理结果"""
        import redis
        
        redis_client = redis.Redis.from_url(self.redis_url)
        start_time = time.time()
        
        while time.time() - start_time < timeout:
            result_key = f"{self.result_queue}:{task_id}"
            result_data = redis_client.get(result_key)
            
            if result_data:
                result = json.loads(result_data)
                return ProcessedImage(
                    data=bytes.fromhex(result["data"]),
                    format=result["format"],
                    original_size=result["original_size"],
                    processed_size=result["processed_size"]
                )
            
            await asyncio.sleep(0.1)
        
        return None
```

1. **配置管理**：
   - 使用YAML进行配置，便于版本控制和环境管理
   - 实现配置验证和自动迁移
   - 支持环境变量和敏感信息管理

2. **性能优化**：
   - 实现渐进式优化策略（质量→尺寸→格式）
   - 添加缓存机制，避免重复处理
   - 支持异步和并发处理

3. **安全隐私**：
   - 默认开启元数据清理
   - 实现内容安全检查
   - 支持数据加密和访问控制

4. **监控运维**：
   - 添加全面的性能统计
   - 实现健康检查和自动恢复
   - 支持日志记录和审计跟踪

### 6.2 常见问题解决

1. **处理速度慢**：
   - **原因**：图像太大或处理器链太长
   - **解决**：优化处理器顺序，添加并发处理
   - **优化**：实现渐进式优化，提前终止不必要处理

2. **内存使用高**：
   - **原因**：大图像缓存或内存泄漏
   - **解决**：实现流式处理，限制缓存大小
   - **优化**：使用内存映射文件处理超大图像

3. **兼容性问题**：
   - **原因**：模型不支持特定格式或大小
   - **解决**：实现智能格式转换和大小调整
   - **优化**：添加模型能力测试和自动适配

4. **安全性问题**：
   - **原因**：用户上传恶意图像或包含隐私数据
   - **解决**：实现内容安全检查和元数据清理
   - **优化**：添加数字签名和访问控制

### 6.3 部署建议

1. **开发环境**：
   - 使用Docker容器化部署
   - 配置本地开发环境变量
   - 实现自动化测试和代码检查

2. **测试环境**：
   - 模拟生产环境配置
   - 实现性能测试和安全测试
   - 添加监控和告警配置

3. **生产环境**：
   - 使用Kubernetes进行容器编排
   - 配置自动扩缩容和负载均衡
   - 实现蓝绿部署或金丝雀发布

4. **监控运维**：
   - 集成Prometheus和Grafana监控
   - 配置日志聚合和异常检测
   - 实现自动化备份和恢复
