#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
模型能力定义和配置数据类
"""

from dataclasses import dataclass, field, asdict
from typing import Dict, Any, List, Optional
import json

@dataclass
class ModelCapabilities:
    """模型能力定义"""
    
    supports_vision: bool = False
    supports_function_calling: bool = False
    supports_long_context: bool = False
    max_context_length: int = 4096
    reasoning_depth: str = "medium"  # shallow, medium, deep
    
    def has_capability(self, capability: str) -> bool:
        """检查是否支持特定能力"""
        capability_map = {
            "vision": self.supports_vision,
            "function_calling": self.supports_function_calling,
            "long_context": self.supports_long_context
        }
        return capability_map.get(capability, False)
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return asdict(self)
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "ModelCapabilities":
        """从字典创建"""
        return cls(**data)

@dataclass
class ModelConfig:
    """模型配置"""
    
    provider: str
    model_name: str
    capabilities: ModelCapabilities
    cost_per_token: float
    config: Dict[str, Any] = field(default_factory=dict)
    
    def __post_init__(self):
        """验证配置"""
        if self.cost_per_token < 0:
            raise ValueError("cost_per_token不能为负数")
        
        # 确保capabilities是ModelCapabilities实例
        if not isinstance(self.capabilities, ModelCapabilities):
            if isinstance(self.capabilities, dict):
                self.capabilities = ModelCapabilities(**self.capabilities)
            else:
                raise TypeError("capabilities必须是ModelCapabilities实例或字典")
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        result = asdict(self)
        result["capabilities"] = self.capabilities.to_dict()
        return result
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "ModelConfig":
        """从字典创建"""
        capabilities_data = data.get("capabilities", {})
        capabilities = ModelCapabilities.from_dict(capabilities_data)
        
        return cls(
            provider=data.get("provider", "unknown"),
            model_name=data.get("model_name", "unknown"),
            capabilities=capabilities,
            cost_per_token=data.get("cost_per_token", 0.001),
            config=data.get("config", {})
        )

# -*- coding: utf-8 -*-
"""
路由器配置管理
"""

from dataclasses import dataclass, field, asdict
from typing import Dict, Any, List, Optional, Union
import json
import yaml
from pathlib import Path
from .model_definitions import ModelConfig, ModelCapabilities

@dataclass
class RoutingDecision:
    """路由决策结果"""
    
    model_name: str
    strategy: str
    confidence: float
    reason: str
    metadata: Dict[str, Any] = field(default_factory=dict)
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return asdict(self)
    
    def __str__(self) -> str:
        """字符串表示"""
        return (f"RoutingDecision(model={self.model_name}, "
                f"strategy={self.strategy}, confidence={self.confidence:.2f}, "
                f"reason='{self.reason}')")
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "RoutingDecision":
        """从字典创建"""
        return cls(**data)

@dataclass
class TaskFeatures:
    """任务特征"""
    
    task_type: Optional[str] = None
    complexity: float = 0.5  # 0-1，表示任务复杂程度
    urgency: float = 0.0     # 0-1，表示紧急程度
    budget: Optional[float] = None
    latency_requirement: Optional[float] = None  # 最大延迟（秒）
    required_capabilities: List[str] = field(default_factory=list)
    
    def validate(self) -> List[str]:
        """验证任务特征，返回错误列表"""
        errors = []
        
        # 复杂度验证
        if self.complexity < 0 or self.complexity > 1:
            errors.append(f"复杂度必须在0-1之间，当前值: {self.complexity}")
        
        # 紧急度验证
        if self.urgency < 0 or self.urgency > 1:
            errors.append(f"紧急度必须在0-1之间，当前值: {self.urgency}")
        
        # 预算验证
        if self.budget is not None and self.budget < 0:
            errors.append(f"预算不能为负数，当前值: {self.budget}")
        
        # 延迟要求验证
        if self.latency_requirement is not None and self.latency_requirement <= 0:
            errors.append(f"延迟要求必须大于0，当前值: {self.latency_requirement}")
        
        return errors
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return asdict(self)
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "TaskFeatures":
        """从字典创建"""
        return cls(**data)

@dataclass
class RouterConfig:
    """路由器配置"""
    
    model_config: Dict[str, Any] = field(default_factory=dict)
    default_model: str = "gpt-3.5-turbo"
    task_routing: Dict[str, List[str]] = field(default_factory=dict)
    models: Dict[str, ModelConfig] = field(default_factory=dict)
    
    @classmethod
    def from_yaml(cls, yaml_content: str) -> "RouterConfig":
        """从YAML内容创建配置"""
        try:
            data = yaml.safe_load(yaml_content)
        except ImportError:
            # 如果yaml不可用，尝试JSON格式
            try:
                data = json.loads(yaml_content)
            except json.JSONDecodeError:
                raise ValueError("配置内容既不是有效的YAML也不是JSON")
        
        # 转换模型配置
        models = {}
        models_data = data.get("models", {})
        
        for model_name, model_data in models_data.items():
            # 处理能力配置
            capabilities_data = model_data.get("capabilities", {})
            capabilities = ModelCapabilities(
                supports_vision=capabilities_data.get("supports_vision", False),
                supports_function_calling=capabilities_data.get("supports_function_calling", False),
                supports_long_context=capabilities_data.get("supports_long_context", False),
                max_context_length=capabilities_data.get("max_context_length", 4096),
                reasoning_depth=capabilities_data.get("reasoning_depth", "medium")
            )
            
            # 创建模型配置
            models[model_name] = ModelConfig(
                provider=model_data.get("provider", "unknown"),
                model_name=model_data.get("model_name", model_name),
                capabilities=capabilities,
                cost_per_token=model_data.get("cost_per_token", 0.001),
                config=model_data.get("config", {})
            )
        
        return cls(
            model_config=data.get("model_config", {}),
            default_model=data.get("default_model", "gpt-3.5-turbo"),
            task_routing=data.get("task_routing", {}),
            models=models
        )
    
    @classmethod
    def from_file(cls, filepath: Union[str, Path]) -> "RouterConfig":
        """从文件创建配置"""
        filepath = Path(filepath)
        if not filepath.exists():
            raise FileNotFoundError(f"配置文件不存在: {filepath}")
        
        with open(filepath, 'r', encoding='utf-8') as f:
            content = f.read()
        
        return cls.from_yaml(content)
    
    def to_yaml(self) -> str:
        """转换为YAML格式"""
        try:
            data = self.to_dict()
            return yaml.dump(data, allow_unicode=True, sort_keys=False)
        except ImportError:
            # 如果yaml不可用，使用JSON
            data = self.to_dict()
            return json.dumps(data, indent=2, ensure_ascii=False)
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        models_dict = {}
        for name, config in self.models.items():
            models_dict[name] = config.to_dict()
        
        return {
            "model_config": self.model_config,
            "default_model": self.default_model,
            "task_routing": self.task_routing,
            "models": models_dict
        }
    
    def validate(self) -> List[str]:
        """验证配置，返回错误列表"""
        errors = []
        
        # 检查默认模型是否存在
        if self.default_model not in self.models:
            errors.append(f"默认模型 '{self.default_model}' 未在模型列表中定义")
        
        # 检查任务路由中的模型是否存在
        for task_type, model_list in self.task_routing.items():
            for model_name in model_list:
                if model_name not in self.models:
                    errors.append(f"任务路由 '{task_type}' 中的模型 '{model_name}' 未定义")
        
        # 检查模型配置
        for model_name, model_config in self.models.items():
            try:
                # 验证模型配置
                if model_config.cost_per_token < 0:
                    errors.append(f"模型 '{model_name}' 的cost_per_token不能为负数")
            except Exception as e:
                errors.append(f"模型 '{model_name}' 配置验证失败: {e}")
        
        return errors

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
使用情况跟踪器
"""

import time
from datetime import datetime
from typing import Dict, Any, List, Optional
from collections import defaultdict
from threading import Lock

class UsageTracker:
    """使用情况跟踪器"""
    
    def __init__(self):
        self.usage_stats: Dict[str, Dict[str, Any]] = defaultdict(self._create_stats_template)
        self.performance_scores: Dict[str, float] = {}
        self.lock = Lock()  # 线程安全锁
        
        # 默认性能得分（用于没有历史数据的模型）
        self.default_scores = {
            "gpt-4": 0.9,
            "gpt-4-turbo": 0.85,
            "gpt-3.5-turbo": 0.7,
            "claude-3": 0.85,
            "claude-2": 0.8,
            "gemini-pro": 0.8,
            "llama-2": 0.65,
            "mixtral": 0.75
        }
    
    def _create_stats_template(self) -> Dict[str, Any]:
        """创建统计模板"""
        return {
            "call_count": 0,
            "total_tokens": 0,
            "success_count": 0,
            "total_response_time": 0.0,
            "last_used": None
        }
    
    def record_usage(self, model_name: str, tokens_used: int = 0, 
                     success: bool = True, response_time: float = 0.0):
        """记录模型使用情况"""
        with self.lock:
            stats = self.usage_stats[model_name]
            stats["call_count"] += 1
            stats["total_tokens"] += tokens_used
            
            if success:
                stats["success_count"] += 1
            
            stats["total_response_time"] += response_time
            stats["last_used"] = datetime.now()
            
            # 清空性能得分缓存
            if model_name in self.performance_scores:
                del self.performance_scores[model_name]
    
    def get_performance_score(self, model_name: str) -> float:
        """获取模型性能得分"""
        # 检查缓存
        if model_name in self.performance_scores:
            return self.performance_scores[model_name]
        
        with self.lock:
            stats = self.usage_stats.get(model_name)
            
            # 如果没有使用记录，返回默认得分
            if not stats or stats["call_count"] == 0:
                score = self.default_scores.get(model_name, 0.5)
                self.performance_scores[model_name] = score
                return score
            
            # 计算成功率
            success_rate = stats["success_count"] / stats["call_count"]
            
            # 计算响应时间得分
            if stats["call_count"] > 0:
                avg_response_time = stats["total_response_time"] / stats["call_count"]
                # 归一化：理想响应时间1秒，最大容忍5秒
                time_score = max(0, 1 - min(avg_response_time, 5) / 5)
            else:
                time_score = 0.5
            
            # 综合得分：成功率60%，响应时间40%
            performance_score = success_rate * 0.6 + time_score * 0.4
            
            # 缓存结果
            self.performance_scores[model_name] = performance_score
            
            return performance_score
    
    def get_usage_summary(self) -> Dict[str, Dict[str, Any]]:
        """获取使用情况摘要"""
        summary = {}
        
        with self.lock:
            for model_name, stats in self.usage_stats.items():
                if stats["call_count"] == 0:
                    continue
                
                success_rate = stats["success_count"] / stats["call_count"]
                avg_tokens = stats["total_tokens"] / stats["call_count"]
                avg_response_time = stats["total_response_time"] / stats["call_count"]
                
                summary[model_name] = {
                    "call_count": stats["call_count"],
                    "success_rate": success_rate,
                    "avg_tokens": avg_tokens,
                    "avg_response_time": avg_response_time,
                    "last_used": stats["last_used"],
                    "performance_score": self.get_performance_score(model_name)
                }
        
        return summary
    
    def reset_stats(self, model_name: Optional[str] = None):
        """重置统计"""
        with self.lock:
            if model_name:
                if model_name in self.usage_stats:
                    self.usage_stats[model_name] = self._create_stats_template()
                if model_name in self.performance_scores:
                    del self.performance_scores[model_name]
            else:
                # 重置所有统计
                self.usage_stats.clear()
                self.performance_scores.clear()
    
    def get_top_performing_models(self, limit: int = 5) -> List[Dict[str, Any]]:
        """获取性能最好的模型"""
        summary = self.get_usage_summary()
        
        # 按性能得分排序
        sorted_models = sorted(
            summary.items(),
            key=lambda x: x[1]["performance_score"],
            reverse=True
        )
        
        result = []
        for model_name, stats in sorted_models[:limit]:
            result.append({
                "model_name": model_name,
                "performance_score": stats["performance_score"],
                "call_count": stats["call_count"],
                "success_rate": stats["success_rate"]
            })
        
        return result
```

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
路由策略实现
"""

import asyncio
from abc import ABC, abstractmethod
from typing import Dict, Any, List, Optional
from dataclasses import dataclass
from .router_config import RoutingDecision
from .usage_tracker import UsageTracker

class RoutingStrategy(ABC):
    """路由策略基类"""
    
    @abstractmethod
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """执行路由决策"""
        pass
    
    def get_strategy_name(self) -> str:
        """获取策略名称"""
        return self.__class__.__name__.replace("Strategy", "").lower()

@dataclass
class ModelInfo:
    """模型信息（用于策略上下文）"""
    name: str
    cost_per_token: float
    capabilities: Any
    config: Dict[str, Any]

class CostOptimizedStrategy(RoutingStrategy):
    """成本优化策略"""
    
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """选择成本最低的可用模型"""
        models = context.get("available_models", [])
        if not models:
            raise ValueError("没有可用的模型")
        
        # 找到成本最低的模型
        cheapest = min(models, key=lambda m: m.cost_per_token)
        
        return RoutingDecision(
            model_name=cheapest.name,
            strategy="cost_optimized",
            confidence=0.8,
            reason=f"选择成本最低的模型（{cheapest.cost_per_token:.6f}/token）"
        )

class PerformanceOptimizedStrategy(RoutingStrategy):
    """性能优化策略"""
    
    def __init__(self, usage_tracker: Optional[UsageTracker] = None):
        self.usage_tracker = usage_tracker or UsageTracker()
    
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """选择性能最好的可用模型"""
        models = context.get("available_models", [])
        if not models:
            raise ValueError("没有可用的模型")
        
        # 计算各模型性能得分
        model_scores = []
        for model in models:
            score = self.usage_tracker.get_performance_score(model.name)
            model_scores.append((model, score))
        
        # 找到性能最好的模型
        best_model, best_score = max(model_scores, key=lambda x: x[1])
        
        return RoutingDecision(
            model_name=best_model.name,
            strategy="performance_optimized",
            confidence=best_score,
            reason=f"选择性能最好的模型（得分: {best_score:.2f}）"
        )

class BalancedStrategy(RoutingStrategy):
    """平衡策略"""
    
    def __init__(self, usage_tracker: Optional[UsageTracker] = None):
        self.usage_tracker = usage_tracker or UsageTracker()
    
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """平衡成本和性能的选择"""
        models = context.get("available_models", [])
        if not models:
            raise ValueError("没有可用的模型")
        
        # 计算综合得分
        best_score = -1
        best_model = None
        
        for model in models:
            # 成本得分（越便宜得分越高）
            cost_score = 1 - min(model.cost_per_token * 1000, 1.0)
            
            # 性能得分
            perf_score = self.usage_tracker.get_performance_score(model.name)
            
            # 平衡权重：成本40%，性能60%
            total_score = cost_score * 0.4 + perf_score * 0.6
            
            if total_score > best_score:
                best_score = total_score
                best_model = model
        
        return RoutingDecision(
            model_name=best_model.name,
            strategy="balanced",
            confidence=best_score,
            reason=f"平衡选择：综合得分{best_score:.2f}（成本{best_model.cost_per_token:.6f}/token，性能{self.usage_tracker.get_performance_score(best_model.name):.2f}）"
        )

class LearningBasedStrategy(RoutingStrategy):
    """学习型策略"""
    
    def __init__(self):
        self.history: List[Dict[str, Any]] = []
        self.success_rates: Dict[str, float] = {}
    
    async def route(self, prompt: str, context: Dict[str, Any]) -> RoutingDecision:
        """基于历史学习的选择"""
        models = context.get("available_models", [])
        if not models:
            raise ValueError("没有可用的模型")
        
        # 更新成功率缓存
        self._update_success_rates()
        
        # 找到成功率最高的模型
        model_names = [model.name for model in models]
        available_rates = {name: rate for name, rate in self.success_rates.items() 
                          if name in model_names}
        
        if not available_rates:
            # 没有历史数据，选择第一个模型
            selected_model = models[0]
            confidence = 0.5
            reason = "没有历史数据，选择默认模型"
        else:
            # 选择成功率最高的模型
            best_model_name = max(available_rates.items(), key=lambda x: x[1])[0]
            selected_model = next(m for m in models if m.name == best_model_name)
            confidence = available_rates[best_model_name]
            reason = f"历史成功率：{confidence:.1%}"
        
        return RoutingDecision(
            model_name=selected_model.name,
            strategy="learning_based",
            confidence=confidence,
            reason=reason
        )
    
    def record_result(self, model_name: str, success: bool):
        """记录模型使用结果"""
        self.history.append({
            "model": model_name,
            "success": success,
            "timestamp": datetime.now()
        })
        
        # 清空成功率缓存
        if model_name in self.success_rates:
            del self.success_rates[model_name]
    
    def _update_success_rates(self):
        """更新成功率缓存"""
        # 找出所有需要计算的模型
        models_needed = set(record["model"] for record in self.history)
        
        for model_name in models_needed:
            if model_name in self.success_rates:
                continue
            
            # 计算该模型的历史成功率
            model_history = [h for h in self.history if h["model"] == model_name]
            if model_history:
                successes = sum(1 for h in model_history if h["success"])
                self.success_rates[model_name] = successes / len(model_history)
            else:
                self.success_rates[model_name] = 0.5

from typing import Dict, Any, List, Optional, Callable
from .router_config import RouterConfig, TaskFeatures, RoutingDecision
from .usage_tracker import UsageTracker
from .model_definitions import ModelConfig

class ModelRouter:
    """模型路由器"""
    
    def __init__(self, config: RouterConfig):
        self.config = config
        self.usage_tracker = UsageTracker()
        self.model_registry: Dict[str, Any] = {}
    
    def register_model(self, name: str, model: Any):
        """注册模型实例"""
        self.model_registry[name] = model
    
    def unregister_model(self, name: str):
        """注销模型实例"""
        if name in self.model_registry:

    def get_registered_models(self) -> List[str]:
        """获取已注册的模型列表"""
        return list(self.model_registry.keys())
    
    async def route(
        self, 
        prompt: str, 
        task_type: str = None,
        budget: float = None,
        latency_requirement: float = None,
        required_capabilities: List[str] = None
    ) -> Any:
        """路由到合适的模型"""
        
        # 1. 获取候选模型
        candidates = self._get_candidate_models(task_type, required_capabilities)
        
        if not candidates:
            raise ValueError("没有符合条件的候选模型")
        
        # 2. 应用过滤规则
        filtered = self._filter_models(candidates, budget, latency_requirement)
        
        if not filtered:
            # 回退到默认模型
            if self.config.default_model not in self.model_registry:
                raise ValueError(f"默认模型 '{self.config.default_model}' 未注册")
            
            self.usage_tracker.record_usage(self.config.default_model)
            return self.model_registry[self.config.default_model]
        
        # 3. 选择最佳模型
        best_model_name = self._select_best_model(filtered, prompt)
        
        # 4. 获取模型实例
        if best_model_name not in self.model_registry:
            raise ValueError(f"模型 '{best_model_name}' 未注册")
        
        best_model = self.model_registry[best_model_name]
        
        # 5. 记录使用情况
        self.usage_tracker.record_usage(best_model_name)
        
        return best_model
    
    async def route_with_decision(
        self,
        prompt: str,
        task_type: str = None,
        budget: float = None,
        latency_requirement: float = None,
        required_capabilities: List[str] = None
    ) -> Dict[str, Any]:
        """路由并返回决策详情"""
        
        # 获取候选模型
        candidates = self._get_candidate_models(task_type, required_capabilities)
        
        if not candidates:
            return {
                "success": False,
                "error": "没有符合条件的候选模型",
                "decision": None
            }
        
        # 应用过滤
        filtered = self._filter_models(candidates, budget, latency_requirement)
        
        if not filtered:

            if self.config.default_model not in self.model_registry:
                return {
                    "success": False,
                    "error": f"默认模型 '{self.config.default_model}' 未注册",
                    "decision": None
                }
            
            return {
                "success": True,
                "model": self.model_registry[self.config.default_model],
                "decision": RoutingDecision(
                    model_name=self.config.default_model,
                    strategy="default_fallback",
                    confidence=0.5,
                    reason="没有符合条件的模型，回退到默认模型"
                )
            }
        
        # 计算得分
        scores = {}
        score_details = {}
        
        for model_name in filtered:
            score, details = self._calculate_model_score(model_name, prompt)
            scores[model_name] = score
            score_details[model_name] = details
        
        # 选择最佳模型
        best_model_name = max(scores.items(), key=lambda x: x[1])[0]
        
        if best_model_name not in self.model_registry:
            return {
                "success": False,
                "error": f"模型 '{best_model_name}' 未注册",
                "decision": None
            }
        
        best_model = self.model_registry[best_model_name]
        
        # 记录使用
        self.usage_tracker.record_usage(best_model_name)
        
        return {
            "success": True,
            "model": best_model,
            "decision": RoutingDecision(
                model_name=best_model_name,
                strategy="scoring_based",
                confidence=scores[best_model_name],
                reason=f"综合得分最高：{scores[best_model_name]:.2f}",
                metadata={
                    "scores": scores,
                    "score_details": score_details,
                    "candidates": candidates,
                    "filtered": filtered
                }
            )
        }
    
    def _get_candidate_models(self, task_type: str = None, 
                             required_capabilities: List[str] = None) -> List[str]:
        """获取候选模型列表"""
        candidates = []
        
        # 基于任务类型选择
        if task_type:
            task_models = self.config.task_routing.get(task_type, [])
            if task_models:
                candidates.extend(task_models)
        
        # 如果没有基于任务类型的候选，使用所有模型
        if not candidates:
            candidates = list(self.config.models.keys())
        
        # 基于能力要求过滤
        if required_capabilities:
            filtered = []
            for model_name in candidates:
                model_config = self.config.models.get(model_name)
                if not model_config:
                    continue
                
                # 检查是否满足所有能力要求
                if all(model_config.capabilities.has_capability(cap) 
                      for cap in required_capabilities):
                    filtered.append(model_name)
            
            candidates = filtered
        
        return candidates
    
    def _filter_models(self, candidates: List[str], budget: float = None, 
                      latency_requirement: float = None) -> List[str]:
        """过滤模型"""
        filtered = []
        
        for model_name in candidates:
            model_config = self.config.models.get(model_name)
            if not model_config:
                continue
            
            # 预算过滤
            if budget is not None:
                # 假设平均每次调用使用1000 tokens
                estimated_cost = model_config.cost_per_token * 1000
                if estimated_cost > budget:
                    continue
            
            # 这里可以添加更多过滤条件
            # 如延迟要求、可用性等
            
            filtered.append(model_name)
        
        return filtered
    
    def _calculate_model_score(self, model_name: str, prompt: str) -> tuple:
        """计算模型综合得分和详情"""
        model_config = self.config.models.get(model_name)
        if not model_config:
            return 0.0, {}
        
        details = {}
        
        # 1. 性能得分（40%）
        perf_score = self.usage_tracker.get_performance_score(model_name)
        details["performance_score"] = perf_score
        details["performance_weight"] = 0.4
        
        # 2. 成本得分（30%）
        cost = model_config.cost_per_token
        cost_score = max(0, 1 - min(cost * 1000, 1.0))
        details["cost_score"] = cost_score
        details["cost"] = cost
        details["cost_weight"] = 0.3
        
        # 3. 适合度得分（30%）
        fit_score = self._calculate_fit_score(model_config, prompt)
        details["fit_score"] = fit_score
        details["fit_weight"] = 0.3
        
        # 综合得分
        total_score = (
            perf_score * 0.4 +
            cost_score * 0.3 +
            fit_score * 0.3
        )
        
        details["total_score"] = total_score
        
        return total_score, details
    
    def _select_best_model(self, candidates: List[str], prompt: str) -> str:
        """选择最佳模型"""
        best_score = -1
        best_model = None
        
        for model_name in candidates:
            score, _ = self._calculate_model_score(model_name, prompt)
            if score > best_score:
                best_score = score
                best_model = model_name
        
        return best_model
    
    def _calculate_fit_score(self, model_config: ModelConfig, prompt: str) -> float:
        """计算模型适合度得分"""
        prompt_length = len(prompt)
        
        # 长度适合度
        if model_config.capabilities.supports_long_context:
            # 长上下文模型适合长提示
            if prompt_length > 2000:
                length_score = 0.9
            elif prompt_length < 500:
                length_score = 0.7
            else:
                length_score = 0.8
        else:
            # 短上下文模型适合短提示
            if prompt_length < 1000:
                length_score = 0.9
            elif prompt_length > 2000:
                length_score = 0.3
            else:
                length_score = 0.6
        
        # 复杂度适合度
        complexity_keywords = ["分析", "解释", "比较", "设计", "实现", "复杂", "困难"]
        complexity = sum(1 for word in complexity_keywords if word in prompt) / len(complexity_keywords)
        
        if model_config.capabilities.reasoning_depth == "deep":
            # 深度推理模型适合复杂任务
            complexity_score = 0.3 + complexity * 0.7
        elif model_config.capabilities.reasoning_depth == "shallow":
            # 浅层推理模型适合简单任务
            complexity_score = 0.7 - complexity * 0.4
        else:
            # 中等推理模型
            complexity_score = 0.5 + (complexity - 0.5) * 0.4
        
        # 综合适合度得分
        fit_score = (length_score + complexity_score) / 2
        
        return fit_score
    
    def get_usage_stats(self) -> Dict[str, Any]:
        """获取使用统计"""
        return self.usage_tracker.get_usage_summary()
    
    def get_performance_scores(self) -> Dict[str, float]:

        scores = {}
        for model_name in self.config.models:
            scores[model_name] = self.usage_tracker.get_performance_score(model_name)
        return scores

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
智能路由器实现
"""

import asyncio
from typing import Dict, Any, List, Optional
from .router_config import RoutingDecision, TaskFeatures
from .usage_tracker import UsageTracker
from .routing_strategies import (
    RoutingStrategy, CostOptimizedStrategy, 
    PerformanceOptimizedStrategy, BalancedStrategy,
    LearningBasedStrategy
)

class SmartRouter:
    """智能路由器"""
    
    def __init__(self):
        self.usage_tracker = UsageTracker()
        self.routing_strategies: Dict[str, RoutingStrategy] = {}
        
        # 初始化内置策略
        self._initialize_builtin_strategies()
    
    def _initialize_builtin_strategies(self):
        """初始化内置策略"""
        self.add_strategy("cost_optimized", CostOptimizedStrategy())
        self.add_strategy("performance_optimized", 
                         PerformanceOptimizedStrategy(self.usage_tracker))
        self.add_strategy("balanced", BalancedStrategy(self.usage_tracker))
        self.add_strategy("learning_based", LearningBasedStrategy())
    
    async def route_with_strategy(
        self,
        strategy_name: str,
        prompt: str,
        context: Dict[str, Any]
    ) -> RoutingDecision:
        """使用指定策略进行路由"""
        if strategy_name not in self.routing_strategies:
            raise ValueError(f"未知的路由策略: {strategy_name}")
        
        strategy = self.routing_strategies[strategy_name]
        return await strategy.route(prompt, context)
    
    async def adaptive_route(
        self,
        prompt: str,
        context: Dict[str, Any],
        task_features: Optional[TaskFeatures] = None
    ) -> RoutingDecision:
        """自适应路由（自动选择最佳策略）"""
        
        # 提取特征
        if task_features:
            features = self._extract_features_from_task_features(task_features)
        else:
            features = self._extract_features_from_prompt(prompt, context)
        
        # 选择最适合的策略
        strategy_name = self._select_strategy_for_features(features)
        
        # 使用选定的策略进行路由
        return await self.route_with_strategy(strategy_name, prompt, context)
    
    def _extract_features_from_prompt(self, prompt: str, context: Dict[str, Any]) -> Dict[str, float]:
        """从提示和上下文中提取特征"""
        features = {}
        
        # 提示长度特征
        features["prompt_length"] = len(prompt) / 1000  # 归一化到0-1
        
        # 复杂度特征
        complexity_indicators = ["分析", "解释", "比较", "设计", "实现", "复杂", "困难"]
        complexity_count = sum(1 for word in complexity_indicators if word in prompt)
        features["complexity"] = complexity_count / len(complexity_indicators)
        
        # 紧急程度特征
        urgency_indicators = ["尽快", "紧急", "马上", "立即", "赶快", "快点"]
        urgency_count = sum(1 for word in urgency_indicators if word in prompt)
        features["urgency"] = urgency_count / len(urgency_indicators)
        
        # 成本敏感度特征
        budget = context.get("budget")
        if budget is not None:
            features["cost_sensitive"] = budget < 5.0
        else:
            features["cost_sensitive"] = False
        
        # 任务类型特征（从上下文获取）
        task_type = context.get("task_type")
        if task_type:
            features["task_type"] = task_type
        
        return features
    
    def _extract_features_from_task_features(self, task_features: TaskFeatures) -> Dict[str, float]:
        """从任务特征中提取路由特征"""
        features = {}
        
        features["complexity"] = task_features.complexity
        features["urgency"] = task_features.urgency
        
        if task_features.budget is not None:
            features["cost_sensitive"] = task_features.budget < 5.0
        else:
            features["cost_sensitive"] = False
        
        if task_features.task_type:
            features["task_type"] = task_features.task_type
        
        return features
    
    def _select_strategy_for_features(self, features: Dict[str, float]) -> str:
        """根据特征选择策略"""
        # 获取特征值
        cost_sensitive = features.get("cost_sensitive", False)
        urgency = features.get("urgency", 0)
        complexity = features.get("complexity", 0)
        task_type = features.get("task_type")
        
        # 决策逻辑
        if cost_sensitive:
            return "cost_optimized"
        elif urgency > 0.7:
            return "performance_optimized"
        elif complexity > 0.6:
            return "balanced"
        elif task_type in ["code_generation", "creative_writing"]:
            # 特定任务类型使用平衡策略
            return "balanced"
        else:
            # 默认策略
            return "balanced"
    
    def add_strategy(self, name: str, strategy: RoutingStrategy):
        """添加自定义策略"""
        if not isinstance(strategy, RoutingStrategy):
            raise TypeError("策略必须是RoutingStrategy的实例")
        
        self.routing_strategies[name] = strategy
    
    def remove_strategy(self, name: str):
        """移除策略"""
        if name in self.routing_strategies:
            del self.routing_strategies[name]
    
    def list_strategies(self) -> List[str]:
        """列出所有可用策略"""
        return list(self.routing_strategies.keys())
    
    def get_strategy_info(self, name: str) -> Dict[str, Any]:
        """获取策略信息"""
        if name not in self.routing_strategies:
            raise ValueError(f"策略不存在: {name}")
        
        strategy = self.routing_strategies[name]
        return {
            "name": name,
            "type": strategy.__class__.__name__,
            "description": strategy.__doc__ or "无描述"
        }
    
    async def evaluate_strategies(
        self,
        prompt: str,
        context: Dict[str, Any],
        strategies: Optional[List[str]] = None
    ) -> Dict[str, Any]:
        """评估不同策略的效果"""
        if strategies is None:
            strategies = self.list_strategies()
        
        results = {}
        for strategy_name in strategies:
            try:
                decision = await self.route_with_strategy(strategy_name, prompt, context)
                results[strategy_name] = {
                    "success": True,
                    "decision": decision,
                    "model": decision.model_name,
                    "confidence": decision.confidence
                }
            except Exception as e:
                results[strategy_name] = {
                    "success": False,
                    "error": str(e),
                    "decision": None
                }
        
        return results
    
    def get_usage_stats(self) -> Dict[str, Any]:
        """获取使用统计"""
        return self.usage_tracker.get_usage_summary()
```
