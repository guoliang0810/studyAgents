## 📝 任务1：理解错误信息优化架构

### 问题答案

1. **用户友好错误信息四大原则**

| 原则 | 描述 | 意义 |
|------|------|------|
| **清晰（Clear）** | 错误消息应该清晰表达问题 | 让开发者立即理解发生了什么问题 |
| **具体（Specific）** | 提供具体的错误位置和原因 | 快速定位问题源头，减少排查时间 |
| **可操作（Actionable）** | 提供具体的修复建议 | 帮助开发者直接解决问题 |
| **友好（Friendly）** | 使用友好的语言 | 改善开发者体验，降低挫败感 |

2. **UserFriendlyError核心属性**

```python
@dataclass
class UserFriendlyError(Exception):
    message: str              # 错误消息
    code: ErrorCode          # 错误代码（唯一标识）
    category: ErrorCategory  # 错误类别
    severity: ErrorSeverity # 严重级别
    suggestions: List[ErrorSuggestion]  # 修复建议
    context: ErrorContext    # 错误上下文
    original_exception: Optional[Exception]  # 原始异常
    details: Dict[str, Any]  # 额外详情
```

每个属性的作用：
- **message**: 直接传达问题本质
- **code**: 机器可读的错误标识
- **category**: 按领域分类错误
- **severity**: 评估错误影响程度
- **suggestions**: 提供具体解决步骤
- **context**: 包含调试所需的信息
- **original_exception**: 保留原始异常信息

## 📝 任务2：扩展UserFriendlyError类

### 完整实现代码

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional, List, Dict, Any
from enum import Enum


class ErrorSeverity(Enum):
    DEBUG = "debug"
    INFO = "info"
    WARNING = "warning"
    ERROR = "error"
    CRITICAL = "critical"


class ErrorCategory(Enum):
    CONFIGURATION = "configuration"
    DEPENDENCY = "dependency"
    VALIDATION = "validation"
    RUNTIME = "runtime"
    UNKNOWN = "unknown"


class ErrorCode(Enum):
    UNKNOWN_ERROR = 9001
    VALIDATION_FAILED = 3001
    DATABASE_ERROR = 5001
    API_ERROR = 6001


@dataclass
class ErrorSuggestion:
    message: str
    action: str
    documentation_url: Optional[str] = None
    priority: int = 1


@dataclass
class ErrorContext:
    timestamp: datetime = field(default_factory=datetime.now)
    module: Optional[str] = None
    function: Optional[str] = None
    line_number: Optional[int] = None
    file_path: Optional[str] = None
    
    def to_dict(self) -> Dict:
        return {
            "timestamp": self.timestamp.isoformat(),
            "module": self.module,
            "function": self.function,
            "line_number": self.line_number,
            "file_path": self.file_path
        }


@dataclass
class ErrorCause:
    """错误原因链"""
    message: str
    timestamp: datetime = field(default_factory=datetime.now)
    parent_error: Optional['ErrorCause'] = None
    
    def to_dict(self) -> Dict:
        result = {
            "message": self.message,
            "timestamp": self.timestamp.isoformat()
        }
        if self.parent_error:
            result["parent_error"] = self.parent_error.to_dict()
        return result


@dataclass
class ErrorImpact:
    """错误影响范围"""
    affected_components: List[str]
    affected_users: Optional[int] = None
    severity_score: float = 0.0
    
    def to_dict(self) -> Dict:
        return {
            "affected_components": self.affected_components,
            "affected_users": self.affected_users,
            "severity_score": self.severity_score
        }


class UserFriendlyError(Exception):
    """用户友好错误基类"""
    
    def __init__(self, 
                 message: str,
                 code: ErrorCode = ErrorCode.UNKNOWN_ERROR,
                 category: ErrorCategory = ErrorCategory.UNKNOWN,
                 severity: ErrorSeverity = ErrorSeverity.ERROR,
                 suggestions: Optional[List[ErrorSuggestion]] = None,
                 context: Optional[ErrorContext] = None,
                 **kwargs):
        super().__init__(message)
        self.message = message
        self.code = code
        self.category = category
        self.severity = severity
        self.suggestions = suggestions or []
        self.context = context or ErrorContext()
        self.details = kwargs
    
    def add_suggestion(self, 
                      message: str, 
                      action: str,
                      documentation_url: Optional[str] = None,
                      priority: int = 1) -> 'UserFriendlyError':
        suggestion = ErrorSuggestion(
            message=message,
            action=action,
            documentation_url=documentation_url,
            priority=priority
        )
        self.suggestions.append(suggestion)
        self.suggestions.sort(key=lambda s: s.priority)
        return self
    
    def to_dict(self) -> Dict:
        return {
            "error": {
                "message": self.message,
                "code": self.code.value,
                "category": self.category.value,
                "severity": self.severity.value,
                "suggestions": [
                    {"message": s.message, "action": s.action}
                    for s in self.suggestions
                ],
                "context": self.context.to_dict()
            }
        }


class ExtendedUserFriendlyError(UserFriendlyError):
    """扩展的用户友好错误"""
    
    def __init__(self, *args, 
                 cause: Optional[ErrorCause] = None,
                 impact: Optional[ErrorImpact] = None,
                 recovery_steps: Optional[List[str]] = None,
                 **kwargs):
        super().__init__(*args, **kwargs)
        self.cause = cause
        self.impact = impact
        self.recovery_steps = recovery_steps or []
    
    def add_recovery_step(self, step: str) -> None:
        self.recovery_steps.append(step)
    
    def get_recovery_plan(self) -> str:
        if not self.recovery_steps:
            return "无自动恢复方案，请手动处理"
        
        plan = ["恢复步骤:"]
        for i, step in enumerate(self.recovery_steps, 1):
            plan.append(f"  {i}. {step}")
        return "\n".join(plan)


class JSONSchemaError(ExtendedUserFriendlyError):
    """JSON Schema验证错误"""
    
    def __init__(self, message: str, schema_path: str, 
                 instance_value: Any,
                 validation_errors: List[Dict], **kwargs):
        super().__init__(
            message=message,
            code=ErrorCode.VALIDATION_FAILED,
            category=ErrorCategory.VALIDATION,
            **kwargs
        )
        self.schema_path = schema_path
        self.instance_value = instance_value
        self.validation_errors = validation_errors
        
        self.add_suggestion(
            message=f"JSON Schema验证失败: {schema_path}",
            action="检查JSON数据是否符合schema定义",
            priority=1
        )


class APIError(ExtendedUserFriendlyError):
    """API错误"""
    
    def __init__(self, message: str, http_status_code: int,
                 endpoint: str, method: str,
                 response_data: Optional[Dict] = None, **kwargs):
        super().__init__(
            message=message,
            code=ErrorCode.API_ERROR,
            category=ErrorCategory.RUNTIME,
            **kwargs
        )
        self.http_status_code = http_status_code
        self.endpoint = endpoint
        self.method = method
        self.response_data = response_data or {}
        
        # 根据HTTP状态码添加建议
        if http_status_code == 401:
            self.add_suggestion("认证失败", "请检查API密钥是否正确")
        elif http_status_code == 403:
            self.add_suggestion("权限不足", "请检查是否有访问该资源的权限")
        elif http_status_code == 404:
            self.add_suggestion("资源不存在", "请检查API端点和参数是否正确")
        elif http_status_code >= 500:
            self.add_suggestion("服务器错误", "请稍后重试或联系API提供商")
```

## 📝 任务3：实现自定义错误转换规则

### 完整实现代码

```python
import re
from typing import Dict, Type, Callable, Optional, Any
from dataclasses import dataclass


class ErrorCategory(Enum):
    CONFIGURATION = "configuration"
    DEPENDENCY = "dependency"
    VALIDATION = "validation"
    NETWORK = "network"
    RUNTIME = "runtime"
    DATABASE = "database"


class ErrorSuggestion:
    def __init__(self, message: str, action: str, priority: int = 1):
        self.message = message
        self.action = action
        self.priority = priority


class UserFriendlyError(Exception):
    def __init__(self, message: str, 
                 category: ErrorCategory = ErrorCategory.UNKNOWN,
                 suggestions: Optional[list] = None):
        super().__init__(message)
        self.message = message
        self.category = category
        self.suggestions = suggestions or []


class ErrorTransformer:
    """自定义错误转换器"""
    
    def __init__(self):
        self.custom_transformers: Dict[Type[Exception], Callable] = {}
        
        # 注册自定义转换器
        self._register_default_transformers()
    
    def _register_default_transformers(self):
        """注册默认转换器"""
        # SQLAlchemy错误
        try:
            import sqlalchemy
            self.register_transformer(
                sqlalchemy.exc.SQLAlchemyError,
                self._transform_sqlalchemy_error
            )
        except ImportError:
            pass
        
        # Pydantic错误
        try:
            import pydantic
            self.register_transformer(
                pydantic.error_wrappers.ValidationError,
                self._transform_pydantic_error
            )
        except ImportError:
            pass
        
        # requests错误
        try:
            import requests
            self.register_transformer(
                requests.exceptions.RequestException,
                self._transform_request_error
            )
        except ImportError:
            pass
        
        # Redis错误
        try:
            import redis
            self.register_transformer(
                redis.exceptions.RedisError,
                self._transform_redis_error
            )
        except ImportError:
            pass
    
    def register_transformer(self, 
                           exception_type: Type[Exception],
                           transformer: Callable[[Exception], UserFriendlyError]):
        self.custom_transformers[exception_type] = transformer
    
    def transform(self, exception: Exception) -> UserFriendlyError:
        # 检查自定义转换器
        for exc_type, transformer in self.custom_transformers.items():
            if isinstance(exception, exc_type):
                return transformer(exception)
        
        # 默认转换
        return UserFriendlyError(
            message=str(exception),
            category=ErrorCategory.UNKNOWN
        )
    
    def _transform_sqlalchemy_error(self, exc) -> UserFriendlyError:
        """转换SQLAlchemy错误"""
        message = str(exc.orig) if hasattr(exc, 'orig') else str(exc)
        
        # 判断错误类型
        try:
            import sqlalchemy
            if isinstance(exc, sqlalchemy.exc.OperationalError):
                return UserFriendlyError(
                    message=f"数据库操作失败: {message}",
                    category=ErrorCategory.DATABASE,
                    suggestions=[
                        ErrorSuggestion(
                            message="数据库连接失败",
                            action="检查数据库服务是否运行",
                            priority=1
                        )
                    ]
                )
            
            if isinstance(exc, sqlalchemy.exc.IntegrityError):
                return UserFriendlyError(
                    message=f"数据完整性错误: {message}",
                    category=ErrorCategory.DATABASE,
                    suggestions=[
                        ErrorSuggestion(
                            message="数据重复或外键约束",
                            action="检查数据是否重复",
                            priority=1
                        )
                    ]
                )
        except ImportError:
            pass
        
        return UserFriendlyError(
            message=f"数据库错误: {message}",
            category=ErrorCategory.DATABASE
        )
    
    def _transform_pydantic_error(self, exc) -> UserFriendlyError:
        """转换Pydantic验证错误"""
        errors = exc.errors()
        
        field_errors = []
        for error in errors:
            loc = ".".join(str(l) for l in error["loc"])
            msg = error["msg"]
            field_errors.append(f"{loc}: {msg}")
        
        message = "; ".join(field_errors)
        
        return UserFriendlyError(
            message=f"Pydantic验证失败: {message}",
            category=ErrorCategory.VALIDATION,
            suggestions=[
                ErrorSuggestion(
                    message="数据模型验证失败",
                    action="检查输入数据是否符合schema定义",
                    priority=1
                )
            ]
        )
    
    def _transform_request_error(self, exc) -> UserFriendlyError:
        """转换requests错误"""
        message = str(exc)
        
        try:
            import requests
            if isinstance(exc, requests.exceptions.Timeout):
                url = exc.request.url if exc.request else "unknown"
                return UserFriendlyError(
                    message=f"请求超时: {url}",
                    category=ErrorCategory.NETWORK,
                    suggestions=[
                        ErrorSuggestion(
                            message="API请求超时",
                            action="增加超时时间或检查网络",
                            priority=1
                        )
                    ]
                )
            
            if isinstance(exc, requests.exceptions.ConnectionError):
                return UserFriendlyError(
                    message=f"连接失败: {message}",
                    category=ErrorCategory.NETWORK,
                    suggestions=[
                        ErrorSuggestion(
                            message="无法连接到服务器",
                            action="检查网络和服务器状态",
                            priority=1
                        )
                    ]
                )
            
            # HTTP错误响应
            if hasattr(exc, 'response') and exc.response:
                status = exc.response.status_code
                return UserFriendlyError(
                    message=f"HTTP错误 {status}: {message}",
                    category=ErrorCategory.NETWORK,
                    suggestions=[
                        ErrorSuggestion(
                            message=f"HTTP {status} 错误",
                            action=f"检查请求URL和参数",
                            priority=1
                        )
                    ]
                )
        except ImportError:
            pass
        
        return UserFriendlyError(
            message=f"请求错误: {message}",
            category=ErrorCategory.NETWORK
        )
    
    def _transform_redis_error(self, exc) -> UserFriendlyError:
        """转换Redis错误"""
        message = str(exc)
        
        if "Connection" in message or "ConnectionRefused" in message:
            return UserFriendlyError(
                message=f"Redis连接失败: {message}",
                category=ErrorCategory.DATABASE,
                suggestions=[
                    ErrorSuggestion(
                        message="Redis服务不可用",
                        action="检查Redis服务是否运行",
                        priority=1
                    )
                ]
            )
        
        if "Timeout" in message:
            return UserFriendlyError(
                message=f"Redis超时: {message}",
                category=ErrorCategory.DATABASE,
                suggestions=[
                    ErrorSuggestion(
                        message="Redis操作超时",
                        action="检查Redis负载和网络",
                        priority=1
                    )
                ]
            )
        
        return UserFriendlyError(
            message=f"Redis错误: {message}",
            category=ErrorCategory.DATABASE
        )
```

## 📝 任务4：实现错误监控面板

### 完整实现代码

```python
import time
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from typing import Dict, List, Optional, Callable, Any
from collections import defaultdict
import json


class ErrorSeverity(Enum):
    DEBUG = "debug"
    INFO = "info"
    WARNING = "warning"
    ERROR = "error"
    CRITICAL = "critical"


class ErrorCategory(Enum):
    UNKNOWN = "unknown"


class ErrorCode(Enum):
    UNKNOWN = 0


class UserFriendlyError(Exception):
    def __init__(self, message: str, code=ErrorCode.UNKNOWN, 
                 category=ErrorCategory.UNKNOWN, severity=ErrorSeverity.ERROR):
        super().__init__(message)
        self.message = message
        self.code = code
        self.category = category
        self.severity = severity
        self.timestamp = datetime.now()


class ErrorHandlingMiddleware:
    """错误处理中间件"""
    
    def __init__(self):
        self.stats = {
            "total_errors": 0,
            "errors_by_category": defaultdict(int),
            "errors_by_severity": defaultdict(int)
        }
        self.error_history: List[Dict] = []
        self.max_history_size = 1000
    
    def record_error(self, error: UserFriendlyError):
        self.stats["total_errors"] += 1
        self.stats["errors_by_category"][error.category.value] += 1
        self.stats["errors_by_severity"][error.severity.value] += 1
        
        self.error_history.append({
            "error": error,
            "timestamp": time.time()
        })
        
        if len(self.error_history) > self.max_history_size:
            self.error_history.pop(0)
    
    def get_stats(self) -> Dict:
        return dict(self.stats)


class AlertRule:
    """告警规则"""
    
    def __init__(self, name: str, 
                 condition: Callable[[UserFriendlyError], bool],
                 severity: ErrorSeverity,
                 message_template: str):
        self.name = name
        self.condition = condition
        self.severity = severity
        self.message_template = message_template
    
    def matches(self, error: UserFriendlyError) -> bool:
        return self.condition(error)


@dataclass
class Alert:
    """告警"""
    message: str
    severity: ErrorSeverity
    errors: List[UserFriendlyError]
    rule: Optional[str] = None
    timestamp: datetime = field(default_factory=datetime.now)


class ErrorMonitoringMiddleware(ErrorHandlingMiddleware):
    """错误监控中间件"""
    
    def __init__(self, *args,
                 alert_threshold: int = 10,
                 alert_window_seconds: int = 60,
                 **kwargs):
        super().__init__(*args, **kwargs)
        
        self.alert_threshold = alert_threshold
        self.alert_window_seconds = alert_window_seconds
        self.alert_callbacks: List[Callable] = []
        self.alert_rules: List[AlertRule] = []
        
        # 时间序列数据
        self.error_timeline: List[Dict] = []
    
    def add_alert_rule(self, rule: AlertRule) -> None:
        self.alert_rules.append(rule)
    
    def register_alert_callback(self, callback: Callable) -> None:
        self.alert_callbacks.append(callback)
    
    def record_error(self, error: UserFriendlyError):
        super().record_error(error)
        
        # 记录时间序列
        self.error_timeline.append({
            "error": error,
            "timestamp": time.time()
        })
        
        # 检查告警
        self._check_alerts(error)
    
    def _check_alerts(self, error: UserFriendlyError):
        """检查是否触发告警"""
        # 检查错误频率
        recent_count = self._get_recent_error_count()
        
        if recent_count >= self.alert_threshold:
            alert = Alert(
                message=f"错误频率超过阈值: {recent_count}个/{self.alert_window_seconds}秒",
                severity=ErrorSeverity.WARNING,
                errors=[error]
            )
            
            for callback in self.alert_callbacks:
                try:
                    callback(alert)
                except Exception as e:
                    print(f"告警回调失败: {e}")
        
        # 检查告警规则
        for rule in self.alert_rules:
            if rule.matches(error):
                alert = Alert(
                    message=rule.message_template.format(
                        code=error.code.value,
                        message=error.message
                    ),
                    severity=rule.severity,
                    errors=[error],
                    rule=rule.name
                )
                
                for callback in self.alert_callbacks:
                    callback(alert)
    
    def _get_recent_error_count(self) -> int:
        """获取最近一段时间的错误数量"""
        cutoff_time = time.time() - self.alert_window_seconds
        
        count = sum(1 for e in self.error_timeline 
                   if e["timestamp"] >= cutoff_time)
        
        return count
    
    def get_error_trends(self, 
                        time_window: str = "1h",
                        interval: int = 300) -> Dict:
        """获取错误趋势"""
        time_windows = {"1h": 3600, "24h": 86400, "7d": 604800}
        
        seconds = time_windows.get(time_window, 3600)
        
        trends = []
        now = time.time()
        
        for i in range(0, seconds, interval):
            start_time = now - seconds + i
            end_time = start_time + interval
            
            count = sum(1 for e in self.error_timeline 
                       if start_time <= e["timestamp"] < end_time)
            
            trends.append({
                "timestamp": start_time,
                "count": count
            })
        
        return {
            "time_window": time_window,
            "interval": interval,
            "trends": trends,
            "total": len(self.error_timeline)
        }
    
    def export_errors(self, 
                    format: str = "json",
                    filters: Optional[Dict] = None) -> str:
        """导出错误数据"""
        errors = [e["error"] for e in self.error_history]
        
        # 应用过滤器
        if filters:
            if "category" in filters:
                errors = [e for e in errors 
                         if e.category.value == filters["category"]]
            if "severity" in filters:
                errors = [e for e in errors 
                         if e.severity.value == filters["severity"]]
        
        if format == "json":
            return json.dumps([
                {
                    "message": e.message,
                    "code": e.code.value,
                    "category": e.category.value,
                    "severity": e.severity.value,
                    "timestamp": e.timestamp.isoformat()
                }
                for e in errors
            ], indent=2)
        
        raise ValueError(f"不支持的格式: {format}")
```

```python
class CachedErrorTransformer:
    """带缓存的错误转换器"""
    
    def __init__(self, cache_size: int = 1000):
        self.cache: Dict[str, UserFriendlyError] = {}
        self.cache_size = cache_size
        self.cache_hits = 0
        self.cache_misses = 0
    
    def transform(self, exception: Exception) -> UserFriendlyError:
        # 生成缓存键
        cache_key = self._generate_cache_key(exception)
        
        # 检查缓存
        if cache_key in self.cache:
            self.cache_hits += 1
            return self.cache[cache_key]
        
        self.cache_misses += 1
        
        # 转换错误
        error = self._do_transform(exception)
        
        # 更新缓存
        if len(self.cache) >= self.cache_size:
            # 简单的FIFO淘汰
            first_key = next(iter(self.cache))
            del self.cache[first_key]
        
        self.cache[cache_key] = error
        
        return error
    
    def _generate_cache_key(self, exception: Exception) -> str:
        return f"{type(exception).__name__}:{str(exception)[:100]}"
```
