```python
@dataclass
class ErrorCause:
    """错误原因链"""
    message: str
    timestamp: datetime
    parent_error: Optional['ErrorCause'] = None

@dataclass  
class ErrorImpact:
    """错误影响范围"""
    affected_components: List[str]
    affected_users: Optional[int] = None
    severity_score: float = 0.0

class ExtendedUserFriendlyError(UserFriendlyError):
    """扩展的用户友好错误"""
    
    def __init__(self, *args, 
                 cause: Optional[ErrorCause] = None,
                 impact: Optional[ErrorImpact] = None,
                 recovery_steps: Optional[List[str]] = None,
                 **kwargs):
        super().__init__(*args, **kwargs)
        self.cause = cause
        self.impact = impact
        self.recovery_steps = recovery_steps or []
    
    def add_recovery_step(self, step: str) -> None:
        """添加恢复步骤"""
        self.recovery_steps.append(step)
    
    def get_recovery_plan(self) -> str:
        """获取恢复计划"""
        if not self.recovery_steps:
            return "无自动恢复方案，请手动处理"
        
        plan = ["恢复步骤:"]
        for i, step in enumerate(self.recovery_steps, 1):
            plan.append(f"  {i}. {step}")
        return "\n".join(plan)


class JSONSchemaError(ExtendedUserFriendlyError):
    """JSON Schema验证错误"""
    
    def __init__(self, message, schema_path, instance_value, 
                 validation_errors, **kwargs):
        super().__init__(
            message=message,
            code=ErrorCode.VALIDATION_FAILED,
            category=ErrorCategory.VALIDATION,
            **kwargs
        )
        self.schema_path = schema_path
        self.instance_value = instance_value
        self.validation_errors = validation_errors
        
        # 添加验证错误建议
        self.add_suggestion(
            message=f"JSON Schema验证失败: {schema_path}",
            action="检查JSON数据是否符合schema定义",
            priority=1
        )


class APIError(ExtendedUserFriendlyError):
    """API错误"""
    
    def __init__(self, message, http_status_code, 
                 endpoint, method, response_data, **kwargs):
        super().__init__(
            message=message,
            code=ErrorCode.RUNTIME_ERROR,
            category=ErrorCategory.RUNTIME,
            **kwargs
        )
        self.http_status_code = http_status_code
        self.endpoint = endpoint
        self.method = method
        self.response_data = response_data
        
        # 根据HTTP状态码添加建议
        if http_status_code == 401:
            self.add_suggestion("认证失败", "请检查API密钥是否正确")
        elif http_status_code == 403:
            self.add_suggestion("权限不足", "请检查是否有访问该资源的权限")
        elif http_status_code == 404:
            self.add_suggestion("资源不存在", "请检查API端点和参数是否正确")
        elif http_status_code >= 500:
            self.add_suggestion("服务器错误", "请稍后重试或联系API提供商")
```

```python
class CustomErrorTransformer(ErrorTransformer):
    """自定义错误转换器"""
    
    def __init__(self):
        super().__init__()
        
        # 注册自定义转换器
        self.register_transformer(
            sqlalchemy.exc.SQLAlchemyError,
            self._transform_sqlalchemy_error
        )
        self.register_transformer(
            pydantic.error_wrappers.ValidationError,
            self._transform_pydantic_error
        )
        self.register_transformer(
            requests.exceptions.RequestException,
            self._transform_request_error
        )
        self.register_transformer(
            redis.exceptions.RedisError,
            self._transform_redis_error
        )
    
    def _transform_sqlalchemy_error(self, exc) -> UserFriendlyError:
        """转换SQLAlchemy错误"""
        # 提取错误信息
        message = str(exc.orig) if exc.orig else str(exc)
        
        # 判断错误类型
        if isinstance(exc, sqlalchemy.exc.OperationalError):
            # 连接错误
            return DependencyError(
                message=f"数据库操作失败: {message}",
                suggestions=[
                    ErrorSuggestion(
                        message="数据库连接失败",
                        action="检查数据库服务是否运行，网络是否正常",
                        priority=1
                    )
                ]
            )
        
        return ValidationError(
            message=f"数据库错误: {message}"
        )
    
    def _transform_pydantic_error(self, exc) -> UserFriendlyError:
        """转换Pydantic验证错误"""
        errors = exc.errors()
        
        # 提取错误信息
        field_errors = []
        for error in errors:
            field = ".".join(str(loc) for loc in error["loc"])
            msg = error["msg"]
            field_errors.append(f"{field}: {msg}")
        
        message = "; ".join(field_errors)
        
        return ValidationError(
            message=f"Pydantic验证失败: {message}",
            suggestions=[
                ErrorSuggestion(
                    message="数据模型验证失败",
                    action="检查输入数据是否符合Pydantic模型定义",
                    priority=1
                )
            ]
        )
    
    def _transform_request_error(self, exc) -> UserFriendlyError:
        """转换requests错误"""
        if isinstance(exc, requests.exceptions.Timeout):
            return NetworkError(
                message="请求超时",
                url=str(exc.request.url) if exc.request else None,
                suggestions=[
                    ErrorSuggestion(
                        message="API请求超时",
                        action="增加超时时间或检查网络状况",
                        priority=1
                    )
                ]
            )
        
        # 其他请求错误
        response = exc.response if hasattr(exc, 'response') else None
        status_code = response.status_code if response else None
        
        return NetworkError(
            message=f"HTTP请求失败: {str(exc)}",
            url=str(exc.request.url) if exc.request else None,
            status_code=status_code
        )
    
    def _transform_redis_error(self, exc) -> UserFriendlyError:
        """转换Redis错误"""
        message = str(exc)
        
        if "Connection" in message:
            return DependencyError(
                message=f"Redis连接失败: {message}",
                suggestions=[
                    ErrorSuggestion(
                        message="Redis服务不可用",
                        action="检查Redis服务是否运行，配置是否正确",
                        priority=1
                    )
                ]
            )
        
        return RuntimeError(message)
```

```python
class ErrorMonitoringMiddleware(ErrorHandlingMiddleware):
    """错误监控中间件"""
    
    def __init__(self, *args, 
                 alert_threshold: int = 10,
                 alert_window_seconds: int = 60,
                 **kwargs):
        super().__init__(*args, **kwargs)
        
        self.alert_threshold = alert_threshold
        self.alert_window_seconds = alert_window_seconds
        self.alert_callbacks: List[Callable] = []
        
        # 错误时间序列
        self.error_timeline: List[Dict] = []
        
        # 告警规则
        self.alert_rules: List[AlertRule] = []
    
    def add_alert_rule(self, rule: 'AlertRule') -> None:
        """添加告警规则"""
        self.alert_rules.append(rule)
    
    def register_alert_callback(self, callback: Callable) -> None:
        """注册告警回调"""
        self.alert_callbacks.append(callback)
    
    def _check_alerts(self, error: UserFriendlyError) -> None:
        """检查是否触发告警"""
        # 检查错误频率
        recent_errors = self._get_recent_errors(self.alert_window_seconds)
        error_count = len(recent_errors)
        
        if error_count >= self.alert_threshold:
            # 触发告警
            alert = Alert(
                message=f"错误频率超过阈值: {error_count}个/{self.alert_window_seconds}秒",
                severity=ErrorSeverity.WARNING,
                errors=recent_errors
            )
            
            for callback in self.alert_callbacks:
                try:
                    callback(alert)
                except Exception as e:
                    logger.error(f"告警回调失败: {e}")
        
        # 检查告警规则
        for rule in self.alert_rules:
            if rule.matches(error):
                alert = rule.create_alert(error)
                for callback in self.alert_callbacks:
                    callback(alert)
    
    def _get_recent_errors(self, seconds: int) -> List[UserFriendlyError]:
        """获取最近一段时间的错误"""
        import time
        cutoff_time = time.time() - seconds
        
        recent = []
        for error_info in self.error_timeline:
            if error_info["timestamp"] >= cutoff_time:
                recent.append(error_info["error"])
        
        return recent
    
    def get_error_trends(self, 
                        time_window: str = "1h",
                        interval: str = "5m") -> Dict:
        """获取错误趋势"""
        # 实现时间窗口解析
        time_windows = {
            "1h": 3600,
            "24h": 86400,
            "7d": 604800
        }
        
        seconds = time_windows.get(time_window, 3600)
        
        # 按时间间隔分组统计
        trends = []
        import time
        now = time.time()
        
        for i in range(0, seconds, 300):  # 5分钟间隔
            start_time = now - seconds + i
            end_time = start_time + 300
            
            count = sum(1 for e in self.error_timeline 
                       if start_time <= e["timestamp"] < end_time)
            
            trends.append({
                "timestamp": start_time,
                "count": count
            })
        
        return {
            "time_window": time_window,
            "interval": interval,
            "trends": trends,
            "total": len(self.error_timeline)
        }
    
    def export_errors(self, 
                    format: str = "json",
                    filters: Optional[Dict] = None) -> str:
        """导出错误数据"""
        errors = self.error_history
        
        # 应用过滤器
        if filters:
            if "category" in filters:
                errors = [e for e in errors 
                         if e.category.value == filters["category"]]
            if "severity" in filters:
                errors = [e for e in errors 
                         if e.severity.value == filters["severity"]]
        
        if format == "json":
            return json.dumps([e.to_dict() for e in errors], indent=2)
        elif format == "csv":
            # CSV导出
            import csv
            import io
            
            output = io.StringIO()
            writer = csv.writer(output)
            
            # 写入表头
            writer.writerow(["timestamp", "category", "severity", 
                           "code", "message", "module"])
            
            # 写入数据
            for error in errors:
                writer.writerow([
                    error.context.timestamp.isoformat(),
                    error.category.value,
                    error.severity.value,
                    error.code.value,
                    error.message,
                    error.context.module
                ])
            
            return output.getvalue()
        
        raise ValueError(f"不支持的导出格式: {format}")


@dataclass
class AlertRule:
    """告警规则"""
    name: str
    condition: Callable[[UserFriendlyError], bool]
    severity: ErrorSeverity
    message_template: str
    
    def matches(self, error: UserFriendlyError) -> bool:
        """检查错误是否匹配规则"""
        return self.condition(error)
    
    def create_alert(self, error: UserFriendlyError) -> 'Alert':
        """创建告警"""
        message = self.message_template.format(
            error_code=error.code.value,
            error_message=error.message,
            category=error.category.value
        )
        
        return Alert(
            message=message,
            severity=self.severity,
            errors=[error],
            rule=self.name
        )


@dataclass
class Alert:
    """告警"""
    message: str
    severity: ErrorSeverity
    errors: List[UserFriendlyError]
    rule: Optional[str] = None
    timestamp: datetime = field(default_factory=datetime.now)
```
