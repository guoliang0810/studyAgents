## 任务1：理解检查点提供者架构

### 1.1 为什么需要抽象Provider？

**Provider抽象的优点：**
1. **解耦存储逻辑和业务逻辑**：业务层不需要关心数据存储在哪里
2. **支持多种存储后端**：可以轻松切换内存、文件、数据库、Redis等
3. **便于测试**：可以使用内存Provider进行单元测试
4. **符合开闭原则**：添加新存储不需要修改已有代码
5. **统一接口**：保证不同存储后端的行为一致性

### 1.2 Provider接口核心方法

| 方法 | 作用 | 重要性 |
|------|------|--------|
| `save(checkpoint)` | 保存检查点数据 | 必须 |
| `load(checkpoint_id)` | 加载检查点数据 | 必须 |
| `delete(checkpoint_id)` | 删除检查点 | 必须 |
| `list()` | 列出所有检查点ID | 必须 |
| `exists(checkpoint_id)` | 检查是否存在 | 可选优化 |

### 1.3 抽象Provider的缺点

1. **性能开销**：抽象层可能带来额外开销
2. **功能限制**：通用接口可能无法利用特定存储的特性
3. **复杂度**：需要维护接口兼容性

### 1.4 思考题答案

**Q: 如果要添加新的存储后端（如Redis），需要修改哪些代码？**

A:
1. 创建新类继承`CheckpointProvider`
2. 实现5个抽象方法
3. 在使用处替换Provider实例
4. 不需要修改`CheckpointManager`或其他业务逻辑

**Q: 如何保证不同Provider的行为一致性？**

A:
1. 严格遵循接口定义
2. 编写统一的测试套件
3. 使用集成测试验证
4. 文档化每个Provider的特性

## 任务2：对象存储检查点提供者

### 2.1 完整实现

```python
import os
import json
import hashlib
from pathlib import Path
from typing import List, Optional, Dict, Any
from dataclasses import dataclass
import asyncio

@dataclass
class ObjectStorageConfig:
    """对象存储配置"""
    bucket: str
    endpoint: str
    region: str
    access_key: str
    secret_key: str
    prefix: str = "checkpoints/"


class ObjectStorageCheckpointProvider(CheckpointProvider):
    """对象存储检查点提供者"""
    
    CHUNK_SIZE = 5 * 1024 * 1024  # 5MB分块
    
    def __init__(self, config: ObjectStorageConfig, local_cache: str = None):
        self.config = config
        self.local_cache = Path(local_cache or ".object_cache")
        self.local_cache.mkdir(parents=True, exist_ok=True)
    
    def _get_object_key(self, checkpoint_id: str) -> str:
        """获取对象存储键"""
        return f"{self.config.prefix}{checkpoint_id}.json"
    
    def _get_local_path(self, checkpoint_id: str) -> Path:
        """获取本地缓存路径"""
        return self.local_cache / f"{checkpoint_id}.json"
    
    def _compute_checksum(self, data: bytes) -> str:
        """计算MD5校验和"""
        return hashlib.md5(data).hexdigest()
    
    async def save(self, checkpoint: Checkpoint) -> None:
        """保存检查点到对象存储"""
        checkpoint_id = checkpoint.metadata.checkpoint_id
        
        # 序列化
        data = json.dumps(checkpoint.to_dict(), ensure_ascii=False)
        data_bytes = data.encode('utf-8')
        
        # 计算校验和
        checksum = self._compute_checksum(data_bytes)
        checkpoint.metadata.checksum = checksum
        checkpoint.metadata.size_bytes = len(data_bytes)
        
        # 分块上传（模拟）
        await self._upload_multipart(checkpoint_id, data_bytes)
        
        # 更新元数据
        checkpoint.metadata.status = CheckpointStatus.AVAILABLE
        
        logger.info(f"检查点已保存到对象存储: {checkpoint_id}")
    
    async def _upload_multipart(self, checkpoint_id: str, data: bytes) -> None:
        """分块上传"""
        chunks = []
        for i in range(0, len(data), self.CHUNK_SIZE):
            chunk = data[i:i + self.CHUNK_SIZE]
            chunks.append(chunk)
        
        local_path = self._get_local_path(checkpoint_id)
        
        # 写入分块信息
        manifest = {
            "checkpoint_id": checkpoint_id,
            "chunks": [],
            "total_size": len(data),
            "chunk_size": self.CHUNK_SIZE
        }
        
        for i, chunk in enumerate(chunks):
            chunk_path = local_path.parent / f"{checkpoint_id}.chunk{i}"
            chunk_path.write_bytes(chunk)
            manifest["chunks"].append({
                "index": i,
                "size": len(chunk),
                "checksum": self._compute_checksum(chunk)
            })
        
        # 写入清单
        manifest_path = local_path
        manifest_path.write_text(json.dumps(manifest), encoding='utf-8')
    
    async def load(self, checkpoint_id: str) -> Optional[Checkpoint]:
        """从对象存储加载检查点"""
        data = await self._download_multipart(checkpoint_id)
        
        if not data:
            return None
        
        checkpoint_dict = json.loads(data)
        checkpoint = Checkpoint.from_dict(checkpoint_dict)
        
        logger.info(f"检查点已从对象存储加载: {checkpoint_id}")
        return checkpoint
    
    async def _download_multipart(self, checkpoint_id: str) -> Optional[bytes]:
        """多部分下载"""
        local_path = self._get_local_path(checkpoint_id)
        
        if not local_path.exists():
            return None
        
        manifest = json.loads(local_path.read_text(encoding='utf-8'))
        
        chunks = []
        for chunk_info in manifest["chunks"]:
            chunk_path = local_path.parent / f"{checkpoint_id}.chunk{chunk_info['index']}"
            if chunk_path.exists():
                chunks.append(chunk_path.read_bytes())
        
        if not chunks:
            return None
        
        return b"".join(chunks)
    
    async def delete(self, checkpoint_id: str) -> bool:
        """从对象存储删除检查点"""
        local_path = self._get_local_path(checkpoint_id)
        
        if not local_path.exists():
            return False
        
        # 删除所有分块
        for chunk_file in local_path.parent.glob(f"{checkpoint_id}.chunk*"):
            chunk_file.unlink()
        
        # 删除清单
        local_path.unlink()
        
        logger.info(f"检查点已从对象存储删除: {checkpoint_id}")
        return True
    
    async def list(self) -> List[str]:
        """列出所有检查点"""
        checkpoint_ids = []
        
        for manifest_file in self.local_cache.glob("*.json"):
            checkpoint_id = manifest_file.stem
            checkpoint_ids.append(checkpoint_id)
        
        return sorted(checkpoint_ids)
    
    async def exists(self, checkpoint_id: str) -> bool:
        """检查检查点是否存在"""
        local_path = self._get_local_path(checkpoint_id)
        return local_path.exists()
```

## 任务3：加密检查点提供者

### 3.1 完整实现

```python
import os
import json
import hashlib
from pathlib import Path
from typing import List, Optional, Dict, Any
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2
import base64


class EncryptedCheckpointProvider(CheckpointProvider):
    """加密检查点提供者"""
    
    def __init__(self, provider: CheckpointProvider, password: str):
        self.provider = provider
        self.fernet = self._derive_key(password)
        self.key_version = 1
        self._key_history: Dict[int, bytes] = {1: self.fernet}
    
    def _derive_key(self, password: str) -> Fernet:
        """从密码派生密钥"""
        salt = b'deercflow_checkpoint_salt'
        kdf = PBKDF2(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000,
        )
        key = base64.urlsafe_b64encode(kdf.derive(password.encode()))
        return Fernet(key)
    
    async def save(self, checkpoint: Checkpoint) -> None:
        """加密并保存检查点"""
        data = json.dumps(checkpoint.to_dict(), ensure_ascii=False)
        data_bytes = data.encode('utf-8')
        
        # 加密
        encrypted_data = self.fernet.encrypt(data_bytes)
        
        # 创建加密检查点
        encrypted_checkpoint = Checkpoint(
            metadata=checkpoint.metadata,
            state={
                "encrypted_data": base64.b64encode(encrypted_data).decode(),
                "key_version": self.key_version,
                "original_size": len(data_bytes)
            }
        )
        
        await self.provider.save(encrypted_checkpoint)
        
        logger.info(f"加密检查点已保存: {checkpoint.metadata.checkpoint_id}")
    
    async def load(self, checkpoint_id: str) -> Optional[Checkpoint]:
        """加载并解密检查点"""
        encrypted_checkpoint = await self.provider.load(checkpoint_id)
        
        if not encrypted_checkpoint:
            return None
        
        encrypted_data_b64 = encrypted_checkpoint.state.get("encrypted_data")
        key_version = encrypted_checkpoint.state.get("key_version", 1)
        
        if not encrypted_data_b64:
            return None
        
        fernet = self._key_history.get(key_version)
        if not fernet:
            raise ValueError(f"未知密钥版本: {key_version}")
        
        encrypted_data = base64.b64decode(encrypted_data_b64)
        decrypted_data = fernet.decrypt(encrypted_data)
        
        checkpoint_dict = json.loads(decrypted_data.decode('utf-8'))
        checkpoint = Checkpoint.from_dict(checkpoint_dict)
        
        return checkpoint
    
    async def rotate_key(self, new_password: str) -> None:
        """轮换密钥"""
        new_fernet = self._derive_key(new_password)
        new_version = self.key_version + 1
        
        self._key_history[self.key_version] = self.fernet
        
        self.fernet = new_fernet
        self.key_version = new_version
        
        logger.info(f"密钥已轮换到版本: {new_version}")
    
    async def secure_delete(self, checkpoint_id: str) -> bool:
        """安全删除（覆写）"""
        checkpoint = await self.provider.load(checkpoint_id)
        
        if not checkpoint:
            return False
        
        if "encrypted_data" in checkpoint.state:
            original_size = checkpoint.state.get("original_size", 1024)
            random_data = os.urandom(original_size)
            
            for _ in range(3):
                checkpoint.state["encrypted_data"] = base64.b64encode(random_data).decode()
                await self.provider.save(checkpoint)
        
        return await self.provider.delete(checkpoint_id)
    
    async def delete(self, checkpoint_id: str) -> bool:
        return await self.provider.delete(checkpoint_id)
    
    async def list(self) -> List[str]:
        return await self.provider.list()
    
    async def exists(self, checkpoint_id: str) -> bool:
        return await self.provider.exists(checkpoint_id)
```

## 任务4：缓存检查点提供者

### 4.1 完整实现

```python
from collections import OrderedDict
from typing import Tuple
from pathlib import Path
import json

class CacheStats:
    """缓存统计"""
    def __init__(self):
        self.hits = 0
        self.misses = 0
        self.evictions = 0
    
    @property
    def hit_rate(self) -> float:
        total = self.hits + self.misses
        return self.hits / total if total > 0 else 0.0


class CachedCheckpointProvider(CheckpointProvider):
    """缓存检查点提供者"""
    
    def __init__(
        self,
        backend: CheckpointProvider,
        l1_size: int = 10,
        l2_size: int = 100
    ):
        self.backend = backend
        self.l1_cache: OrderedDict[str, Checkpoint] = OrderedDict()
        self.l1_size = l1_size
        self.l2_cache_dir = Path(".l2_cache")
        self.l2_cache_dir.mkdir(exist_ok=True)
        self.l2_size = l2_size
        self.stats = CacheStats()
    
    def _get_l2_path(self, checkpoint_id: str) -> Path:
        return self.l2_cache_dir / f"{checkpoint_id}.json"
    
    def _get_from_l1(self, checkpoint_id: str) -> Optional[Checkpoint]:
        if checkpoint_id in self.l1_cache:
            self.l1_cache.move_to_end(checkpoint_id)
            self.stats.hits += 1
            return self.l1_cache[checkpoint_id]
        
        self.stats.misses += 1
        return None
    
    def _put_to_l1(self, checkpoint: Checkpoint) -> None:
        checkpoint_id = checkpoint.metadata.checkpoint_id
        
        if checkpoint_id in self.l1_cache:
            del self.l1_cache[checkpoint_id]
        
        self.l1_cache[checkpoint_id] = checkpoint
        
        while len(self.l1_cache) > self.l1_size:
            oldest = next(iter(self.l1_cache))
            del self.l1_cache[oldest]
            self.stats.evictions += 1
    
    def _invalidate_l1(self, checkpoint_id: str) -> None:
        if checkpoint_id in self.l1_cache:
            del self.l1_cache[checkpoint_id]
    
    async def _get_from_l2(self, checkpoint_id: str) -> Optional[Checkpoint]:
        l2_path = self._get_l2_path(checkpoint_id)
        
        if not l2_path.exists():
            return None
        
        try:
            data = json.loads(l2_path.read_text(encoding='utf-8'))
            checkpoint = Checkpoint.from_dict(data)
            self._put_to_l1(checkpoint)
            return checkpoint
        except Exception as e:
            logger.warning(f"L2缓存读取失败: {checkpoint_id}, {e}")
            return None
    
    async def _put_to_l2(self, checkpoint: Checkpoint) -> None:
        checkpoint_id = checkpoint.metadata.checkpoint_id
        l2_path = self._get_l2_path(checkpoint_id)
        
        try:
            data = json.dumps(checkpoint.to_dict(), ensure_ascii=False)
            l2_path.write_text(data, encoding='utf-8')
        except Exception as e:
            logger.warning(f"L2缓存写入失败: {checkpoint_id}, {e}")
    
    async def _get_cached(self, checkpoint_id: str) -> Tuple[Optional[Checkpoint], str]:
        checkpoint = self._get_from_l1(checkpoint_id)
        if checkpoint:
            return checkpoint, "L1"
        
        checkpoint = await self._get_from_l2(checkpoint_id)
        if checkpoint:
            return checkpoint, "L2"
        
        return None, "MISS"
    
    async def save(self, checkpoint: Checkpoint) -> None:
        await self.backend.save(checkpoint)
        self._put_to_l1(checkpoint)
        await self._put_to_l2(checkpoint)
    
    async def load(self, checkpoint_id: str) -> Optional[Checkpoint]:
        checkpoint, source = await self._get_cached(checkpoint_id)
        
        if checkpoint:
            logger.debug(f"缓存命中: {checkpoint_id} ({source})")
            return checkpoint
        
        checkpoint = await self.backend.load(checkpoint_id)
        
        if checkpoint:
            self._put_to_l1(checkpoint)
            await self._put_to_l2(checkpoint)
        
        return checkpoint
    
    async def delete(self, checkpoint_id: str) -> bool:
        self._invalidate_l1(checkpoint_id)
        
        l2_path = self._get_l2_path(checkpoint_id)
        if l2_path.exists():
            l2_path.unlink()
        
        return await self.backend.delete(checkpoint_id)
    
    async def list(self) -> List[str]:
        return await self.backend.list()
    
    async def exists(self, checkpoint_id: str) -> bool:
        if await self._get_from_l1(checkpoint_id):
            return True
        
        if self._get_l2_path(checkpoint_id).exists():
            return True
        
        return await self.backend.exists(checkpoint_id)
    
    async def warm_up(self, checkpoint_ids: List[str]) -> None:
        logger.info(f"开始缓存预热: {len(checkpoint_ids)} 个检查点")
        
        for checkpoint_id in checkpoint_ids:
            if not await self._get_from_l1(checkpoint_id):
                checkpoint = await self.backend.load(checkpoint_id)
                if checkpoint:
                    self._put_to_l1(checkpoint)
        
        logger.info("缓存预热完成")
    
    def get_stats(self) -> Dict[str, Any]:
        return {
            "l1_size": len(self.l1_cache),
            "l1_max": self.l1_size,
            "l2_files": len(list(self.l2_cache_dir.glob("*.json"))),
            "hits": self.stats.hits,
            "misses": self.stats.misses,
            "hit_rate": f"{self.stats.hit_rate:.2%}",
            "evictions": self.stats.evictions
        }
```

### 挑战1：分布式检查点提供者

```python
import hashlib

class DistributedCheckpointProvider(CheckpointProvider):
    """分布式检查点提供者"""
    
    def __init__(self, nodes: List[CheckpointProvider]):
        self.nodes = nodes
        self.node_count = len(nodes)
    
    def _get_node_index(self, checkpoint_id: str) -> int:
        """一致性哈希"""
        hash_value = int(hashlib.md5(checkpoint_id.encode()).hexdigest(), 16)
        return hash_value % self.node_count
    
    def _get_node(self, checkpoint_id: str) -> CheckpointProvider:
        index = self._get_node_index(checkpoint_id)
        return self.nodes[index]
    
    async def save(self, checkpoint: Checkpoint) -> None:
        node = self._get_node(checkpoint.metadata.checkpoint_id)
        await node.save(checkpoint)
    
    async def load(self, checkpoint_id: str) -> Optional[Checkpoint]:
        node = self._get_node(checkpoint_id)
        return await node.load(checkpoint_id)
    
    async def delete(self, checkpoint_id: str) -> bool:
        node = self._get_node(checkpoint_id)
        return await node.delete(checkpoint_id)
    
    async def list(self) -> List[str]:
        all_ids = []
        for node in self.nodes:
            ids = await node.list()
            all_ids.extend(ids)
        return list(set(all_ids))
    
    async def exists(self, checkpoint_id: str) -> bool:
        node = self._get_node(checkpoint_id)
        return await node.exists(checkpoint_id)
```

## 常见问题与解决方案

### Q1: 分块上传失败怎么办？
A:
1. 实现重试机制
2. 记录已上传的分块
3. 支持断点续传
4. 清理失败的分块

### Q2: 密钥丢失怎么办？
A:
1. 密钥应该安全备份
2. 实现密钥恢复机制
3. 记录密钥版本历史
4. 不要删除旧密钥

### Q3: 缓存一致性问题
A:
1. 删除时同时清除各级缓存
2. 更新时同时更新各级缓存
3. 设置合理的缓存过期时间
4. 实现缓存失效策略
