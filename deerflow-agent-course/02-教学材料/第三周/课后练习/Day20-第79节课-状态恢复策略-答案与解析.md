## 任务1：理解状态恢复策略架构

### 1.1 长时间任务为什么需要检查点策略？

1. **防止数据丢失**：长时间任务（如模型训练）如果中途失败，没有检查点会导致所有进度丢失
2. **资源恢复**：从检查点恢复比从头开始节省大量计算资源
3. **用户体验**：支持任务中断后继续，提升用户体验

### 1.2 清理策略类型

| 类型 | 优点 | 缺点 |
|------|------|------|
| 时间策略 | 简单直接 | 可能删除有用的检查点 |
| 数量策略 | 控制存储增长 | 可能删除最新检查点 |
| 大小策略 | 直接控制存储 | 计算开销大 |
| 频率策略 | 保留常用 | 需要跟踪访问 |

### 1.3 状态迁移原子性

使用两阶段提交：
1. 先在目标创建临时检查点
2. 验证成功后更新元数据
3. 删除源检查点

## 任务2：事件触发检查点策略

### 完整实现

```python
from abc import ABC, abstractmethod
from typing import List, Dict, Any, Optional, Callable
from datetime import datetime

class CheckpointTrigger(ABC):
    @abstractmethod
    def should_checkpoint(self, context: Dict[str, Any]) -> bool:
        pass

class ProgressBasedTrigger(CheckpointTrigger):
    def __init__(self, threshold: int = 10):
        self.threshold = threshold
        self.last_progress = 0
    
    def should_checkpoint(self, context: Dict[str, Any]) -> bool:
        current = context.get("progress", 0)
        delta = current - self.last_progress
        
        if delta >= self.threshold:
            self.last_progress = current
            return True
        
        return False

class EventBasedTrigger(CheckpointTrigger):
    def __init__(self, events: List[str]):
        self.trigger_events = set(events)
        self.last_event = None
    
    def should_checkpoint(self, context: Dict[str, Any]) -> bool:
        event = context.get("event")
        
        if event and event in self.trigger_events:
            self.last_event = event
            return True
        
        return False

class TimeBasedTrigger(CheckpointTrigger):
    def __init__(self, interval_seconds: int = 300):
        self.interval = interval_seconds
        self.last_checkpoint_time = datetime.now()
    
    def should_checkpoint(self, context: Dict[str, Any]) -> bool:
        now = datetime.now()
        elapsed = (now - self.last_checkpoint_time).total_seconds()
        
        if elapsed >= self.interval:
            self.last_checkpoint_time = now
            return True
        
        return False

class CompositeTrigger(CheckpointTrigger):
    def __init__(self, triggers: List[CheckpointTrigger], mode: str = "any"):
        self.triggers = triggers
        self.mode = mode  # "any" or "all"
    
    def should_checkpoint(self, context: Dict[str, Any]) -> bool:
        if self.mode == "any":
            return any(t.should_checkpoint(context) for t in self.triggers)
        else:
            return all(t.should_checkpoint(context) for t in self.triggers)
```

### 完整实现

from typing import List, Tuple

class WeightedCleanupStrategy:
    def __init__(self, strategies: List[CheckpointCleanupStrategy], weights: Dict[str, float] = None):
        self.strategies = strategies
        self.weights = weights or {s.__class__.__name__: 1.0 for s in strategies}
    
    async def should_cleanup(self, checkpoint: Checkpoint, all_checkpoints: List[Checkpoint]) -> bool:
        for strategy in self.strategies:
            if await strategy.should_cleanup(checkpoint, all_checkpoints):
                return True
        return False
    
    def get_weighted_priority(self, checkpoint: Checkpoint) -> float:
        total = 0.0
        for strategy in self.strategies:
            weight = self.weights.get(strategy.__class__.__name__, 1.0)
            priority = strategy.get_priority(checkpoint)
            total += weight * priority
        return total
    
    async def get_cleanup_candidates(
        self, manager: CheckpointManager, limit: int = None
    ) -> List[Checkpoint]:
        checkpoints = await manager.list_checkpoints()
        
        scored = []
        for cp in checkpoints:
            if await self.should_cleanup(cp, checkpoints):
                priority = self.get_weighted_priority(cp)
                scored.append((cp, priority))
        
        scored.sort(key=lambda x: x[1], reverse=True)
        
        if limit:
            return [cp for cp, _ in scored[:limit]]
        
        return [cp for cp, _ in scored]


class DryRunCleanupStrategy:
    def __init__(self, wrapped: SmartCleanupStrategy):
        self.wrapped = wrapped
        self.dry_run_results: List[Checkpoint] = []
    
    async def cleanup(self, manager: CheckpointManager) -> int:

        self.dry_run_results = candidates
        
        return len(candidates)
    
    async def get_preview(self) -> List[Dict[str, Any]]:
        return [
            {
                "checkpoint_id": cp.metadata.checkpoint_id,
                "name": cp.metadata.name,
                "created_at": cp.metadata.created_at.isoformat(),
                "size_bytes": cp.metadata.size_bytes
            }
            for cp in self.dry_run_results
        ]

## 任务4：任务恢复UI

```python
from typing import Dict, Any, Optional
import json

class TaskRecoveryUI:
    def __init__(self, manager: CheckpointManager):
        self.manager = manager
    
    async def show_task_status(self, task_name: str) -> Dict[str, Any]:
        checkpoints = await self.manager.list_checkpoints()
        
        task_checkpoints = [
            cp for cp in checkpoints
            if cp.metadata.name == task_name
        ]
        
        task_checkpoints.sort(key=lambda c: c.metadata.created_at, reverse=True)
        
        return {
            "task_name": task_name,
            "total_checkpoints": len(task_checkpoints),
            "latest_checkpoint": task_checkpoints[0].metadata.checkpoint_id if task_checkpoints else None,
            "checkpoints": [
                {
                    "id": cp.metadata.checkpoint_id,
                    "version": cp.metadata.version,
                    "created_at": cp.metadata.created_at.isoformat(),
                    "size_bytes": cp.metadata.size_bytes
                }
                for cp in task_checkpoints
            ]
        }
    
    async def select_checkpoint(
        self, task_name: str, version: Optional[int] = None
    ) -> Optional[Checkpoint]:
        checkpoints = await self.manager.list_checkpoints()
        
        task_checkpoints = [
            cp for cp in checkpoints
            if cp.metadata.name == task_name
        ]
        
        if version:
            for cp in task_checkpoints:
                if cp.metadata.version == version:
                    return cp
        else:
            task_checkpoints.sort(key=lambda c: c.metadata.created_at, reverse=True)
            return task_checkpoints[0] if task_checkpoints else None
    
    async def restore_with_progress(
        self, checkpoint_id: str, progress_callback: Callable[[float], None]
    ) -> Optional[Checkpoint]:
        progress_callback(0.1)
        
        checkpoint = await self.manager.restore_checkpoint(checkpoint_id)
        
        progress_callback(0.5)
        
        if checkpoint:
            checkpoint.metadata.metadata["restored_at"] = datetime.now().isoformat()
        
        progress_callback(1.0)
        
        return checkpoint
    
    def render_checkpoints_table(self, checkpoints: List[Checkpoint]) -> str:
        lines = [
            "| Version | Created At | Size |",
            "|---------|------------|------|"
        ]
        
        for cp in checkpoints:
            created = cp.metadata.created_at.strftime("%Y-%m-%d %H:%M:%S")
            size_kb = cp.metadata.size_bytes / 1024
            lines.append(f"| {cp.metadata.version} | {created} | {size_kb:.1f}KB |")
        
        return "\n".join(lines)
```

## 常见问题

### Q: 如何选择检查点保存频率？
A: 
- 太频繁：增加存储开销
- 太稀少：丢失过多进度
- 建议：根据任务性质和资源情况，通常5-15分钟一次

### Q: 清理策略冲突怎么办？
A:
- 使用权重综合评分
- 或者按优先级依次判断
