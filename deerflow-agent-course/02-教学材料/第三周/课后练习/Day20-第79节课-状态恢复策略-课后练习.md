```python
class ProgressBasedCheckpointStrategy:
    def __init__(self, threshold: int = 10):
        self.threshold = threshold
        self.last_checkpoint_progress = 0
    
    def should_checkpoint(self, current_progress: int) -> bool:
        delta = current_progress - self.last_checkpoint_progress
        return delta >= self.threshold

class EventBasedCheckpointStrategy:
    def __init__(self, events: List[str]):
        self.trigger_events = set(events)
    
    def should_checkpoint(self, event: str) -> bool:
        return event in self.trigger_events
```
