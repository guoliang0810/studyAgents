## 任务1：差异压缩检查点

```python
class DiffCompressionManager:
    def __init__(self, provider, compressor=None):
        self.provider = provider
        self.compressor = compressor or CheckpointCompressor()
        self._history: Dict[str, Checkpoint] = {}
    
    async def save_with_diff(self, name, state):
        # 比较历史状态
        last = self._history.get(name)
        
        if last:
            diff = self._compute_diff(last.state, state)
            
            if len(diff) < len(state) * 0.5:
                # 差异小，使用增量
                return await self._save_incremental(name, diff, last)
        
        # 差异大，使用完整
        return await self._save_full(name, state)
    
    def _compute_diff(self, old, new):
        return {k: v for k, v in new.items() if old.get(k) != v}
```

## 任务2：自适应压缩

```python
class AdaptiveCompressor:
    COMPRESSORS = {
        "gzip": lambda: CheckpointCompressor("gzip"),
        "zlib": lambda: CheckpointCompressor("zlib"),
        "none": lambda: CheckpointCompressor()
    }
    
    def __init__(self):
        self.selected = "gzip"
        self._stats: Dict[str, float] = {}
    
    def select_compressor(self, data_size: int) -> str:
        if data_size > 1000000:
            return "gzip"
        elif data_size > 100000:
            return "zlib"
        return "none"
```
