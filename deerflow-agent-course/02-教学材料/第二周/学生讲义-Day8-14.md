## Day 8：沙箱抽象层设计

### 第29节：Sandbox抽象接口

#### 核心概念
沙箱（Sandbox）是DeerFlow中用于安全执行外部代码的环境抽象。它提供以下核心能力：
1. **隔离执行**：限制代码对主机系统的访问
2. **资源控制**：限制CPU、内存、磁盘使用
3. **路径虚拟化**：将虚拟路径映射到实际文件系统
4. **超时管理**：防止无限执行

#### 代码分析：`sandbox/base.py`
```python
class Sandbox(ABC):
    """沙箱抽象基类"""
    
    @abstractmethod
    async def execute(
        self, 
        command: str, 
        args: List[str], 
        env: Dict[str, str] = None,
        timeout: float = None
    ) -> ExecutionResult:
        """执行命令并返回结果"""
        pass
    
    @abstractmethod
    def translate_path(self, virtual_path: str) -> str:
        """将虚拟路径转换为实际路径"""
        pass
    
    @abstractmethod
    async def cleanup(self):
        """清理沙箱资源"""
        pass
```

```python
class LocalSandboxProvider:
    def create(self, config: SandboxConfig) -> Sandbox:

            workdir=config.workdir,
            timeout=config.timeout,

        )

class LocalSandbox(Sandbox):
    def __init__(self, workdir: str, timeout: float, env: Dict[str, str]):
        self.workdir = workdir
        self.timeout = timeout
        self.env = env or {}
    
    async def execute(self, command: str, args: List[str], env: Dict[str, str] = None, timeout: float = None):
        # 合并环境变量
        full_env = {**self.env, **(env or {})}
        
        # 使用asyncio创建子进程
        process = await asyncio.create_subprocess_exec(
            command, *args,
            cwd=self.workdir,
            env=full_env,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        
        try:
            stdout, stderr = await asyncio.wait_for(
                process.communicate(),
                timeout=timeout or self.timeout
            )
            return ExecutionResult(
                return_code=process.returncode,
                stdout=stdout.decode(),
                stderr=stderr.decode(),
            )
        except asyncio.TimeoutError:
            process.kill()
            await process.wait()
            raise TimeoutError(f"Command timed out after {timeout} seconds")
```

## Day 9：虚拟路径系统与安全

### 第33节：虚拟路径架构

#### 路径映射原理
```
虚拟路径: /workspace/user123/project/src/main.py
实际路径: /tmp/sandbox_abc123/user123/project/src/main.py
```

#### 代码实现
```python
class VirtualPathSystem:
    def __init__(self, base_dir: str):
        self.base_dir = base_dir
        self.mappings: Dict[str, str] = {}
    
    def translate(self, virtual_path: str) -> str:
        """翻译虚拟路径到实际路径"""
        # 1. 标准化路径
        normalized = os.path.normpath(virtual_path)
        
        # 2. 检查路径遍历攻击
        if ".." in normalized.split("/"):
            raise SecurityError("Path traversal detected")
        
        # 3. 应用映射规则
        for prefix, target in self.mappings.items():
            if normalized.startswith(prefix):
                remaining = normalized[len(prefix):]
                return os.path.join(target, remaining)
        
        # 4. 默认映射到基础目录
        return os.path.join(self.base_dir, normalized[1:] if normalized.startswith("/") else normalized)
```

```python
        """调度任务到合适的Worker"""
        # 1. 过滤可用Worker
        available = [w for w in self.workers if w.can_accept(task)]
```

#### Server-Sent Events
```python
class EventStream:
    def __init__(self):
        self.listeners: List[asyncio.Queue] = []
    
    async def publish(self, event: Event):
        """发布事件"""
        for queue in self.listeners:
            await queue.put(event)
    
    async def subscribe(self) -> AsyncIterator[Event]:
        """订阅事件流"""
        queue = asyncio.Queue()
        self.listeners.append(queue)
        
        try:
            while True:
                event = await queue.get()
                yield event
        finally:
            self.listeners.remove(queue)
```

```python
def decide_delegation(task: Task, available_agents: List[SubagentInfo]) -> str:
    """决定将任务委派给哪个子代理"""
    
    # 1. 基于任务类型匹配
    task_type = classify_task(task.description)
    matching_agents = [
        agent for agent in available_agents 
        if task_type in agent.capabilities
    ]
    
    if matching_agents:
        # 选择负载最低的匹配代理

    # 2. 基于历史性能选择

    return "generic"
```

```python
class PerformanceMonitor:
    def __init__(self):
        self.metrics: Dict[str, List[float]] = {
            "execution_time": [],

            "success_rate": [],
        }
    
    def record_metric(self, metric_name: str, value: float):

        self.metrics[metric_name].append(value)
        
        # 维护滑动窗口（最近100个点）
        if len(self.metrics[metric_name]) > 100:
            self.metrics[metric_name].pop(0)
    
    def get_performance_score(self, agent_name: str) -> float:
        """计算性能综合得分"""
        metrics = self.metrics[agent_name]
        if not metrics:
            return 0.0
        
        weights = {
            "execution_time": 0.3,
            "memory_usage": 0.2,
            "success_rate": 0.5,
        }
        
        score = 0.0
        for metric, weight in weights.items():
            values = metrics.get(metric, [])
            if values:
                # 归一化处理
                normalized = self.normalize(values, metric)
                score += normalized * weight
        
        return score
```

```python
class MemoryUpdateQueue:
    def __init__(self):
        self.queue = asyncio.Queue()
        self.processing = False
    
    async def put(self, update: MemoryUpdate):
        """放入更新"""
        await self.queue.put(update)
    
    async def process(self):
        """处理队列"""
        self.processing = True
        
        try:
            while True:
                try:
                    update = await asyncio.wait_for(
                        self.queue.get(),
                        timeout=1.0
                    )
                    await self.process_update(update)
                except asyncio.TimeoutError:
                    # 检查是否需要继续运行
                    if self.should_stop():
                        break
        finally:
            self.processing = False
    
    async def process_update(self, update: MemoryUpdate):
        """处理单个更新"""
        try:
            # 应用更新到存储
            await self.store.apply_update(update)
            
            # 记录成功
            logger.info(f"Memory update processed: {update.id}")
        except Exception as e:
            # 失败重试逻辑
            await self.handle_failure(update, e)
```

```python
                FactModel.content.match(query)
            ).limit(limit)
            
            results = await session.execute(stmt)
            db_facts = results.scalars().all()
            
            return [fact.to_fact() for fact in db_facts]
```

```python
class MemoryConfig:
    def __init__(self, base_config: Dict[str, Any]):
        self.base_config = base_config
        self.overrides: Dict[str, Any] = {}
    
    def get(self, key: str, default: Any = None) -> Any:
        """获取配置值（支持嵌套）"""
        # 1. 检查覆盖
        if key in self.overrides:
            return self.overrides[key]
        
        # 2. 从基础配置获取
        value = self.base_config
        for part in key.split("."):
            if isinstance(value, dict) and part in value:
                value = value[part]
            else:
                return default
        
        return value
    
    def set_override(self, key: str, value: Any):
        """设置配置覆盖"""
        self.overrides[key] = value
```

```python
class FactLifecycleManager:
    def __init__(self):
        self.facts: Dict[str, Fact] = {}
        self.access_tracker: Dict[str, List[float]] = {}
```

### 第56节：内存性能优化

#### 缓存策略
```python
class MemoryCache:
    def __init__(self, max_size: int = 1000):
        self.cache: Dict[str, Fact] = {}
        self.access_order: List[str] = []
        self.max_size = max_size
    
    async def get(self, fact_id: str) -> Optional[Fact]:
        """从缓存获取事实"""
        if fact_id in self.cache:
            # 更新访问顺序
            self.access_order.remove(fact_id)
            self.access_order.append(fact_id)
            return self.cache[fact_id]
        return None
    
    async def set(self, fact: Fact):
        """缓存事实"""
        if len(self.cache) >= self.max_size:
            # 移除最久未使用的
            oldest = self.access_order.pop(0)
            del self.cache[oldest]
        
        self.cache[fact.id] = fact
        self.access_order.append(fact.id)
```

```python
class MemoryPerformanceMonitor:
    METRICS = [
        "extraction_time",
        "injection_time",
        "cache_hit_rate",
        "storage_latency",
    ]
    
    def __init__(self):
        self.metrics: Dict[str, List[float]] = {m: [] for m in self.METRICS}
    
    def record(self, metric: str, value: float):
        """记录性能指标"""
        self.metrics[metric].append(value)
    
    def get_report(self) -> Dict[str, Any]:
        """生成性能报告"""
        report = {}
        
        for metric, values in self.metrics.items():
            if values:
                report[metric] = {
                    "avg": sum(values) / len(values),
                    "min": min(values),
                    "max": max(values),
                    "count": len(values),
                    "p95": self.percentile(values, 95),
                }
        
        return report
```
