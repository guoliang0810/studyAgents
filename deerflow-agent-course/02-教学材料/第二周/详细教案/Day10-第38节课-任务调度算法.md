### 附录3：调度器实现代码示例
```python
from dataclasses import dataclass, field
from typing import List, Dict, Optional, Tuple
import asyncio
import time
from enum import Enum
import random

class TaskPriority(Enum):
    """任务优先级"""
    LOW = 1
    MEDIUM = 2
    HIGH = 3
    CRITICAL = 4

@dataclass
class ScheduledTask:
    """调度任务"""
    id: str
    subagent_name: str
    description: str
    priority: TaskPriority = TaskPriority.MEDIUM
    estimated_duration: float = 10.0  # 预估执行时间（秒）
    created_at: float = field(default_factory=time.time)
    scheduled_at: Optional[float] = None
    started_at: Optional[float] = None
    completed_at: Optional[float] = None
    
    def waiting_time(self) -> float:
        """等待时间"""
        if self.started_at:
            return self.started_at - self.created_at
        return time.time() - self.created_at
    
    def execution_time(self) -> Optional[float]:
        """执行时间"""
        if self.started_at and self.completed_at:
            return self.completed_at - self.started_at
        return None

@dataclass
class Worker:
    """工作节点"""
    id: str
    name: str
    capabilities: List[str]  # 支持的能力
    max_concurrent_tasks: int = 5  # 最大并发任务数
    current_tasks: List[ScheduledTask] = field(default_factory=list)
    cpu_usage: float = 0.0  # CPU使用率（0-1）
    memory_usage: float = 0.0  # 内存使用率（0-1）
    last_update_time: float = field(default_factory=time.time)
    
    def can_accept(self, task: ScheduledTask) -> bool:
        """是否可以接受新任务"""
        # 检查能力是否匹配
        if task.subagent_name not in self.capabilities:
            return False
        
        # 检查并发限制
        if len(self.current_tasks) >= self.max_concurrent_tasks:
            return False
        
        # 检查资源限制（简化）
        if self.cpu_usage > 0.8 or self.memory_usage > 0.8:
            return False
        
        return True
    
    def current_load(self) -> float:
        """计算当前负载（0-1）"""
        # 基于并发任务数、CPU和内存使用率综合计算
        task_load = len(self.current_tasks) / self.max_concurrent_tasks
        resource_load = (self.cpu_usage + self.memory_usage) / 2
        
        # 加权平均
        return 0.6 * task_load + 0.4 * resource_load
    
    def add_task(self, task: ScheduledTask):
        """添加任务"""
        self.current_tasks.append(task)
        # 模拟资源使用增加
        self.cpu_usage = min(1.0, self.cpu_usage + 0.1)
        self.memory_usage = min(1.0, self.memory_usage + 0.05)
        self.last_update_time = time.time()
    
    def complete_task(self, task_id: str):
        """完成任务"""
        self.current_tasks = [t for t in self.current_tasks if t.id != task_id]
        # 模拟资源使用减少
        self.cpu_usage = max(0.0, self.cpu_usage - 0.08)
        self.memory_usage = max(0.0, self.memory_usage - 0.04)
        self.last_update_time = time.time()

class Scheduler:
    """任务调度器"""
    
    def __init__(self):
        self.workers: Dict[str, Worker] = {}
        self.task_queue: List[ScheduledTask] = []
        self.scheduling_history: List[Dict] = []
    
    def register_worker(self, worker: Worker):
        """注册工作节点"""
        self.workers[worker.id] = worker
        print(f"✅ 注册Worker: {worker.name} (ID: {worker.id})")
    
    def submit_task(self, task: ScheduledTask):
        """提交任务"""
        self.task_queue.append(task)
        print(f"📥 提交任务: {task.description} (优先级: {task.priority.name})")
    
    def schedule_fifo(self) -> List[Tuple[ScheduledTask, Optional[Worker]]]:
        """FIFO调度
        
        Returns:
            调度决策列表：[(任务, 分配的Worker)]
        """
        decisions = []
        
        # 按提交时间排序
        self.task_queue.sort(key=lambda t: t.created_at)
        
        for task in self.task_queue[:]:  # 复制列表以便修改
            assigned = False
            
            # 查找可用Worker
            for worker in self.workers.values():
                if worker.can_accept(task):
                    decisions.append((task, worker))
                    worker.add_task(task)
                    self.task_queue.remove(task)
                    assigned = True
                    break
            
            if not assigned:
                decisions.append((task, None))
        
        return decisions
    
    def schedule_priority(self) -> List[Tuple[ScheduledTask, Optional[Worker]]]:
        """优先级调度"""
        decisions = []
        
        # 按优先级排序（高优先级在前）
        self.task_queue.sort(key=lambda t: t.priority.value, reverse=True)
        
        for task in self.task_queue[:]:
            assigned = False
            
            # 查找可用Worker
            for worker in self.workers.values():
                if worker.can_accept(task):
                    decisions.append((task, worker))
                    worker.add_task(task)
                    self.task_queue.remove(task)
                    assigned = True
                    break
            
            if not assigned:
                decisions.append((task, None))
        
        return decisions
    
    def schedule_resource_aware(self) -> List[Tuple[ScheduledTask, Optional[Worker]]]:
        """资源感知调度"""
        decisions = []
        
        # 按优先级和等待时间综合排序
        def task_score(task: ScheduledTask) -> float:
            priority_score = task.priority.value * 0.7
            wait_score = min(task.waiting_time() / 60, 1.0) * 0.3  # 等待时间影响（归一化）
            return priority_score + wait_score
        
        self.task_queue.sort(key=task_score, reverse=True)
        
        for task in self.task_queue[:]:
            assigned = False
            
            # 查找可用Worker并按负载排序
            available_workers = [
                w for w in self.workers.values() 
                if w.can_accept(task)
            ]
            
            if available_workers:
                # 选择负载最低的Worker
                available_workers.sort(key=lambda w: w.current_load())
                selected_worker = available_workers[0]
                
                decisions.append((task, selected_worker))
                selected_worker.add_task(task)
                self.task_queue.remove(task)
                assigned = True
            
            if not assigned:
                decisions.append((task, None))
        
        return decisions
    
    def schedule_round_robin(self) -> List[Tuple[ScheduledTask, Optional[Worker]]]:
        """轮询调度"""
        decisions = []
        
        if not self.workers:
            return decisions
        
        # 获取Worker ID列表
        worker_ids = list(self.workers.keys())
        current_worker_index = 0
        
        for task in self.task_queue[:]:
            assigned = False
            attempts = 0
            
            while attempts < len(worker_ids):
                worker = self.workers[worker_ids[current_worker_index]]
                
                if worker.can_accept(task):
                    decisions.append((task, worker))
                    worker.add_task(task)
                    self.task_queue.remove(task)
                    assigned = True
                    break
                
                # 尝试下一个Worker
                current_worker_index = (current_worker_index + 1) % len(worker_ids)
                attempts += 1
            
            if not assigned:
                decisions.append((task, None))
            
            # 移动到下一个Worker
            current_worker_index = (current_worker_index + 1) % len(worker_ids)
        
        return decisions
    
    def log_scheduling(self, algorithm: str, decisions: List[Tuple[ScheduledTask, Optional[Worker]]]):
        """记录调度决策"""
        for task, worker in decisions:
            entry = {
                "timestamp": time.time(),
                "task_id": task.id,
                "algorithm": algorithm,
                "worker_assigned": worker.id if worker else None,
                "task_priority": task.priority.name,
                "waiting_time": task.waiting_time()
            }
            self.scheduling_history.append(entry)
    
    def get_statistics(self) -> Dict:
        """获取调度统计"""
        if not self.scheduling_history:
            return {}
        
        successful = [e for e in self.scheduling_history if e["worker_assigned"]]
        failed = [e for e in self.scheduling_history if not e["worker_assigned"]]
        
        avg_waiting_time = sum(e["waiting_time"] for e in successful) / len(successful) if successful else 0
        
        return {
            "total_scheduled": len(self.scheduling_history),
            "successful": len(successful),
            "failed": len(failed),
            "success_rate": len(successful) / len(self.scheduling_history) if self.scheduling_history else 0,
            "avg_waiting_time": avg_waiting_time
        }
```

```python
class LoadEvaluator:
    """负载评估器"""
    
    @staticmethod
    def evaluate_cpu_load(worker: Worker) -> float:
        """评估CPU负载"""
        # 实际中会从系统获取，这里使用模拟值
        return worker.cpu_usage
    
    @staticmethod
    def evaluate_memory_load(worker: Worker) -> float:
        """评估内存负载"""
        return worker.memory_usage
    
    @staticmethod
    def evaluate_task_load(worker: Worker) -> float:
        """评估任务负载"""
        return len(worker.current_tasks) / worker.max_concurrent_tasks
    
    @staticmethod
    def evaluate_network_load(worker: Worker) -> float:
        """评估网络负载（简化）"""
        # 模拟网络负载
        return min(1.0, len(worker.current_tasks) * 0.1)
    
    @staticmethod
    def comprehensive_load_score(worker: Worker) -> float:
        """综合负载评分"""
        cpu = LoadEvaluator.evaluate_cpu_load(worker) * 0.3
        memory = LoadEvaluator.evaluate_memory_load(worker) * 0.3
        task = LoadEvaluator.evaluate_task_load(worker) * 0.3
        network = LoadEvaluator.evaluate_network_load(worker) * 0.1
        
        return cpu + memory + task + network
    
    @staticmethod
    def predict_finish_time(worker: Worker, new_task: ScheduledTask) -> float:
        """预测完成新任务后的完成时间"""
        current_load = LoadEvaluator.comprehensive_load_score(worker)
        
        # 简单预测：当前负载越高，新任务执行时间越长
        load_factor = 1.0 + current_load * 0.5
        predicted_time = new_task.estimated_duration * load_factor
        
        # 考虑现有任务的剩余时间
        existing_tasks_time = sum(
            t.estimated_duration * 0.7  # 假设已完成30%
            for t in worker.current_tasks
        )
        
        return existing_tasks_time + predicted_time
```
