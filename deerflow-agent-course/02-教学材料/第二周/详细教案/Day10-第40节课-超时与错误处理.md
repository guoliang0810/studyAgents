### 附录3：错误处理实现代码示例
```python
import asyncio
import time
import random
from dataclasses import dataclass, field
from typing import Any, Callable, Optional, Dict, List
from enum import Enum
import logging

# 设置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class ErrorType(Enum):
    """错误类型"""
    TIMEOUT = "timeout"
    NETWORK = "network"
    RESOURCE = "resource"
    BUSINESS = "business"
    DEPENDENCY = "dependency"
    CONFIGURATION = "configuration"
    CONCURRENCY = "concurrency"
    SYSTEM = "system"

@dataclass
class ErrorContext:
    """错误上下文"""
    error_type: ErrorType
    error_message: str
    timestamp: float = field(default_factory=time.time)
    retry_count: int = 0
    operation: str = ""
    resource: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

@dataclass
class RetryConfig:
    """重试配置"""
    max_retries: int = 3
    initial_delay: float = 1.0  # 初始延迟（秒）
    max_delay: float = 60.0     # 最大延迟（秒）
    backoff_factor: float = 2.0  # 退避因子
    jitter: bool = True         # 是否添加随机抖动
    retry_on_errors: List[ErrorType] = field(default_factory=list)  # 重试的错误类型
    
    def calculate_delay(self, retry_count: int) -> float:
        """计算延迟时间"""
        delay = self.initial_delay * (self.backoff_factor ** retry_count)
        delay = min(delay, self.max_delay)
        
        if self.jitter:
            # 添加±10%的随机抖动
            jitter = random.uniform(0.9, 1.1)
            delay *= jitter
        
        return delay

class TimeoutError(Exception):
    """超时错误"""
    def __init__(self, message: str, timeout: float):
        super().__init__(message)
        self.timeout = timeout
        self.message = message

class RetryableError(Exception):
    """可重试错误"""
    def __init__(self, message: str, error_type: ErrorType, retryable: bool = True):
        super().__init__(message)
        self.error_type = error_type
        self.retryable = retryable
        self.message = message

class ErrorHandler:
    """错误处理器"""
    
    def __init__(self, retry_config: RetryConfig = None):
        self.retry_config = retry_config or RetryConfig()
        self.error_history: List[ErrorContext] = []
        self.max_history = 100
    
    async def execute_with_timeout(
        self,
        coro,
        timeout: float,
        operation: str = "",
        resource: str = ""
    ) -> Any:
        """带超时执行
        
        Args:
            coro: 协程或可调用对象
            timeout: 超时时间（秒）
            operation: 操作名称
            resource: 资源标识
            
        Returns:
            执行结果
            
        Raises:
            TimeoutError: 超时错误
        """
        start_time = time.time()
        
        try:
            if asyncio.iscoroutine(coro):
                result = await asyncio.wait_for(coro, timeout=timeout)
            else:
                # 如果是普通函数，在线程池中执行
                loop = asyncio.get_event_loop()
                result = await asyncio.wait_for(
                    loop.run_in_executor(None, coro),
                    timeout=timeout
                )
            
            execution_time = time.time() - start_time
            logger.info(f"操作 {operation} 成功完成，耗时: {execution_time:.2f}s")
            return result
            
        except asyncio.TimeoutError:
            execution_time = time.time() - start_time
            error_msg = f"操作 {operation} 超时 ({timeout}s)"
            
            error_context = ErrorContext(
                error_type=ErrorType.TIMEOUT,
                error_message=error_msg,
                operation=operation,
                resource=resource,
                metadata={
                    "timeout": timeout,
                    "execution_time": execution_time
                }
            )
            self._record_error(error_context)
            
            logger.warning(f"{error_msg}, 实际执行: {execution_time:.2f}s")
            raise TimeoutError(error_msg, timeout)
            
        except Exception as e:
            execution_time = time.time() - start_time
            error_type = self._classify_error(e)
            
            error_context = ErrorContext(
                error_type=error_type,
                error_message=str(e),
                operation=operation,
                resource=resource,
                metadata={
                    "execution_time": execution_time,
                    "exception_type": type(e).__name__
                }
            )
            self._record_error(error_context)
            
            logger.error(f"操作 {operation} 失败: {e}", exc_info=True)
            raise
    
    async def execute_with_retry(
        self,
        coro_func: Callable,
        operation: str = "",
        resource: str = "",
        retry_config: RetryConfig = None
    ) -> Any:
        """带重试执行
        
        Args:
            coro_func: 协程函数（可接受retry_count参数）
            operation: 操作名称
            resource: 资源标识
            retry_config: 重试配置，None使用默认
            
        Returns:
            执行结果
            
        Raises:
            最后一次尝试的异常
        """
        config = retry_config or self.retry_config
        last_error = None
        
        for retry_count in range(config.max_retries + 1):
            try:
                # 调用函数，传入重试次数
                if asyncio.iscoroutinefunction(coro_func):
                    result = await coro_func(retry_count=retry_count)
                else:
                    result = coro_func(retry_count=retry_count)
                
                if retry_count > 0:
                    logger.info(f"操作 {operation} 在第{retry_count}次重试后成功")
                
                return result
                
            except Exception as e:
                error_type = self._classify_error(e)
                last_error = e
                
                # 检查是否应该重试
                should_retry = self._should_retry(e, error_type, retry_count, config)
                
                if not should_retry or retry_count >= config.max_retries:
                    # 不重试或达到最大重试次数
                    error_msg = f"操作 {operation} 失败"
                    if retry_count > 0:
                        error_msg += f"（重试{retry_count}次后）"
                    
                    error_context = ErrorContext(
                        error_type=error_type,
                        error_message=error_msg,
                        retry_count=retry_count,
                        operation=operation,
                        resource=resource,
                        metadata={
                            "exception": str(e),
                            "exception_type": type(e).__name__,
                            "final_attempt": True
                        }
                    )
                    self._record_error(error_context)
                    
                    logger.error(f"{error_msg}: {e}")
                    raise last_error
                
                # 计算延迟并等待
                delay = config.calculate_delay(retry_count)
                logger.warning(
                    f"操作 {operation} 失败，{delay:.1f}s后重试 "
                    f"({retry_count+1}/{config.max_retries}): {e}"
                )
                
                error_context = ErrorContext(
                    error_type=error_type,
                    error_message=str(e),
                    retry_count=retry_count,
                    operation=operation,
                    resource=resource,
                    metadata={
                        "delay": delay,
                        "exception_type": type(e).__name__
                    }
                )
                self._record_error(error_context)
                
                await asyncio.sleep(delay)
        
        # 理论上不会执行到这里
        raise last_error
    
    def _classify_error(self, error: Exception) -> ErrorType:
        """分类错误"""
        error_str = str(error).lower()
        
        if isinstance(error, TimeoutError):
            return ErrorType.TIMEOUT
        
        if "connection" in error_str or "network" in error_str or "socket" in error_str:
            return ErrorType.NETWORK
        
        if "memory" in error_str or "disk" in error_str or "resource" in error_str:
            return ErrorType.RESOURCE
        
        if "timeout" in error_str:
            return ErrorType.TIMEOUT
        
        # 更多分类逻辑...
        return ErrorType.SYSTEM
    
    def _should_retry(
        self,
        error: Exception,
        error_type: ErrorType,
        retry_count: int,
        config: RetryConfig
    ) -> bool:
        """判断是否应该重试"""
        # 检查错误类型是否在重试列表中
        if config.retry_on_errors and error_type not in config.retry_on_errors:
            return False
        
        # 检查是否达到最大重试次数
        if retry_count >= config.max_retries:
            return False
        
        # 检查错误是否可重试
        if isinstance(error, RetryableError):
            return error.retryable
        
        # 默认重试网络和超时错误
        return error_type in [ErrorType.NETWORK, ErrorType.TIMEOUT, ErrorType.DEPENDENCY]
    
    def _record_error(self, error_context: ErrorContext):
        """记录错误"""
        self.error_history.append(error_context)
        if len(self.error_history) > self.max_history:
            self.error_history = self.error_history[-self.max_history:]
    
    def get_error_statistics(self) -> Dict:
        """获取错误统计"""
        if not self.error_history:
            return {}
        
        stats = {}
        for error_type in ErrorType:
            count = len([e for e in self.error_history if e.error_type == error_type])
            if count > 0:
                stats[error_type.value] = count
        
        total_errors = len(self.error_history)
        stats["total_errors"] = total_errors
        
        if total_errors > 0:
            # 计算错误率（按时间窗口）
            time_window = 3600  # 1小时
            recent_errors = [
                e for e in self.error_history
                if time.time() - e.timestamp < time_window
            ]
            stats["recent_error_rate"] = len(recent_errors) / (time_window / 3600)  # 错误/小时
        
        return stats

class FallbackHandler:
    """降级处理器"""
    
    def __init__(self, primary_func, fallback_func, fallback_condition=None):
        """
        Args:
            primary_func: 主函数
            fallback_func: 降级函数
            fallback_condition: 降级条件函数，返回True时触发降级
        """
        self.primary_func = primary_func
        self.fallback_func = fallback_func
        self.fallback_condition = fallback_condition or (lambda e: True)
        self.fallback_count = 0
    
    async def execute(self, *args, **kwargs):
        """执行，失败时降级"""
        try:
            if asyncio.iscoroutinefunction(self.primary_func):
                return await self.primary_func(*args, **kwargs)
            else:
                return self.primary_func(*args, **kwargs)
                
        except Exception as e:
            if self.fallback_condition(e):
                self.fallback_count += 1
                logger.warning(f"主服务失败，触发降级 (第{self.fallback_count}次): {e}")
                
                try:
                    if asyncio.iscoroutinefunction(self.fallback_func):
                        return await self.fallback_func(*args, **kwargs)
                    else:
                        return self.fallback_func(*args, **kwargs)
                        
                except Exception as fallback_error:
                    logger.error(f"降级服务也失败: {fallback_error}")
                    raise fallback_error
            else:
                # 不符合降级条件，直接抛出异常
                raise

class CircuitBreaker:
    """断路器"""
    
    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: float = 30.0,
        half_open_max_attempts: int = 3
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_attempts = half_open_max_attempts
        
        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
        self.failure_count = 0
        self.last_failure_time = 0
        self.half_open_attempts = 0
    
    async def execute(self, coro_func, *args, **kwargs):
        """通过断路器执行"""
        if self.state == "OPEN":
            # 检查是否应该进入半开状态
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = "HALF_OPEN"
                self.half_open_attempts = 0
                logger.info("断路器进入半开状态")
            else:
                raise Exception("断路器打开，拒绝执行")
        
        try:
            if asyncio.iscoroutinefunction(coro_func):
                result = await coro_func(*args, **kwargs)
            else:
                result = coro_func(*args, **kwargs)
            
            # 执行成功，重置状态
            if self.state == "HALF_OPEN":
                self.half_open_attempts += 1
                if self.half_open_attempts >= self.half_open_max_attempts:
                    self.state = "CLOSED"
                    self.failure_count = 0
                    logger.info("断路器关闭，恢复正常")
            
            return result
            
        except Exception as e:
            # 执行失败
            if self.state == "HALF_OPEN":
                # 半开状态下失败，立即打开
                self.state = "OPEN"
                self.last_failure_time = time.time()
                logger.warning("半开状态下执行失败，断路器重新打开")
            else:
                # 关闭状态下失败
                self.failure_count += 1
                if self.failure_count >= self.failure_threshold:
                    self.state = "OPEN"
                    self.last_failure_time = time.time()
                    logger.warning(f"失败次数达到阈值，断路器打开 ({self.failure_count}/{self.failure_threshold})")
            
            raise
```
