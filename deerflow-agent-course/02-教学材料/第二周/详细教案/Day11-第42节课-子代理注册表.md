**基础注册表示例：**
```python
from typing import Dict, Optional
import logging

logger = logging.getLogger(__name__)

class SubagentRegistry:
    """子代理注册表"""
    
    def __init__(self):
        self.agents: Dict[str, SubagentInfo] = {}
    
    def register(self, name: str, agent: SubagentInfo):
        """注册子代理"""
        if name in self.agents:
            raise ValueError(f"Subagent {name} already registered")
        
        self.agents[name] = agent
        logger.info(f"Registered subagent: {name}")
    
    def get(self, name: str) -> SubagentInfo:
        """获取子代理"""
        agent = self.agents.get(name)
        if not agent:
            raise KeyError(f"Subagent {name} not found")
        return agent
    
    def list(self) -> List[str]:
        """列出所有子代理"""
        return list(self.agents.keys())
    
    def unregister(self, name: str):
        """注销子代理"""
        if name in self.agents:
            del self.agents[name]
            logger.info(f"Unregistered subagent: {name}")
```

**动态加载扩展演示：**
```python
import yaml

class EnhancedSubagentRegistry(SubagentRegistry):
    """增强版子代理注册表"""
    
    def __init__(self):
        super().__init__()
        self._settings: Dict[str, SubagentSettings] = {}
        self._configs: Dict[str, Dict] = {}
    
    def load_from_yaml(self, config_path: str):
        """从YAML加载配置并自动注册"""
        with open(config_path, 'r') as f:
            config = yaml.safe_load(f)
        
        for name, settings_dict in config.get('subagents', {}).items():
            self._configs[name] = settings_dict
            self._settings[name] = SubagentSettings(**settings_dict)
            print(f"📥 加载配置: {name}")
    
    def get_settings(self, name: str) -> SubagentSettings:
        """获取子代理设置"""
        return self._settings.get(name, SubagentSettings())
    
    def update_settings(self, name: str, settings: SubagentSettings):
        """更新设置"""
        self._settings[name] = settings
        print(f"⚙️ 更新设置: {name}")
```
