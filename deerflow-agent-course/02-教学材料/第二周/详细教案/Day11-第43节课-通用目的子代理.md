**通用子代理基础实现：**
```python
from typing import Dict, Any
import asyncio

class GenericSubagent(Subagent):
    """通用目的子代理"""
    
    def __init__(self, config: GenericSubagentConfig):
        self.config = config
        self.sandbox = create_sandbox(config.sandbox_config)
        self.memory = create_memory(config.memory_config)
    
    async def execute(self, task: Task) -> TaskResult:
        """执行任务 - 三阶段流程"""
        # 阶段1: 准备执行环境
        await self.prepare_environment(task)
        
        # 阶段2: 执行主要逻辑
        result = await self.execute_main(task)
        
        # 阶段3: 清理资源
        await self.cleanup(task)
        
        return result
    
    async def prepare_environment(self, task: Task):
        """准备执行环境"""
        # 1. 初始化沙箱环境
        await self.sandbox.initialize()
        
        # 2. 加载必要资源
        await self.load_resources(task)
        
        # 3. 设置执行上下文
        self.context = self.create_context(task)
    
    async def execute_main(self, task: Task) -> TaskResult:
        """执行主要逻辑"""
        # 根据任务类型选择执行策略
        if task.type == "code_execution":
            return await self.execute_code(task)
        elif task.type == "data_analysis":
            return await self.analyze_data(task)
        elif task.type == "api_call":
            return await self.call_api(task)
        else:
            raise ValueError(f"Unsupported task type: {task.type}")
    
    async def cleanup(self, task: Task):
        """清理资源"""
        # 1. 释放沙箱资源
        await self.sandbox.cleanup()
        
        # 2. 保存执行结果到内存
        await self.memory.store(task.id, self.context)
        
        # 3. 清理临时文件
        await self.cleanup_temp_files()
```

async def execute_file_operation(self, task: Task) -> TaskResult:

    if operation == "read":

        content = task.params.get("content")

**能力扩展机制演示：**
```python
class CapabilityRegistry:
    """能力注册表"""
    
    def __init__(self):
        self.capabilities: Dict[str, Callable] = {}
    
    def register(self, name: str, handler: Callable):
        """注册能力处理器"""
        self.capabilities[name] = handler
    
    def get(self, name: str) -> Callable:
        """获取能力处理器"""
        return self.capabilities.get(name)

class ExtensibleGenericSubagent(GenericSubagent):
    """可扩展的通用子代理"""
    
    def __init__(self, config: GenericSubagentConfig):
        super().__init__(config)
        self.capabilities = CapabilityRegistry()
        self.register_default_capabilities()
    
    def register_default_capabilities(self):
        """注册默认能力"""
        self.capabilities.register("code_execution", self.execute_code)
        self.capabilities.register("data_analysis", self.analyze_data)
        self.capabilities.register("api_call", self.call_api)
    
    async def execute_main(self, task: Task) -> TaskResult:
        """通过能力注册表动态执行"""
        handler = self.capabilities.get(task.type)
        if handler:
            return await handler(task)
        else:
            # 尝试动态加载能力
            return await self.try_dynamic_capability(task)
```
