**基础委派决策算法：**
```python
from typing import List, Dict, Optional
from enum import Enum

class DelegationStrategy(Enum):
    """委派策略"""
    TOOL = "tool"              # 使用工具
    SUBAGENT = "subagent"      # 使用子代理
    MULTI_SUBAGENT = "multi"  # 多子代理并行
    HYBRID = "hybrid"          # 混合模式

def decide_delegation(task: Task, available_agents: List[SubagentInfo]) -> str:
    """决定将任务委派给哪个子代理"""
    
    # 1. 基于任务类型匹配
    task_type = classify_task(task.description)
    matching_agents = [
        agent for agent in available_agents 
        if task_type in agent.capabilities
    ]
    
    if matching_agents:
        # 选择负载最低的匹配代理
        return min(matching_agents, key=lambda a: a.current_load()).name
    
    # 2. 基于历史性能选择
    historical_success = get_historical_success_rate(available_agents, task_type)
    if historical_success:
        return max(historical_success.items(), key=lambda x: x[1])[0]
    
    # 3. 默认选择通用代理
    return "generic"

def classify_task(description: str) -> str:
    """分类任务类型"""
    description_lower = description.lower()
    
    if any(word in description_lower for word in ["代码", "编程", "开发"]):
        return "code"
    elif any(word in description_lower for word in ["数据", "分析", "统计"]):
        return "data"
    elif any(word in description_lower for word in ["文件", "目录", "系统"]):
        return "system"
    elif any(word in description_lower for word in ["网络", "请求", "api"]):
        return "network"
    else:
        return "general"

class TaskComplexityAnalyzer:
    """任务复杂度分析器"""
    
    COMPLEXITY_INDICATORS = {
        "high": ["分析", "设计", "优化", "比较", "评估", "架构"],
        "medium": ["生成", "创建", "查找", "获取", "转换"],
        "low": ["计算", "格式化", "排序", "过滤", "验证"]
    }
    
    def analyze(self, task: str) -> str:
        """分析任务复杂度"""
        task_lower = task.lower()
        
        for indicator in self.COMPLEXITY_INDICATORS["high"]:
            if indicator in task_lower:
                return "high"
        
        for indicator in self.COMPLEXITY_INDICATORS["medium"]:
            if indicator in task_lower:
                return "medium"
        
        return "low"
```

# 学生练习：添加响应时间预测到委派决策
class ResponseTimePredictor:
    """响应时间预测器"""
    
    def __init__(self):
        self.history: Dict[str, List[float]] = {}  # 代理名称 -> 历史执行时间列表
    
    def record_execution_time(self, agent_name: str, execution_time: float):
        """记录执行时间"""
        if agent_name not in self.history:
            self.history[agent_name] = []
        self.history[agent_name].append(execution_time)
        
        # 保持最近100条记录
        if len(self.history[agent_name]) > 100:
            self.history[agent_name].pop(0)
    
    def predict_response_time(self, agent_name: str, task_type: str) -> float:
        """预测响应时间"""
        if agent_name not in self.history or not self.history[agent_name]:
            return 10.0  # 默认值
        
        # 计算历史平均时间
        avg_time = sum(self.history[agent_name]) / len(self.history[agent_name])
        
        # 根据任务类型调整（假设不同类型任务执行时间不同）
        type_multiplier = {
            "code": 1.5,
            "data": 1.2,
            "system": 1.0,
            "network": 0.8,
            "general": 1.0
        }
        
        return avg_time * type_multiplier.get(task_type, 1.0)

**智能委派器完整实现演示：**
```python
class SmartDelegator:
    """智能任务委派器"""
    
    def __init__(self, executor: SubagentExecutor, registry: SubagentRegistry):
        self.executor = executor
        self.registry = registry
        self.analyzer = TaskComplexityAnalyzer()
        self.predictor = ResponseTimePredictor()
    
    async def delegate(self, task: str) -> Dict:
        """委派任务"""
        complexity = self.analyzer.analyze(task)
        
        if complexity == "low":
            return await self._delegate_to_tool(task)
        elif complexity == "medium":
            return await self._delegate_to_subagent(task)
        else:
            return await self._delegate_to_multi(task)
    
    async def _delegate_to_tool(self, task: str) -> Dict:
        """委派给工具"""
        print(f"🔧 委派给工具: {task}")
        return {"strategy": "tool", "result": "工具执行完成"}
    
    async def _delegate_to_subagent(self, task: str) -> Dict:
        """委派给子代理"""
        print(f"🤖 委派给子代理: {task}")
        
        # 选择合适的子代理
        subagent = self._select_subagent(task)
        if subagent:
            result = await self.executor.execute_single(
                subagent.config.name,
                {"description": task}
            )
            return {"strategy": "subagent", "result": result.output}
        
        return {"strategy": "subagent", "result": "无合适的子代理"}
    
    async def _delegate_to_multi(self, task: str) -> Dict:
        """委派给多个子代理"""
        print(f"🚀 委派给多个子代理: {task}")
        return {"strategy": "multi", "result": "多代理协作完成"}
    
    def _select_subagent(self, task: str) -> Optional[BaseSubagent]:
        """选择合适的子代理"""
        task_lower = task.lower()
        
        # 收集所有匹配的子代理
        matching_agents = []
        for name in self.registry.list_all():
            subagent = self.registry.get(name)
            for cap in subagent.config.capabilities:
                if cap in task_lower:
                    matching_agents.append((subagent, cap))
        
        if not matching_agents:
            return None
        
        # 选择最佳匹配（考虑能力和负载）
        task_type = classify_task(task)
        best_agent = None
        best_score = -1
        
        for agent, capability in matching_agents:
            # 计算综合得分：能力匹配度 + 负载系数 + 响应时间预测
            capability_score = 1.0 if capability == task_type else 0.7
            load_score = 1.0 / (1.0 + agent.current_load())  # 负载越低得分越高
            time_score = 1.0 / (1.0 + self.predictor.predict_response_time(agent.config.name, task_type))
            
            total_score = capability_score * 0.5 + load_score * 0.3 + time_score * 0.2
            
            if total_score > best_score:
                best_score = total_score
                best_agent = agent
        
        return best_agent
```

### 附录4：常见决策错误及解决方法
| 错误现象 | 可能原因 | 解决方法 |
|----------|----------|----------|
| 总是选择同一代理 | 权重设置不合理 | 调整权重，增加负载因素比重 |
| 决策结果不稳定 | 历史数据不足 | 增加默认值，平滑决策 |
| 响应时间过长 | 预测模型不准 | 使用移动平均，定期校准 |
| 负载不均衡 | 负载计算不准确 | 改进负载度量，考虑资源使用率 |
| 类型匹配失败 | 分类算法太简单 | 使用更先进的NLP分类方法 |
