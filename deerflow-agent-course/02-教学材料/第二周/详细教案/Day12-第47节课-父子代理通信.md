**父子代理通信协议基础实现：**
```python
from typing import Dict, Any, Optional
from dataclasses import dataclass
from enum import Enum
import json
import asyncio
import uuid
from datetime import datetime

class MessageType(Enum):
    """消息类型"""
    INSTRUCTION = "instruction"    # 指令
    RESULT = "result"              # 结果
    QUESTION = "question"          # 问题
    PROGRESS = "progress"          # 进度
    ERROR = "error"                # 错误
    HEARTBEAT = "heartbeat"        # 心跳
    CANCEL = "cancel"              # 取消

@dataclass
class ParentChildMessage:
    """父子代理通信消息"""
    message_id: str                # 消息ID
    sender: str                    # 发送者："parent" 或 "child"
    receiver: str                  # 接收者
    message_type: MessageType      # 消息类型
    content: Dict[str, Any]        # 消息内容
    timestamp: float               # 时间戳
    correlation_id: Optional[str] = None  # 关联ID（用于请求-响应）
    priority: int = 1              # 优先级
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return {
            "message_id": self.message_id,
            "sender": self.sender,
            "receiver": self.receiver,
            "message_type": self.message_type.value,
            "content": self.content,
            "timestamp": self.timestamp,
            "correlation_id": self.correlation_id,
            "priority": self.priority
        }
    
    def to_json(self) -> str:
        """转换为JSON字符串"""
        return json.dumps(self.to_dict(), ensure_ascii=False)
    
    @classmethod
    def from_json(cls, json_str: str) -> "ParentChildMessage":
        """从JSON字符串创建消息"""
        data = json.loads(json_str)
        return cls(
            message_id=data["message_id"],
            sender=data["sender"],
            receiver=data["receiver"],
            message_type=MessageType(data["message_type"]),
            content=data["content"],
            timestamp=data["timestamp"],
            correlation_id=data.get("correlation_id"),
            priority=data.get("priority", 1)
        )

class CommunicationProtocol:
    """通信协议基类"""
    
    def __init__(self):
        self.message_handlers: Dict[MessageType, callable] = {}
        self.register_default_handlers()
    
    def register_default_handlers(self):
        """注册默认消息处理器"""
        self.message_handlers[MessageType.INSTRUCTION] = self.handle_instruction
        self.message_handlers[MessageType.RESULT] = self.handle_result
        self.message_handlers[MessageType.QUESTION] = self.handle_question
        self.message_handlers[MessageType.PROGRESS] = self.handle_progress
        self.message_handlers[MessageType.ERROR] = self.handle_error
    
    async def handle_message(self, message: ParentChildMessage) -> Optional[ParentChildMessage]:
        """处理消息"""
        handler = self.message_handlers.get(message.message_type)
        if handler:
            return await handler(message)
        else:
            print(f"⚠️ 未知消息类型: {message.message_type}")
            return None
    
    async def handle_instruction(self, message: ParentChildMessage) -> ParentChildMessage:
        """处理指令消息"""
        print(f"📨 收到指令: {message.content.get('task')}")
        # 执行指令并返回结果
        result = await self.execute_instruction(message.content)
        
        return ParentChildMessage(
            message_id=str(uuid.uuid4()),
            sender=message.receiver,  # 发送者是接收指令的一方
            receiver=message.sender,   # 接收者是指令发送方
            message_type=MessageType.RESULT,
            content={"result": result},
            timestamp=datetime.now().timestamp(),
            correlation_id=message.message_id  # 关联原始消息
        )
    
    async def handle_result(self, message: ParentChildMessage) -> None:
        """处理结果消息"""
        print(f"✅ 收到结果: {message.content.get('result')}")
        # 处理结果，可能更新任务状态等
        await self.process_result(message.content)
    
    async def handle_error(self, message: ParentChildMessage) -> Optional[ParentChildMessage]:
        """处理错误消息"""
        print(f"❌ 收到错误: {message.content.get('error')}")
        # 根据错误类型决定是否重试或放弃
        should_retry = await self.handle_error_strategy(message.content)
        
        if should_retry:
            return ParentChildMessage(
                message_id=str(uuid.uuid4()),
                sender=message.receiver,
                receiver=message.sender,
                message_type=MessageType.INSTRUCTION,
                content={"retry": True, "original_task": message.content},
                timestamp=datetime.now().timestamp(),
                correlation_id=message.correlation_id
            )
        return None
    
    async def execute_instruction(self, content: Dict[str, Any]) -> Any:
        """执行指令（子类实现）"""
        raise NotImplementedError
    
    async def process_result(self, content: Dict[str, Any]) -> None:
        """处理结果（子类实现）"""
        raise NotImplementedError
    
    async def handle_error_strategy(self, content: Dict[str, Any]) -> bool:
        """错误处理策略（子类实现）"""
        return False
```

class ProgressReportingProtocol(CommunicationProtocol):
    """支持进度报告的通信协议"""
    
    def __init__(self, progress_callback: Optional[callable] = None):
        super().__init__()
        self.progress_callback = progress_callback
        self.message_handlers[MessageType.PROGRESS] = self.handle_progress
    
    async def handle_progress(self, message: ParentChildMessage) -> None:
        """处理进度报告"""
        progress = message.content.get("progress", 0)
        status = message.content.get("status", "")
        
        print(f"📊 进度报告: {progress}% - {status}")
        
        # 调用进度回调函数
        if self.progress_callback:
            await self.progress_callback(progress, status, message.content)
    
    async def send_progress_update(self, receiver: str, progress: float, status: str, 
                                  details: Dict[str, Any] = None) -> ParentChildMessage:
        """发送进度更新"""
        message = ParentChildMessage(
            message_id=str(uuid.uuid4()),
            sender=self.agent_name,
            receiver=receiver,
            message_type=MessageType.PROGRESS,
            content={
                "progress": progress,
                "status": status,
                "details": details or {},
                "timestamp": datetime.now().timestamp()
            },
            timestamp=datetime.now().timestamp()
        )
        
        await self.send_message(message)
        return message

### 附录4：常见通信问题及解决方法
| 问题现象 | 可能原因 | 解决方法 |
|----------|----------|----------|
| 消息丢失 | 网络问题或队列满 | 添加消息确认机制，实现重发 |
| 响应超时 | 子代理处理过慢 | 设置合理超时，添加超时处理 |
| 消息乱序 | 异步并发处理 | 使用关联ID和顺序号保证顺序 |
| 内存泄漏 | 消息未及时清理 | 添加消息生命周期管理 |
| 性能瓶颈 | 序列化开销大 | 使用二进制协议或压缩 |
| 协议不兼容 | 版本升级 | 添加版本号，支持向后兼容 |
