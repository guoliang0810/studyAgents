### 附录3：动态加载代码示例
```python
import sys
import importlib
from pathlib import Path
from typing import Optional, Type

def load_skill(skill_name: str, skills_base_path: str = "/skills") -> Optional[Type]:
    """动态加载技能
    
    Args:
        skill_name: 技能名称，如 "calculator"
        skills_base_path: 技能基础路径
        
    Returns:
        技能类，如果加载失败返回None
    """
    try:
        # 构建技能路径
        skill_path = Path(skills_base_path) / skill_name
        
        # 安全检查：路径是否存在且为目录
        if not skill_path.exists():
            raise FileNotFoundError(f"技能路径不存在: {skill_path}")
        if not skill_path.is_dir():
            raise NotADirectoryError(f"技能路径不是目录: {skill_path}")
        
        # 路径遍历检查
        resolved_path = skill_path.resolve()
        base_path = Path(skills_base_path).resolve()
        if not str(resolved_path).startswith(str(base_path)):
            raise SecurityError(f"路径遍历尝试: {skill_path}")
        
        # 添加到Python路径（临时）
        sys.path.insert(0, str(skill_path.parent))
        
        try:
            # 动态导入技能模块
            module_name = f"skills.{skill_name}"
            module = importlib.import_module(module_name)
            
            # 获取技能类（约定技能类名为Skill）
            if hasattr(module, 'Skill'):
                skill_class = getattr(module, 'Skill')
                print(f"✅ 成功加载技能: {skill_name}")
                return skill_class
            else:
                raise AttributeError(f"模块中没有Skill类: {module_name}")
                
        finally:
            # 清理Python路径
            if str(skill_path.parent) in sys.path:
                sys.path.remove(str(skill_path.parent))
                
    except Exception as e:
        print(f"❌ 加载技能失败: {skill_name}, 错误: {e}")
        return None


def register_skill(skill_name: str, skill_class: Type, skill_registry: dict):
    """注册技能到系统
    
    Args:
        skill_name: 技能名称
        skill_class: 技能类
        skill_registry: 技能注册表
    """
    if skill_name in skill_registry:
        print(f"⚠️ 技能已存在: {skill_name}, 将覆盖")
    
    skill_registry[skill_name] = {
        'class': skill_class,
        'loaded_at': datetime.now(),
        'active': True
    }
    print(f"✅ 技能注册成功: {skill_name}")


# 使用示例
if __name__ == "__main__":
    # 加载计算器技能
    calculator_class = load_skill("calculator")
    
    if calculator_class:
        # 注册技能
        skill_registry = {}
        register_skill("calculator", calculator_class, skill_registry)
        
        # 使用技能
        skill_instance = calculator_class()
        result = skill_instance.add(5, 3)
        print(f"计算器技能测试: 5 + 3 = {result}")
```
