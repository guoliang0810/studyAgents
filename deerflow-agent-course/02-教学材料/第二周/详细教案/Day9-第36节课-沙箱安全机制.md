### 附录3：安全审计代码示例
```python
import json
import time
from dataclasses import dataclass, asdict
from datetime import datetime
from typing import Optional, Dict, List
import os

@dataclass
class SecurityAuditLog:
    """安全审计日志"""
    timestamp: str
    operation: str
    resource: str
    user_id: str
    success: bool
    details: Optional[Dict] = None
    risk_level: str = "low"  # low, medium, high
    
    def to_dict(self):
        return asdict(self)

class SecurityAuditor:
    """安全审计器"""
    
    def __init__(self, log_file: str = "security_audit.log"):
        self.log_file = log_file
        self._ensure_log_file()
    
    def _ensure_log_file(self):
        """确保日志文件存在"""
        log_dir = os.path.dirname(self.log_file)
        if log_dir and not os.path.exists(log_dir):
            os.makedirs(log_dir)
    
    def audit_sandbox(self, sandbox_config: dict) -> tuple[bool, List[str]]:
        """审计沙箱配置
        
        Args:
            sandbox_config: 沙箱配置字典
            
        Returns:
            (通过与否, 问题列表)
        """
        issues = []
        
        # 检查资源限制
        if not self._check_resource_limits(sandbox_config):
            issues.append("资源限制配置不足")
        
        # 检查路径隔离
        if not self._check_path_isolation(sandbox_config):
            issues.append("路径隔离配置不足")
        
        # 检查网络访问
        if not self._check_network_access(sandbox_config):
            issues.append("网络访问控制不足")
        
        # 检查系统调用过滤
        if not self._check_syscall_filters(sandbox_config):
            issues.append("系统调用过滤不足")
        
        # 记录审计结果
        self.log(
            user_id="system",
            operation="sandbox_audit",
            resource=sandbox_config.get('name', 'unknown'),
            success=len(issues) == 0,
            details={"issues": issues, "config": sandbox_config}
        )
        
        return len(issues) == 0, issues
    
    def _check_resource_limits(self, config: dict) -> bool:
        """检查资源限制"""
        required_limits = ['cpu', 'memory', 'pids']
        limits = config.get('resource_limits', {})
        
        for limit in required_limits:
            if limit not in limits:
                return False
        
        # 检查限制值是否合理
        if limits.get('memory', 0) > 1024 * 1024 * 1024:  # 1GB
            return False
        
        return True
    
    def _check_path_isolation(self, config: dict) -> bool:
        """检查路径隔离"""
        mounts = config.get('mounts', [])
        
        # 必须至少有一个只读挂载点
        has_readonly = any(mount.get('readonly', False) for mount in mounts)
        if not has_readonly:
            return False
        
        # 检查敏感路径防护
        sensitive_paths = ['/etc/passwd', '/etc/shadow', '/root']
        for mount in mounts:
            for sensitive in sensitive_paths:
                if sensitive.startswith(mount.get('target', '')):
                    if not mount.get('readonly', False):
                        return False
        
        return True
    
    def _check_network_access(self, config: dict) -> bool:
        """检查网络访问"""
        network = config.get('network', {})
        
        # 默认应该禁止网络访问，除非显式允许
        if network.get('enabled', False):
            # 如果有网络访问，必须有防火墙规则
            if not network.get('firewall_rules'):
                return False
        
        return True
    
    def _check_syscall_filters(self, config: dict) -> bool:
        """检查系统调用过滤"""
        seccomp = config.get('seccomp', {})
        
        # 必须启用seccomp
        if not seccomp.get('enabled', False):
            return False
        
        # 必须有策略文件或规则
        if not seccomp.get('policy') and not seccomp.get('rules'):
            return False
        
        return True
    
    def log(
        self,
        user_id: str,
        operation: str,
        resource: str,
        success: bool,
        details: Dict = None
    ):
        """记录安全审计日志"""
        entry = SecurityAuditLog(
            timestamp=datetime.now().isoformat(),
            operation=operation,
            resource=resource,
            user_id=user_id,
            success=success,
            details=details,
            risk_level=self._calculate_risk_level(operation, success)
        )
        
        with open(self.log_file, 'a') as f:
            f.write(json.dumps(entry.to_dict()) + '\n')
    
    def _calculate_risk_level(self, operation: str, success: bool) -> str:
        """计算风险等级"""
        high_risk_ops = ['file_write', 'network_connect', 'syscall_exec']
        
        if operation in high_risk_ops and not success:
            return "high"
        elif operation in high_risk_ops:
            return "medium"
        else:
            return "low"
    
    def query_logs(
        self,
        user_id: str = None,
        operation: str = None,
        risk_level: str = None,
        start_time: str = None,
        end_time: str = None
    ) -> List[SecurityAuditLog]:
        """查询审计日志"""
        logs = []
        
        if not os.path.exists(self.log_file):
            return logs
        
        with open(self.log_file, 'r') as f:
            for line in f:
                data = json.loads(line)
                entry = SecurityAuditLog(**data)
                
                # 应用过滤条件
                if user_id and entry.user_id != user_id:
                    continue
                if operation and entry.operation != operation:
                    continue
                if risk_level and entry.risk_level != risk_level:
                    continue
                
                logs.append(entry)
        
        return logs


# 使用示例
if __name__ == "__main__":
    auditor = SecurityAuditor()
    
    # 测试沙箱配置审计
    sandbox_config = {
        'name': 'test_sandbox',
        'resource_limits': {
            'cpu': 0.5,
            'memory': 512 * 1024 * 1024,  # 512MB
            'pids': 100
        },
        'mounts': [
            {'source': '/tmp/sandbox', 'target': '/app', 'readonly': True}
        ],
        'network': {'enabled': False},
        'seccomp': {'enabled': True, 'policy': 'default'}
    }
    
    passed, issues = auditor.audit_sandbox(sandbox_config)
    print(f"沙箱审计结果: 通过={passed}, 问题={issues}")
    
    # 查询高风险日志
    high_risk_logs = auditor.query_logs(risk_level="high")
    print(f"高风险日志数量: {len(high_risk_logs)}")
```

### 附录4：cgroups配置示例
```bash
#!/bin/bash
# cgroups配置脚本示例

# 创建cgroup
sudo cgcreate -g cpu,memory,pids:/sandbox_group

# 设置CPU限制（最多使用50%的CPU）
echo "50000" | sudo tee /sys/fs/cgroup/cpu/sandbox_group/cpu.cfs_quota_us
echo "100000" | sudo tee /sys/fs/cgroup/cpu/sandbox_group/cpu.cfs_period_us

# 设置内存限制（最大1GB）
echo "1G" | sudo tee /sys/fs/cgroup/memory/sandbox_group/memory.limit_in_bytes

# 设置进程数限制（最多100个进程）
```

### 附录5：seccomp策略示例
```python
import seccomp

# 创建seccomp过滤器
def create_seccomp_filter():
    """创建seccomp过滤器，允许基本系统调用"""
    
    filter = seccomp.SyscallFilter(defaction=seccomp.ALLOW)
    
    # 禁止危险系统调用
    dangerous_syscalls = [
        "execve", "fork", "clone", "kill", "ptrace",
        "mount", "umount", "chroot", "setuid", "setgid"
    ]
    
    for syscall in dangerous_syscalls:
        try:
            filter.add_rule(seccomp.KILL, syscall)
        except:
            pass  # 系统调用可能不存在
    
    # 限制网络相关系统调用
    network_syscalls = [
        "socket", "connect", "bind", "listen", "accept"
    ]
    
    for syscall in network_syscalls:
        try:
            filter.add_rule(seccomp.ERRNO(1), syscall)  # 返回错误
        except:
            pass
    
    # 允许必要的系统调用
    allowed_syscalls = [
        "read", "write", "open", "close", "fstat",
        "lseek", "mmap", "munmap", "mprotect", "brk",
        "rt_sigaction", "rt_sigprocmask", "exit_group"
    ]
    
    # 实际不需要特别允许，因为默认是ALLOW
    
    return filter

# 使用seccomp过滤器
def run_sandboxed(code):
    """在seccomp保护下运行代码"""
    
    # 创建并加载过滤器
    filter = create_seccomp_filter()
    filter.load()
    
    try:
        # 执行代码
        exec(code)
    except Exception as e:
        print(f"执行失败: {e}")
    finally:
        # 注意：seccomp过滤器一旦加载就不能卸载
        pass

# 测试
if __name__ == "__main__":
    test_code = """
print("Hello from sandbox")
# 尝试危险操作（会被阻止）
import os
try:
    os.system("ls")  # 需要execve，会被阻止
except Exception as e:
    print(f"危险操作被阻止: {e}")
"""
    
    run_sandboxed(test_code)
```
