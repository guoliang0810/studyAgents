```python
class ConfigFileWatcher:
    """配置文件监视器"""

class EnhancedMemoryConfig(MemoryConfig):
    """增强的内存配置类"""
    
    def __init__(self):
        super().__init__()
        self.validators: Dict[str, Callable[[Any], bool]] = {}
        self.required_keys: List[str] = []
        self.watcher = ConfigFileWatcher(self)
        
    def add_validator(self, key: str, validator: Callable[[Any], bool]):
        """添加配置验证器"""
        self.validators[key] = validator
        
    def set_required(self, keys: List[str]):
        """设置必填配置项"""
        self.required_keys = keys
        
    def validate(self) -> List[str]:
        """验证配置，返回错误列表"""
        errors = []
        
        # 检查必填项
        for key in self.required_keys:
            if key not in self._configs and key not in self._defaults:
                errors.append(f"缺少必填配置项: {key}")
        
        # 检查验证器
        for key, validator in self.validators.items():
            value = self.get(key)
            if value is not None and not validator(value):
                errors.append(f"配置项 {key} 验证失败: {value}")
        
        return errors
    
    def load_from_file(self, filepath: str, format: str = "auto"):
        """从文件加载配置"""
        if format == "auto":
            if filepath.endswith(".yaml") or filepath.endswith(".yml"):
                format = "yaml"
            elif filepath.endswith(".json"):
                format = "json"
        
        with open(filepath, 'r', encoding='utf-8') as f:
            if format == "yaml":
                data = yaml.safe_load(f)
            elif format == "json":
                data = json.load(f)
            else:
                raise ValueError(f"不支持的格式: {format}")
        
        # 转换为配置源
        for key, value in data.items():
            self.set(key, value, ConfigSource.FILE)
        
        # 添加文件监视
        def reload_callback():
            print(f"配置文件 {filepath} 已修改，重新加载...")
            self.load_from_file(filepath, format)
            self.reload(ConfigSource.FILE)
        
        self.watcher.watch_file(filepath, reload_callback)
        
    def start_watching(self):
        """启动配置文件监视"""
        self.watcher.start()
```

**关键点解析**：
1. **热重载机制**：使用watchdog库监控文件变化，文件修改时自动回调
2. **格式支持**：支持YAML和JSON两种常用配置文件格式
3. **验证机制**：可自定义验证器，确保配置值的有效性
4. **错误处理**：验证失败时返回详细错误信息
5. **自动检测**：根据文件后缀自动识别格式

### 练习2：事实生命周期优化

**参考实现**：

```python
class EnhancedFactLifecycleManager(FactLifecycleManager):
    """增强的事实生命周期管理器"""
    
    def __init__(self, config: MemoryConfig):
        super().__init__(config)
        self.usage_counts: Dict[str, int] = {}  # 事实ID -> 使用次数
        self.last_used: Dict[str, float] = {}   # 事实ID -> 最后使用时间
        self.importance_scores: Dict[str, float] = {}  # 事实ID -> 重要性分数
        
    def record_usage(self, fact_id: str):
        """记录事实使用"""
        self.usage_counts[fact_id] = self.usage_counts.get(fact_id, 0) + 1
        self.last_used[fact_id] = time.time()
        self._update_importance(fact_id)
        
    def _update_importance(self, fact_id: str):
        """更新重要性分数"""
        if fact_id not in self._facts:
            return
            
        fact = self._facts[fact_id]
        age_days = (time.time() - fact.created_at) / (24 * 3600)
        
        # 年龄因子：越新越重要
        age_factor = max(0.1, 1.0 / (1 + age_days * 0.1))
        
        # 使用频率因子
        usage_count = self.usage_counts.get(fact_id, 1)
        usage_factor = min(1.0, usage_count / 10.0)  # 上限10次
        
        # 计算重要性
        importance = fact.confidence * age_factor * usage_factor
        self.importance_scores[fact_id] = importance
        
    def get_important_facts(self, limit: int = 10) -> List[Fact]:
        """获取最重要的事实"""
        sorted_ids = sorted(
            self.importance_scores.items(),
            key=lambda x: x[1],
            reverse=True
        )[:limit]
        
        return [self._facts[fact_id] for fact_id, _ in sorted_ids if fact_id in self._facts]
    
    def find_related_facts(self, fact_id: str) -> List[Fact]:
        """查找相关事实"""
        if fact_id not in self._facts:
            return []
            
        target = self._facts[fact_id]
        related = []
        
        for other_id, other_fact in self._facts.items():
            if other_id == fact_id:
                continue
                
            # 计算相关性分数
            similarity = self._calculate_similarity(target, other_fact)
            if similarity > 0.5:  # 相似度阈值
                related.append(other_fact)
                
        return related
    
    def _calculate_similarity(self, fact1: Fact, fact2: Fact) -> float:
        """计算事实相似度"""
        # 标签相似度
        tag_similarity = len(set(fact1.tags) & set(fact2.tags)) / max(1, len(set(fact1.tags) | set(fact2.tags)))
        
        # 来源相似度
        source_similarity = 1.0 if fact1.source == fact2.source else 0.0
        
        # 内容相似度（简化的Jaccard相似度）
        words1 = set(fact1.content.lower().split())
        words2 = set(fact2.content.lower().split())
        content_similarity = len(words1 & words2) / max(1, len(words1 | words2))
        
        # 加权平均
        return 0.3 * tag_similarity + 0.2 * source_similarity + 0.5 * content_similarity
    
    def smart_archive(self, threshold: float = 0.3):
        """智能归档"""
        to_archive = []
        
        for fact_id, fact in self._facts.items():
            if fact.status == FactStatus.ACTIVE and not fact.is_expired():
                importance = self.importance_scores.get(fact_id, 0.5)
                if importance < threshold:
                    self.update_status(fact_id, FactStatus.ARCHIVED, f"重要性过低: {importance:.2f}")
                    to_archive.append(fact_id)
        
        return to_archive
```

**关键点解析**：
1. **重要性计算**：结合置信度、年龄、使用频率的综合评分
2. **相关性分析**：基于标签、来源、内容的相似度计算
3. **智能归档**：根据重要性阈值自动归档低价值事实
4. **使用追踪**：记录事实使用情况，用于优化重要性计算

### 练习3：记忆注入策略扩展

**参考实现**：

```python
class AdaptiveMemoryInjectionMiddleware(MemoryInjectionMiddleware):
    """自适应记忆注入中间件"""
    
    def __init__(self, config: MemoryConfig, lifecycle_manager: FactLifecycleManager):
        super().__init__(config, lifecycle_manager)
        self.strategies = {
            "exact_match": self._exact_match_strategy,
            "semantic_match": self._semantic_match_strategy,
            "contextual": self._contextual_strategy,
            "adaptive": self._adaptive_strategy
        }
        self.strategy_performance: Dict[str, Dict[str, float]] = {}
        self.query_analyzer = QueryAnalyzer()
        
    def inject(self, query: str, context: Dict[str, Any] = None) -> str:
        """自适应注入"""
        # 分析查询复杂度
        complexity = self.query_analyzer.analyze_complexity(query)
        
        # 选择策略
        strategy = self._select_strategy(complexity, context)
        
        # 执行注入
        start_time = time.time()
        result = self.strategies[strategy](query, context)
        elapsed = time.time() - start_time
        
        # 记录性能（简化版）
        if strategy not in self.strategy_performance:
            self.strategy_performance[strategy] = {"total_time": 0, "count": 0}
        
        self.strategy_performance[strategy]["total_time"] += elapsed
        self.strategy_performance[strategy]["count"] += 1
        
        return result
    
    def _select_strategy(self, complexity: float, context: Dict[str, Any]) -> str:
        """选择注入策略"""
        # 简单查询使用精确匹配
        if complexity < 0.3:
            return "exact_match"
        # 中等复杂度使用语义匹配
        elif complexity < 0.6:
            return "semantic_match"
        # 复杂查询使用上下文感知
        elif complexity < 0.8:
            return "contextual"
        # 非常复杂的查询使用自适应策略
        else:
            return "adaptive"
    
    def _exact_match_strategy(self, query: str, context: Dict[str, Any]) -> str:
        """精确匹配策略"""
        # 简化实现：关键词完全匹配
        words = query.lower().split()
        relevant = []
        
        for fact in self.lifecycle_manager._facts.values():
            if fact.status == FactStatus.ACTIVE and not fact.is_expired():
                content_lower = fact.content.lower()
                if any(word in content_lower for word in words):
                    relevant.append(fact)
        
        return self._build_enhanced_query(query, relevant[:3], context)
    
    def _semantic_match_strategy(self, query: str, context: Dict[str, Any]) -> str:
        """语义匹配策略"""
        # 简化实现：扩展匹配范围
        return super()._build_enhanced_query(
            query, 
            self._extract_relevant_facts(query, context),
            context
        )
    
    def _contextual_strategy(self, query: str, context: Dict[str, Any]) -> str:
        """上下文感知策略"""
        # 结合上下文信息
        all_facts = []
        
        # 查询相关事实
        query_facts = self._extract_relevant_facts(query, context)
        all_facts.extend(query_facts)
        
        # 上下文相关事实
        if context and "previous_queries" in context:
            for prev_query in context["previous_queries"][-3:]:
                prev_facts = self._extract_relevant_facts(prev_query, {})
                all_facts.extend(prev_facts)
        
        # 去重
        seen_ids = set()
        unique_facts = []
        for fact in all_facts:
            if fact.id not in seen_ids:
                seen_ids.add(fact.id)
                unique_facts.append(fact)
        
        return self._build_enhanced_query(query, unique_facts[:5], context)
    
    def _adaptive_strategy(self, query: str, context: Dict[str, Any]) -> str:
        """自适应策略"""
        # 基于历史性能选择最佳策略
        best_strategy = "semantic_match"
        best_avg_time = float('inf')
        
        for strategy, perf in self.strategy_performance.items():
            if perf["count"] > 0:
                avg_time = perf["total_time"] / perf["count"]
                if avg_time < best_avg_time:
                    best_avg_time = avg_time
                    best_strategy = strategy
        
        return self.strategies[best_strategy](query, context)

class QueryAnalyzer:
    """查询分析器"""
    
    def analyze_complexity(self, query: str) -> float:
        """分析查询复杂度"""
        # 长度因子
        length_factor = min(1.0, len(query) / 100.0)
        
        # 关键词数量
        words = query.split()
        keyword_factor = min(1.0, len(words) / 20.0)
        
        # 句子结构（简化：标点符号数量）
        punctuation_count = sum(1 for c in query if c in ".?!,;:")
        structure_factor = min(1.0, punctuation_count / 5.0)
        
        # 综合复杂度
        complexity = 0.4 * length_factor + 0.3 * keyword_factor + 0.3 * structure_factor
        return min(1.0, complexity)
```

**关键点解析**：
1. **策略选择**：根据查询复杂度自动选择最佳注入策略
2. **性能追踪**：记录各策略的执行时间，用于自适应优化
3. **查询分析**：分析查询长度、关键词、结构等特征
4. **上下文利用**：结合历史查询，提供更相关的记忆注入
