**参考实现**：

```python
from abc import ABC, abstractmethod
from typing import Dict, Any, List
import time

# 基础模板类
class BaseTemplate(ABC):
    @abstractmethod
    def render(self, context: Dict[str, Any]) -> str:
        pass

# 1. 继承扩展
class InheritedMemoryTemplate(BaseTemplate):
    """继承扩展：通过继承添加记忆功能"""
    
    def __init__(self, template: str):
        super().__init__()
        self.template = template
        self.memories: List[Dict] = []
        self.cache: Dict[str, str] = {}
    
    def add_memory(self, content: str, source: str, confidence: float = 1.0):
        """添加记忆"""
        self.memories.append({
            "content": content,
            "source": source,
            "confidence": confidence,
            "timestamp": time.time()
        })
    
    def search_memories(self, query: str, limit: int = 3) -> List[Dict]:
        """搜索相关记忆"""
        # 简化搜索：基于关键词匹配
        results = []
        for mem in self.memories:
            if query.lower() in mem["content"].lower():
                results.append(mem)
        return results[:limit]
    
    def render(self, context: Dict[str, Any]) -> str:
        """渲染模板（带记忆注入）"""
        # 从上下文提取查询
        query = context.get("query", "")
        
        # 搜索相关记忆
        memories = self.search_memories(query)
        
        # 构建上下文
        context_text = ""
        if memories:
            context_text = "相关记忆：\n"
            for i, mem in enumerate(memories, 1):
                context_text += f"{i}. {mem['content']} (来源: {mem['source']})\n"
            context_text += "\n"
        
        # 渲染模板
        rendered = self.template
        for key, value in context.items():
            placeholder = f"{{{{{key}}}}}"
            rendered = rendered.replace(placeholder, str(value))
        
        return context_text + rendered

# 2. 组合扩展
class ComposedMemoryTemplate:
    """组合扩展：通过组合添加记忆功能"""
    
    def __init__(self, base_template: BaseTemplate):
        self.base_template = base_template
        self.memory_store = MemoryStore()
        self.injector = ContextInjector()
    
    def add_memory(self, content: str, source: str, confidence: float = 1.0):
        """添加记忆"""
        self.memory_store.add(content, source, confidence)
    
    def render(self, context: Dict[str, Any]) -> str:
        """渲染模板"""
        # 先渲染基础模板
        base_result = self.base_template.render(context)
        
        # 搜索并注入记忆
        query = context.get("query", "")
        memories = self.memory_store.search(query)
        final_result = self.injector.inject(base_result, memories)
        
        return final_result

# 3. 装饰器扩展
def memory_injection_decorator(template_func):
    """装饰器扩展：通过装饰器添加记忆功能"""
    
    def wrapper(context: Dict[str, Any], memory_store=None, *args, **kwargs):
        # 执行原始模板函数
        result = template_func(context, *args, **kwargs)
        
        # 如果提供了记忆存储，注入记忆
        if memory_store:
            query = context.get("query", "")
            memories = memory_store.search(query)
            if memories:
                context_text = "\n相关上下文：\n"
                for mem in memories:
                    context_text += f"- {mem['content']}\n"
                result = context_text + "\n" + result
        
        return result
    
    return wrapper

# 使用装饰器
@memory_injection_decorator
def render_template(context: Dict[str, Any]) -> str:
    """基础模板函数"""
    template = "问题：{{query}}\n回答：{{answer}}"
    for key, value in context.items():
        template = template.replace(f"{{{{{key}}}}}", str(value))
    return template

# 辅助类
class MemoryStore:
    """记忆存储"""
    
    def __init__(self):
        self.memories = []
    
    def add(self, content: str, source: str, confidence: float):
        self.memories.append({
            "content": content,
            "source": source,
            "confidence": confidence
        })
    
    def search(self, query: str) -> List[Dict]:
        return [m for m in self.memories if query.lower() in m["content"].lower()][:3]

class ContextInjector:
    """上下文注入器"""
    
    def inject(self, text: str, memories: List[Dict]) -> str:
        if not memories:
            return text
        
        context = "相关上下文：\n"
        for mem in memories:
            context += f"- {mem['content']} (来源: {mem['source']})\n"
        
        return context + "\n" + text
```

**对比分析报告**：

| 扩展方法 | 优点 | 缺点 | 适用场景 |
|----------|------|------|----------|
| **继承扩展** | 1. 代码结构清晰，逻辑集中<br>2. 便于理解和维护<br>3. 类型安全，IDE支持好 | 1. 多重继承可能导致复杂度<br>2. 父类修改可能影响子类<br>3. 灵活性相对较差 | 1. 功能扩展明确且稳定<br>2. 需要重用父类大部分功能<br>3. 团队熟悉面向对象设计 |
| **组合扩展** | 1. 灵活性高，可动态组合<br>2. 松耦合，便于测试<br>3. 可复用现有组件 | 1. 代码量可能较多<br>2. 需要管理多个组件关系<br>3. 接口设计需要精心考虑 | 1. 需要动态配置功能<br>2. 系统组件需要独立演化<br>3. 需要支持多种配置组合 |
| **装饰器扩展** | 1. 非侵入式，不改动原代码<br>2. 可叠加多个装饰器<br>3. 运行时动态添加功能 | 1. 调试和理解难度较大<br>2. 可能影响性能<br>3. 装饰器顺序敏感 | 1. 需要为现有函数添加功能<br>2. 功能需要可插拔<br>3. AOP（面向切面编程）场景 |

**推荐选择**：
- **简单项目**：继承扩展，结构清晰易于维护
- **复杂系统**：组合扩展，灵活支持多种配置
- **框架开发**：装饰器扩展，提供非侵入式扩展点

### 练习2：上下文注入策略优化

**参考实现**：

```python
import time
from typing import List, Dict, Any
from dataclasses import dataclass
from enum import Enum

@dataclass
class Memory:
    content: str
    source: str
    confidence: float
    timestamp: float
    tags: List[str]
    
    def age_in_days(self) -> float:
        """计算记忆年龄（天）"""
        return (time.time() - self.timestamp) / (24 * 3600)

class InjectionStrategy(Enum):
    PREPEND = "prepend"
    APPEND = "append"
    INTERLEAVE = "interleave"
    CONDITIONAL = "conditional"
    SMART = "smart"  # 智能策略

class SmartContextInjector:
    """智能上下文注入器"""
    
    def __init__(self, decay_rate: float = 0.1):
        self.decay_rate = decay_rate
        self.strategy_weights = {
            InjectionStrategy.PREPEND: 0.3,
            InjectionStrategy.APPEND: 0.2,
            InjectionStrategy.INTERLEAVE: 0.1,
            InjectionStrategy.CONDITIONAL: 0.4
        }
    
    def calculate_relevance(self, memory: Memory, query: str) -> float:
        """计算相关性"""
        # 关键词匹配
        query_words = set(query.lower().split())
        content_words = set(memory.content.lower().split())
        
        if not query_words:
            return 0.0
        
        # Jaccard相似度
        intersection = len(query_words & content_words)
        union = len(query_words | content_words)
        keyword_similarity = intersection / union if union > 0 else 0
        
        # 标签匹配加分
        tag_bonus = 0
        for tag in memory.tags:
            if tag.lower() in query.lower():
                tag_bonus += 0.1
        
        return min(1.0, keyword_similarity + tag_bonus)
    
    def calculate_freshness(self, memory: Memory) -> float:
        """计算新鲜度"""
        age = memory.age_in_days()
        freshness = 1.0 / (1.0 + age * self.decay_rate)
        return freshness
    
    def calculate_score(self, memory: Memory, query: str) -> float:
        """计算综合评分"""
        relevance = self.calculate_relevance(memory, query)
        freshness = self.calculate_freshness(memory)
        confidence = memory.confidence
        
        # 加权综合
        score = (
            relevance * 0.5 +    # 相关性最重要
            freshness * 0.3 +    # 新鲜度次之
            confidence * 0.2     # 置信度作为补充
        )
        
        return score
    
    def select_memories(self, memories: List[Memory], query: str, 
                       max_count: int = 5, min_score: float = 0.3) -> List[Memory]:
        """选择高质量记忆"""
        scored_memories = []
        
        for mem in memories:
            score = self.calculate_score(mem, query)
            if score >= min_score:
                scored_memories.append((mem, score))
        
        # 按评分排序
        scored_memories.sort(key=lambda x: x[1], reverse=True)
        
        # 选择前N个
        selected = [mem for mem, score in scored_memories[:max_count]]
        return selected
    
    def select_strategy(self, memories: List[Memory], query: str) -> InjectionStrategy:
        """选择注入策略"""
        if not memories:
            return InjectionStrategy.CONDITIONAL
        
        # 计算平均质量
        avg_score = sum(self.calculate_score(m, query) for m in memories) / len(memories)
        
        if avg_score > 0.7 and len(memories) >= 3:
            # 高质量多记忆 → 前置注入
            return InjectionStrategy.PREPEND
        elif avg_score > 0.5:
            # 中等质量 → 后置注入
            return InjectionStrategy.APPEND
        elif avg_score > 0.3 and len(memories) <= 2:
            # 低质量少记忆 → 交错注入
            return InjectionStrategy.INTERLEAVE
        else:
            # 质量过低 → 条件注入（可能不注入）
            return InjectionStrategy.CONDITIONAL
    
    def format_context(self, memories: List[Memory], format_type: str = "list") -> str:
        """格式化上下文"""
        if not memories:
            return ""
        
        if format_type == "list":
            # 列表式
            lines = ["相关上下文："]
            for i, mem in enumerate(memories, 1):
                lines.append(f"{i}. {mem.content} (来源: {mem.source}, 置信度: {mem.confidence:.2f})")
            return "\n".join(lines)
        
        elif format_type == "summary":
            # 摘要式
            contents = [mem.content for mem in memories]
            summary = "，".join(contents[:3])
            if len(contents) > 3:
                summary += f"等{len(contents)}条相关信息"
            return f"根据以下信息：{summary}"
        
        elif format_type == "dialogue":
            # 对话式
            lines = ["历史信息："]
            for mem in memories:
                lines.append(f"- {mem.source}: {mem.content}")
            return "\n".join(lines)
        
        else:
            return "\n".join([mem.content for mem in memories])
    
    def inject(self, prompt: str, memories: List[Memory], query: str) -> str:
        """智能注入"""
        # 选择记忆
        selected = self.select_memories(memories, query)
        
        # 选择策略
        strategy = self.select_strategy(selected, query)
        
        # 选择格式（基于策略）
        if strategy == InjectionStrategy.PREPEND:
            format_type = "list"
        elif strategy == InjectionStrategy.APPEND:
            format_type = "summary"
        elif strategy == InjectionStrategy.INTERLEAVE:
            format_type = "dialogue"
        else:
            format_type = "list"
        
        # 格式化上下文
        context = self.format_context(selected, format_type)
        
        # 根据策略组合
        if strategy == InjectionStrategy.PREPEND:
            return f"{context}\n\n{prompt}"
        elif strategy == InjectionStrategy.APPEND:
            return f"{prompt}\n\n{context}"
        elif strategy == InjectionStrategy.INTERLEAVE:
            lines = prompt.split('\n')
            if len(lines) > 1:
                mid = len(lines) // 2
                return '\n'.join(lines[:mid] + [context] + lines[mid:])
            else:
                return f"{context}\n\n{prompt}"
        elif strategy == InjectionStrategy.CONDITIONAL:
            # 只有高质量记忆才注入
            if selected and self.calculate_score(selected[0], query) > 0.5:
                return f"{context}\n\n{prompt}"
            else:
                return prompt
        else:
            return prompt
```

**关键优化点**：
1. **综合评分系统**：结合相关性、新鲜度、置信度三个维度
2. **策略选择算法**：基于记忆质量和数量动态选择最佳注入策略
3. **多样化格式**：支持列表式、摘要式、对话式等多种上下文格式
4. **质量阈值控制**：设置最低质量阈值，过滤低质量记忆

### 练习3：模板变量系统增强

**参考实现**：

```python
import ast
from typing import Dict, Any, List, Optional, Callable, Set
from enum import Enum
from collections import defaultdict

class VariableType(Enum):
    STRING = "string"
    INTEGER = "integer"
    FLOAT = "float"
    BOOLEAN = "boolean"
    LIST = "list"
    DICT = "dict"
    ANY = "any"

class VariableScope(Enum):
    GLOBAL = "global"
    LOCAL = "local"
    TEMPORARY = "temporary"

class EnhancedTemplateVariableSystem:
    """增强的模板变量系统"""
    
    def __init__(self):
        self.variables: Dict[str, Dict] = {}
        self.values: Dict[str, Any] = {}
        self.scopes: Dict[str, VariableScope] = {}
        self.dependencies: Dict[str, Set[str]] = defaultdict(set)
        self.reverse_deps: Dict[str, Set[str]] = defaultdict(set)
        self.validators: Dict[str, List[Callable]] = defaultdict(list)
        self.converters: Dict[VariableType, Callable] = {
            VariableType.STRING: str,
            VariableType.INTEGER: int,
            VariableType.FLOAT: float,
            VariableType.BOOLEAN: bool,
            VariableType.LIST: list,
            VariableType.DICT: dict
        }
    
    def register_variable(self, name: str, var_type: VariableType,
                         default: Any = None, scope: VariableScope = VariableScope.LOCAL,
                         depends_on: List[str] = None):
        """注册变量（支持依赖关系）"""
        self.variables[name] = {
            "type": var_type,
            "default": default,
            "scope": scope
        }
        self.scopes[name] = scope
        
        # 设置依赖关系
        if depends_on:
            for dep in depends_on:
                self.dependencies[name].add(dep)
                self.reverse_deps[dep].add(name)
    
    def add_validator(self, name: str, validator: Callable[[Any], bool]):
        """添加验证器"""
        self.validators[name].append(validator)
    
    def set_value(self, name: str, value: Any, scope: VariableScope = None,
                 validate: bool = True, convert: bool = True):
        """设置变量值"""
        if name not in self.variables:
            # 自动注册
            self.register_variable(
                name, VariableType.ANY, scope=scope or VariableScope.TEMPORARY
            )
        
        var_info = self.variables[name]
        
        # 类型转换
        if convert and var_info["type"] != VariableType.ANY:
            converter = self.converters.get(var_info["type"])
            if converter:
                try:
                    value = converter(value)
                except (ValueError, TypeError):
                    raise ValueError(f"变量 {name} 类型转换失败: {value} -> {var_info['type']}")
        
        # 验证
        if validate:
            errors = self._validate_value(name, value)
            if errors:
                raise ValueError(f"变量 {name} 验证失败: {errors}")
        
        # 设置值
        self.values[name] = value
        
        # 更新依赖此变量的其他变量
        self._update_dependents(name)
    
    def _validate_value(self, name: str, value: Any) -> List[str]:
        """验证变量值"""
        errors = []
        
        if name in self.variables:
            var_info = self.variables[name]
            
            # 类型检查
            if var_info["type"] != VariableType.ANY:
                expected_type = var_info["type"].value
                if var_info["type"] == VariableType.INTEGER:
                    if not isinstance(value, int):
                        errors.append(f"应为整数类型，实际为 {type(value).__name__}")
                elif var_info["type"] == VariableType.FLOAT:
                    if not isinstance(value, (int, float)):
                        errors.append(f"应为浮点数类型，实际为 {type(value).__name__}")
                elif var_info["type"] == VariableType.BOOLEAN:
                    if not isinstance(value, bool):
                        errors.append(f"应为布尔类型，实际为 {type(value).__name__}")
                elif var_info["type"] == VariableType.LIST:
                    if not isinstance(value, list):
                        errors.append(f"应为列表类型，实际为 {type(value).__name__}")
                elif var_info["type"] == VariableType.DICT:
                    if not isinstance(value, dict):
                        errors.append(f"应为字典类型，实际为 {type(value).__name__}")
            
            # 自定义验证器
            for validator in self.validators.get(name, []):
                if not validator(value):
                    errors.append("自定义验证器失败")
        
        return errors
    
    def _update_dependents(self, changed_var: str):
        """更新依赖变量"""
        if changed_var not in self.reverse_deps:
            return
        
        for dep_name in self.reverse_deps[changed_var]:
            # 检查依赖是否满足
            if self._check_dependencies(dep_name):
                # 重新计算表达式（如果有）
                if hasattr(self, f"_compute_{dep_name}"):
                    compute_func = getattr(self, f"_compute_{dep_name}")
                    new_value = compute_func()
                    self.set_value(dep_name, new_value, validate=False, convert=False)
    
    def _check_dependencies(self, var_name: str) -> bool:
        """检查变量依赖是否满足"""
        if var_name not in self.dependencies:
            return True
        
        for dep in self.dependencies[var_name]:
            if dep not in self.values:
                return False
        
        return True
    
    def get_value(self, name: str, default: Any = None, 
                 scope_filter: Optional[VariableScope] = None) -> Any:
        """获取变量值（支持作用域过滤）"""
        if name not in self.values:
            return default
        
        if scope_filter and self.scopes.get(name) != scope_filter:
            return default
        
        return self.values[name]
    
    def get_values_by_scope(self, scope: VariableScope) -> Dict[str, Any]:
        """按作用域获取变量值"""
        result = {}
        for name, value in self.values.items():
            if self.scopes.get(name) == scope:
                result[name] = value
        return result
    
    def interpolate(self, text: str, 
                   scope_filter: Optional[VariableScope] = None) -> str:
        """插值变量（支持作用域过滤）"""
        result = text
        
        # 提取所有变量名
        import re
        pattern = r"\{\{(\w+)\}\}"
        matches = re.findall(pattern, text)
        
        for var_name in matches:
            value = self.get_value(var_name, scope_filter=scope_filter)
            if value is not None:
                placeholder = f"{{{{{var_name}}}}}"
                result = result.replace(placeholder, str(value))
        
        return result
    
    def clear_scope(self, scope: VariableScope):
        """清空指定作用域的变量"""
        to_remove = []
        for name in self.values:
            if self.scopes.get(name) == scope:
                to_remove.append(name)
        
        for name in to_remove:
            del self.values[name]
            if name in self.scopes:
                del self.scopes[name]
    
    def validate_all(self) -> Dict[str, List[str]]:
        """验证所有变量"""
        errors = {}
        
        for name in self.variables:
            value = self.values.get(name)
            if value is not None:
                var_errors = self._validate_value(name, value)
                if var_errors:
                    errors[name] = var_errors
        
        return errors
```
