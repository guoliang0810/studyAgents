```python
class ConfigVersionManager:
    """配置版本管理器"""
    
    def __init__(self, max_versions: int = 100):
        self.max_versions = max_versions
        self.versions = []  # 按时间排序的版本列表
        self.version_index = {}  # 版本ID到版本的映射
        
    def create_version(self, config_snapshot: Dict[str, Any], 
                      author: str, reason: str) -> str:
        """创建新版本"""
        version_id = self._generate_version_id()
        version = ConfigVersion(
            id=version_id,
            timestamp=time.time(),
            author=author,
            reason=reason,
            snapshot=copy.deepcopy(config_snapshot),
            checksum=self._calculate_checksum(config_snapshot)
        )
        
        # 添加到版本列表
        self.versions.append(version)
        self.version_index[version_id] = version
        
        # 清理旧版本
        self._cleanup_old_versions()
        
        return version_id
    
    def get_version(self, version_id: str) -> Optional[ConfigVersion]:
        """获取指定版本"""
        return self.version_index.get(version_id)
    
    def rollback(self, version_id: str) -> bool:
        """回滚到指定版本"""
        version = self.get_version(version_id)
        if not version:
            return False
        
        # 应用版本快照
        self.config.apply_snapshot(version.snapshot)
        
        # 记录回滚操作
        self.create_version(
            version.snapshot,
            author="system",
            reason=f"Rollback to version {version_id}"
        )
        
        return True
    
    def diff(self, version_id1: str, version_id2: str) -> Dict[str, Any]:
        """对比两个版本差异"""
        v1 = self.get_version(version_id1)
        v2 = self.get_version(version_id2)
        
        if not v1 or not v2:
            return {}
        
        return self._calculate_diff(v1.snapshot, v2.snapshot)
    
    def _generate_version_id(self) -> str:
        """生成版本ID（时间戳+随机数）"""
        timestamp = int(time.time() * 1000)
        random_str = ''.join(random.choices('abcdef0123456789', k=8))
        return f"v{timestamp}_{random_str}"
    
    def _calculate_checksum(self, snapshot: Dict[str, Any]) -> str:
        """计算配置快照的校验和"""
        snapshot_str = json.dumps(snapshot, sort_keys=True)
        return hashlib.sha256(snapshot_str.encode()).hexdigest()
    
    def _calculate_diff(self, snapshot1: Dict[str, Any], 
                       snapshot2: Dict[str, Any]) -> Dict[str, Any]:
        """计算两个快照的差异"""
        diff_result = {
            "added": {},
            "removed": {},
            "modified": {}
        }
        
        # 找出新增和修改的配置项
        for key, value2 in snapshot2.items():
            if key not in snapshot1:
                diff_result["added"][key] = value2
            elif snapshot1[key] != value2:
                diff_result["modified"][key] = {
                    "old": snapshot1[key],
                    "new": value2
                }
        
        # 找出删除的配置项
        for key, value1 in snapshot1.items():
            if key not in snapshot2:
                diff_result["removed"][key] = value1
        
        return diff_result
    
    def _cleanup_old_versions(self):
        """清理旧版本，保持最多max_versions个版本"""
        if len(self.versions) <= self.max_versions:
            return
        
        # 删除最旧的版本
        while len(self.versions) > self.max_versions:
            old_version = self.versions.pop(0)
            del self.version_index[old_version.id]
```

```python
class ConfigTemplateEngine:
    """配置模板引擎"""
    
    def __init__(self):
        self.env = jinja2.Environment(
            loader=jinja2.DictLoader({}),
            autoescape=False,
            trim_blocks=True,
            lstrip_blocks=True
        )
        
        # 注册自定义过滤器
        self.env.filters['to_json'] = self._to_json_filter
        self.env.filters['to_yaml'] = self._to_yaml_filter
        self.env.filters['encrypt'] = self._encrypt_filter
        
    def render_template(self, template_str: str, context: Dict[str, Any]) -> Dict[str, Any]:
        """渲染模板字符串为配置字典"""
        try:
            template = self.env.from_string(template_str)
            rendered = template.render(**context)
            
            # 解析为配置字典
            if template_str.strip().startswith('{'):
                return json.loads(rendered)
            else:
                return yaml.safe_load(rendered)
        except Exception as e:
            raise ConfigTemplateError(f"模板渲染失败: {e}")
    
    def load_template(self, template_path: str) -> Template:
        """加载模板文件"""
        with open(template_path, 'r', encoding='utf-8') as f:
            template_content = f.read()
        
        return self.env.from_string(template_content)
    
    def register_template(self, name: str, template_str: str):
        """注册命名模板"""
        self.env.loader.mapping[name] = template_str
    
    def _to_json_filter(self, value: Any) -> str:
        """转换为JSON字符串的过滤器"""
        return json.dumps(value, ensure_ascii=False)
    
    def _to_yaml_filter(self, value: Any) -> str:
        """转换为YAML字符串的过滤器"""
        return yaml.dump(value, default_flow_style=False, allow_unicode=True)
    
    def _encrypt_filter(self, value: str, key_name: str = 'default') -> str:
        """加密值的过滤器"""
        encryption_key = self._get_encryption_key(key_name)
        encrypted = self._encrypt_value(value, encryption_key)
        return f"encrypted:{encrypted}"
```

```python
class ConfigTemplateSystem:
    """配置模板系统"""
    
    def __init__(self, template_dir: str):
        self.template_dir = Path(template_dir)
        self.template_engine = ConfigTemplateEngine()
        self.template_cache = {}
        
    def generate_config(self, base_template: str, 
                       environment: str = 'development',
                       overrides: Dict[str, Any] = None) -> Dict[str, Any]:
        """生成配置"""
        
        # 加载基础模板
        base_content = self._load_template(base_template)
        
        # 加载环境特定模板
        env_template = f"{base_template}.{environment}"
        env_content = None
        if self._template_exists(env_template):
            env_content = self._load_template(env_template)
        
        # 合并模板
        if env_content:
            # 环境模板扩展基础模板
            merged_content = self._merge_templates(base_content, env_content)
        else:
            merged_content = base_content
        
        # 准备模板上下文
        context = self._build_context(environment, overrides)
        
        # 渲染模板
        config_dict = self.template_engine.render_template(merged_content, context)
        
        return config_dict
    
    def _load_template(self, template_name: str) -> str:
        """加载模板内容"""
        if template_name in self.template_cache:
            return self.template_cache[template_name]
        
        template_path = self.template_dir / f"{template_name}.j2"
        if not template_path.exists():
            raise ConfigTemplateError(f"模板不存在: {template_name}")
        
        with open(template_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        self.template_cache[template_name] = content
        return content
    
    def _merge_templates(self, base_template: str, 
                        extension_template: str) -> str:
        """合并两个模板（扩展机制）"""
        # 使用Jinja2的模板继承机制
        # 基础模板定义块，扩展模板填充块
        merged = "{% extends 'base' %}\n"
        merged += extension_template
        
        # 临时注册基础模板
        self.template_engine.register_template('base', base_template)
        
        return merged
    
    def _build_context(self, environment: str, 
                      overrides: Dict[str, Any]) -> Dict[str, Any]:
        """构建模板上下文"""
        context = {
            'env': environment,
            'overrides': overrides or {},
            'timestamp': int(time.time()),
            'hostname': socket.gethostname(),
            'env_vars': dict(os.environ)
        }
        
        # 添加环境特定变量
        if environment == 'production':
            context['debug'] = False
            context['log_level'] = 'INFO'
        elif environment == 'development':
            context['debug'] = True
            context['log_level'] = 'DEBUG'
        elif environment == 'testing':
            context['debug'] = True
            context['log_level'] = 'DEBUG'
            context['test_mode'] = True
        
        return context
```

```python
class ConfigSyncManager:
    """配置同步管理器"""
    
    def __init__(self, node_id: str, sync_strategy: str = 'publish_subscribe'):
        self.node_id = node_id
        self.sync_strategy = sync_strategy
        self.peers = {}  # 节点ID -> 连接信息
        self.config_state = {}  # 配置键 -> 版本信息
        self.conflict_resolver = ConfigConflictResolver()
        
    def add_peer(self, peer_id: str, connection_info: Dict[str, Any]):
        """添加对等节点"""
        self.peers[peer_id] = {
            'info': connection_info,
            'state': 'disconnected',
            'last_sync': None,
            'config_version': {}
        }
    
    def sync_config(self, config_key: str, config_value: Any) -> bool:
        """同步配置到所有节点"""
        # 生成配置版本
        version_info = self._create_version_info(config_key, config_value)
        
        # 检查冲突
        conflicts = self._detect_conflicts(config_key, version_info)
        if conflicts:
            resolved = self.conflict_resolver.resolve(conflicts)
            if not resolved:
                return False
        
        # 同步到所有对等节点
        sync_results = []
        for peer_id, peer_info in self.peers.items():
            if peer_info['state'] == 'connected':
                result = self._sync_to_peer(peer_id, config_key, 
                                          config_value, version_info)
                sync_results.append(result)
        
        # 更新本地状态
        self.config_state[config_key] = version_info
        
        return all(sync_results)
    
    def receive_sync(self, sender_id: str, config_key: str, 
                    config_value: Any, version_info: Dict[str, Any]) -> bool:
        """接收其他节点的配置同步"""
        # 验证发送者
        if sender_id not in self.peers:
            return False
        
        # 检查版本
        local_version = self.config_state.get(config_key)
        if local_version:
            # 版本比较
            if self._compare_versions(local_version, version_info) < 0:
                # 远程版本更新，接受同步
                self.config.set(config_key, config_value)
                self.config_state[config_key] = version_info
                return True
            else:
                # 本地版本更新或相同，忽略或引发冲突
                conflict = ConfigConflict(
                    local_config=local_version,
                    remote_config=version_info,
                    key=config_key
                )
                return self.conflict_resolver.resolve([conflict])
        else:
            # 本地没有该配置，直接接受
            self.config.set(config_key, config_value)
            self.config_state[config_key] = version_info
            return True
    
    def _create_version_info(self, config_key: str, config_value: Any) -> Dict[str, Any]:
        """创建版本信息"""
        return {
            'version': str(uuid.uuid4()),
            'timestamp': time.time(),
            'node_id': self.node_id,
            'value_hash': self._calculate_value_hash(config_value),
            'previous_version': self.config_state.get(config_key, {}).get('version')
        }
    
    def _detect_conflicts(self, config_key: str, 
                         new_version: Dict[str, Any]) -> List[ConfigConflict]:
        """检测配置冲突"""
        conflicts = []
        
        local_version = self.config_state.get(config_key)
        if local_version:
            # 检查是否有其他节点正在修改同一配置
            for peer_id, peer_info in self.peers.items():
                peer_version = peer_info['config_version'].get(config_key)
                if peer_version and peer_version.get('node_id') != self.node_id:
                    # 检测到并发修改
                    conflict = ConfigConflict(
                        local_config=local_version,
                        remote_config=peer_version,
                        key=config_key,
                        conflict_type='concurrent_modification'
                    )
                    conflicts.append(conflict)
        
        return conflicts
```

#### 冲突解决策略
```python
class ConfigConflictResolver:
    """配置冲突解决器"""
    
    def __init__(self):
        self.strategies = {
            'last_write_wins': self._last_write_wins,
            'manual_intervention': self._manual_intervention,
            'value_merge': self._value_merge,
            'node_priority': self._node_priority
        }
        
    def resolve(self, conflicts: List[ConfigConflict]) -> bool:
        """解决冲突"""
        if not conflicts:
            return True
        
        for conflict in conflicts:
            # 根据冲突类型选择解决策略
            strategy = self._select_strategy(conflict)
            resolved = strategy(conflict)
            
            if not resolved:
                return False
        
        return True
    
    def _select_strategy(self, conflict: ConfigConflict) -> Callable:
        """选择解决策略"""
        conflict_type = conflict.conflict_type
        
        if conflict_type == 'concurrent_modification':
            # 并发修改，使用最后写入胜出或节点优先级
            return self.strategies['last_write_wins']
        elif conflict_type == 'value_incompatible':
            # 值不兼容，需要人工干预
            return self.strategies['manual_intervention']
        elif conflict_type == 'value_mergeable':
            # 值可合并，使用合并策略
            return self.strategies['value_merge']
        else:
            # 默认使用最后写入胜出
            return self.strategies['last_write_wins']
    
    def _last_write_wins(self, conflict: ConfigConflict) -> bool:
        """最后写入胜出策略"""
        local_time = conflict.local_config.get('timestamp', 0)
        remote_time = conflict.remote_config.get('timestamp', 0)
        
        if remote_time > local_time:
            # 远程版本更新，接受远程配置
            return True
        else:
            # 本地版本更新，保持本地配置
            return True
    
    def _value_merge(self, conflict: ConfigConflict) -> bool:
        """值合并策略"""
        local_value = conflict.local_config.get('value')
        remote_value = conflict.remote_config.get('value')
        
        if isinstance(local_value, dict) and isinstance(remote_value, dict):
            # 字典合并
            merged = {**local_value, **remote_value}
            conflict.resolved_value = merged
            return True
        elif isinstance(local_value, list) and isinstance(remote_value, list):
            # 列表合并（去重）
            merged = list(set(local_value + remote_value))
            conflict.resolved_value = merged
            return True
        else:
            # 无法自动合并，需要人工干预
            return self._manual_intervention(conflict)
```

```python
class ConfigHealthChecker:
    """配置健康检查器"""
    
    def __init__(self, config_manager):
        self.config_manager = config_manager
        self.check_rules = self._load_default_rules()
        self.check_history = []
        
    def add_check_rule(self, rule_name: str, rule_func: Callable, 
                      severity: str = 'warning'):
        """添加检查规则"""
        self.check_rules[rule_name] = {
            'function': rule_func,
            'severity': severity,
            'enabled': True
        }
    
    def run_health_check(self) -> HealthCheckReport:
        """运行健康检查"""
        report = HealthCheckReport()
        
        for rule_name, rule_info in self.check_rules.items():
            if not rule_info['enabled']:
                continue
                
            try:
                result = rule_info['function'](self.config_manager)
                
                if not result['passed']:
                    issue = HealthIssue(
                        rule_name=rule_name,
                        severity=rule_info['severity'],
                        message=result['message'],
                        details=result.get('details', {}),
                        timestamp=time.time()
                    )
                    report.add_issue(issue)
            except Exception as e:
                # 规则执行失败
                issue = HealthIssue(
                    rule_name=rule_name,
                    severity='error',
                    message=f"规则执行失败: {e}",
                    details={'exception': str(e)},
                    timestamp=time.time()
                )
                report.add_issue(issue)
        
        # 计算健康分数
        report.calculate_health_score()
        
        # 保存检查历史
        self.check_history.append(report)
        
        return report
    
    def _load_default_rules(self) -> Dict[str, Any]:
        """加载默认检查规则"""
        return {
            'required_configs': self._check_required_configs,
            'config_types': self._check_config_types,
            'value_ranges': self._check_value_ranges,
            'security_configs': self._check_security_configs,
            'dependency_checks': self._check_dependencies,
            'performance_configs': self._check_performance_configs
        }
    
    def _check_required_configs(self, config_manager) -> Dict[str, Any]:
        """检查必需配置项"""
        required_keys = [
            'injection.max_facts',
            'cache.ttl',
            'security.encryption_key',
            'database.host',
            'database.port'
        ]
        
        missing = []
        for key in required_keys:
            if config_manager.get(key) is None:
                missing.append(key)
        
        if missing:
            return {
                'passed': False,
                'message': f"缺少必需配置项: {missing}",
                'details': {'missing_keys': missing}
            }
        else:
            return {'passed': True, 'message': "必需配置项完整"}
    
    def _check_config_types(self, config_manager) -> Dict[str, Any]:
        """检查配置项类型"""
        type_rules = {
            'injection.max_facts': (int,),
            'cache.ttl': (int, float),
            'security.encryption': (bool,),
            'database.port': (int,)
        }
        
        type_errors = []
        for key, expected_types in type_rules.items():
            value = config_manager.get(key)
            if value is not None and not isinstance(value, expected_types):
                type_errors.append({
                    'key': key,
                    'expected': expected_types,
                    'actual': type(value).__name__
                })
        
        if type_errors:
            return {
                'passed': False,
                'message': f"配置项类型错误: {len(type_errors)}处",
                'details': {'type_errors': type_errors}
            }
        else:
            return {'passed': True, 'message': "配置项类型正确"}
```

### 2.2 配置性能监控实现

```python
class ConfigPerformanceMonitor:
    """配置性能监控器"""
    
    def __init__(self):
        self.metrics = {
            'load_time': [],      # 配置加载时间
            'get_time': [],       # 配置获取时间
            'set_time': [],       # 配置设置时间
            'cache_hit_rate': 0,  # 缓存命中率
            'memory_usage': 0,    # 内存使用
            'config_size': 0      # 配置大小
        }
        
        self.thresholds = {
            'load_time': 1.0,     # 加载时间阈值（秒）
            'get_time': 0.01,     # 获取时间阈值（秒）
            'cache_hit_rate': 0.8, # 缓存命中率阈值
            'memory_usage_mb': 100 # 内存使用阈值（MB）
        }
        
        self.alerts = []
        
    def record_load_time(self, duration: float):
        """记录配置加载时间"""
        self.metrics['load_time'].append(duration)
        
        # 检查是否超过阈值
        if duration > self.thresholds['load_time']:
            self._trigger_alert(
                'load_time_exceeded',
                f"配置加载时间过长: {duration:.3f}s",
                {'duration': duration, 'threshold': self.thresholds['load_time']}
            )
    
    def record_get_time(self, key: str, duration: float):
        """记录配置获取时间"""
        self.metrics['get_time'].append(duration)
        
        # 检查是否超过阈值
        if duration > self.thresholds['get_time']:
            self._trigger_alert(
                'get_time_exceeded',
                f"配置获取时间过长: {duration:.3f}s (key: {key})",
                {'key': key, 'duration': duration, 
                 'threshold': self.thresholds['get_time']}
            )
    
    def update_cache_metrics(self, hits: int, misses: int):
        """更新缓存指标"""
        total = hits + misses
        if total > 0:
            hit_rate = hits / total
            self.metrics['cache_hit_rate'] = hit_rate
            
            # 检查缓存命中率
            if hit_rate < self.thresholds['cache_hit_rate']:

                    {'hit_rate': hit_rate,

    def get_performance_report(self) -> PerformanceReport:

        report.memory_usage_mb = self.metrics['memory_usage']
        report.config_size_kb = self.metrics['config_size']
        
        # 添加性能建议
        report.suggestions = self._generate_suggestions()
        
        # 添加告警
```

```python
                index = handler(index)
                time.sleep(1)  # 防止过于频繁的请求
```

```python
class EtcdConfigProvider:
    """etcd配置提供器"""
    
    def __init__(self, hosts: List[str], username: Optional[str] = None,
                 password: Optional[str] = None):
        self.hosts = hosts
        self.username = username
        self.password = password
        self.client = etcd3.client(host=hosts, username=username, 
                                  password=password)
        self.lease_cache = {}
        
    def get_config(self, key: str, default: Any = None) -> Any:
        """从etcd获取配置"""
        try:
            value, metadata = self.client.get(key)
            if value:
                # 解析配置值
                try:
                    return json.loads(value.decode('utf-8'))
                except json.JSONDecodeError:
                    return value.decode('utf-8')
            
            return default
        except Exception as e:
            logging.error(f"从etcd获取配置失败: {e}")
            return default
    
    def set_config_with_lease(self, key: str, value: Any, ttl: int = 60):
        """设置配置并附加租约"""
        try:
            # 获取或创建租约
            if ttl not in self.lease_cache:
                lease = self.client.lease(ttl)
                self.lease_cache[ttl] = lease
            else:
                lease = self.lease_cache[ttl]
            
            # 转换为字符串
            if isinstance(value, (dict, list)):
                value_str = json.dumps(value, ensure_ascii=False)
            else:
                value_str = str(value)
            
            self.client.put(key, value_str, lease=lease)
        except Exception as e:
            logging.error(f"设置配置到etcd失败: {e}")
            raise
    
    def watch_config_prefix(self, prefix: str, callback: Callable):
        """监视前缀下的所有配置变更"""
        try:
            events_iterator, cancel = self.client.watch_prefix(prefix)
            
            def watch_loop():
                for event in events_iterator:
                    if isinstance(event, etcd3.events.PutEvent):
                        # 配置添加或更新
                        key = event.key.decode('utf-8')
                        value = event.value.decode('utf-8')
                        callback('put', key, value)
                    elif isinstance(event, etcd3.events.DeleteEvent):
                        # 配置删除
                        key = event.key.decode('utf-8')
                        callback('delete', key, None)
            
            # 启动监视线程
            thread = threading.Thread(target=watch_loop, daemon=True)
            thread.start()
            
            return cancel
        except Exception as e:
            logging.error(f"启动配置监视失败: {e}")
            return None
    
    def transaction(self, operations: List[Dict[str, Any]]) -> bool:
        """执行配置事务"""
        try:
            # 构建事务条件
            compare = []
            success = []
            failure = []
            
            for op in operations:
                if op['type'] == 'get':
                    # 获取操作
                    value, metadata = self.client.get(op['key'])
                    compare.append(
                        self.transactions.value(op['key']) == value
                    )
                elif op['type'] == 'put':
                    # 设置操作
                    success.append(
                        self.transactions.put(op['key'], op['value'])
                    )
                elif op['type'] == 'delete':
                    # 删除操作
                    success.append(
                        self.transactions.delete(op['key'])
                    )
            
            # 执行事务
            result = self.client.transaction(
                compare=compare,
                success=success,
                failure=failure
            )
            
            return result[0]  # 事务是否成功
        except Exception as e:
            logging.error(f"执行配置事务失败: {e}")
            return False
```

```python
class MultiSourceConfigManager:
    """多配置源管理器"""
    
    def __init__(self, sources: List[Dict[str, Any]]):
        self.sources = []
        self.source_weights = {}
        self.source_health = {}
        
        # 初始化配置源
        for source_config in sources:
            source = self._create_source(source_config)
            if source:
                self.sources.append(source)
                self.source_weights[source.name] = source_config.get('weight', 1.0)
                self.source_health[source.name] = 'healthy'
    
    def get_config(self, key: str, default: Any = None) -> Any:
        """从多个配置源获取配置（融合）"""
        values = []
        weights = []
        
        for source in self.sources:
            if self.source_health[source.name] == 'healthy':
                try:
                    value = source.get_config(key)
                    if value is not None:
                        values.append(value)
                        weights.append(self.source_weights[source.name])
                except Exception as e:
                    # 标记源为不健康
                    self.source_health[source.name] = 'unhealthy'
                    logging.error(f"配置源{source.name}获取失败: {e}")
        
        if not values:
            return default
        
        # 融合策略：加权平均或优先级选择
        if self._is_numeric_key(key):
            # 数值型配置使用加权平均
            return self._weighted_average(values, weights)
        else:
            # 非数值型配置使用优先级选择
            # 按权重排序，选择权重最高的值
            max_weight_index = weights.index(max(weights))
            return values[max_weight_index]
    
    def set_config(self, key: str, value: Any):
        """设置配置到所有健康的配置源"""
        results = {}
        
        for source in self.sources:
            if self.source_health[source.name] == 'healthy':
                try:
                    source.set_config(key, value)
                    results[source.name] = 'success'
                except Exception as e:
                    results[source.name] = f"failed: {e}"
                    self.source_health[source.name] = 'unhealthy'
        
        return results
    
    def _create_source(self, config: Dict[str, Any]) -> Optional[BaseConfigSource]:
        """创建配置源实例"""
        source_type = config.get('type')
        
        if source_type == 'consul':
            return ConsulConfigSource(
                host=config.get('host', 'localhost'),
                port=config.get('port', 8500),
                token=config.get('token')
            )
        elif source_type == 'etcd':
            return EtcdConfigSource(
                hosts=config.get('hosts', ['localhost:2379']),
                username=config.get('username'),
                password=config.get('password')
            )
        elif source_type == 'file':
            return FileConfigSource(
                path=config.get('path'),
                format=config.get('format', 'yaml')
            )
        elif source_type == 'env':
            return EnvConfigSource(
                prefix=config.get('prefix', 'APP_')
            )
        else:
            logging.error(f"未知的配置源类型: {source_type}")
            return None
    
    def _weighted_average(self, values: List[Any], weights: List[float]) -> float:
        """加权平均"""
        if len(values) != len(weights):
            raise ValueError("值和权重数量不一致")
        
        total_weight = sum(weights)
        if total_weight == 0:
            return sum(values) / len(values)
        
        weighted_sum = sum(v * w for v, w in zip(values, weights))
        return weighted_sum / total_weight
    
    def _is_numeric_key(self, key: str) -> bool:
        """判断是否为数值型配置键"""
        numeric_patterns = [
            '.*timeout.*', '.*interval.*', '.*delay.*',
            '.*threshold.*', '.*limit.*', '.*size.*',
            '.*count.*', '.*port.*', '.*weight.*'
        ]
        
        for pattern in numeric_patterns:
            if re.match(pattern, key, re.IGNORECASE):
                return True
        
        return False
```

```python
class OptimizedConfigCache:
    """优化版配置缓存"""
    
    def __init__(self, max_size: int = 1000, ttl: int = 300):
        self.max_size = max_size
        self.ttl = ttl
        self.cache = OrderedDict()  # LRU缓存
        self.hits = 0
        self.misses = 0
        
    def get(self, key: str) -> Optional[Any]:
        """获取缓存值（带LRU更新）"""
        if key in self.cache:
            entry = self.cache[key]
            
            # 检查是否过期
            if time.time() - entry['timestamp'] > self.ttl:
                del self.cache[key]
                self.misses += 1
                return None
            
            # 更新LRU顺序
            self.cache.move_to_end(key)
            self.hits += 1
            return entry['value']
        
        self.misses += 1
        return None
    
    def set(self, key: str, value: Any):
        """设置缓存值（带LRU淘汰）"""
        entry = {
            'value': value,
            'timestamp': time.time()
        }
        
        self.cache[key] = entry
        self.cache.move_to_end(key)
        
        # 检查是否超过最大大小
        if len(self.cache) > self.max_size:
            # 删除最久未使用的条目
            self.cache.popitem(last=False)
    
    def get_hit_rate(self) -> float:
        """获取缓存命中率"""
        total = self.hits + self.misses
        if total == 0:
            return 0.0
        return self.hits / total
```

```python
class LazyConfigLoader:
    """懒加载配置器"""
    
    def __init__(self, loader_func: Callable):
        self.loader_func = loader_func
        self._config = None
        self._loaded = False
        self._lock = threading.RLock()
        
    @property
    def config(self) -> Dict[str, Any]:
        """获取配置（懒加载）"""
        with self._lock:
            if not self._loaded:
                self._config = self.loader_func()
                self._loaded = True
            return self._config
    
    def get(self, key: str, default: Any = None) -> Any:
        """获取配置项"""
        config_dict = self.config
        return config_dict.get(key, default)
    
    def reload(self):
        """重新加载配置"""
        with self._lock:
            self._config = self.loader_func()
```

```python
class SecureConfigManager:
    """安全配置管理器"""
    
    def __init__(self, master_key: Optional[str] = None):
        self.master_key = master_key or self._generate_master_key()
        self.encryption = Fernet(self._derive_encryption_key(self.master_key))
        
    def encrypt_sensitive(self, key: str, value: Any) -> str:
        """加密敏感配置"""
        if isinstance(value, (dict, list)):
            value_str = json.dumps(value, ensure_ascii=False)
        else:
            value_str = str(value)
        
        encrypted = self.encryption.encrypt(value_str.encode())
        return f"encrypted:{encrypted.decode()}"
    
    def decrypt_sensitive(self, encrypted_value: str) -> Any:
        """解密敏感配置"""
        if not encrypted_value.startswith('encrypted:'):
            return encrypted_value
        
        encrypted = encrypted_value[10:]  # 移除'encrypted:'前缀
        decrypted = self.encryption.decrypt(encrypted.encode())
        
        # 尝试解析为JSON
        try:
            return json.loads(decrypted.decode())
        except json.JSONDecodeError:
            return decrypted.decode()
    
    def _generate_master_key(self) -> str:
        """生成主密钥"""
        return secrets.token_urlsafe(32)
    
    def _derive_encryption_key(self, master_key: str) -> bytes:
        """从主密钥派生加密密钥"""
        salt = b'deerflow_config_salt'
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000
        )
        return base64.urlsafe_b64encode(kdf.derive(master_key.encode()))
```
