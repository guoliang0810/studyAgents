### 第86节：缓存策略优化

#### 多级缓存架构
DeerFlow Agent系统适合采用三级缓存：
1. **内存缓存**：高频、小数据量，使用LRU策略
2. **Redis缓存**：中频、中等数据量，设置TTL
3. **持久化缓存**：低频、大数据量，使用数据库

#### 智能缓存实现
```python
from typing import Optional, Any
import hashlib
import json
from datetime import datetime, timedelta
from functools import wraps

class IntelligentCache:
    """智能缓存系统"""
    
    def __init__(self, redis_client=None):
        self.memory_cache = {}
        self.redis = redis_client
        
    def _generate_key(self, func_name: str, *args, **kwargs) -> str:
        """生成缓存键"""
        # 将参数序列化为JSON字符串
        args_str = json.dumps(args, sort_keys=True)
        kwargs_str = json.dumps(kwargs, sort_keys=True)
        
        # 生成MD5哈希作为键
        content = f"{func_name}:{args_str}:{kwargs_str}"
        return hashlib.md5(content.encode()).hexdigest()
    
    def cache_memory(self, ttl_seconds: int = 300):
        """内存缓存装饰器"""
        def decorator(func):
            @wraps(func)
            async def wrapper(*args, **kwargs):
                key = self._generate_key(func.__name__, *args, **kwargs)
                
                # 检查内存缓存
                if key in self.memory_cache:
                    entry = self.memory_cache[key]
                    if datetime.now() < entry['expires']:
                        return entry['value']
                
                # 执行函数
                result = await func(*args, **kwargs)
                
                # 更新缓存
                self.memory_cache[key] = {
                    'value': result,
                    'expires': datetime.now() + timedelta(seconds=ttl_seconds)
                }
                
                # 清理过期条目（简单实现）
                self._clean_expired_memory()
                
                return result
            return wrapper
        return decorator
    
    def _clean_expired_memory(self):
        """清理过期内存缓存"""
        now = datetime.now()
        expired_keys = [
            key for key, entry in self.memory_cache.items()
            if entry['expires'] < now
        ]
        for key in expired_keys:
            del self.memory_cache[key]
```

```python
T = TypeVar('T')

class AsyncConnectionPool(Generic[T]):

        self,
        create_connection,
        max_size: int = 10,
        min_size: int = 2

        self.max_size = max_size
        self.min_size = min_size

            # 如果没有可用连接且未达到最大大小，创建新连接
            elif len(self.in_use) + len(self.pool) < self.max_size:
                conn = await self.create_connection()
            else:

                # 实际实现应使用条件变量
                pass
        
        if conn is None:
            # 实际实现应等待
            raise Exception("No connections available")
        
        self.in_use.add(conn)
        
        try:
            yield conn
```

```python
from typing import Any, Dict, List
import re
from html import escape

class InputValidator:
    """多层级输入验证器"""
    
    def __init__(self, security_config: Dict):
        self.config = security_config
        
    async def validate_agent_input(self, input_data: Dict) -> Dict:
        """验证Agent输入"""
        
        validated = {}
        
        # 第一层：类型和结构验证
        validated = self._validate_structure(input_data)
        
        # 第二层：内容验证
        validated = await self._validate_content(validated)
        
        # 第三层：业务逻辑验证
        validated = await self._validate_business_rules(validated)
        
        # 第四层：清理和转义
        validated = self._sanitize_input(validated)
        
        return validated
    
    def _validate_structure(self, input_data: Dict) -> Dict:
        """验证输入结构"""
        
        required_fields = self.config.get('required_fields', [])
        for field in required_fields:
            if field not in input_data:
                raise ValidationError(f"Missing required field: {field}")
        
        # 类型检查
        type_rules = self.config.get('type_rules', {})
        for field, expected_type in type_rules.items():
            if field in input_data:
                if not isinstance(input_data[field], expected_type):
                    raise ValidationError(
                        f"Field {field} must be {expected_type.__name__}"
                    )
        
        return input_data.copy()
    
    async def _validate_content(self, input_data: Dict) -> Dict:
        """验证输入内容"""
        
        # 长度限制
        max_lengths = self.config.get('max_lengths', {})
        for field, max_len in max_lengths.items():
            if field in input_data:
                value = str(input_data[field])
                if len(value) > max_len:
                    raise ValidationError(
                        f"Field {field} exceeds maximum length {max_len}"
                    )
        
        # 正则表达式模式验证
        patterns = self.config.get('patterns', {})
        for field, pattern in patterns.items():
            if field in input_data:
                value = str(input_data[field])
                if not re.match(pattern, value):
                    raise ValidationError(
                        f"Field {field} does not match required pattern"
                    )
        
        # 敏感信息检测
        if self.config.get('detect_sensitive_info', False):
            await self._detect_sensitive_info(input_data)
        
        return input_data
    
    def _sanitize_input(self, input_data: Dict) -> Dict:
        """清理输入数据"""
        
        sanitized = {}
        
        for key, value in input_data.items():
            if isinstance(value, str):
                # HTML转义
                sanitized[key] = escape(value)
                
                # 移除危险字符
                dangerous_patterns = [
                    r'<script.*?>.*?</script>',
                    r'on\w+=".*?"',
                    r'javascript:'
                ]
                
                for pattern in dangerous_patterns:
                    sanitized[key] = re.sub(
                        pattern, '', sanitized[key], flags=re.IGNORECASE
                    )
            elif isinstance(value, dict):
                sanitized[key] = self._sanitize_input(value)
            elif isinstance(value, list):
                sanitized[key] = [
                    self._sanitize_input(item) if isinstance(item, dict)
                    else escape(str(item)) if isinstance(item, str)
                    else item
                    for item in value
                ]
            else:
                sanitized[key] = value
        
        return sanitized
```

```python
import re
from typing import List, Tuple

class SensitiveInfoDetector:
    """敏感信息检测器"""
    
    PATTERNS = {
        'credit_card': r'\b(?:\d[ -]*?){13,16}\b',
        'ssn': r'\b\d{3}[-\s]?\d{2}[-\s]?\d{4}\b',
        'email': r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b',
        'phone': r'\b(?:\+?1[-.\s]?)?\(?[0-9]{3}\)?[-.\s]?[0-9]{3}[-.\s]?[0-9]{4}\b',
        'api_key': r'\b(?:sk-|AKIA|AIza)[A-Za-z0-9]{20,}\b'
    }
    
    def detect(self, text: str) -> List[Tuple[str, str]]:
        """检测敏感信息"""
        
        findings = []
        
        for info_type, pattern in self.PATTERNS.items():
            matches = re.finditer(pattern, text, re.IGNORECASE)
            for match in matches:
                # 模糊化敏感信息（只显示部分）
                sensitive_text = match.group()
                masked = self._mask_sensitive_info(info_type, sensitive_text)
                
                findings.append((info_type, masked))
        
        return findings
    
    def _mask_sensitive_info(self, info_type: str, text: str) -> str:
        """模糊化敏感信息"""
        
        if info_type == 'credit_card':
            if len(text) >= 4:
                return f"****-****-****-{text[-4:]}"
        elif info_type == 'ssn':
            if len(text) >= 4:
                return f"***-**-{text[-4:]}"
        elif info_type == 'api_key':
            if len(text) >= 8:
                return f"{text[:4]}...{text[-4:]}"
        
        # 默认模糊化
        if len(text) > 8:
            return f"{text[:2]}...{text[-2:]}"
        else:
            return "***"
```

```python
from typing import Set, List
import os

class SandboxSecurityPolicy:
    """沙箱安全策略"""
    
    def __init__(self, config: Dict):
        self.config = config
        self.allowed_commands = set(config.get('allowed_commands', []))
        self.blocked_commands = set(config.get('blocked_commands', []))
        self.max_execution_time = config.get('max_execution_time', 30)
        self.max_memory_mb = config.get('max_memory_mb', 512)
        self.read_only_paths = config.get('read_only_paths', [])
        self.writable_paths = config.get('writable_paths', [])
    
    def validate_command(self, command: str, args: List[str]) -> bool:
        """验证命令是否允许执行"""
        
        full_command = f"{command} {' '.join(args)}"
        
        # 检查是否在阻止列表中
        for blocked in self.blocked_commands:
            if blocked in full_command:
                return False
        
        # 检查是否在允许列表中
        if self.allowed_commands:
            for allowed in self.allowed_commands:
                if allowed in command:
                    return True
            return False
        
        return True
    
    def resolve_path(self, virtual_path: str) -> str:
        """解析路径，确保在允许的目录内"""
        
        # 规范化路径
        normalized = os.path.normpath(virtual_path)
        
        # 检查是否试图访问上级目录
        if normalized.startswith("..") or normalized.startswith("/"):
            raise SecurityError("Path traversal attempt detected")
        
        # 映射到沙箱内的安全位置
        sandbox_root = self.config.get('sandbox_root', '/sandbox')
        resolved = os.path.join(sandbox_root, normalized)
        
        # 确保路径在沙箱根目录内
        if not resolved.startswith(sandbox_root):
            raise SecurityError("Path outside sandbox detected")
        
        return resolved
```

```python
import resource
import signal
from contextlib import contextmanager

@contextmanager
def resource_limits(max_memory_mb: int, max_cpu_time: int):
    """资源限制上下文管理器"""
    
    # 设置内存限制
    soft_limit = max_memory_mb * 1024 * 1024  # 转换为字节
    hard_limit = soft_limit * 2
    
    resource.setrlimit(
        resource.RLIMIT_AS,
        (soft_limit, hard_limit)
    )
    
    # 设置CPU时间限制
    resource.setrlimit(
        resource.RLIMIT_CPU,
        (max_cpu_time, max_cpu_time + 5)  # 5秒宽限期
    )
    
    # 设置文件大小限制
    resource.setrlimit(
        resource.RLIMIT_FSIZE,
        (100 * 1024 * 1024, 200 * 1024 * 1024)  # 100MB文件大小限制
    )
    
    try:
        yield
    except MemoryError:
        raise ResourceLimitExceeded("Memory limit exceeded")
    except resource.error as e:
        if e.errno == resource.RLIMIT_CPU:
            raise ResourceLimitExceeded("CPU time limit exceeded")
        raise
    finally:
        # 恢复默认限制
        resource.setrlimit(resource.RLIMIT_AS, (resource.RLIM_INFINITY, resource.RLIM_INFINITY))
        resource.setrlimit(resource.RLIMIT_CPU, (resource.RLIM_INFINITY, resource.RLIM_INFINITY))
        resource.setrlimit(resource.RLIMIT_FSIZE, (resource.RLIM_INFINITY, resource.RLIM_INFINITY))

class TimeoutException(Exception):
    pass

def timeout_handler(signum, frame):
    """超时信号处理器"""
    raise TimeoutException("Execution timeout")

@contextmanager
def execution_timeout(seconds: int):
    """执行超时上下文管理器"""
    
    # 设置超时信号
    signal.signal(signal.SIGALRM, timeout_handler)
```

```python
            expire = datetime.utcnow() + expires_delta
        else:
            expire = datetime.utcnow() + timedelta(hours=1)
        
        payload = {
            "sub": user_id,
            "roles": roles,
            "exp": expire,
            "iat": datetime.utcnow(),
            "type": "access"
        }
        
        return jwt.encode(
            payload,
            self.secret_key,
            algorithm=self.algorithm
        )
    
    def create_refresh_token(self, user_id: str) -> str:
        """创建刷新令牌"""
        
        expire = datetime.utcnow() + timedelta(days=30)
        
        payload = {
            "sub": user_id,
            "exp": expire,
            "iat": datetime.utcnow(),
            "type": "refresh"
        }
        
        return jwt.encode(
            payload,
            self.secret_key,
            algorithm=self.algorithm
        )
    
    def verify_token(self, token: str) -> Optional[Dict[str, Any]]:
        """验证令牌"""
        
        try:
            payload = jwt.decode(
                token,
                self.secret_key,
                algorithms=[self.algorithm]
            )
            return payload
        except jwt.ExpiredSignatureError:
            raise AuthenticationError("Token expired")
        except jwt.InvalidTokenError:
            raise AuthenticationError("Invalid token")
    
    def refresh_access_token(self, refresh_token: str) -> str:
        """使用刷新令牌获取新的访问令牌"""
        
        payload = self.verify_token(refresh_token)
        
        if payload.get("type") != "refresh":
            raise AuthenticationError("Not a refresh token")
        
        # 获取用户信息（实际应从数据库获取）
        user_id = payload.get("sub")
        
        # 创建新的访问令牌
        return self.create_access_token(user_id, payload.get("roles", []))
```

```python
    ) -> str:

            salt = os.urandom(16)
        
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000,
        )
        
        key = base64.urlsafe_b64encode(kdf.derive(password.encode()))
        return key.decode()

        return base64.urlsafe_b64encode(encrypted).decode()

        decrypted = self.cipher.decrypt(encrypted_bytes)
        return decrypted.decode()
```

```python
                masked_local = local[0] + '*' * (len(local) - 2) + local[-1]
            else:
                masked_local = local[0] + '*'
            
            # 脱敏域名
            domain_parts = domain.split('.')
            if len(domain_parts) >= 2:
                masked_domain = domain_parts[0][0] + '*' * (len(domain_parts[0]) - 1)
                for part in domain_parts[1:]:
                    masked_domain += f".{part}"
            else:
                masked_domain = domain
            
            return f"{masked_local}@{masked_domain}"
        return email

            if '-' in phone:
                return f"{masked[:3]}-***-{masked[-4:]}"
            elif ' ' in phone:

            else:
```

```python
from typing import Set, List, Dict

class SkillDependencyManager:
    """技能依赖管理器"""
    
    def __init__(self, skill_registry: SkillRegistry):
        self.registry = skill_registry
        self.dependencies: Dict[str, Set[str]] = {}
        self.reverse_dependencies: Dict[str, Set[str]] = {}
    
    def add_dependency(self, skill_name: str, depends_on: List[str]):
        """添加技能依赖"""
        
        if skill_name not in self.dependencies:
            self.dependencies[skill_name] = set()
        
        for dependency in depends_on:
            self.dependencies[skill_name].add(dependency)
            
            if dependency not in self.reverse_dependencies:
                self.reverse_dependencies[dependency] = set()
            self.reverse_dependencies[dependency].add(skill_name)
    
    def get_execution_order(self, skill_names: List[str]) -> List[str]:
        """获取技能执行顺序（考虑依赖）"""
        
        # 构建完整的依赖图
        all_skills = set(skill_names)
        for skill in skill_names:
            all_skills.update(self._get_all_dependencies(skill))
        
        # 拓扑排序
        visited = set()
        temp_visited = set()
        order = []
        
        def visit(skill):
            if skill in temp_visited:
                raise CircularDependencyError(
                    f"Circular dependency detected involving {skill}"
                )
            if skill not in visited:
                temp_visited.add(skill)
                
                # 先访问依赖
                for dependency in self.dependencies.get(skill, set()):
                    if dependency in all_skills:
                        visit(dependency)
                
                temp_visited.remove(skill)
                visited.add(skill)
                order.append(skill)
        
        for skill in all_skills:
            if skill not in visited:
                visit(skill)
        
        # 只保留请求的技能
        return [s for s in order if s in skill_names]
    
    def _get_all_dependencies(self, skill_name: str) -> Set[str]:
        """获取技能的所有依赖（递归）"""
        
        dependencies = set()
        
        def collect_deps(skill):
            for dep in self.dependencies.get(skill, set()):
                if dep not in dependencies:
                    dependencies.add(dep)
                    collect_deps(dep)
        
        collect_deps(skill_name)
        return dependencies
    
    def detect_circular_dependencies(self) -> List[List[str]]:
        """检测循环依赖"""
        
        cycles = []
        visited = set()
        path = []
        
        def dfs(skill):
            if skill in path:
                # 找到循环
                cycle_start = path.index(skill)
                cycle = path[cycle_start:]
                cycles.append(cycle.copy())
                return
            
            if skill in visited:
                return
            
            visited.add(skill)
            path.append(skill)
            
            for dependency in self.dependencies.get(skill, set()):
                dfs(dependency)
            
            path.pop()
        
        for skill in self.dependencies:
            if skill not in visited:
                dfs(skill)
        
        return cycles
```

```python
            patch, pre_release = patch_part.split('-', 1)
        else:
            patch = patch_part
            pre_release = ''
        
        patch_num = int(patch) if patch.isdigit() else 0
        
        # 返回可排序的元组
        return (major, minor, patch_num, pre_release)
```

### 第95节：中间件扩展

#### 可插拔中间件系统
```python
from typing import List, Dict, Optional, Callable
from abc import ABC, abstractmethod

class Middleware(ABC):
    """中间件基类"""
    
    def __init__(self, config: Optional[Dict] = None):
        self.config = config or {}
        self.priority = self.config.get('priority', 100)
    
    @abstractmethod
    async def before_agent(self, state: Dict) -> Dict:
        """Agent执行前调用"""
        pass
    
    @abstractmethod
    async def after_agent(self, response: Dict) -> Dict:
        """Agent执行后调用"""
        pass
    
    async def wrap_tool_call(
        self,
        tool_name: str,
        tool_args: Dict,
        config: Optional[Dict] = None
    ) -> tuple:
        """包装工具调用"""
        # 默认实现：直接返回工具名和参数
        return tool_name, tool_args

class MiddlewareChain:
    """中间件链"""
    
    def __init__(self):
        self.middlewares: List[Middleware] = []
    
    def add_middleware(self, middleware: Middleware):
        """添加中间件"""
        self.middlewares.append(middleware)
        # 按优先级排序
        self.middlewares.sort(key=lambda m: m.priority)
    
    def remove_middleware(self, middleware_name: str):
        """移除中间件"""
        self.middlewares = [
            m for m in self.middlewares
            if m.__class__.__name__ != middleware_name
        ]
    
    async def execute_before_agent(self, state: Dict) -> Dict:
        """执行所有中间件的before_agent"""
        
        for middleware in self.middlewares:
            state = await middleware.before_agent(state)
        
        return state
    
    async def execute_after_agent(self, response: Dict) -> Dict:
        """执行所有中间件的after_agent"""
        
        for middleware in reversed(self.middlewares):
            response = await middleware.after_agent(response)
        
        return response
    
    async def wrap_tool_call(
        self,
        tool_name: str,
        tool_args: Dict,
        config: Optional[Dict] = None
    ) -> tuple:
        """包装工具调用"""
        
        wrapped_name = tool_name
        wrapped_args = tool_args
        
        for middleware in self.middlewares:
            wrapped_name, wrapped_args = await middleware.wrap_tool_call(
                wrapped_name, wrapped_args, config
            )
        
        return wrapped_name, wrapped_args

class MiddlewareRegistry:
    """中间件注册表"""
    
    def __init__(self):
        self.middleware_classes: Dict[str, type] = {}
        self.middleware_configs: Dict[str, Dict] = {}
    
    def register(
        self,
        middleware_class: type,
        name: Optional[str] = None,
        default_config: Optional[Dict] = None
    ):
        """注册中间件类"""
        
        middleware_name = name or middleware_class.__name__
        
        if middleware_name in self.middleware_classes:
            raise ValueError(f"Middleware {middleware_name} already registered")
        
        self.middleware_classes[middleware_name] = middleware_class
        self.middleware_configs[middleware_name] = default_config or {}
        
        return middleware_name
    
    def create_middleware(
        self,
        name: str,
        config: Optional[Dict] = None
    ) -> Middleware:
        """创建中间件实例"""
        
        if name not in self.middleware_classes:
            raise ValueError(f"Middleware {name} not found")
        
        middleware_class = self.middleware_classes[name]
        
        # 合并默认配置和提供的配置
        default_config = self.middleware_configs.get(name, {})
        merged_config = {**default_config, **(config or {})}
        
        return middleware_class(merged_config)
    
    def discover_middlewares(self, package_path: str):
        """自动发现中间件"""
        
        package = importlib.import_module(package_path)
        
        for _, module_name, is_pkg in pkgutil.iter_modules(
            package.__path__, package.__name__ + '.'
        ):
            if not is_pkg:
                module = importlib.import_module(module_name)
                
                for attr_name in dir(module):
                    attr = getattr(module, attr_name)
                    
                    if (
                        isinstance(attr, type) and
                        issubclass(attr, Middleware) and
                        attr != Middleware
                    ):
                        self.register(attr)

# 示例中间件
class LoggingMiddleware(Middleware):
    """日志中间件"""
    
    def __init__(self, config: Optional[Dict] = None):
        super().__init__(config)
        self.logger = config.get('logger') if config else None
    
    async def before_agent(self, state: Dict) -> Dict:
        """Agent执行前记录日志"""
        
        if self.logger:
            self.logger.info(
                "Agent execution started",
                thread_id=state.get('thread_id'),
                input=state.get('input')
            )
        
        return state
    
    async def after_agent(self, response: Dict) -> Dict:
        """Agent执行后记录日志"""
        
        if self.logger:
            self.logger.info(
                "Agent execution completed",
                thread_id=response.get('thread_id'),
                response_length=len(str(response))
            )
        
        return response
    
    async def wrap_tool_call(
        self,
        tool_name: str,
        tool_args: Dict,
        config: Optional[Dict] = None
    ) -> tuple:
        """包装工具调用并记录日志"""
        
        if self.logger:
            self.logger.info(
                "Tool call started",
                tool_name=tool_name,
                tool_args=tool_args
            )
        
        # 调用父类实现
        result = await super().wrap_tool_call(tool_name, tool_args, config)
        
        if self.logger:
            self.logger.info(
                "Tool call completed",
                tool_name=result[0]
            )
        
        return result
```

```python
from typing import Dict, List, Optional, Type
from dataclasses import dataclass

@dataclass
class SubagentConfig:
    """子代理配置"""
    
    name: str
    agent_type: str
    description: str
    capabilities: List[str]
    config: Dict
    dependencies: List[str]
    max_concurrent: int = 1
    timeout_seconds: int = 300
    retry_policy: Dict = None
    
    def __post_init__(self):
        if self.retry_policy is None:
            self.retry_policy = {
                'max_retries': 3,
                'backoff_factor': 1.0,
                'retry_on': ['timeout', 'tool_error']
            }

class SubagentFactory:
    """子代理工厂"""
    
    def __init__(self, agent_registry: 'AgentRegistry'):
        self.agent_registry = agent_registry
        self.subagent_templates: Dict[str, SubagentConfig] = {}
    
    def register_template(
        self,
        template_name: str,
        config: SubagentConfig
    ):
        """注册子代理模板"""
        
        self.subagent_templates[template_name] = config
    
    def create_subagent(
        self,
        template_name: str,
        overrides: Optional[Dict] = None
    ) -> SubagentConfig:
        """创建子代理实例"""
        
        if template_name not in self.subagent_templates:
            raise ValueError(f"Template {template_name} not found")
        
        template = self.subagent_templates[template_name]
        
        # 应用覆盖
        config_dict = asdict(template)
        if overrides:
            for key, value in overrides.items():
                if key in config_dict:
                    config_dict[key] = value
        
        return SubagentConfig(**config_dict)
    
    def create_specialized_subagent(
        self,
        parent_agent: str,
        task_description: str,
        available_tools: List[str]
    ) -> SubagentConfig:
        """创建专门化的子代理"""
        
        # 分析任务描述，确定需要的功能
        capabilities = self._analyze_task_requirements(task_description)
        
        # 选择最合适的代理类型
        agent_type = self._select_agent_type(capabilities, available_tools)
        
        # 生成配置
        config = SubagentConfig(
            name=f"{parent_agent}_sub_{hash(task_description)}",
            agent_type=agent_type,
            description=f"Specialized subagent for: {task_description}",
            capabilities=capabilities,
            config={
                'tools': available_tools,
                'model': 'gpt-4',
                'temperature': 0.1
            },
            dependencies=[parent_agent]
        )
        
        return config
    
    def _analyze_task_requirements(self, task_description: str) -> List[str]:
        """分析任务需求"""
        
        # 简单的关键词匹配
        capabilities = []
        
        task_lower = task_description.lower()
        
        if any(word in task_lower for word in ['计算', '数学', '公式']):
            capabilities.append('calculation')
        
        if any(word in task_lower for word in ['搜索', '查询', '查找']):
            capabilities.append('search')
        
        if any(word in task_lower for word in ['分析', '统计', '数据']):
            capabilities.append('data_analysis')
        
        if any(word in task_lower for word in ['代码', '编程', '开发']):
            capabilities.append('coding')
        
        if any(word in task_lower for word in ['写作', '生成', '创作']):
            capabilities.append('writing')
        
        return capabilities
    
    def _select_agent_type(
        self,
        capabilities: List[str],
        available_tools: List[str]
    ) -> str:
        """选择代理类型"""
        
        # 基于功能和可用工具选择代理类型
        if 'coding' in capabilities:
            return 'coding_agent'
        elif 'data_analysis' in capabilities:
            return 'analysis_agent'
        elif 'search' in capabilities:
            return 'research_agent'
        else:
            return 'general_agent'

class SubagentOrchestrator:
    """子代理编排器"""
    
    def __init__(self, subagent_factory: SubagentFactory):
        self.factory = subagent_factory
        self.active_subagents: Dict[str, SubagentConfig] = {}
        self.task_queue: asyncio.Queue = asyncio.Queue()
        self.results: Dict[str, Dict] = {}
    
    async def delegate_task(
        self,
        parent_agent: str,
        task_description: str,
        available_tools: List[str]
    ) -> str:
        """委派任务给子代理"""
        
        # 创建专门化子代理
        subagent_config = self.factory.create_specialized_subagent(
            parent_agent,
            task_description,
            available_tools
        )
        
        # 注册子代理
        subagent_id = subagent_config.name
        self.active_subagents[subagent_id] = subagent_config
        
        # 创建任务
        task = {
            'subagent_id': subagent_id,
            'task_description': task_description,
            'parent_agent': parent_agent,
            'created_at': datetime.now()
        }
        
        await self.task_queue.put(task)
        
        return subagent_id
    
    async def process_tasks(self):
        """处理任务队列"""
        
        while True:
            try:
                task = await self.task_queue.get()
                
                subagent_id = task['subagent_id']
                subagent_config = self.active_subagents.get(subagent_id)
                
                if not subagent_config:
                    continue
                
                # 执行子代理任务
                result = await self._execute_subagent_task(
                    subagent_config,
                    task['task_description']
                )
                
                # 存储结果
                self.results[subagent_id] = result
                
                # 清理子代理
                del self.active_subagents[subagent_id]
                
                self.task_queue.task_done()
                
            except asyncio.CancelledError:
                break
            except Exception as e:
                print(f"Error processing task: {e}")
                continue
    
    async def _execute_subagent_task(
        self,
        config: SubagentConfig,
        task_description: str
    ) -> Dict:
        """执行子代理任务"""
        
        # 创建代理实例
        agent = await self.factory.agent_registry.create_agent(
            config.agent_type,
            config.config
        )
        
        # 执行任务
        result = await agent.run({
            'input': task_description,
            'capabilities': config.capabilities
        })
        
        return {
            'result': result,
            'subagent_config': config,
            'completed_at': datetime.now()
        }
```

```python
# tests/integration/test_deployment.py
import pytest
import requests
import time
from typing import Dict

class TestDeployment:
    
    BASE_URL = "http://deerflow-agent-service.deerflow.svc.cluster.local"

        """测试健康检查端点"""

        assert response.status_code == 200
        assert response.json()["status"] == "healthy"
    
    def test_agent_endpoint(self):
        """测试Agent端点"""

            "input": "你好，你是谁？",
            "thread_id": "test_thread"
        }

            f"{self.BASE_URL}/api/agent/run",
            json=payload,
            timeout=30
        )
        
        assert response.status_code == 200
        data = response.json()
        assert "response" in data
        assert "thread_id" in data
    
    def test_concurrent_requests(self):
        """测试并发请求"""
        import concurrent.futures
        
        def make_request(i):
            payload = {
                "input": f"测试请求 {i}",
                "thread_id": f"thread_{i}"
            }
            
            try:
                response = requests.post(
                    f"{self.BASE_URL}/api/agent/run",
                    json=payload,
                    timeout=10
                )
                return response.status_code
            except Exception as e:
                return str(e)
        
        # 发送10个并发请求
        with concurrent.futures.ThreadPoolExecutor(max_workers=10) as executor:
            futures = [executor.submit(make_request, i) for i in range(10)]
            results = [future.result() for future in concurrent.futures.as_completed(futures)]
        
        # 检查所有请求都成功
        success_count = sum(1 for r in results if r == 200)
        assert success_count >= 8, f"Only {success_count}/10 requests succeeded"
```
