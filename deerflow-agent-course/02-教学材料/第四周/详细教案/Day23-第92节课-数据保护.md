## 🎯 教学目标

### 知识目标（学生将知道/理解）
1. **数据保护层次**: 理解数据在存储、传输、处理各环节的保护需求和措施
2. **加密技术原理**: 掌握对称加密、非对称加密、哈希算法在数据保护中的应用
3. **数据脱敏策略**: 了解不同类型数据（PII、金融数据、健康数据）的脱敏规则和法规要求
4. **隐私保护框架**: 理解隐私设计原则、数据最小化、目的限制等隐私保护核心概念

### 技能目标（学生将能够）
1. **实现数据加密**: 能够使用cryptography库实现数据的加密存储和解密访问
2. **设计脱敏系统**: 能够根据数据类型和场景设计合理的数据脱敏规则
3. **实施隐私保护**: 能够在系统设计中应用隐私保护原则和最佳实践
4. **构建数据安全**: 能够整合加密、脱敏、访问控制构建完整的数据保护体系

### 情感/态度目标
1. **培养数据隐私意识**: 认识到数据隐私保护的法律和伦理重要性
2. **建立合规思维**: 理解数据保护法规（如GDPR、CCPA）对技术设计的影响
3. **增强数据安全自信**: 通过实现生产级数据保护系统，提升对敏感数据处理的信心

## 📚 教学重点与难点
- **教学重点**:

```python
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
import base64
import os
from typing import Optional

class DataEncryptor:
    """数据加密器"""
    
    def __init__(self, master_key: Optional[str] = None):
        if master_key:
            # 使用提供的密钥
            key = base64.urlsafe_b64decode(master_key)
        else:
            # 生成新密钥
            key = Fernet.generate_key()
        
        self.cipher = Fernet(key)
    
    @staticmethod
    def generate_key_from_password(
        password: str,
        salt: Optional[bytes] = None
    ) -> str:
        """从密码生成密钥"""
        
        if salt is None:
            salt = os.urandom(16)
        
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000,
        )
        
        key = base64.urlsafe_b64encode(kdf.derive(password.encode()))
        return key.decode()
    
    def encrypt(self, data: str) -> str:
        """加密数据"""
        encrypted = self.cipher.encrypt(data.encode())
        return base64.urlsafe_b64encode(encrypted).decode()
    
    def decrypt(self, encrypted_data: str) -> str:
        """解密数据"""
        encrypted_bytes = base64.urlsafe_b64decode(encrypted_data.encode())
        decrypted = self.cipher.decrypt(encrypted_bytes)
        return decrypted.decode()
    
    def encrypt_dict(self, data: Dict) -> Dict:
        """加密字典中的敏感字段"""
        
        # 定义敏感字段
        sensitive_fields = {
            'password', 'token', 'api_key', 'secret',
            'credit_card', 'ssn', 'email', 'phone'
        }
        
        encrypted_dict = data.copy()
        
        for key, value in data.items():
            if key in sensitive_fields and isinstance(value, str):
                encrypted_dict[key] = self.encrypt(value)
            elif isinstance(value, dict):
                encrypted_dict[key] = self.encrypt_dict(value)
            elif isinstance(value, list):
                encrypted_dict[key] = [
                    self.encrypt_dict(item) if isinstance(item, dict)
                    else self.encrypt(item) if isinstance(item, str) and key in sensitive_fields
                    else item
                    for item in value
                ]
        
        return encrypted_dict

class DataMasker:
    """数据脱敏器"""
    
    MASKING_RULES = {
        'email': lambda x: self._mask_email(x),
        'phone': lambda x: self._mask_phone(x),
        'credit_card': lambda x: self._mask_credit_card(x),
        'ssn': lambda x: self._mask_ssn(x),
        'name': lambda x: self._mask_name(x),
        'address': lambda x: self._mask_address(x)
    }
    
    @staticmethod
    def _mask_email(email: str) -> str:
        """脱敏邮箱"""
        if '@' in email:
            local, domain = email.split('@')
            if len(local) > 2:
                masked_local = local[0] + '*' * (len(local) - 2) + local[-1]
            else:
                masked_local = local[0] + '*'
            
            # 脱敏域名
            domain_parts = domain.split('.')
            if len(domain_parts) >= 2:
                masked_domain = domain_parts[0][0] + '*' * (len(domain_parts[0]) - 1)
                for part in domain_parts[1:]:
                    masked_domain += f".{part}"
            else:
                masked_domain = domain
            
            return f"{masked_local}@{masked_domain}"
        return email
    
    @staticmethod
    def _mask_phone(phone: str) -> str:
        """脱敏电话号码"""
        digits = ''.join(filter(str.isdigit, phone))
        
        if len(digits) >= 10:
            masked = f"{digits[:3]}***{digits[-4:]}"
            
            # 保留原始格式（如果可能）
            if '-' in phone:
                return f"{masked[:3]}-***-{masked[-4:]}"
            elif ' ' in phone:
                return f"{masked[:3]} *** {masked[-4:]}"
            else:
                return masked
        return phone
    
    def mask_data(self, data: Dict, fields_to_mask: List[str]) -> Dict:
        """脱敏数据"""
        
        masked_data = data.copy()
        
        for field in fields_to_mask:
            if field in masked_data:
                value = masked_data[field]
                
                if isinstance(value, str):
                    # 根据字段类型选择脱敏规则
                    mask_func = self.MASKING_RULES.get(
                        field,
                        lambda x: self._default_mask(x)
                    )
                    masked_data[field] = mask_func(value)
                elif isinstance(value, list):
                    masked_data[field] = [
                        self.mask_data(item, fields_to_mask) if isinstance(item, dict)
                        else mask_func(str(item)) if isinstance(item, str) and field in self.MASKING_RULES
                        else item
                        for item in value
                    ]
                elif isinstance(value, dict):
                    masked_data[field] = self.mask_data(value, fields_to_mask)
        
        return masked_data
```
