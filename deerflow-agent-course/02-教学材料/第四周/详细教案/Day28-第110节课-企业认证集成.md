## 🎯 教学目标

2. 掌握SSO（单点登录）的工作原理和实现方案
3. 了解企业级认证系统的安全要求和最佳实践

### 技能目标（学生将能够）
1. 实现基本的SSO集成类（SSOIntegration）
2. 配置OAuth 2.0提供商集成
3. 设计支持多认证提供商的企业认证系统

### 情感/态度目标
1. 培养学生对企业安全标准的重视
2. 增强身份认证和访问控制的安全意识

### 附录2：核心代码示例
```python
from typing import Dict, Optional
import aiohttp
from dataclasses import dataclass

@dataclass
class SSOProviderConfig:
    """SSO提供商配置"""
    name: str
    client_id: str
    client_secret: str
    auth_url: str
    token_url: str
    userinfo_url: str
    scope: str = "openid profile email"
    redirect_uri: str = "http://localhost:8000/auth/callback"

class SSOIntegration:
    """SSO集成系统"""
    
    def __init__(self):
        self.providers: Dict[str, SSOProviderConfig] = {}
        self.sessions: Dict[str, Dict] = {}  # 会话存储
    
    def add_provider(self, name: str, config: Dict):
        """添加SSO提供商"""
        provider_config = SSOProviderConfig(
            name=name,
            client_id=config.get("client_id", ""),
            client_secret=config.get("client_secret", ""),
            auth_url=config.get("auth_url", ""),
            token_url=config.get("token_url", ""),
            userinfo_url=config.get("userinfo_url", ""),
            scope=config.get("scope", "openid profile email"),
            redirect_uri=config.get("redirect_uri", "http://localhost:8000/auth/callback")
        )
        self.providers[name] = provider_config
        print(f"🔐 添加SSO提供商: {name}")
        return provider_config
    
    def get_auth_url(self, provider: str, state: str = None) -> Optional[str]:
        """获取认证URL"""
        if provider not in self.providers:
            raise ValueError(f"未知提供商: {provider}")
        
        config = self.providers[provider]
        import urllib.parse
        
        params = {
            "client_id": config.client_id,
            "redirect_uri": config.redirect_uri,
            "response_type": "code",
            "scope": config.scope,
            "state": state or "default_state"
        }
        
        auth_url = f"{config.auth_url}?{urllib.parse.urlencode(params)}"
        return auth_url
    
    async def authenticate(self, provider: str, code: str) -> Dict:
        """SSO认证 - 使用授权码交换令牌"""
        if provider not in self.providers:
            raise ValueError(f"未知提供商: {provider}")
        
        config = self.providers[provider]
        
        # 交换令牌
        token_data = {
            "client_id": config.client_id,
            "client_secret": config.client_secret,
            "code": code,
            "redirect_uri": config.redirect_uri,
            "grant_type": "authorization_code"
        }
        
        async with aiohttp.ClientSession() as session:
            # 获取访问令牌
            async with session.post(config.token_url, data=token_data) as response:
                if response.status != 200:
                    error_text = await response.text()
                    raise ValueError(f"令牌请求失败: {response.status} - {error_text}")
                
                token_response = await response.json()
                access_token = token_response.get("access_token")
                
                if not access_token:
                    raise ValueError("未获取到访问令牌")
            
            # 获取用户信息
            headers = {"Authorization": f"Bearer {access_token}"}
            async with session.get(config.userinfo_url, headers=headers) as response:
                if response.status != 200:
                    error_text = await response.text()
                    raise ValueError(f"用户信息请求失败: {response.status} - {error_text}")
                
                user_info = await response.json()
        
        # 返回认证结果
        result = {
            "user_id": user_info.get("sub") or user_info.get("id"),
            "email": user_info.get("email"),
            "name": user_info.get("name") or user_info.get("login"),
            "access_token": access_token,
            "provider": provider,
            "raw_user_info": user_info
        }
        
        # 存储会话
        session_id = f"session_{hash(str(result))}"
        self.sessions[session_id] = {
            "user_info": result,
            "created_at": "2024-04-21T10:00:00Z",  # 实际使用datetime.now()
            "expires_at": "2024-04-21T11:00:00Z"   # 1小时后过期
        }
        
        return result
    
    def validate_session(self, session_id: str) -> bool:
        """验证会话有效性"""
        session = self.sessions.get(session_id)
        if not session:
            return False
        
        # 检查是否过期（简化版）
        # 实际应比较当前时间和expires_at
        return True
    
    def get_providers_list(self) -> list:
        """获取可用提供商列表"""
        return list(self.providers.keys())
```
