```python
from dataclasses import dataclass, asdict
from datetime import datetime
from typing import Dict, List, Optional, Any
import json
import uuid

@dataclass
class AuditEvent:
    """审计事件"""
    event_id: str
    timestamp: str
    event_type: str  # authentication, data_access, configuration_change, etc.
    user_id: Optional[str]
    tenant_id: Optional[str]
    action: str
    resource_type: str
    resource_id: Optional[str]
    details: Dict[str, Any]
    ip_address: Optional[str]
    user_agent: Optional[str]
    status: str  # success, failure, partial
    error_message: Optional[str]

class AuditLogger:
    """审计日志系统"""
    
    def __init__(self, storage_backend: str = "file"):
        self.storage_backend = storage_backend
        self.events_buffer: List[AuditEvent] = []
        self.buffer_size = 100
        
    def log_event(self, 
                 event_type: str,
                 user_id: Optional[str],
                 tenant_id: Optional[str],
                 action: str,
                 resource_type: str,
                 resource_id: Optional[str] = None,
                 details: Optional[Dict] = None,
                 ip_address: Optional[str] = None,
                 user_agent: Optional[str] = None,
                 status: str = "success",
                 error_message: Optional[str] = None) -> str:
        """记录审计事件"""
        
        event = AuditEvent(
            event_id=f"audit_{uuid.uuid4().hex[:8]}",
            timestamp=datetime.utcnow().isoformat() + "Z",
            event_type=event_type,
            user_id=user_id,
            tenant_id=tenant_id,
            action=action,
            resource_type=resource_type,
            resource_id=resource_id,
            details=details or {},
            ip_address=ip_address,
            user_agent=user_agent,
            status=status,
            error_message=error_message
        )
        
        self.events_buffer.append(event)
        
        # 缓冲区满时持久化
        if len(self.events_buffer) >= self.buffer_size:
            self._flush_buffer()
        
        return event.event_id
    
    def _flush_buffer(self):
        """刷新缓冲区到存储"""
        if self.storage_backend == "file":
            self._save_to_file()
        elif self.storage_backend == "database":
            self._save_to_database()
        # 其他存储后端...
        
        self.events_buffer.clear()
    
    def _save_to_file(self):
        """保存到文件"""
        if not self.events_buffer:
            return
        
        filename = f"audit_log_{datetime.utcnow().strftime('%Y%m%d')}.json"
        events_data = [asdict(event) for event in self.events_buffer]
        
        try:
            # 追加模式写入
            with open(filename, 'a') as f:
                for event_data in events_data:
                    f.write(json.dumps(event_data) + '\n')
            print(f"📝 审计日志已保存到 {filename}，共 {len(events_data)} 条记录")
        except Exception as e:
            print(f"❌ 保存审计日志失败: {e}")
    
    def query_events(self,
                    start_time: Optional[str] = None,
                    end_time: Optional[str] = None,
                    event_type: Optional[str] = None,
                    user_id: Optional[str] = None,
                    tenant_id: Optional[str] = None) -> List[AuditEvent]:
        """查询审计事件"""
        # 简化版查询，实际应使用数据库查询
        results = []
        
        # 这里应该从持久化存储查询
        # 为演示目的，只查询缓冲区中的事件
        for event in self.events_buffer:
            match = True
            
            if start_time and event.timestamp < start_time:
                match = False
            if end_time and event.timestamp > end_time:
                match = False
            if event_type and event.event_type != event_type:
                match = False
            if user_id and event.user_id != user_id:
                match = False
            if tenant_id and event.tenant_id != tenant_id:
                match = False
            
            if match:
                results.append(event)
        
        return results

class ComplianceChecker:
    """合规检查器"""
    
    def __init__(self):
        self.rules = self._load_default_rules()
    
    def _load_default_rules(self) -> Dict:
        """加载默认合规规则"""
        return {
            "gdpr": {
                "data_retention": {
                    "max_days": 730,  # 2年
                    "description": "GDPR数据保留期限"
                },
                "user_consent": {
                    "required": True,
                    "explicit": True,
                    "description": "需要明确用户同意"
                },
                "data_subject_rights": {
                    "access": True,
                    "rectification": True,
                    "erasure": True,
                    "portability": True,
                    "description": "数据主体权利"
                }
            },
            "ccpa": {
                "opt_out": {
                    "required": True,
                    "description": "CCPA退出权"
                },
                "data_sale": {
                    "tracking_required": True,
                    "description": "数据销售跟踪"
                }
            }
        }
    
    def check_gdpr_compliance(self, system_config: Dict) -> Dict:
        """检查GDPR合规性"""
        violations = []
        
        # 检查数据保留策略
        retention_days = system_config.get("data_retention_days", 0)
        if retention_days > self.rules["gdpr"]["data_retention"]["max_days"]:
            violations.append({
                "rule": "data_retention",
                "description": f"数据保留期限超过GDPR要求：{retention_days}天 > {self.rules['gdpr']['data_retention']['max_days']}天",
                "severity": "high"
            })
        
        # 检查用户同意机制
        if not system_config.get("user_consent_enabled", False):
            violations.append({
                "rule": "user_consent",
                "description": "未启用用户同意机制",
                "severity": "high"
            })
        
        # 检查数据主体权利支持
        rights_supported = system_config.get("data_subject_rights_supported", {})
        required_rights = ["access", "rectification", "erasure", "portability"]
        
        for right in required_rights:
            if not rights_supported.get(right, False):
                violations.append({
                    "rule": f"data_subject_right_{right}",
                    "description": f"未支持GDPR数据主体权利：{right}",
                    "severity": "medium"
                })
        
        return {
            "compliant": len(violations) == 0,
            "violations": violations,
            "score": 100 - (len(violations) * 10)  # 简化评分
        }
    
    def generate_compliance_report(self, system_config: Dict) -> Dict:
        """生成合规报告"""
        gdpr_result = self.check_gdpr_compliance(system_config)
        ccpa_result = self.check_ccpa_compliance(system_config)
        
        return {
            "report_id": f"compliance_report_{uuid.uuid4().hex[:8]}",
            "generated_at": datetime.utcnow().isoformat() + "Z",
            "systems_checked": ["gdpr", "ccpa"],
            "overall_compliant": gdpr_result["compliant"] and ccpa_result["compliant"],
            "details": {
                "gdpr": gdpr_result,
                "ccpa": ccpa_result
            },
            "recommendations": self._generate_recommendations(gdpr_result, ccpa_result)
        }
    
    def check_ccpa_compliance(self, system_config: Dict) -> Dict:
        """检查CCPA合规性"""
        violations = []
        
        # 检查退出权支持
        if not system_config.get("opt_out_enabled", False):
            violations.append({
                "rule": "opt_out",
                "description": "未支持CCPA退出权",
                "severity": "high"
            })
        
        # 检查数据销售跟踪
        if system_config.get("data_sale_enabled", False) and not system_config.get("data_sale_tracking_enabled", False):
            violations.append({
                "rule": "data_sale_tracking",
                "description": "数据销售启用但未启用跟踪",
                "severity": "medium"
            })
        
        return {
            "compliant": len(violations) == 0,
            "violations": violations,
            "score": 100 - (len(violations) * 10)
        }
    
    def _generate_recommendations(self, gdpr_result: Dict, ccpa_result: Dict) -> List[str]:
        """生成改进建议"""
        recommendations = []
        
        for violation in gdpr_result["violations"]:
            recommendations.append(f"GDPR违规: {violation['description']}")
        
        for violation in ccpa_result["violations"]:
            recommendations.append(f"CCPA违规: {violation['description']}")
        
        if not recommendations:
            recommendations.append("系统目前符合GDPR和CCPA要求，继续保持！")
        
        return recommendations
```
