### 技能目标（学生将能够）
1. 设计并实现品牌定制系统（Logo、颜色、字体）
2. 配置功能模块的按需启用/禁用机制
3. 创建动态UI主题和样式系统

### 情感/态度目标
1. 培养学生对产品定制化价值的认识
2. 增强客户导向的产品设计思维
3. 建立可扩展和可配置的系统架构理念

from dataclasses import dataclass
from typing import Dict, List, Optional, Any
import json
from enum import Enum

class ThemeMode(Enum):
    """主题模式"""
    LIGHT = "light"
    DARK = "dark"
    AUTO = "auto"

@dataclass
class BrandConfig:
    """品牌配置"""
    tenant_id: str
    brand_name: str
    logo_url: str
    favicon_url: str
    primary_color: str  # 十六进制颜色代码，如 "#3b82f6"
    secondary_color: str
    accent_color: str
    font_family: str
    font_url: Optional[str]
    theme_mode: ThemeMode = ThemeMode.LIGHT
    custom_css: Optional[str] = None
    custom_js: Optional[str] = None
    enabled_features: List[str] = None
    
    def __post_init__(self):
        if self.enabled_features is None:
            self.enabled_features = ["dashboard", "analytics", "reports"]
    
    def to_css_variables(self) -> Dict[str, str]:
        """转换为CSS变量"""
        return {
            "--primary-color": self.primary_color,
            "--secondary-color": self.secondary_color,
            "--accent-color": self.accent_color,
            "--font-family": self.font_family,
            "--theme-mode": self.theme_mode.value
        }
    
    def to_json(self) -> str:
        """转换为JSON"""
        config_dict = {
            "tenant_id": self.tenant_id,
            "brand_name": self.brand_name,
            "logo_url": self.logo_url,
            "favicon_url": self.favicon_url,

        }
        return json.dumps(config_dict, ensure_ascii=False, indent=2)

class BrandManager:
    """品牌管理器"""
    
    def __init__(self):
        self.brand_configs: Dict[str, BrandConfig] = {}
        self.default_config = self._create_default_config()
    
    def _create_default_config(self) -> BrandConfig:
        """创建默认配置"""
        return BrandConfig(

class FeatureToggle:
    """功能开关系统"""
    
    def __init__(self):
        self.features: Dict[str, Dict] = {}
        self._initialize_default_features()
    
    def _initialize_default_features(self):
        """初始化默认功能"""
        default_features = {
            "dashboard": {
                "name": "仪表板",
                "description": "主控制面板",
                "enabled_by_default": True,
                "dependent_features": []
            },
            "analytics": {

                "enabled_by_default": False,

                "description": "品牌定制功能",
                "enabled_by_default": False,
                "dependent_features": []
            }
        }
        
        for feature_id, config in default_features.items():
            self.register_feature(feature_id, config)
    
    def register_feature(self, feature_id: str, config: Dict):
        """注册功能"""
        self.features[feature_id] = {
            "name": config.get("name", feature_id),
            "description": config.get("description", ""),
            "enabled_by_default": config.get("enabled_by_default", False),
            "dependent_features": config.get("dependent_features", []),
            "enabled_for_tenants": {}  # tenant_id -> bool
        }
        print(f"⚙️ 注册功能: {feature_id} - {config.get('name', '')}")
    
    def is_feature_enabled(self, feature_id: str, tenant_id: str = "default") -> bool:
        """检查功能是否启用"""
        if feature_id not in self.features:
            return False
        
        feature = self.features[feature_id]
        
        # 检查租户特定设置
        if tenant_id in feature["enabled_for_tenants"]:
            return feature["enabled_for_tenants"][tenant_id]
        
        # 返回默认设置
        return feature["enabled_by_default"]
    
    def enable_feature(self, feature_id: str, tenant_id: str = "all"):
        """启用功能"""
        if feature_id not in self.features:
            raise ValueError(f"未知功能: {feature_id}")
        
        if tenant_id == "all":
            # 为所有租户启用
            for tenant in self.get_all_tenants():
                self.features[feature_id]["enabled_for_tenants"][tenant] = True
            print(f"✅ 为所有租户启用功能: {feature_id}")
        else:
            self.features[feature_id]["enabled_for_tenants"][tenant_id] = True
            print(f"✅ 为租户 {tenant_id} 启用功能: {feature_id}")
    
    def disable_feature(self, feature_id: str, tenant_id: str = "all"):
        """禁用功能"""
        if feature_id not in self.features:
            raise ValueError(f"未知功能: {feature_id}")
        
        if tenant_id == "all":
            # 为所有租户禁用
            for tenant in self.get_all_tenants():
                self.features[feature_id]["enabled_for_tenants"][tenant] = False
            print(f"❌ 为所有租户禁用功能: {feature_id}")
        else:
            self.features[feature_id]["enabled_for_tenants"][tenant_id] = False
            print(f"❌ 为租户 {tenant_id} 禁用功能: {feature_id}")
    
    def get_all_tenants(self) -> List[str]:
        """获取所有租户列表"""
        tenants = set()
        for feature_id, feature in self.features.items():
            tenants.update(feature["enabled_for_tenants"].keys())
        return list(tenants)
    
    def get_tenant_features(self, tenant_id: str) -> Dict[str, bool]:
        """获取租户的所有功能状态"""
        result = {}
        for feature_id in self.features:
            result[feature_id] = self.is_feature_enabled(feature_id, tenant_id)
        return result
    
    def validate_feature_dependencies(self, tenant_id: str) -> List[str]:
        """验证功能依赖关系"""
        issues = []
        tenant_features = self.get_tenant_features(tenant_id)
        
        for feature_id, enabled in tenant_features.items():
            if not enabled:
                continue
            
            feature = self.features[feature_id]
            for dep_feature_id in feature["dependent_features"]:
                if dep_feature_id not in tenant_features or not tenant_features[dep_feature_id]:
                    issues.append(f"功能 '{feature_id}' 依赖的功能 '{dep_feature_id}' 未启用")
        
        return issues

class WhiteLabelSystem:
    """白标系统整合"""
    
    def __init__(self):
        self.brand_manager = BrandManager()
        self.feature_toggle = FeatureToggle()
        self.tenant_configs: Dict[str, Dict] = {}
    
    def get_tenant_ui_config(self, tenant_id: str) -> Dict[str, Any]:
        """获取租户UI配置"""
        brand_config = self.brand_manager.get_brand_config(tenant_id)
        features = self.feature_toggle.get_tenant_features(tenant_id)
        
        # 检查白标功能是否启用
        white_label_enabled = self.feature_toggle.is_feature_enabled("white_label", tenant_id)
        
        config = {
            "tenant_id": tenant_id,
            "brand": {
                "name": brand_config.brand_name,
                "logo_url": brand_config.logo_url if white_label_enabled else "/static/default-logo.png",
                "favicon_url": brand_config.favicon_url if white_label_enabled else "/static/default-favicon.ico",
                "primary_color": brand_config.primary_color if white_label_enabled else "#3b82f6",
                "secondary_color": brand_config.secondary_color if white_label_enabled else "#10b981",
                "accent_color": brand_config.accent_color if white_label_enabled else "#f59e0b",
                "font_family": brand_config.font_family,
                "theme_mode": brand_config.theme_mode.value
            },
            "features": features,
            "white_label_enabled": white_label_enabled,
            "css_variables": brand_config.to_css_variables() if white_label_enabled else {},
            "customizations": {
                "custom_css": brand_config.custom_css if white_label_enabled else None,
                "custom_js": brand_config.custom_js if white_label_enabled else None
            }
        }
        
        return config
    
    def update_tenant_config(self, tenant_id: str, config_updates: Dict) -> Dict:
        """更新租户配置"""
        # 更新品牌配置
        if "brand" in config_updates:
            self.brand_manager.update_brand_config(tenant_id, **config_updates["brand"])
        
        # 更新功能配置
        if "features" in config_updates:
            for feature_id, enabled in config_updates["features"].items():
                if enabled:
                    self.feature_toggle.enable_feature(feature_id, tenant_id)
                else:
                    self.feature_toggle.disable_feature(feature_id, tenant_id)
        
        # 验证依赖关系
        dependency_issues = self.feature_toggle.validate_feature_dependencies(tenant_id)
        if dependency_issues:
            print(f"⚠️  依赖问题: {dependency_issues}")
        
        return self.get_tenant_ui_config(tenant_id)
