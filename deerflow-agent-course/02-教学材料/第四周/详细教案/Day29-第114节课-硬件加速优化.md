```python
import torch
import torch.nn as nn
import torch.quantization
from typing import Dict, Any
import time
import numpy as np

class SimpleCNN(nn.Module):
    """简单的CNN模型用于演示"""
    def __init__(self):
        super(SimpleCNN, self).__init__()
        self.conv1 = nn.Conv2d(3, 16, kernel_size=3, padding=1)
        self.relu1 = nn.ReLU()
        self.pool1 = nn.MaxPool2d(2)
        self.conv2 = nn.Conv2d(16, 32, kernel_size=3, padding=1)
        self.relu2 = nn.ReLU()
        self.pool2 = nn.MaxPool2d(2)
        self.fc1 = nn.Linear(32 * 8 * 8, 128)
        self.relu3 = nn.ReLU()
        self.fc2 = nn.Linear(128, 10)
    
    def forward(self, x):
        x = self.conv1(x)
        x = self.relu1(x)
        x = self.pool1(x)
        x = self.conv2(x)
        x = self.relu2(x)
        x = self.pool2(x)
        x = x.view(x.size(0), -1)
        x = self.fc1(x)
        x = self.relu3(x)
        x = self.fc2(x)
        return x

class HardwareAccelerationDemo:
    """硬件加速演示类"""
    
    def __init__(self):
        self.model = SimpleCNN()
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        print(f"🚀 使用设备: {self.device}")
        
        # 移动到指定设备
        self.model.to(self.device)
        
        # 创建测试数据
        self.batch_size = 32
        self.input_tensor = torch.randn(self.batch_size, 3, 32, 32).to(self.device)
    
    def benchmark_inference(self, model, warmup_iters=10, test_iters=100) -> Dict[str, float]:
        """基准测试推理性能"""
        # 预热
        with torch.no_grad():
            for _ in range(warmup_iters):
                _ = model(self.input_tensor)
        
        # 同步GPU（如果使用）
        if self.device.type == "cuda":
            torch.cuda.synchronize()
        
        # 正式测试
        start_time = time.time()
        
        with torch.no_grad():
            for _ in range(test_iters):
                _ = model(self.input_tensor)
        
        # 同步GPU（如果使用）
        if self.device.type == "cuda":
            torch.cuda.synchronize()
        
        end_time = time.time()
        
        total_time = end_time - start_time
        avg_time_per_batch = total_time / test_iters
        fps = self.batch_size * test_iters / total_time
        
        return {
            "total_time": total_time,
            "avg_time_per_batch": avg_time_per_batch,
            "fps": fps,
            "device": str(self.device)
        }
    
    def quantize_model(self, model):
        """量化模型"""
        # 设置为评估模式
        model.eval()
        
        # 准备量化配置
        model.qconfig = torch.quantization.get_default_qconfig('fbgemm')
        
        # 准备模型进行量化
        torch.quantization.prepare(model, inplace=True)
        
        # 校准（使用一些数据）
        with torch.no_grad():
            for _ in range(10):
                _ = model(self.input_tensor)
        
        # 转换为量化模型
        quantized_model = torch.quantization.convert(model, inplace=False)
        
        return quantized_model
    
    def compare_performance(self):
        """比较不同优化级别的性能"""
        results = {}
        
        # 1. 原始模型（CPU）
        if torch.cuda.is_available():
            cpu_model = SimpleCNN().to("cpu")
            cpu_input = self.input_tensor.cpu()
            
            # 临时替换输入进行CPU测试
            original_input = self.input_tensor
            self.input_tensor = cpu_input
            
            results["cpu_original"] = self.benchmark_inference(cpu_model)
            
            # 恢复原始输入
            self.input_tensor = original_input
        
        # 2. 原始模型（GPU）
        results["gpu_original"] = self.benchmark_inference(self.model)
        
        # 3. 量化模型
        quantized_model = self.quantize_model(SimpleCNN().to("cpu"))
        if torch.cuda.is_available():
            # 临时替换输入和模型进行量化测试
            original_model = self.model
            original_input = self.input_tensor
            
            cpu_input = original_input.cpu()
            self.model = quantized_model
            self.input_tensor = cpu_input
            self.device = torch.device("cpu")
            
            results["cpu_quantized"] = self.benchmark_inference(quantized_model)
            
            # 恢复
            self.model = original_model
            self.input_tensor = original_input
            self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        
        # 4. 混合精度（如果使用GPU）
        if self.device.type == "cuda":
            from torch.cuda.amp import autocast
            
            mixed_model = SimpleCNN().to(self.device)
            
            with autocast():
                results["gpu_mixed_precision"] = self.benchmark_inference(mixed_model)
        
        return results
    
    def analyze_model_size(self, model, model_name: str) -> Dict[str, Any]:
        """分析模型大小"""
        # 保存模型到临时文件
        import tempfile
        import os
        
        with tempfile.NamedTemporaryFile(suffix='.pth', delete=False) as tmp:
            torch.save(model.state_dict(), tmp.name)
            
            # 获取文件大小
            file_size = os.path.getsize(tmp.name)
            
            # 清理
            os.unlink(tmp.name)
        
        # 计算参数量
        total_params = sum(p.numel() for p in model.parameters())
        trainable_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
        
        return {
            "model_name": model_name,
            "file_size_mb": file_size / (1024 * 1024),
            "total_params": total_params,
            "trainable_params": trainable_params,
            "params_millions": total_params / 1_000_000
        }
    
    def generate_optimization_report(self) -> str:
        """生成优化报告"""
        # 性能比较
        perf_results = self.compare_performance()
        
        # 模型大小分析
        original_model = SimpleCNN()
        quantized_model = self.quantize_model(SimpleCNN())
        
        size_results = {
            "original": self.analyze_model_size(original_model, "原始模型"),
            "quantized": self.analyze_model_size(quantized_model, "量化模型")
        }
        
        # 生成报告
        report_lines = []
        report_lines.append("=" * 60)
        report_lines.append("AI模型硬件加速优化报告")
        report_lines.append("=" * 60)
        report_lines.append("")
        
        report_lines.append("1. 性能测试结果:")
        report_lines.append("-" * 40)
        
        for config, result in perf_results.items():
            report_lines.append(f"  {config}:")
            report_lines.append(f"    设备: {result.get('device', 'N/A')}")
            report_lines.append(f"    总时间: {result.get('total_time', 0):.3f}s")
            report_lines.append(f"    每批平均时间: {result.get('avg_time_per_batch', 0)*1000:.2f}ms")
            report_lines.append(f"    FPS: {result.get('fps', 0):.2f}")
        
        report_lines.append("")
        report_lines.append("2. 模型大小分析:")

    def evaluate_option(self, option_name: str, custom_weights: Dict = None) -> Dict[str, Any]:

        if option_name not in self.hardware_options:
            raise ValueError(f"未知硬件选项: {option_name}")
        
        option = self.hardware_options[option_name]
        weights = custom_weights or {c: self.criteria[c]["weight"] for c in self.criteria}
        
        # 计算加权得分
        weighted_score = 0
        criteria_scores = {}
        
        for criterion, weight in weights.items():
            if criterion in self.criteria and criterion in option:
                criterion_score = option[criterion] * weight
                weighted_score += criterion_score
                criteria_scores[criterion] = {
                    "raw_score": option[criterion],
                    "weighted_score": criterion_score,
                    "weight": weight
                }
        
        return {
            "option_name": option["name"],
            "description": option["description"],
            "weighted_score": weighted_score,
            "criteria_scores": criteria_scores,
            "raw_scores": {k: option[k] for k in option if k not in ["name", "description"]}
        }

                "future_proof": 0.1
            }
        }
        
        weights = scenario_weights.get(scenario, self.criteria)
        
        # 评估所有选项
        evaluations = {}
        for option_name in self.hardware_options:
            evaluations[option_name] = self.evaluate_option(option_name, weights)
        
        # 排序
        sorted_options = sorted(
            evaluations.items(),
            key=lambda x: x[1]["weighted_score"],
            reverse=True
        )
        
        return {
            "scenario": scenario,
            "top_recommendation": sorted_options[0] if sorted_options else None,
            "all_evaluations": evaluations,
            "sorted_options": sorted_options
        }
```
