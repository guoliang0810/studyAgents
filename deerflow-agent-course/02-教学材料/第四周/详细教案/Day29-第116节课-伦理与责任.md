```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from fairlearn.metrics import (
    demographic_parity_difference,
    equalized_odds_difference,
    selection_rate,
    MetricFrame
)
from fairlearn.postprocessing import ThresholdOptimizer
from fairlearn.reductions import ExponentiatedGradient, DemographicParity
import matplotlib.pyplot as plt
from typing import Dict, Any, List, Tuple

class AIEthicsAnalyzer:
    """AI伦理分析器"""
    
    def __init__(self, model=None, sensitive_features=None):
        self.model = model
        self.sensitive_features = sensitive_features
        self.fairness_metrics = {}
        self.bias_mitigation_results = {}
        
    def load_sample_dataset(self):
        """加载示例数据集（模拟招聘数据）"""
        np.random.seed(42)
        
        n_samples = 2000
        data = {
            'age': np.random.randint(20, 60, n_samples),
            'education_years': np.random.randint(8, 20, n_samples),
            'experience_years': np.random.randint(0, 30, n_samples),
            'test_score': np.random.normal(70, 15, n_samples),
            'gender': np.random.choice(['male', 'female'], n_samples, p=[0.6, 0.4]),
            'race': np.random.choice(['group_a', 'group_b', 'group_c'], n_samples, p=[0.5, 0.3, 0.2])
        }
        
        # 创建有偏见的标签（模拟历史偏见）
        def calculate_hire_probability(row):
            base_prob = 0.3
            # 教育和工作经验加分
            base_prob += row['education_years'] * 0.01
            base_prob += row['experience_years'] * 0.02
            base_prob += (row['test_score'] - 70) * 0.005
            
            # 添加偏见（模拟历史歧视）
            if row['gender'] == 'female':
                base_prob -= 0.1  # 女性降低10%概率
            if row['race'] == 'group_c':
                base_prob -= 0.15  # 特定种族组降低15%概率
                
            return 1 if np.random.random() < base_prob else 0
        
        data['hired'] = [calculate_hire_probability(pd.Series({k: data[k][i] for k in data})) 
                         for i in range(n_samples)]
        
        df = pd.DataFrame(data)
        
        # 特征和标签
        features = ['age', 'education_years', 'experience_years', 'test_score']
        sensitive_attrs = ['gender', 'race']
        
        X = df[features]
        y = df['hired']
        A = df[sensitive_attrs]
        
        return X, y, A, features, sensitive_attrs
    
    def train_model_with_bias(self, X_train, y_train):
        """训练可能包含偏见的模型"""
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        self.model = model
        return model
    
    def evaluate_fairness(self, X_test, y_test, A_test, sensitive_features=None):
        """评估模型公平性"""
        if self.model is None:
            raise ValueError("请先训练模型")
        
        if sensitive_features is None:
            sensitive_features = self.sensitive_features
        
        # 获取预测
        y_pred = self.model.predict(X_test)
        y_pred_proba = self.model.predict_proba(X_test)[:, 1]
        
        # 计算公平性指标
        metrics = {
            'accuracy': lambda y_true, y_pred: np.mean(y_true == y_pred),
            'selection_rate': lambda y_true, y_pred: selection_rate(y_true, y_pred),
            'demographic_parity_diff': lambda y_true, y_pred: demographic_parity_difference(
                y_true, y_pred, sensitive_features=A_test
            ),
            'equalized_odds_diff': lambda y_true, y_pred: equalized_odds_difference(
                y_true, y_pred, sensitive_features=A_test
            )
        }
        
        # 创建MetricFrame
        metric_frame = MetricFrame(
            metrics=metrics,
            y_true=y_test,
            y_pred=y_pred,
            sensitive_features=A_test
        )
        
        # 计算分组指标
        by_group = metric_frame.by_group
        
        # 提取关键公平性问题
        fairness_issues = self._identify_fairness_issues(metric_frame, by_group, A_test)
        
        self.fairness_metrics = {
            'overall': metric_frame.overall,
            'by_group': by_group.to_dict(),
            'fairness_issues': fairness_issues,
            'demographic_parity_difference': float(metric_frame.overall['demographic_parity_diff']),
            'equalized_odds_difference': float(metric_frame.overall['equalized_odds_diff'])
        }
        
        return self.fairness_metrics
    
    def _identify_fairness_issues(self, metric_frame, by_group, A_test):
        """识别公平性问题"""
        issues = []
        
        # 检查人口统计均等差异
        dp_diff = metric_frame.overall['demographic_parity_diff']
        if abs(dp_diff) > 0.1:  # 阈值10%
            issues.append({
                'type': 'demographic_parity_violation',
                'severity': 'high' if abs(dp_diff) > 0.2 else 'medium',
                'description': f'人口统计均等差异达{dp_diff:.3f}，超过0.1阈值',
                'metric_value': dp_diff
            })
        
        # 检查均衡赔率差异
        eo_diff = metric_frame.overall['equalized_odds_diff']
        if abs(eo_diff) > 0.1:  # 阈值10%
            issues.append({
                'type': 'equalized_odds_violation',
                'severity': 'high' if abs(eo_diff) > 0.2 else 'medium',
                'description': f'均衡赔率差异达{eo_diff:.3f}，超过0.1阈值',
                'metric_value': eo_diff
            })
        
        # 检查各组选择率差异
        selection_rates = by_group['selection_rate']
        max_sr = max(selection_rates.values())
        min_sr = min(selection_rates.values())
        sr_ratio = max_sr / min_sr if min_sr > 0 else float('inf')
        
        if sr_ratio > 2.0:  # 选择率比例超过2倍
            worst_group = min(selection_rates, key=selection_rates.get)
            best_group = max(selection_rates, key=selection_rates.get)
            issues.append({
                'type': 'selection_rate_disparity',
                'severity': 'high' if sr_ratio > 4.0 else 'medium',
                'description': f'选择率差异大：{best_group}={max_sr:.3f} vs {worst_group}={min_sr:.3f}，比例{sr_ratio:.1f}倍',
                'worst_group': worst_group,
                'best_group': best_group,
                'ratio': sr_ratio
            })
        
        return issues
    
    def mitigate_bias_postprocessing(self, X_train, y_train, A_train, X_test, A_test):
        """后处理偏见缓解"""
        # 使用ThresholdOptimizer进行后处理
        postprocessor = ThresholdOptimizer(
            estimator=self.model,
            constraints='demographic_parity',  # 或'equalized_odds'
            prefit=True
        )
        
        # 在训练数据上拟合后处理器
        postprocessor.fit(X_train, y_train, sensitive_features=A_train)
        
        # 在测试数据上应用
        y_pred_fair = postprocessor.predict(X_test, sensitive_features=A_test)
        
        # 评估缓解效果
        original_metrics = self.fairness_metrics
        mitigated_metrics = self.evaluate_fairness(X_test, y_pred_fair, A_test)
        
        comparison = {
            'original': {
                'demographic_parity_diff': original_metrics.get('demographic_parity_difference', 0),
                'equalized_odds_diff': original_metrics.get('equalized_odds_difference', 0),
                'accuracy': original_metrics['overall']['accuracy']
            },
            'mitigated': {
                'demographic_parity_diff': mitigated_metrics.get('demographic_parity_difference', 0),
                'equalized_odds_diff': mitigated_metrics.get('equalized_odds_difference', 0),
                'accuracy': mitigated_metrics['overall']['accuracy']
            },
            'improvement': {
                'dp_improvement': original_metrics.get('demographic_parity_difference', 0) - 
                                 mitigated_metrics.get('demographic_parity_difference', 0),
                'eo_improvement': original_metrics.get('equalized_odds_difference', 0) - 
                                 mitigated_metrics.get('equalized_odds_difference', 0),
                'accuracy_change': mitigated_metrics['overall']['accuracy'] - 
                                  original_metrics['overall']['accuracy']
            }
        }
        
        self.bias_mitigation_results['postprocessing'] = {
            'postprocessor': postprocessor,
            'predictions': y_pred_fair,
            'comparison': comparison
        }
        
        return comparison
    
    def mitigate_bias_reduction(self, X_train, y_train, A_train, X_test, y_test, A_test):
        """减少方法偏见缓解"""
        # 使用ExponentiatedGradient减少偏见
        estimator = RandomForestClassifier(n_estimators=50, random_state=42)
        
        # 定义公平性约束
        constraint = DemographicParity()
        
        # 创建减少对象
        mitigator = ExponentiatedGradient(
            estimator=estimator,
            constraints=constraint,
            eps=0.01  # 容忍度
        )
        
        # 拟合缓解器
        mitigator.fit(X_train, y_train, sensitive_features=A_train)
        
        # 预测
        y_pred_mitigated = mitigator.predict(X_test)
        
        # 评估
        mitigated_metrics = self.evaluate_fairness(X_test, y_pred_mitigated, A_test)
        
        # 与原始模型比较
        original_y_pred = self.model.predict(X_test)
        original_metrics = self.evaluate_fairness(X_test, original_y_pred, A_test)
        
        comparison = {
            'original': {
                'demographic_parity_diff': original_metrics.get('demographic_parity_difference', 0),
                'accuracy': original_metrics['overall']['accuracy']
            },
            'mitigated': {
                'demographic_parity_diff': mitigated_metrics.get('demographic_parity_difference', 0),
                'accuracy': mitigated_metrics['overall']['accuracy']
            }
        }
        
        self.bias_mitigation_results['reduction'] = {
            'mitigator': mitigator,
            'predictions': y_pred_mitigated,
            'comparison': comparison
        }
        
        return comparison

    def _generate_recommendations(self) -> List[str]:

        if dp_diff > 0.2 or eo_diff > 0.2:

        # 基于问题类型的建议

            for issue in metrics['fairness_issues']:
                if issue['type'] == 'demographic_parity_violation':
                    recommendations.append("实施人口统计均等约束，如使用后处理或减少方法")

        recommendations.append("提供模型决策解释，增加透明度")
        recommendations.append("建立用户反馈和申诉机制")
        
        return recommendations

    def _estimate_risk_level(self, system_desc: str, category: str) -> str:
        """估计风险等级"""
        # 简化版风险估计逻辑
        keyword_mapping = {
            'fairness': ['招聘', '贷款', '医疗', '司法', '信用'],
            'privacy': ['个人数据', '敏感信息', '医疗记录', '财务信息'],
            'safety': ['自动驾驶', '医疗设备', '工业控制', '物理交互'],
            'transparency': ['黑盒', '复杂模型', '深度神经网络', '自动决策'],
            'accountability': ['高风险', '法律后果', '人身伤害', '重大损失']
        }
        
        keywords = keyword_mapping.get(category, [])
        matches = sum(1 for kw in keywords if kw in system_desc)
        
        if matches >= 3:
            return 'high'
        elif matches >= 1:
            return 'medium'
        else:
            return 'low'

        avg_score = sum(scores) / len(scores)
        
        if avg_score >= 3:
            return 'high'
        elif avg_score >= 2:
            return 'medium'
        else:
            return 'low'
```
