**题目**: 实现一个性能监控装饰器，用于测量函数执行时间。

**要求**:

4. 应用到Agent的关键函数上

**代码模板**:

```python
from functools import wraps
from typing import Dict, List

class PerformanceMonitor:
    """性能监控器"""
    
    def __init__(self):
        self.stats: Dict[str, List[float]] = {}
    
    def record(self, func_name: str, duration: float):
        """记录执行时间"""
        pass
    
    def get_stats(self, func_name: str) -> Dict:
        """获取函数统计信息"""
        pass
    
    def print_summary(self):
        """打印性能统计摘要"""
        pass

def monitor(monitor: PerformanceMonitor):
    """性能监控装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            start = time.time()
            result = func(*args, **kwargs)
            duration = time.time() - start
            monitor.record(func.__name__, duration)
            return result
        return wrapper
    return decorator
```
