### 练习3：实现连接池
**题目**: 实现一个简单的数据库连接池。

**要求**:
1. 支持最小/最大连接数
2. 支持连接获取和归还
3. 自动回收空闲连接
4. 线程安全

**代码框架**:
```python
import threading
import time
from queue import Queue, Empty
from typing import Optional

class Connection:
    """数据库连接模拟"""
    def __init__(self, conn_id: int):
        self.conn_id = conn_id
        self.created_at = time.time()
        self.last_used = time.time()
    
    def is_valid(self, max_idle_time: float) -> bool:
        return time.time() - self.last_used < max_idle_time
    
    def execute(self, query: str):
        """模拟执行查询"""
        self.last_used = time.time()
        return f"Result from connection {self.conn_id}"

class ConnectionPool:
    """连接池实现"""
    
    def __init__(
        self, 
        min_size: int = 2, 
        max_size: int = 10,
        max_idle_time: float = 300,
        connection_timeout: float = 30
    ):
        self.min_size = min_size
        self.max_size = max_size
        self.max_idle_time = max_idle_time
        self.connection_timeout = connection_timeout
        
        self._pool: Queue = Queue(maxsize=max_size)
        self._lock = threading.Lock()
        self._total_connections = 0
        self._shutdown = False
        
        # 初始化最小连接数
        self._init_connections()
    
    def _init_connections(self):
        """初始化连接"""
        for _ in range(self.min_size):
            conn = Connection(self._total_connections)
            self._total_connections += 1
            self._pool.put(conn)
    
    def get_connection(self, timeout: Optional[float] = None) -> Connection:
        """获取连接"""
        timeout = timeout or self.connection_timeout
        
        try:
            conn = self._pool.get(timeout=timeout)
            if conn.is_valid(self.max_idle_time):
                return conn
            # 连接已过期，创建新的
        except Empty:
            # 创建新连接
            with self._lock:
                if self._total_connections < self.max_size:
                    conn = Connection(self._total_connections)
                    self._total_connections += 1
                    return conn
        
        # 返回新创建的连接
        with self._lock:
            conn = Connection(self._total_connections)
            self._total_connections += 1
            return conn
    
    def return_connection(self, conn: Connection):
        """归还连接"""
        if self._shutdown:
            return
        self._pool.put(conn)
    
    def close(self):
        """关闭连接池"""
        self._shutdown = True
        while not self._pool.empty():
            try:
                self._pool.get_nowait()
            except Empty:
                break
```

**知识点**: 连接池、线程安全、资源管理
**难度**: 进阶
**建议时间**: 15分钟

### 练习4：连接池上下文管理器
**题目**: 实现连接池的上下文管理器，自动管理连接获取和归还。

**要求**:
1. 使用contextmanager实现
2. 自动处理异常情况
3. 支持连接复用

**代码框架**:
```python
from contextlib import contextmanager
from typing import Generator

class PooledConnection:
    """池化连接的上下文管理器"""
    
    def __init__(self, pool: ConnectionPool):
        self._pool = pool
        self._conn: Optional[Connection] = None
    
    def __enter__(self) -> Connection:
        self._conn = self._pool.get_connection()
        return self._conn
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self._conn:
            self._pool.return_connection(self._conn)
        return False

@contextmanager
def get_pooled_connection(pool: ConnectionPool) -> Generator[Connection, None, None]:
    """获取池化连接的上下文管理器"""
    conn = pool.get_connection()
    try:
        yield conn
    finally:
        pool.return_connection(conn)
```

**知识点**: 上下文管理器、异常处理、资源管理
