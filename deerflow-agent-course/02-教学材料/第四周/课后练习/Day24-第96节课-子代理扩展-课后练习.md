### 练习3：子代理管理器
**题目**: 实现子代理的创建和管理。

```python
import uuid
from typing import Dict, Any, List
from dataclasses import dataclass

@dataclass
class SubAgent:
    id: str
    name: str
    config: Dict[str, Any]
    status: str = "idle"

class SubAgentManager:
    def __init__(self, max_agents: int = 5):
        self.max_agents = max_agents
        self.agents: Dict[str, SubAgent] = {}
    
    def create_agent(self, name: str, config: Dict[str, Any]) -> str:
        agent_id = str(uuid.uuid4())
        self.agents[agent_id] = SubAgent(agent_id, name, config)
        return agent_id
    
    def get_agent(self, agent_id: str) -> SubAgent:
        return self.agents.get(agent_id)
```
