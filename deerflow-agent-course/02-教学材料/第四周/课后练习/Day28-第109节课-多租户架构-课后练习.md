tenant_context: ContextVar[str] = ContextVar('tenant_id')

class TenantContext:
    @staticmethod
    def set(tenant_id: str):
        tenant_context.set(tenant_id)
    
    @staticmethod
    def get() -> str:
        return tenant_context.get()
