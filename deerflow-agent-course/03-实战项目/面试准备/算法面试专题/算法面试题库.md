```python
import time
from functools import wraps


def timing_decorator(func):
    """记录函数执行时间的装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} 执行耗时: {end - start:.4f}秒")
        return result
    return wrapper


@timing_decorator
def slow_function():
    time.sleep(1)
    return "完成"
```

**题目**：
实现一个上下文管理器，用于管理数据库连接。

**参考解答**：

```python
from contextlib import contextmanager


class DatabaseConnection:
    """数据库连接"""
    
    def __init__(self, connection_string: str):
        self.connection_string = connection_string
        self.connected = False
    
    def connect(self):
        print(f"连接数据库: {self.connection_string}")
        self.connected = True
    
    def disconnect(self):
        print("断开数据库连接")
        self.connected = False
    
    def execute(self, query: str):
        if not self.connected:
            raise RuntimeError("未连接数据库")
        print(f"执行查询: {query}")
        return f"结果: {query}"


class DatabaseManager:
    """数据库管理器"""
    
    def __init__(self, connection_string: str):
        self.connection_string = connection_string
        self.connection = None
    
    def __enter__(self):
        self.connection = DatabaseConnection(self.connection_string)
        self.connection.connect()
        return self.connection
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.connection:
            self.connection.disconnect()
        return False  # 不吞掉异常


@contextmanager
def db_transaction(connection_string: str):
    """事务上下文管理器"""
    conn = DatabaseConnection(connection_string)
    try:
        conn.connect()
        yield conn
        print("提交事务")
    except Exception as e:
        print(f"回滚事务: {e}")
        raise
    finally:
        conn.disconnect()


# 使用示例
with DatabaseManager("mysql://localhost/test") as db:
    result = db.execute("SELECT * FROM users")
    print(result)


# 使用contextmanager
with db_transaction("mysql://localhost/test") as db:
    result = db.execute("INSERT INTO users VALUES (1, 'John')")
```

### 题目4：异步编程

**难度**：⭐⭐⭐⭐

**题目**：
实现一个异步任务调度器，支持并发执行和超时控制。

**参考解答**：

```python
import asyncio
from typing import Callable, Any, Optional
from dataclasses import dataclass
from enum import Enum
import time


class TaskStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    TIMEOUT = "timeout"


@dataclass
class Task:
    task_id: str
    func: Callable
    args: tuple = ()
    kwargs: dict = None
    status: TaskStatus = TaskStatus.PENDING
    result: Any = None
    error: Optional[str] = None
    start_time: float = 0
    end_time: float = 0
    
    def __post_init__(self):
        if self.kwargs is None:
            self.kwargs = {}


class AsyncTaskScheduler:
    """异步任务调度器"""
    
    def __init__(self, max_concurrent: int = 10):
        self.max_concurrent = max_concurrent
        self.tasks: dict[str, Task] = {}
        self.semaphore = asyncio.Semaphore(max_concurrent)
    
    async def execute_task(self, task: Task, timeout: Optional[float] = None) -> Any:
        """执行单个任务"""
        async with self.semaphore:
            task.status = TaskStatus.RUNNING
            task.start_time = time.time()
            
            try:
                if asyncio.iscoroutinefunction(task.func):
                    if timeout:
                        task.result = await asyncio.wait_for(
                            task.func(*task.args, **task.kwargs),
                            timeout=timeout
                        )
                    else:
                        task.result = await task.func(*task.args, **task.kwargs)
                else:
                    # 同步函数在线程池中执行
                    loop = asyncio.get_event_loop()
                    task.result = await loop.run_in_executor(
                        None,
                        lambda: task.func(*task.args, **task.kwargs)
                    )
                
                task.status = TaskStatus.COMPLETED
                return task.result
                
            except asyncio.TimeoutError:
                task.status = TaskStatus.TIMEOUT
                task.error = f"任务超时 ({timeout}秒)"
            except Exception as e:
                task.status = TaskStatus.FAILED
                task.error = str(e)
            finally:
                task.end_time = time.time()
    
    async def submit(
        self,
        func: Callable,
        *args,
        task_id: Optional[str] = None,
        timeout: Optional[float] = None,
        **kwargs
    ) -> str:
        """提交任务"""
        if task_id is None:
            task_id = f"task_{len(self.tasks)}"
        
        task = Task(
            task_id=task_id,
            func=func,
            args=args,
            kwargs=kwargs
        )
        self.tasks[task_id] = task
        
        asyncio.create_task(self.execute_task(task, timeout))
        return task_id
    
    async def wait_all(self) -> dict[str, Task]:
        """等待所有任务完成"""
        while any(t.status in (TaskStatus.PENDING, TaskStatus.RUNNING) 
                 for t in self.tasks.values()):
            await asyncio.sleep(0.1)
        return self.tasks
    
    def get_result(self, task_id: str) -> Optional[Any]:
        """获取任务结果"""
        task = self.tasks.get(task_id)
        return task.result if task else None


# 使用示例
async def main():
    async def fetch_data(url: str, delay: float):
        await asyncio.sleep(delay)
        return f"数据: {url}"
    
    scheduler = AsyncTaskScheduler(max_concurrent=5)
    
    # 提交多个任务
    task_ids = []
    for i in range(5):
        task_id = await scheduler.submit(
            fetch_data,
            f"http://example.com/{i}",
            float(i + 1),
            timeout=10.0
        )
        task_ids.append(task_id)
        print(f"提交任务: {task_id}")
    
    # 等待完成
    results = await scheduler.wait_all()
    
    for task_id, task in results.items():
        print(f"{task_id}: {task.status.value} - {task.result or task.error}")


asyncio.run(main())
```

### 题目5：LRU缓存实现

**难度**：⭐⭐⭐

**题目**：
实现一个LRU（最近最少使用）缓存。

**参考解答**：

```python
from collections import OrderedDict
from typing import TypeVar, Generic, Optional, Callable
import threading


T = TypeVar('T')


class LRUCache(Generic[T]):
    """线程安全的LRU缓存"""
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache: OrderedDict[str, T] = OrderedDict()
        self.lock = threading.RLock()
    
    def get(self, key: str) -> Optional[T]:
        """获取值"""
        with self.lock:
            if key not in self.cache:
                return None
            
            self.cache.move_to_end(key)
            return self.cache[key]
    
    def put(self, key: str, value: T) -> None:
        """存入值"""
        with self.lock:
            if key in self.cache:
                self.cache.move_to_end(key)
            else:
                if len(self.cache) >= self.capacity:
                    self.cache.popitem(last=False)
            
            self.cache[key] = value
    
    def delete(self, key: str) -> bool:
        """删除键"""
        with self.lock:
            if key in self.cache:
                del self.cache[key]
                return True
            return False
    
    def clear(self) -> None:
        """清空缓存"""
        with self.lock:
            self.cache.clear()
    
    def __len__(self) -> int:
        return len(self.cache)
    
    def __contains__(self, key: str) -> bool:
        return key in self.cache


class MemoizedFunction(Generic[T]):
    """带LRU缓存的函数"""
    
    def __init__(self, func: Callable, capacity: int = 128):
        self.func = func
        self.cache = LRUCache(capacity)
    
    def __call__(self, *args, **kwargs) -> T:
        # 构建缓存键
        key = str(args) + str(sorted(kwargs.items()))
        
        # 尝试从缓存获取
        result = self.cache.get(key)
        if result is not None:
            return result
        
        # 计算并缓存
        result = self.func(*args, **kwargs)
        self.cache.put(key, result)
        return result


# 使用示例
cache = LRUCache[str](capacity=3)
cache.put("a", "value_a")
cache.put("b", "value_b")
cache.put("c", "value_c")

print(cache.get("a"))  # value_a
cache.put("d", "value_d")  # 淘汰 'b'
print(cache.get("b"))  # None


# 装饰器版本
@MemoizedFunction(capacity=100)
def expensive_computation(n: int) -> int:
    print(f"计算中: {n}")
    return n * n


print(expensive_computation(5))  # 计算中: 5, 25
print(expensive_computation(5))  # 25 (使用缓存)
```

### 题目6： Trie树实现

**难度**：⭐⭐⭐⭐

**题目**：
实现一个Trie树（前缀树），支持插入、搜索和前缀匹配。

**参考解答**：

```python
from typing import Optional


class TrieNode:
    """Trie树节点"""
    
    def __init__(self):
        self.children: dict[str, TrieNode] = {}
        self.is_end: bool = False
        self.count: int = 0  # 以此节点结束的词的数量


class Trie:
    """前缀树"""
    
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word: str) -> None:
        """插入单词"""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
        node.count += 1
    
    def search(self, word: str) -> bool:
        """搜索单词"""
        node = self._search_prefix(word)
        return node is not None and node.is_end
    
    def starts_with(self, prefix: str) -> bool:
        """检查前缀"""
        return self._search_prefix(prefix) is not None
    
    def _search_prefix(self, prefix: str) -> Optional[TrieNode]:
        """查找前缀对应的节点"""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return None
            node = node.children[char]
        return node
    
    def autocomplete(self, prefix: str) -> list[str]:
        """自动补全"""
        node = self._search_prefix(prefix)
        if not node:
            return []
        
        results = []
        self._dfs(node, prefix, results)
        return results
    
    def _dfs(self, node: TrieNode, current: str, results: list[str]) -> None:
        """深度优先搜索收集所有单词"""
        if node.is_end:
            results.append(current)
        
        for char, child in node.children.items():
            self._dfs(child, current + char, results)
    
    def word_count(self, word: str) -> int:
        """返回单词出现次数"""
        node = self._search_prefix(word)
        return node.count if node and node.is_end else 0
    
    def delete(self, word: str) -> bool:
        """删除单词"""
        def _delete(node: TrieNode, word: str, depth: int) -> bool:
            if depth == len(word):
                if not node.is_end:
                    return False
                node.is_end = False
                node.count = max(0, node.count - 1)
                return len(node.children) == 0
            
            char = word[depth]
            if char not in node.children:
                return False
            
            child = node.children[char]
            should_delete_child = _delete(child, word, depth + 1)
            
            if should_delete_child:
                del node.children[char]
                return len(node.children) == 0 and not node.is_end
            
            return False
        
        if not self.search(word):
            return False
        
        _delete(self.root, word, 0)
        return True


# 使用示例
trie = Trie()

words = ["apple", "app", "application", "apply", "banana", "band"]
for word in words:
    trie.insert(word)

print(trie.search("apple"))      # True
print(trie.search("app"))        # True
print(trie.search("apps"))       # False
print(trie.starts_with("app"))   # True
print(trie.starts_with("ban"))   # True

print(trie.autocomplete("app"))  # ['app', 'apple', 'application', 'apply']
print(trie.autocomplete("ban"))  # ['banana', 'band']
```

### 题目7：优先队列实现

**难度**：⭐⭐⭐⭐

**题目**：
实现一个支持自定义优先级的优先队列。

**参考解答**：

```python
import heapq
from typing import Any, Optional
from dataclasses import dataclass, field
from enum import Enum


class Priority(Enum):
    LOW = 3
    NORMAL = 2
    HIGH = 1
    CRITICAL = 0


@dataclass(order=True)
class PrioritizedItem:
    """优先队列项"""
    priority: int
    sequence: int = field(compare=False)
    data: Any = field(compare=False)


class PriorityQueue:
    """优先队列"""
    
    def __init__(self):
        self._heap: list[PrioritizedItem] = []
        self._counter = 0
        self._lock = __import__('threading').RLock()
    
    def enqueue(self, item: Any, priority: int = Priority.NORMAL.value) -> None:
        """入队"""
        with self._lock:
            prioritized = PrioritizedItem(
                priority=priority,
                sequence=self._counter,
                data=item
            )
            heapq.heappush(self._heap, prioritized)
            self._counter += 1
    
    def dequeue(self) -> Optional[Any]:
        """出队"""
        with self._lock:
            if not self._heap:
                return None
            item = heapq.heappop(self._heap)
            return item.data
    
    def peek(self) -> Optional[Any]:
        """查看队首元素"""
        with self._lock:
            if not self._heap:
                return None
            return self._heap[0].data
    
    def size(self) -> int:
        """队列大小"""
        with self._lock:
            return len(self._heap)
    
    def is_empty(self) -> bool:
        """是否为空"""
        return self.size() == 0
    
    def clear(self) -> None:
        """清空队列"""
        with self._lock:
            self._heap.clear()
            self._counter = 0


class AgentTaskQueue(PriorityQueue):
    """Agent任务队列"""
    
    def enqueue_task(self, task: dict, priority: Priority = Priority.NORMAL) -> None:
        """入队任务"""
        self.enqueue(task, priority.value)
    
    def get_high_priority_tasks(self, count: int = 10) -> list[dict]:
        """获取高优先级任务"""
        tasks = []
        original_heap = self._heap.copy()
        
        for _ in range(min(count, len(self._heap))):
            item = self.dequeue()
            if item:
                tasks.append(item)
        
        # 恢复队列
        for item in original_heap:
            if item not in self._heap:
                heapq.heappush(self._heap, item)
        
        return tasks


# 使用示例
queue = AgentTaskQueue()

tasks = [
    {"id": 1, "type": "analysis"},
    {"id": 2, "type": "report"},
    {"id": 3, "type": "urgent"},
]

queue.enqueue_task(tasks[0], Priority.NORMAL)
queue.enqueue_task(tasks[1], Priority.LOW)
queue.enqueue_task(tasks[2], Priority.HIGH)

print(f"队列大小: {queue.size()}")
print(f"下一个任务: {queue.peek()}")

while not queue.is_empty():
    task = queue.dequeue()
    print(f"处理任务: {task}")
```

### 题目8：并发任务合并

**难度**：⭐⭐⭐⭐

**题目**：
实现一个函数，合并多个异步任务的返回结果。

**参考解答**：

```python
import asyncio
from typing import List, Callable, Any, TypeVar, Dict
from dataclasses import dataclass
from enum import Enum


T = TypeVar('T')


class TaskStrategy(Enum):
    SEQUENTIAL = "sequential"
    PARALLEL = "parallel"
    BATCH = "batch"


@dataclass
class TaskResult:
    """任务结果"""
    index: int
    success: bool
    data: Any = None
    error: str = None


async def merge_task_results(
    tasks: List[Callable],
    strategy: TaskStrategy = TaskStrategy.PARALLEL,
    batch_size: int = 5
) -> List[TaskResult]:
    """合并任务结果"""
    
    if strategy == TaskStrategy.SEQUENTIAL:
        return await _sequential_execution(tasks)
    
    elif strategy == TaskStrategy.PARALLEL:
        return await _parallel_execution(tasks)
    
    elif strategy == TaskStrategy.BATCH:
        return await _batch_execution(tasks, batch_size)
    
    raise ValueError(f"Unknown strategy: {strategy}")


async def _sequential_execution(tasks: List[Callable]) -> List[TaskResult]:
    """顺序执行"""
    results = []
    for i, task in enumerate(tasks):
        try:
            result = task() if not asyncio.iscoroutinefunction(task) else await task()
            results.append(TaskResult(index=i, success=True, data=result))
        except Exception as e:
            results.append(TaskResult(index=i, success=False, error=str(e)))
    return results


async def _parallel_execution(tasks: List[Callable]) -> List[TaskResult]:
    """并行执行"""
    async def safe_execute(i: int, task: Callable) -> TaskResult:
        try:
            result = task() if not asyncio.iscoroutinefunction(task) else await task()
            return TaskResult(index=i, success=True, data=result)
        except Exception as e:
            return TaskResult(index=i, success=False, error=str(e))
    
    coroutines = [safe_execute(i, task) for i, task in enumerate(tasks)]
    return await asyncio.gather(*coroutines)


async def _batch_execution(tasks: List[Callable], batch_size: int) -> List[TaskResult]:
    """批量执行"""
    results = []
    
    for i in range(0, len(tasks), batch_size):
        batch = tasks[i:i + batch_size]
        batch_results = await _parallel_execution(batch)
        results.extend(batch_results)
    
    return results


# 使用示例
async def main():
    async def fetch_data(i: int) -> str:
        await asyncio.sleep(0.1)
        return f"data_{i}"
    
    tasks = [lambda i=i: fetch_data(i) for i in range(10)]
    
    # 并行执行
    results = await merge_task_results(tasks, TaskStrategy.PARALLEL)
    for r in results:
        status = "✓" if r.success else "✗"
        print(f"[{status}] Task {r.index}: {r.data or r.error}")
    
    # 批量执行
    results = await merge_task_results(tasks, TaskStrategy.BATCH, batch_size=3)
    print(f"\n批量执行完成，共 {len(results)} 个结果")


asyncio.run(main())
```

### 题目9：状态机转换

**难度**：⭐⭐⭐⭐⭐

**题目**：
实现一个通用的状态机，支持复杂的状态转换逻辑。

**参考解答**：

```python
from typing import Callable, Dict, List, Optional, Set, Any
from dataclasses import dataclass, field
from enum import Enum, auto


class State(Enum):
    """状态枚举"""
    IDLE = auto()
    RUNNING = auto()
    PAUSED = auto()
    STOPPED = auto()
    ERROR = auto()


class Event(Enum):
    """事件枚举"""
    START = auto()
    STOP = auto()
    PAUSE = auto()
    RESUME = auto()
    FAIL = auto()
    RESET = auto()


@dataclass
class Transition:
    """状态转换定义"""
    from_state: State
    to_state: State
    event: Event
    condition: Optional[Callable[[], bool]] = None
    action: Optional[Callable] = None


@dataclass
class StateContext:
    """状态上下文"""
    current_state: State
    data: dict = field(default_factory=dict)
    history: List[State] = field(default_factory=list)


class StateMachineError(Exception):
    """状态机异常"""
    pass


class StateMachine:
    """通用状态机"""
    
    def __init__(self, initial_state: State):
        self._initial_state = initial_state
        self._context = StateContext(current_state=initial_state)
        self._transitions: Dict[tuple, List[Transition]] = {}
        self._global_handlers: Dict[Event, List[Callable]] = {}
        self._state_entry_handlers: Dict[State, List[Callable]] = {}
        self._state_exit_handlers: Dict[State, List[Callable]] = {}
    
    def add_transition(
        self,
        from_state: State,
        to_state: State,
        event: Event,
        condition: Optional[Callable] = None,
        action: Optional[Callable] = None
    ) -> "StateMachine":
        """添加转换"""
        key = (from_state, event)
        if key not in self._transitions:
            self._transitions[key] = []
        
        self._transitions[key].append(Transition(
            from_state=from_state,
            to_state=to_state,
            event=event,
            condition=condition,
            action=action
        ))
        return self
    
    def on_event(self, event: Event) -> bool:
        """触发事件"""
        key = (self._context.current_state, event)
        
        if key not in self._transitions:
            # 尝试全局处理器
            if event in self._global_handlers:
                for handler in self._global_handlers[event]:
                    handler(self._context)
            return False
        
        for transition in self._transitions[key]:
            # 检查条件
            if transition.condition and not transition.condition():
                continue
            
            # 执行退出动作
            self._execute_handlers(
                self._state_exit_handlers.get(self._context.current_state, [])
            )
            
            # 记录历史
            self._context.history.append(self._context.current_state)
            
            # 执行转换动作
            if transition.action:
                transition.action(self._context)
            
            # 更新状态
            self._context.current_state = transition.to_state
            
            # 执行进入动作
            self._execute_handlers(
                self._state_entry_handlers.get(transition.to_state, [])
            )
            
            return True
        
        return False
    
    def on_enter(self, state: State) -> Callable:
        """进入状态装饰器"""
        def decorator(func: Callable) -> Callable:
            if state not in self._state_entry_handlers:
                self._state_entry_handlers[state] = []
            self._state_entry_handlers[state].append(func)
            return func
        return decorator
    
    def on_exit(self, state: State) -> Callable:
        """退出状态装饰器"""
        def decorator(func: Callable) -> Callable:
            if state not in self._state_exit_handlers:
                self._state_exit_handlers[state] = []
            self._state_exit_handlers[state].append(func)
            return func
        return decorator
    
    def _execute_handlers(self, handlers: List[Callable]):
        """执行处理器"""
        for handler in handlers:
            handler(self._context)
    
    @property
    def current_state(self) -> State:
        return self._context.current_state
    
    def reset(self):
        """重置状态机"""
        self._context = StateContext(current_state=self._initial_state)
        self._context.history.clear()
```
