```python
import time
from functools import wraps


def timing_decorator(func):
    """记录函数执行时间的装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} 执行耗时: {end - start:.4f}秒")
        return result
    return wrapper


@timing_decorator
def slow_function():
    time.sleep(1)
    return "完成"
```

**题目**：
实现一个上下文管理器，用于管理数据库连接。

**参考解答**：

```python
from contextlib import contextmanager


class DatabaseConnection:
    """数据库连接"""
    
    def __init__(self, connection_string: str):
        self.connection_string = connection_string
        self.connected = False
    
    def connect(self):
        print(f"连接数据库: {self.connection_string}")
        self.connected = True
    
    def disconnect(self):
        print("断开数据库连接")
        self.connected = False
    
    def execute(self, query: str):
        if not self.connected:
            raise RuntimeError("未连接数据库")
        print(f"执行查询: {query}")
        return f"结果: {query}"


class DatabaseManager:
    """数据库管理器"""
    
    def __init__(self, connection_string: str):
        self.connection_string = connection_string
        self.connection = None
    
    def __enter__(self):
        self.connection = DatabaseConnection(self.connection_string)
        self.connection.connect()
        return self.connection
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.connection:
            self.connection.disconnect()
        return False  # 不吞掉异常


@contextmanager
def db_transaction(connection_string: str):
    """事务上下文管理器"""
    conn = DatabaseConnection(connection_string)
    try:
        conn.connect()
        yield conn
        print("提交事务")
    except Exception as e:
        print(f"回滚事务: {e}")
        raise
    finally:
        conn.disconnect()


# 使用示例
with DatabaseManager("mysql://localhost/test") as db:
    result = db.execute("SELECT * FROM users")
    print(result)


# 使用contextmanager
with db_transaction("mysql://localhost/test") as db:
    result = db.execute("INSERT INTO users VALUES (1, 'John')")
```

### 题目4：异步编程

**难度**：⭐⭐⭐⭐

**题目**：
实现一个异步任务调度器，支持并发执行和超时控制。

**参考解答**：

```python
import asyncio
from typing import Callable, Any, Optional
from dataclasses import dataclass
from enum import Enum
import time


class TaskStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    TIMEOUT = "timeout"


@dataclass
class Task:
    task_id: str
    func: Callable
    args: tuple = ()
    kwargs: dict = None
    status: TaskStatus = TaskStatus.PENDING
    result: Any = None
    error: Optional[str] = None
    start_time: float = 0
    end_time: float = 0
    
    def __post_init__(self):
        if self.kwargs is None:
            self.kwargs = {}


class AsyncTaskScheduler:
    """异步任务调度器"""
    
    def __init__(self, max_concurrent: int = 10):
        self.max_concurrent = max_concurrent
        self.tasks: dict[str, Task] = {}
        self.semaphore = asyncio.Semaphore(max_concurrent)
    
    async def execute_task(self, task: Task, timeout: Optional[float] = None) -> Any:
        """执行单个任务"""
        async with self.semaphore:
            task.status = TaskStatus.RUNNING
            task.start_time = time.time()
            
            try:
                if asyncio.iscoroutinefunction(task.func):
                    if timeout:
                        task.result = await asyncio.wait_for(
                            task.func(*task.args, **task.kwargs),
                            timeout=timeout
                        )
                    else:
                        task.result = await task.func(*task.args, **task.kwargs)
                else:
                    # 同步函数在线程池中执行
                    loop = asyncio.get_event_loop()
                    task.result = await loop.run_in_executor(
                        None,
                        lambda: task.func(*task.args, **task.kwargs)
                    )
                
                task.status = TaskStatus.COMPLETED
                return task.result
                
            except asyncio.TimeoutError:
                task.status = TaskStatus.TIMEOUT
                task.error = f"任务超时 ({timeout}秒)"
            except Exception as e:
                task.status = TaskStatus.FAILED
                task.error = str(e)
            finally:
                task.end_time = time.time()
    
    async def submit(
        self,
        func: Callable,
        *args,
        task_id: Optional[str] = None,
        timeout: Optional[float] = None,
        **kwargs
    ) -> str:
        """提交任务"""
        if task_id is None:
            task_id = f"task_{len(self.tasks)}"
        
        task = Task(
            task_id=task_id,
            func=func,
            args=args,
            kwargs=kwargs
        )
        self.tasks[task_id] = task
        
        asyncio.create_task(self.execute_task(task, timeout))
        return task_id
    
    async def wait_all(self) -> dict[str, Task]:
        """等待所有任务完成"""
        while any(t.status in (TaskStatus.PENDING, TaskStatus.RUNNING) 
                 for t in self.tasks.values()):
            await asyncio.sleep(0.1)
        return self.tasks
    
    def get_result(self, task_id: str) -> Optional[Any]:
        """获取任务结果"""
        task = self.tasks.get(task_id)
        return task.result if task else None


# 使用示例
async def main():
    async def fetch_data(url: str, delay: float):
        await asyncio.sleep(delay)
        return f"数据: {url}"
    
    scheduler = AsyncTaskScheduler(max_concurrent=5)
    
    # 提交多个任务
    task_ids = []
    for i in range(5):
        task_id = await scheduler.submit(
            fetch_data,
            f"http://example.com/{i}",
            float(i + 1),
            timeout=10.0
        )
        task_ids.append(task_id)
        print(f"提交任务: {task_id}")
    
    # 等待完成
    results = await scheduler.wait_all()
    
    for task_id, task in results.items():
        print(f"{task_id}: {task.status.value} - {task.result or task.error}")


asyncio.run(main())
```

### 题目5：LRU缓存实现

**难度**：⭐⭐⭐

**题目**：
实现一个LRU（最近最少使用）缓存。

**参考解答**：

```python
from collections import OrderedDict
from typing import TypeVar, Generic, Optional, Callable
import threading


T = TypeVar('T')


class LRUCache(Generic[T]):
    """线程安全的LRU缓存"""
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache: OrderedDict[str, T] = OrderedDict()
        self.lock = threading.RLock()
    
    def get(self, key: str) -> Optional[T]:
        """获取值"""
        with self.lock:
            if key not in self.cache:
                return None
            
            self.cache.move_to_end(key)
            return self.cache[key]
    
    def put(self, key: str, value: T) -> None:
        """存入值"""
        with self.lock:
            if key in self.cache:
                self.cache.move_to_end(key)
            else:
                if len(self.cache) >= self.capacity:
                    self.cache.popitem(last=False)
            
            self.cache[key] = value
    
    def delete(self, key: str) -> bool:
        """删除键"""
        with self.lock:
            if key in self.cache:
                del self.cache[key]
                return True
            return False
    
    def clear(self) -> None:
        """清空缓存"""
        with self.lock:
            self.cache.clear()
    
    def __len__(self) -> int:
        return len(self.cache)
    
    def __contains__(self, key: str) -> bool:
        return key in self.cache


class MemoizedFunction(Generic[T]):
    """带LRU缓存的函数"""
    
    def __init__(self, func: Callable, capacity: int = 128):
        self.func = func
        self.cache = LRUCache(capacity)
    
    def __call__(self, *args, **kwargs) -> T:
        # 构建缓存键
        key = str(args) + str(sorted(kwargs.items()))
        
        # 尝试从缓存获取
        result = self.cache.get(key)
        if result is not None:
            return result
        
        # 计算并缓存
        result = self.func(*args, **kwargs)
        self.cache.put(key, result)
        return result


# 使用示例
cache = LRUCache[str](capacity=3)
cache.put("a", "value_a")
cache.put("b", "value_b")
cache.put("c", "value_c")

print(cache.get("a"))  # value_a
cache.put("d", "value_d")  # 淘汰 'b'
print(cache.get("b"))  # None


# 装饰器版本
@MemoizedFunction(capacity=100)
def expensive_computation(n: int) -> int:
    print(f"计算中: {n}")
    return n * n


print(expensive_computation(5))  # 计算中: 5, 25
print(expensive_computation(5))  # 25 (使用缓存)
```

### 题目6： Trie树实现

**难度**：⭐⭐⭐⭐

**题目**：
实现一个Trie树（前缀树），支持插入、搜索和前缀匹配。

**参考解答**：

```python
from typing import Optional


class TrieNode:
    """Trie树节点"""
    
    def __init__(self):
        self.children: dict[str, TrieNode] = {}
        self.is_end: bool = False
        self.count: int = 0  # 以此节点结束的词的数量


class Trie:
    """前缀树"""
    
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word: str) -> None:
        """插入单词"""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
        node.count += 1
    
    def search(self, word: str) -> bool:
        """搜索单词"""
        node = self._search_prefix(word)
        return node is not None and node.is_end
    
    def starts_with(self, prefix: str) -> bool:
        """检查前缀"""
        return self._search_prefix(prefix) is not None
    
    def _search_prefix(self, prefix: str) -> Optional[TrieNode]:
        """查找前缀对应的节点"""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return None
            node = node.children[char]
        return node
    
    def autocomplete(self, prefix: str) -> list[str]:
        """自动补全"""
        node = self._search_prefix(prefix)
        if not node:
            return []
        
        results = []
        self._dfs(node, prefix, results)
        return results
    
    def _dfs(self, node: TrieNode, current: str, results: list[str]) -> None:
        """深度优先搜索收集所有单词"""
        if node.is_end:
            results.append(current)
        
        for char, child in node.children.items():
            self._dfs(child, current + char, results)
    
    def word_count(self, word: str) -> int:
        """返回单词出现次数"""
        node = self._search_prefix(word)
        return node.count if node and node.is_end else 0
    
    def delete(self, word: str) -> bool:
        """删除单词"""
        def _delete(node: TrieNode, word: str, depth: int) -> bool:
            if depth == len(word):
                if not node.is_end:
                    return False
                node.is_end = False
                node.count = max(0, node.count - 1)
                return len(node.children) == 0
            
            char = word[depth]
            if char not in node.children:
                return False
            
            child = node.children[char]
            should_delete_child = _delete(child, word, depth + 1)
            
            if should_delete_child:
                del node.children[char]
                return len(node.children) == 0 and not node.is_end
            
            return False
        
        if not self.search(word):
            return False
        
        _delete(self.root, word, 0)
        return True


# 使用示例
trie = Trie()

words = ["apple", "app", "application", "apply", "banana", "band"]
for word in words:
    trie.insert(word)

print(trie.search("apple"))      # True
print(trie.search("app"))        # True
print(trie.search("apps"))       # False
print(trie.starts_with("app"))   # True
print(trie.starts_with("ban"))   # True

print(trie.autocomplete("app"))  # ['app', 'apple', 'application', 'apply']
print(trie.autocomplete("ban"))  # ['banana', 'band']
```

### 题目7：优先队列实现

**难度**：⭐⭐⭐⭐

**题目**：
实现一个支持自定义优先级的优先队列。

**参考解答**：

```python
import heapq
from typing import Any, Optional
from dataclasses import dataclass, field
from enum import Enum


class Priority(Enum):
    LOW = 3
    NORMAL = 2
    HIGH = 1
    CRITICAL = 0


@dataclass(order=True)
class PrioritizedItem:
    """优先队列项"""
    priority: int
    sequence: int = field(compare=False)
    data: Any = field(compare=False)


class PriorityQueue:
    """优先队列"""
    
    def __init__(self):
        self._heap: list[PrioritizedItem] = []
        self._counter = 0
        self._lock = __import__('threading').RLock()
    
    def enqueue(self, item: Any, priority: int = Priority.NORMAL.value) -> None:
        """入队"""
        with self._lock:
            prioritized = PrioritizedItem(
                priority=priority,
                sequence=self._counter,
                data=item
            )
            heapq.heappush(self._heap, prioritized)
            self._counter += 1
    
    def dequeue(self) -> Optional[Any]:
        """出队"""
        with self._lock:
            if not self._heap:
                return None
            item = heapq.heappop(self._heap)
            return item.data
    
    def peek(self) -> Optional[Any]:
        """查看队首元素"""
        with self._lock:
            if not self._heap:
                return None
            return self._heap[0].data
    
    def size(self) -> int:
        """队列大小"""
        with self._lock:
            return len(self._heap)
    
    def is_empty(self) -> bool:
        """是否为空"""
        return self.size() == 0
    
    def clear(self) -> None:
        """清空队列"""
        with self._lock:
            self._heap.clear()
            self._counter = 0


class AgentTaskQueue(PriorityQueue):
    """Agent任务队列"""
    
    def enqueue_task(self, task: dict, priority: Priority = Priority.NORMAL) -> None:
        """入队任务"""
        self.enqueue(task, priority.value)
    
    def get_high_priority_tasks(self, count: int = 10) -> list[dict]:
        """获取高优先级任务"""
        tasks = []
        original_heap = self._heap.copy()
        
        for _ in range(min(count, len(self._heap))):
            item = self.dequeue()
            if item:
                tasks.append(item)
        
        # 恢复队列
        for item in original_heap:
            if item not in self._heap:
                heapq.heappush(self._heap, item)
        
        return tasks


# 使用示例
queue = AgentTaskQueue()

tasks = [
    {"id": 1, "type": "analysis"},
    {"id": 2, "type": "report"},
    {"id": 3, "type": "urgent"},
]

queue.enqueue_task(tasks[0], Priority.NORMAL)
queue.enqueue_task(tasks[1], Priority.LOW)
queue.enqueue_task(tasks[2], Priority.HIGH)

print(f"队列大小: {queue.size()}")
print(f"下一个任务: {queue.peek()}")

while not queue.is_empty():
    task = queue.dequeue()
    print(f"处理任务: {task}")
```

### 题目8：并发任务合并

**难度**：⭐⭐⭐⭐

**题目**：
实现一个函数，合并多个异步任务的返回结果。

**参考解答**：

```python
import asyncio
from typing import List, Callable, Any, TypeVar, Dict
from dataclasses import dataclass
from enum import Enum


T = TypeVar('T')


class TaskStrategy(Enum):
    SEQUENTIAL = "sequential"
    PARALLEL = "parallel"
    BATCH = "batch"


@dataclass
class TaskResult:
    """任务结果"""
    index: int
    success: bool
    data: Any = None
    error: str = None


async def merge_task_results(
    tasks: List[Callable],
    strategy: TaskStrategy = TaskStrategy.PARALLEL,
    batch_size: int = 5
) -> List[TaskResult]:
    """合并任务结果"""
    
    if strategy == TaskStrategy.SEQUENTIAL:
        return await _sequential_execution(tasks)
    
    elif strategy == TaskStrategy.PARALLEL:
        return await _parallel_execution(tasks)
    
    elif strategy == TaskStrategy.BATCH:
        return await _batch_execution(tasks, batch_size)
    
    raise ValueError(f"Unknown strategy: {strategy}")


async def _sequential_execution(tasks: List[Callable]) -> List[TaskResult]:
    """顺序执行"""
    results = []
    for i, task in enumerate(tasks):
        try:
            result = task() if not asyncio.iscoroutinefunction(task) else await task()
            results.append(TaskResult(index=i, success=True, data=result))
        except Exception as e:
            results.append(TaskResult(index=i, success=False, error=str(e)))
    return results


async def _parallel_execution(tasks: List[Callable]) -> List[TaskResult]:
    """并行执行"""
    async def safe_execute(i: int, task: Callable) -> TaskResult:
        try:
            result = task() if not asyncio.iscoroutinefunction(task) else await task()
            return TaskResult(index=i, success=True, data=result)
        except Exception as e:
            return TaskResult(index=i, success=False, error=str(e))
    
    coroutines = [safe_execute(i, task) for i, task in enumerate(tasks)]
    return await asyncio.gather(*coroutines)


async def _batch_execution(tasks: List[Callable], batch_size: int) -> List[TaskResult]:
    """批量执行"""
    results = []
    
    for i in range(0, len(tasks), batch_size):
        batch = tasks[i:i + batch_size]
        batch_results = await _parallel_execution(batch)
        results.extend(batch_results)
    
    return results


# 使用示例
async def main():
    async def fetch_data(i: int) -> str:
        await asyncio.sleep(0.1)
        return f"data_{i}"
    
    tasks = [lambda i=i: fetch_data(i) for i in range(10)]
    
    # 并行执行
    results = await merge_task_results(tasks, TaskStrategy.PARALLEL)
    for r in results:
        status = "✓" if r.success else "✗"
        print(f"[{status}] Task {r.index}: {r.data or r.error}")
    
    # 批量执行
    results = await merge_task_results(tasks, TaskStrategy.BATCH, batch_size=3)
    print(f"\n批量执行完成，共 {len(results)} 个结果")


asyncio.run(main())
```

### 题目9：状态机转换

**难度**：⭐⭐⭐⭐⭐

**题目**：
实现一个通用的状态机，支持复杂的状态转换逻辑。

**参考解答**：

```python
from typing import Callable, Dict, List, Optional, Set, Any
from dataclasses import dataclass, field
from enum import Enum, auto


class State(Enum):
    """状态枚举"""
    IDLE = auto()
    RUNNING = auto()
    PAUSED = auto()
    STOPPED = auto()
    ERROR = auto()


class Event(Enum):
    """事件枚举"""
    START = auto()
    STOP = auto()
    PAUSE = auto()
    RESUME = auto()
    FAIL = auto()
    RESET = auto()


@dataclass
class Transition:
    """状态转换定义"""
    from_state: State
    to_state: State
    event: Event
    condition: Optional[Callable[[], bool]] = None
    action: Optional[Callable] = None


@dataclass
class StateContext:
    """状态上下文"""
    current_state: State
    data: dict = field(default_factory=dict)
    history: List[State] = field(default_factory=list)


class StateMachineError(Exception):
    """状态机异常"""
    pass


class StateMachine:
    """通用状态机"""
    
    def __init__(self, initial_state: State):
        self._initial_state = initial_state
        self._context = StateContext(current_state=initial_state)
        self._transitions: Dict[tuple, List[Transition]] = {}
        self._global_handlers: Dict[Event, List[Callable]] = {}
        self._state_entry_handlers: Dict[State, List[Callable]] = {}
        self._state_exit_handlers: Dict[State, List[Callable]] = {}
    
    def add_transition(
        self,
        from_state: State,
        to_state: State,
        event: Event,
        condition: Optional[Callable] = None,
        action: Optional[Callable] = None
    ) -> "StateMachine":
        """添加转换"""
        key = (from_state, event)
        if key not in self._transitions:
            self._transitions[key] = []
        
        self._transitions[key].append(Transition(
            from_state=from_state,
            to_state=to_state,
            event=event,
            condition=condition,
            action=action
        ))
        return self
    
    def on_event(self, event: Event) -> bool:
        """触发事件"""
        key = (self._context.current_state, event)
        
        if key not in self._transitions:
            # 尝试全局处理器
            if event in self._global_handlers:
                for handler in self._global_handlers[event]:
                    handler(self._context)
            return False
        
        for transition in self._transitions[key]:
            # 检查条件
            if transition.condition and not transition.condition():
                continue
            
            # 执行退出动作
            self._execute_handlers(
                self._state_exit_handlers.get(self._context.current_state, [])
            )
            
            # 记录历史
            self._context.history.append(self._context.current_state)
            
            # 执行转换动作
            if transition.action:
                transition.action(self._context)
            
            # 更新状态
            self._context.current_state = transition.to_state
            
            # 执行进入动作
            self._execute_handlers(
                self._state_entry_handlers.get(transition.to_state, [])
            )
            
            return True
        
        return False
    
    def on_enter(self, state: State) -> Callable:
        """进入状态装饰器"""
        def decorator(func: Callable) -> Callable:
            if state not in self._state_entry_handlers:
                self._state_entry_handlers[state] = []
            self._state_entry_handlers[state].append(func)
            return func
        return decorator
    
    def on_exit(self, state: State) -> Callable:
        """退出状态装饰器"""
        def decorator(func: Callable) -> Callable:
            if state not in self._state_exit_handlers:
                self._state_exit_handlers[state] = []
            self._state_exit_handlers[state].append(func)
            return func
        return decorator
    
    def _execute_handlers(self, handlers: List[Callable]):
        """执行处理器"""
        for handler in handlers:
            handler(self._context)
    
    @property
    def current_state(self) -> State:
        return self._context.current_state
    
    def reset(self):
        """重置状态机"""
        self._context = StateContext(current_state=self._initial_state)
        self._context.history.clear()
```

### 题目10：Agent任务调度算法

**难度**：⭐⭐⭐⭐  
**应用场景**：多Agent系统中任务分配和调度

**问题描述**：
设计一个任务调度算法，将N个任务分配给M个Agent执行。每个任务有：
- 预估执行时间 t_i
- 优先级 p_i (1-5, 5最高)
- 所需资源 r_i (CPU、内存、GPU)
每个Agent有：
- 处理能力 c_j
- 当前负载 l_j
- 支持的工具集 T_j

目标：最小化总体完成时间，同时考虑优先级和资源约束。

**考察重点**：
1. 多目标优化问题建模
2. 贪心算法与启发式策略
3. 资源约束调度
4. 实时调度与动态调整

**参考解答**：

```python
from typing import List, Dict, Tuple
from dataclasses import dataclass
from enum import Enum
import heapq
from collections import defaultdict


class ResourceType(Enum):
    CPU = "cpu"
    MEMORY = "memory"
    GPU = "gpu"


@dataclass
class Task:
    task_id: str
    estimated_time: float  # 预估执行时间
    priority: int          # 优先级 1-5
    resources: Dict[ResourceType, float]  # 资源需求
    required_tools: List[str]  # 所需工具


@dataclass
class Agent:
    agent_id: str
    capacity: Dict[ResourceType, float]  # 资源容量
    current_load: Dict[ResourceType, float]  # 当前负载
    available_tools: List[str]  # 可用工具
    processing_tasks: List[str]  # 正在处理的任务


class TaskScheduler:
    """任务调度器"""
    
    def __init__(self, agents: List[Agent]):
        self.agents = {a.agent_id: a for a in agents}
        self.task_queue = []  # 优先队列 (优先级, 到达时间, 任务)
        self._counter = 0
    
    def add_task(self, task: Task):
        """添加任务到队列"""
        # 优先级高的先出队，所以用负值
        heapq.heappush(self.task_queue, (-task.priority, self._counter, task))
        self._counter += 1
    
    def can_assign(self, agent: Agent, task: Task) -> bool:
        """检查Agent是否能执行任务"""
        # 检查工具
        if not all(tool in agent.available_tools for tool in task.required_tools):
            return False
        
        # 检查资源
        for res_type, need in task.resources.items():
            available = agent.capacity[res_type] - agent.current_load.get(res_type, 0)
            if available < need:
                return False
        
        return True
    
    def schedule(self) -> Dict[str, List[Task]]:
        """调度任务"""
        assignments = defaultdict(list)
        
        # 复制队列以便遍历
        temp_queue = self.task_queue.copy()
        scheduled_tasks = []
        
        while temp_queue:
            _, _, task = heapq.heappop(temp_queue)
            best_agent = None
            best_score = float('-inf')
            
            for agent in self.agents.values():
                if not self.can_assign(agent, task):
                    continue
                
                # 评分函数：考虑负载均衡和优先级
                score = self._calculate_score(agent, task)
                if score > best_score:
                    best_score = score
                    best_agent = agent
            
            if best_agent:
                # 分配任务
                assignments[best_agent.agent_id].append(task)
                # 更新Agent负载
                for res_type, need in task.resources.items():
                    best_agent.current_load[res_type] = \
                        best_agent.current_load.get(res_type, 0) + need
                best_agent.processing_tasks.append(task.task_id)
                scheduled_tasks.append(task)
        
        # 从原队列中移除已调度的任务
        self.task_queue = [(p, c, t) for (p, c, t) in self.task_queue 
                          if t not in scheduled_tasks]
        
        return assignments
    
    def _calculate_score(self, agent: Agent, task: Task) -> float:
        """计算Agent对任务的适配分数"""
        # 基础分数：优先级权重
        base_score = task.priority * 10
        
        # 资源利用率分数：偏好资源利用率低的Agent
        resource_utilization = 0
        for res_type in ResourceType:
            current = agent.current_load.get(res_type, 0)
            capacity = agent.capacity.get(res_type, 0)
            if capacity > 0:
                resource_utilization += current / capacity
        
        # 资源利用率越低越好
        utilization_score = (1 - resource_utilization / len(ResourceType)) * 5
        
        # 工具匹配分数：完全匹配加分
        tool_match = len(set(task.required_tools) & set(agent.available_tools))
        tool_score = tool_match * 2
        
        return base_score + utilization_score + tool_score
    
    def dynamic_adjust(self, completed_task: Task, agent_id: str):
        """动态调整：任务完成后释放资源"""
        agent = self.agents[agent_id]
        for res_type, need in completed_task.resources.items():
            agent.current_load[res_type] = max(0, 
                agent.current_load.get(res_type, 0) - need)
        
        if completed_task.task_id in agent.processing_tasks:
            agent.processing_tasks.remove(completed_task.task_id)


# 使用示例
def main():
    # 创建Agent
    agents = [
        Agent(
            agent_id="agent1",
            capacity={ResourceType.CPU: 4.0, ResourceType.MEMORY: 16.0, ResourceType.GPU: 1.0},
            current_load={},
            available_tools=["search", "calculator", "database"],
            processing_tasks=[]
        ),
        Agent(
            agent_id="agent2",
            capacity={ResourceType.CPU: 8.0, ResourceType.MEMORY: 32.0, ResourceType.GPU: 2.0},
            current_load={ResourceType.CPU: 2.0},
            available_tools=["search", "visualization"],
            processing_tasks=[]
        )
    ]
    
    scheduler = TaskScheduler(agents)
    
    # 添加任务
    tasks = [
        Task("task1", 5.0, 3, {ResourceType.CPU: 2.0, ResourceType.MEMORY: 4.0}, ["search"]),
        Task("task2", 3.0, 5, {ResourceType.CPU: 1.0, ResourceType.GPU: 0.5}, ["visualization"]),
        Task("task3", 8.0, 2, {ResourceType.CPU: 4.0, ResourceType.MEMORY: 8.0}, ["search", "database"]),
    ]
    
    for task in tasks:
        scheduler.add_task(task)
    
    # 调度
    assignments = scheduler.schedule()
    
    for agent_id, tasks in assignments.items():
        print(f"{agent_id}: {[t.task_id for t in tasks]}")


if __name__ == "__main__":
    main()
```

**算法分析**：
- **时间复杂度**：O(N * M)，N为任务数，M为Agent数
- **空间复杂度**：O(N + M)
- **优化方向**：
  1. 使用更复杂的评分函数（机器学习预测）
  2. 考虑任务间的依赖关系
  3. 支持抢占式调度
  4. 实时调整和重调度

```python
        Args:
            compression_ratio: 压缩比例 (0.1表示压缩到10%)
            max_length: 最大摘要长度（字符）

        self.max_length = max_length

        category = self._categorize_message(message)
        
        # 提取关键信息
        key_info = self._extract_key_information(message, category)
        
        if importance > 1.0:  # 重要性阈值
            self.key_points.append({
                "content": key_info,
                "importance": importance,
                "timestamp": message.timestamp,
                "category": category
            })
        
        # 定期重新生成摘要
        if len(self.message_history) % 5 == 0:
            self._regenerate_summary()
    
    def _calculate_importance(self, message: Message) -> float:
        """计算消息重要性"""
        content = message.content.lower()
        
        # 基础重要性
        base_importance = 1.0
        
        # 消息类型权重
        type_weights = {
            MessageType.USER: 1.2,
            MessageType.SYSTEM: 1.5,
            MessageType.ASSISTANT: 1.0,
            MessageType.TOOL: 1.3
        }
        base_importance *= type_weights.get(message.type, 1.0)
        
        # 内容特征
        features = {
            "question": bool(re.search(r'\?|如何|怎么|为什么', content)),
            "decision": bool(re.search(r'决定|选择|建议|推荐', content)),
            "fact": bool(re.search(r'\d+|是|有|在|到', content)) and len(content) > 20,
            "instruction": bool(re.search(r'请|帮我|需要|要求', content)),
            "confirmation": bool(re.search(r'好的|是的|不是|对|错', content)),
            "small_talk": bool(re.search(r'你好|谢谢|再见|早上好', content)),
        }
        
        # 加权计算
        total_importance = base_importance
        for feature, weight in self.importance_weights.items():
            if features.get(feature, False):
                total_importance *= weight
        
        # 长度调整：过短的消息可能不重要
        if len(content) < 10:
            total_importance *= 0.5
        
        return total_importance
    
    def _categorize_message(self, message: Message) -> str:
        """消息分类"""
        content = message.content
        
        if re.search(r'\?|如何|怎么|为什么', content):
            return "question"
        elif re.search(r'决定|选择|建议|推荐', content):
            return "decision"
        elif re.search(r'\d+|具体|详细', content) and len(content) > 30:
            return "fact"
        elif re.search(r'请|帮我|需要|要求', content):
            return "instruction"
        elif re.search(r'好的|是的|不是|对|错', content):
            return "confirmation"
        else:
            return "general"
    
    def _extract_key_information(self, message: Message, category: str) -> str:
        """提取关键信息"""
        content = message.content
        
        if category == "question":
            # 提取问题核心
            match = re.search(r'(如何|怎么|为什么|什么是|在哪里)\s*(.*?)[？?]', content)
            return match.group(0) if match else content[:50]
        
        elif category == "decision":
            # 提取决策点
            match = re.search(r'(决定|选择|建议|推荐)\s*(.*?)[。.]', content)
            return match.group(0) if match else content[:50]
        
        elif category == "fact":
            # 提取数字和关键事实
            numbers = re.findall(r'\d+\.?\d*', content)
            if numbers:
                return f"关键数据: {', '.join(numbers[:3])}"
            return content[:100]
        
        else:
            # 通用提取：前100字符
            return content[:100]
    
    def _regenerate_summary(self):
        """重新生成摘要"""
        if not self.key_points:
            self.summary = "对话摘要：暂无关键信息"
            return
        
        # 按重要性排序
        sorted_points = sorted(self.key_points, 
                              key=lambda x: x["importance"], 
                              reverse=True)
        
        # 计算目标长度
        target_length = min(
            int(len(''.join(m.content for m in self.message_history)) * self.compression_ratio),
            self.max_length
        )
        
        # 生成摘要
        summary_parts = []
        current_length = 0
        
        for point in sorted_points:
            point_text = f"[{point['category']}] {point['content']}"
            if current_length + len(point_text) <= target_length:
                summary_parts.append(point_text)
                current_length += len(point_text)
            else:
                break
        
        self.summary = "对话摘要：\n" + "\n".join(summary_parts)
    
    def get_summary(self) -> str:
        """获取当前摘要"""
        if not self.summary:
            self._regenerate_summary()
        return self.summary
```

#### 12.1 实现带优先级的任务队列（LeetCode 1845）

**原题**：设计一个支持优先级的任务队列

**AI Agent场景**：Agent系统中任务调度，高优先级任务优先处理

**扩展要求**：
1. 支持动态调整优先级
2. 支持任务依赖关系
3. 支持资源约束
4. 实现公平性防止饿死

```python
import heapq
from typing import List, Optional
from dataclasses import dataclass
from enum import Enum


class Priority(Enum):
    LOW = 1
    MEDIUM = 2
    HIGH = 3
    CRITICAL = 4


@dataclass
class AgentTask:
    task_id: str
    priority: Priority
    estimated_time: float
    dependencies: List[str]  # 依赖的任务ID
    required_resources: dict
    
    def __lt__(self, other):
        # 优先队列按优先级排序
        return self.priority.value > other.priority.value


class AdvancedTaskQueue:
    def __init__(self):
        self.heap = []
        self.tasks = {}  # task_id -> task
        self.completed = set()
        self.resource_available = {"cpu": 10.0, "memory": 32.0, "gpu": 2.0}
    
    def add_task(self, task: AgentTask) -> bool:
        """添加任务"""
        # 检查资源是否足够
        if not self._check_resources(task):
            return False
        
        self.tasks[task.task_id] = task
        heapq.heappush(self.heap, task)
        return True
    
    def get_next_task(self) -> Optional[AgentTask]:
        """获取下一个可执行任务"""
        temp_heap = []
        
        while self.heap:
            task = heapq.heappop(self.heap)
            
            # 检查依赖是否满足
            if all(dep in self.completed for dep in task.dependencies):
                # 检查资源
                if self._check_resources(task):
                    # 消耗资源
                    self._allocate_resources(task)
                    return task
                else:
                    # 资源不足，放回队列
                    temp_heap.append(task)
            else:
                temp_heap.append(task)
        
        # 恢复堆
        for t in temp_heap:
            heapq.heappush(self.heap, t)
        
        return None
    
    def complete_task(self, task_id: str):
        """标记任务完成"""
        if task_id in self.tasks:
            task = self.tasks[task_id]
            # 释放资源
            self._release_resources(task)
            self.completed.add(task_id)
    
    def _check_resources(self, task: AgentTask) -> bool:
        for resource, amount in task.required_resources.items():
            if self.resource_available.get(resource, 0) < amount:
                return False
        return True
    
    def _allocate_resources(self, task: AgentTask):
        for resource, amount in task.required_resources.items():
            self.resource_available[resource] -= amount
    
    def _release_resources(self, task: AgentTask):
        for resource, amount in task.required_resources.items():
            self.resource_available[resource] += amount


# 使用示例
queue = AdvancedTaskQueue()

task1 = AgentTask("t1", Priority.HIGH, 5.0, [], {"cpu": 2.0})
task2 = AgentTask("t2", Priority.MEDIUM, 3.0, ["t1"], {"memory": 8.0})
```

#### 12.2 实现LRU缓存变体（LeetCode 146）

**原题**：实现LRU缓存

**AI Agent场景**：Agent工具调用结果缓存，考虑工具调用成本和新鲜度

**扩展要求**：
1. 考虑缓存项的不同新鲜度要求
2. 支持基于成本的缓存淘汰（计算成本高的优先保留）
3. 支持基于重要性的缓存保留
4. 实现多级缓存（内存+磁盘）

```python
from collections import OrderedDict
from typing import Any, Optional
import time
from dataclasses import dataclass
from enum import Enum


class CachePriority(Enum):
    LOW = 1      # 计算结果容易获得
    MEDIUM = 2   # 中等计算成本
    HIGH = 3     # 高计算成本或重要结果


@dataclass
class CacheItem:
    key: str
    value: Any
    priority: CachePriority
    cost: float  # 计算成本（秒）
    timestamp: float
    access_count: int = 0


class SmartLRUCache:
    """智能LRU缓存，考虑优先级和成本"""
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()  # key -> CacheItem
        self.total_cost_saved = 0.0
    
    def get(self, key: str) -> Optional[Any]:
        if key not in self.cache:
            return None
        
        item = self.cache[key]
        item.access_count += 1
        item.timestamp = time.time()
        
        # 移动到最新位置
        self.cache.move_to_end(key)
        
        return item.value
    
    def put(self, key: str, value: Any, priority: CachePriority, cost: float):
        if key in self.cache:
            # 更新现有项
            item = self.cache[key]
            item.value = value
            item.priority = priority
            item.cost = cost
            item.timestamp = time.time()
            self.cache.move_to_end(key)
        else:
            # 添加新项
            item = CacheItem(
                key=key,
                value=value,
                priority=priority,
                cost=cost,
                timestamp=time.time()
            )
            self.cache[key] = item
            self.total_cost_saved += cost
            
            # 检查容量
            if len(self.cache) > self.capacity:
                self._evict()
    
    def _evict(self):
        """智能淘汰策略"""
        if not self.cache:
            return
        
        # 计算每个项的淘汰分数
        scores = {}
        current_time = time.time()
        
        for key, item in self.cache.items():
            # 分数 = 优先级 * 成本 / (时间衰减 * 访问次数)
            time_decay = max(1, current_time - item.timestamp) / 3600  # 小时
            access_factor = max(1, item.access_count)
            
            score = (item.priority.value * item.cost) / (time_decay * access_factor)
            scores[key] = score
        
        # 淘汰分数最低的项
        evict_key = min(scores, key=scores.get)
        evicted_item = self.cache[evict_key]
        self.total_cost_saved -= evicted_item.cost
        del self.cache[evict_key]
        
        print(f"淘汰缓存项: {evict_key} (分数: {scores[evict_key]:.2f})")
    
    def clear_low_priority(self):
        """清除低优先级项"""
        keys_to_remove = [
            key for key, item in self.cache.items()
            if item.priority == CachePriority.LOW
        ]
        
        for key in keys_to_remove:
            item = self.cache[key]
            self.total_cost_saved -= item.cost
            del self.cache[key]
        
        print(f"清除了 {len(keys_to_remove)} 个低优先级缓存项")
    
    def stats(self) -> dict:
        """缓存统计"""
        total_items = len(self.cache)
        total_size = sum(len(str(item.value)) for item in self.cache.values())
        
        priority_dist = {}
        for item in self.cache.values():
            priority_dist[item.priority.name] = priority_dist.get(item.priority.name, 0) + 1
        
        return {
            "total_items": total_items,
            "total_size_bytes": total_size,
            "priority_distribution": priority_dist,
            "total_cost_saved": self.total_cost_saved,
            "hit_rate": self._calculate_hit_rate() if hasattr(self, 'access_count') else 0
        }


# 使用示例
cache = SmartLRUCache(capacity=5)

# 添加缓存项
cache.put("user_123_profile", {"name": "张三", "age": 30}, CachePriority.HIGH, 2.5)
cache.put("search_results_python", ["结果1", "结果2"], CachePriority.MEDIUM, 1.0)
cache.put("weather_beijing", "晴 25°C", CachePriority.LOW, 0.5)

# 访问
profile = cache.get("user_123_profile")
print(f"获取用户资料: {profile}")

# 统计
print("缓存统计:", cache.stats())
```

#### 12.3 实现最小栈变体（LeetCode 155）

**原题**：实现最小栈

**AI Agent场景**：Agent决策过程中的多目标优化，维护当前最优解集

**扩展要求**：
1. 支持多维度比较（多个目标函数）
2. 支持Pareto最优解维护
3. 支持解的空间限制
4. 支持解的多样性和覆盖度

```python
from typing import List, Tuple, Any
import heapq
from dataclasses import dataclass
from functools import total_ordering


@total_ordering
class Solution:
    """多目标优化解"""
    
    def __init__(self, id: str, values: List[float], data: Any = None):
        self.id = id
        self.values = values  # 多个目标函数值
        self.data = data
    
    def dominates(self, other: 'Solution') -> bool:
        """判断是否支配另一个解"""
        # 在所有目标上都不差，且至少一个目标更好
        not_worse = all(v1 <= v2 for v1, v2 in zip(self.values, other.values))
        better = any(v1 < v2 for v1, v2 in zip(self.values, other.values))
        return not_worse and better
    
    def __eq__(self, other):
        return self.values == other.values
    
    def __lt__(self, other):
        # 按第一个目标函数比较（用于堆排序）
        return self.values[0] < other.values[0]
    
    def __repr__(self):
        return f"Solution({self.id}, values={self.values})"


class ParetoFront:
    """Pareto最优解集"""
    
    def __init__(self, max_size: int = 10):
        self.max_size = max_size
        self.solutions = []  # Pareto前沿解
        self.all_solutions = []  # 所有解（用于多样性维护）
    
    def add_solution(self, solution: Solution) -> bool:
        """添加解，返回是否加入Pareto前沿"""
        # 检查是否被现有解支配
        for existing in self.solutions:
            if existing.dominates(solution):
                return False  # 被支配，不加入
        
        # 移除被新解支配的现有解
        self.solutions = [s for s in self.solutions if not solution.dominates(s)]
        
        # 添加新解
        self.solutions.append(solution)
        self.all_solutions.append(solution)
        
        # 维护大小限制
        if len(self.solutions) > self.max_size:
            self._prune_solutions()
        
        return True
    
    def _prune_solutions(self):
        """修剪解集，保持多样性"""
        if len(self.solutions) <= self.max_size:
            return
        
        # 计算解的拥挤距离
        distances = self._calculate_crowding_distance()
        
        # 移除拥挤距离最小的解
        min_idx = min(range(len(self.solutions)), key=lambda i: distances[i])
        removed = self.solutions.pop(min_idx)
        print(f"移除解 {removed.id} 以保持多样性")
    
    def _calculate_crowding_distance(self) -> List[float]:
        """计算拥挤距离（NSGA-II算法）"""
        if len(self.solutions) <= 2:
            return [float('inf')] * len(self.solutions)
        
        n_obj = len(self.solutions[0].values)
        distances = [0.0] * len(self.solutions)
        
        for obj_idx in range(n_obj):
            # 按当前目标排序
            sorted_indices = sorted(
                range(len(self.solutions)),
                key=lambda i: self.solutions[i].values[obj_idx]
            )
            
            # 边界解距离为无穷大
            distances[sorted_indices[0]] = float('inf')
            distances[sorted_indices[-1]] = float('inf')
            
            # 计算中间解的拥挤距离
            min_val = self.solutions[sorted_indices[0]].values[obj_idx]
            max_val = self.solutions[sorted_indices[-1]].values[obj_idx]
            
            if max_val - min_val < 1e-10:  # 防止除零
                continue
            
            for i in range(1, len(sorted_indices) - 1):
                idx = sorted_indices[i]
                next_val = self.solutions[sorted_indices[i + 1]].values[obj_idx]
                prev_val = self.solutions[sorted_indices[i - 1]].values[obj_idx]
                
                distances[idx] += (next_val - prev_val) / (max_val - min_val)
        
        return distances
    
    def get_best_solutions(self, n: int = 3) -> List[Solution]:
        """获取最优解（按第一个目标排序）"""
        return sorted(self.solutions, key=lambda s: s.values[0])[:n]
    
    def get_diverse_solutions(self, n: int = 3) -> List[Solution]:
        """获取多样性最好的解"""
        if len(self.solutions) <= n:
            return self.solutions.copy()
        
        # 使用拥挤距离选择
        distances = self._calculate_crowding_distance()
        indices = sorted(range(len(self.solutions)), 
                        key=lambda i: distances[i], 
                        reverse=True)
        
        return [self.solutions[i] for i in indices[:n]]
    
    def stats(self) -> dict:
        """统计信息"""
        return {
            "pareto_size": len(self.solutions),
            "total_solutions": len(self.all_solutions),
            "objective_ranges": self._get_objective_ranges(),
            "diversity_score": self._calculate_diversity_score()
        }
    
    def _get_objective_ranges(self) -> List[Tuple[float, float]]:
        """获取各目标函数值范围"""
        if not self.solutions:
            return []
        
        n_obj = len(self.solutions[0].values)
        ranges = []
        
        for obj_idx in range(n_obj):
            values = [s.values[obj_idx] for s in self.solutions]
            ranges.append((min(values), max(values)))
        
        return ranges
    
    def _calculate_diversity_score(self) -> float:
        """计算多样性分数"""
        if len(self.solutions) <= 1:
            return 0.0
        
        total_distance = 0.0
        count = 0
        
        for i in range(len(self.solutions)):
            for j in range(i + 1, len(self.solutions)):
                # 欧氏距离
                dist = sum((a - b) ** 2 for a, b in 
                          zip(self.solutions[i].values, self.solutions[j].values)) ** 0.5
                total_distance += dist
                count += 1
        
        return total_distance / count if count > 0 else 0.0


# 使用示例
front = ParetoFront(max_size=5)

# 添加解
solutions = [
    Solution("s1", [1.0, 10.0], {"config": "A"}),
    Solution("s2", [2.0, 8.0], {"config": "B"}),
    Solution("s3", [3.0, 6.0], {"config": "C"}),
    Solution("s4", [4.0, 4.0], {"config": "D"}),
    Solution("s5", [5.0, 2.0], {"config": "E"}),
    Solution("s6", [1.5, 9.0], {"config": "F"}),  # 被s1支配
]

for sol in solutions:
    added = front.add_solution(sol)
    print(f"添加解 {sol.id}: {'成功' if added else '失败'}")

print("\nPareto前沿解:")
for sol in front.solutions:
    print(f"  {sol}")

print("\n最优解（按目标1）:")
for sol in front.get_best_solutions(2):
    print(f"  {sol}")

print("\n统计信息:", front.stats())
```
