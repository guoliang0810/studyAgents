```python
from typing import TypedDict, List, Callable, Any, Optional
from enum import Enum
from dataclasses import dataclass


class AgentState(TypedDict):
    """Agent状态定义"""
    messages: List[str]
    current_action: str
    context: dict


class ActionResult:
    """动作执行结果"""
    success: bool
    data: Any
    error: Optional[str] = None


class BaseAgent:
    """基础Agent类"""
    
    def __init__(self, name: str):
        self.name = name
        self.state: AgentState = {
            "messages": [],
            "current_action": "idle",
            "context": {}
        }
        self.tools: dict[str, Callable] = {}
        self.policies: List[Callable] = []
    
    def register_tool(self, name: str, tool: Callable):
        """注册工具"""
        self.tools[name] = tool
    
    def add_policy(self, policy: Callable):
        """添加策略"""
        self.policies.append(policy)
    
    def decide(self) -> str:
        """决策引擎"""
        for policy in self.policies:
            action = policy(self.state)
            if action:
                return action
        return "respond"
    
    def execute(self, action: str) -> ActionResult:
        """执行动作"""
        if action in self.tools:
            return ActionResult(
                success=True,
                data=self.tools[action]()
            )
        return ActionResult(success=True, data=f"Executed {action}")
    
    def run(self, message: str) -> str:
        """运行Agent"""
        self.state["messages"].append(message)
        action = self.decide()
        self.state["current_action"] = action
        result = self.execute(action)
        return result.data


# 使用示例
agent = BaseAgent("Assistant")

@agent.register_tool
def search(query: str) -> str:
    return f"搜索结果: {query}"

agent.add_policy(lambda state: "search" if "搜索" in state["messages"][-1] else None)

response = agent.run("帮我搜索Python")
print(response)
```

### 题目2：设计一个状态机管理对话流程

**难度**：⭐⭐⭐⭐

```python
from enum import Enum, auto
from typing import Callable, Optional
from dataclasses import dataclass, field


class ConversationState(Enum):
    """对话状态枚举"""
    IDLE = auto()
    WAITING_INPUT = auto()
    PROCESSING = auto()
    WAITING_TOOL_RESULT = auto()
    RESPONDING = auto()
    ERROR = auto()


@dataclass
class StateTransition:
    """状态转换"""
    from_state: ConversationState
    to_state: ConversationState
    condition: Callable[["ConversationContext"], bool]
    action: Optional[Callable] = None


@dataclass
class ConversationContext:
    """对话上下文"""
    current_state: ConversationState = ConversationState.IDLE
    messages: list = field(default_factory=list)
    tool_results: dict = field(default_factory=dict)
    error: Optional[str] = None


class StateMachine:
    """状态机"""
    
    def __init__(self):
        self.context = ConversationContext()
        self.transitions: list[StateTransition] = []
        self._setup_default_transitions()
    
    def _setup_default_transitions(self):
        """设置默认转换规则"""
        self.transitions = [
            StateTransition(
                ConversationState.IDLE,
                ConversationState.WAITING_INPUT,
                lambda ctx: True
            ),
            StateTransition(
                ConversationState.WAITING_INPUT,
                ConversationState.PROCESSING,
                lambda ctx: len(ctx.messages) > 0
            ),
            StateTransition(
                ConversationState.PROCESSING,
                ConversationState.WAITING_TOOL_RESULT,
                lambda ctx: "needs_tool" in ctx.messages[-1]
            ),
            StateTransition(
                ConversationState.WAITING_TOOL_RESULT,
                ConversationState.PROCESSING,
                lambda ctx: len(ctx.tool_results) > 0
            ),
            StateTransition(
                ConversationState.PROCESSING,
                ConversationState.RESPONDING,
                lambda ctx: "needs_tool" not in ctx.messages[-1]
            ),
            StateTransition(
                ConversationState.RESPONDING,
                ConversationState.IDLE,
                lambda ctx: True
            ),
        ]
    
    def can_transition(self, to_state: ConversationState) -> bool:
        """检查是否可以转换"""
        for transition in self.transitions:
            if (transition.from_state == self.context.current_state and
                transition.to_state == to_state):
                return transition.condition(self.context)
        return False
    
    def transition(self, to_state: ConversationState) -> bool:
        """执行状态转换"""
        if self.can_transition(to_state):
            self.context.current_state = to_state
            return True
        return False
    
    def step(self) -> bool:
        """执行一步"""
        current = self.context.current_state
        
        if current == ConversationState.IDLE:
            return self.transition(ConversationState.WAITING_INPUT)
        
        elif current == ConversationState.WAITING_INPUT:
            if len(self.context.messages) > 0:
                return self.transition(ConversationState.PROCESSING)
        
        elif current == ConversationState.PROCESSING:
            if self.context.messages[-1].get("needs_tool"):
                return self.transition(ConversationState.WAITING_TOOL_RESULT)
            return self.transition(ConversationState.RESPONDING)
        
        elif current == ConversationState.WAITING_TOOL_RESULT:
            if len(self.context.tool_results) > 0:
                return self.transition(ConversationState.PROCESSING)
        
        elif current == ConversationState.RESPONDING:
            return self.transition(ConversationState.IDLE)
        
        return False


# 使用示例
sm = StateMachine()
print(f"初始状态: {sm.context.current_state}")

sm.context.messages.append({"content": "Hello"})
sm.step()
print(f"添加消息后: {sm.context.current_state}")

sm.step()
print(f"处理后: {sm.context.current_state}")
```

```python
from typing import Any, Callable, Dict, Optional, Type
from pydantic import BaseModel, ValidationError, create_model
from dataclasses import dataclass
from enum import Enum
import time
import asyncio


class ToolError(Exception):
    """工具异常"""
    pass


class ParameterValidationError(ToolError):
    """参数验证异常"""
    pass


class ToolExecutionError(ToolError):
    """工具执行异常"""
    pass


@dataclass
class ToolMetadata:
    """工具元数据"""
    name: str
    description: str
    parameters: Type[BaseModel]
    retry_count: int = 3
    timeout: int = 30


class Tool:
    """工具基类"""
    
    def __init__(self, metadata: ToolMetadata, handler: Callable):
        self.metadata = metadata
        self.handler = handler
    
    def validate_params(self, params: dict) -> BaseModel:
        """验证参数"""
        try:
            return self.metadata.parameters(**params)
        except ValidationError as e:
            raise ParameterValidationError(f"参数验证失败: {e}")
    
    def execute(self, params: dict) -> Any:
        """执行工具"""
        validated = self.validate_params(params)
        return self.handler(validated)


class ToolRegistry:
    """工具注册表"""
    
    def __init__(self):
        self._tools: Dict[str, Tool] = {}
    
    def register(
        self,
        name: str,
        description: str,
        params_model: Type[BaseModel],
        handler: Callable,
        retry_count: int = 3
    ):
        """注册工具"""
        metadata = ToolMetadata(
            name=name,
            description=description,
            parameters=params_model,
            retry_count=retry_count
        )
        self._tools[name] = Tool(metadata, handler)
    
    def get(self, name: str) -> Optional[Tool]:
        """获取工具"""
        return self._tools.get(name)
    
    def list_all(self) -> list[str]:
        """列出所有工具"""
        return list(self._tools.keys())
    
    def search(self, keyword: str) -> list[str]:
        """搜索工具"""
        return [
            name for name, tool in self._tools.items()
            if keyword.lower() in tool.metadata.description.lower()
        ]


class ToolExecutor:
    """工具执行器"""
    
    def __init__(self, registry: ToolRegistry):
        self.registry = registry
    
    def execute_with_retry(
        self,
        name: str,
        params: dict,
        retry_count: Optional[int] = None
    ) -> Any:
        """带重试的执行"""
        tool = self.registry.get(name)
        if not tool:
            raise ToolError(f"工具 '{name}' 不存在")
        
        max_retries = retry_count or tool.metadata.retry_count
        
        for attempt in range(max_retries):
            try:
                return tool.execute(params)
            except Exception as e:
                if attempt == max_retries - 1:
                    raise ToolExecutionError(f"执行失败: {e}")
                time.sleep(2 ** attempt)  # 指数退避
        
        raise ToolExecutionError("执行失败")


# 使用示例
class CalculatorParams(BaseModel):
    expression: str


def calculator_handler(params: CalculatorParams) -> float:
    return eval(params.expression)


registry = ToolRegistry()
registry.register("calculator", "执行数学计算", CalculatorParams, calculator_handler)

executor = ToolExecutor(registry)
result = executor.execute_with_retry("calculator", {"expression": "2 + 3 * 4"})
print(f"计算结果: {result}")
```

```python
from typing import Any, Callable, List, TypeVar, Generic
from dataclasses import dataclass
from abc import ABC, abstractmethod
import asyncio


T = TypeVar('T')


@dataclass
class MiddlewareContext:
    """中间件上下文"""
    request_id: str
    data: dict
    errors: List[str] = None
    
    def __post_init__(self):
        if self.errors is None:
            self.errors = []


class BaseMiddleware(ABC, Generic[T]):
    """中间件基类"""
    
    @abstractmethod
    async def process(self, context: MiddlewareContext, next_handler: Callable) -> T:
        """处理请求"""
        pass


class LoggingMiddleware(BaseMiddleware):
    """日志中间件"""
    
    async def process(self, context: MiddlewareContext, next_handler: Callable) -> Any:
        print(f"[LOG] Request: {context.request_id}")
        try:
            result = await next_handler()
            print(f"[LOG] Success: {context.request_id}")
            return result
        except Exception as e:
            context.errors.append(str(e))
            print(f"[LOG] Error: {e}")
            raise


class ValidationMiddleware(BaseMiddleware):
    """验证中间件"""
    
    async def process(self, context: MiddlewareContext, next_handler: Callable) -> Any:
        if "required_field" not in context.data:
            raise ValueError("缺少必需字段")
        return await next_handler()


class RateLimitMiddleware(BaseMiddleware):
    """限流中间件"""
    
    def __init__(self):
        self.request_counts: dict[str, int] = {}
    
    async def process(self, context: MiddlewareContext, next_handler: Callable) -> Any:
        client_id = context.data.get("client_id", "default")
        self.request_counts[client_id] = self.request_counts.get(client_id, 0) + 1
        
        if self.request_counts[client_id] > 100:
            raise ValueError("请求过于频繁")
        
        return await next_handler()


class MiddlewareChain:
    """中间件链"""
    
    def __init__(self):
        self.middlewares: List[BaseMiddleware] = []
    
    def add(self, middleware: BaseMiddleware) -> "MiddlewareChain":
        self.middlewares.append(middleware)
        return self
    
    async def execute(self, context: MiddlewareContext) -> Any:
        """执行中间件链"""
        
        async def handler(index: int = 0) -> Any:
            if index >= len(self.middlewares):
                return {"status": "success", "data": context.data}
            
            middleware = self.middlewares[index]
            return await middleware.process(context, lambda: handler(index + 1))
        
        return await handler()


# 使用示例
async def main():
    chain = MiddlewareChain()
    chain.add(LoggingMiddleware())
    chain.add(RateLimitMiddleware())
    chain.add(ValidationMiddleware())
    
    context = MiddlewareContext(
        request_id="req_001",
        data={"required_field": "value", "client_id": "client_1"}
    )
    
    result = await chain.execute(context)
    print(f"结果: {result}")


asyncio.run(main())
```

```python
from typing import Any, Callable, Dict, List, Optional
from dataclasses import dataclass, field
from enum import Enum
import asyncio
from abc import ABC, abstractmethod


class TaskStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    TIMEOUT = "timeout"


@dataclass
class Task:
    """任务"""
    task_id: str
    agent_name: str
    payload: dict
    status: TaskStatus = TaskStatus.PENDING
    result: Any = None
    error: Optional[str] = None


@dataclass
class AgentCapability:
    """Agent能力"""
    name: str
    description: str
    input_schema: dict
    output_schema: dict


class BaseAgent(ABC):
    """Agent基类"""
    
    def __init__(self, name: str):
        self.name = name
        self.capabilities: List[AgentCapability] = []
    
    @abstractmethod
    async def execute(self, task: Task) -> Any:
        """执行任务"""
        pass
    
    def register_capability(self, capability: AgentCapability):
        """注册能力"""
        self.capabilities.append(capability)


class AgentRegistry:
    """Agent注册表"""
    
    def __init__(self):
        self._agents: Dict[str, BaseAgent] = {}
    
    def register(self, agent: BaseAgent):
        """注册Agent"""
        self._agents[agent.name] = agent
    
    def get(self, name: str) -> Optional[BaseAgent]:
        """获取Agent"""
        return self._agents.get(name)
    
    def find_by_capability(self, capability_name: str) -> List[BaseAgent]:
        """按能力查找Agent"""
        return [
            agent for agent in self._agents.values()
            if any(c.name == capability_name for c in agent.capabilities)
        ]


class TaskScheduler:
    """任务调度器"""
    
    def __init__(self, registry: AgentRegistry):
        self.registry = registry
        self.tasks: Dict[str, Task] = {}
    
    async def submit(self, agent_name: str, payload: dict, timeout: int = 60) -> Task:
        """提交任务"""
        task_id = f"task_{len(self.tasks)}"
        task = Task(
            task_id=task_id,
            agent_name=agent_name,
            payload=payload
        )
        self.tasks[task_id] = task
        
        asyncio.create_task(self._execute(task, timeout))
        return task
    
    async def _execute(self, task: Task, timeout: int):
        """执行任务"""
        agent = self.registry.get(task.agent_name)
        if not agent:
            task.status = TaskStatus.FAILED
            task.error = f"Agent '{task.agent_name}' 不存在"
            return
        
        task.status = TaskStatus.RUNNING
        
        try:
            result = await asyncio.wait_for(
                agent.execute(task),
                timeout=timeout
            )
            task.status = TaskStatus.COMPLETED
            task.result = result
        except asyncio.TimeoutError:
            task.status = TaskStatus.TIMEOUT
            task.error = "任务超时"
        except Exception as e:
            task.status = TaskStatus.FAILED
            task.error = str(e)
    
    def get_result(self, task_id: str) -> Optional[Task]:
        """获取结果"""
        return self.tasks.get(task_id)


# 使用示例
class AnalysisAgent(BaseAgent):
    """分析Agent"""
    
    def __init__(self):
        super().__init__("analysis")
        self.register_capability(AgentCapability(
            name="analyze",
            description="数据分析",
            input_schema={"data": "array"},
            output_schema={"result": "object"}
        ))
    
    async def execute(self, task: Task) -> dict:
        await asyncio.sleep(1)  # 模拟处理
        return {"mean": 50, "sum": 500}


async def main():
    registry = AgentRegistry()
    registry.register(AnalysisAgent())
    
    scheduler = TaskScheduler(registry)
    task = await scheduler.submit("analysis", {"data": [1, 2, 3, 4, 5]})
    
    await asyncio.sleep(2)
    result = scheduler.get_result(task.task_id)
    print(f"任务状态: {result.status.value if result else 'not found'}")
    print(f"结果: {result.result if result else None}")


asyncio.run(main())
```

```python
from typing import Any, Dict, List, Optional
from dataclasses import dataclass, field
from enum import Enum
import asyncio
import time
from collections import deque


class CircuitState(Enum):
    CLOSED = "closed"      # 正常
    OPEN = "open"          # 熔断
    HALF_OPEN = "half_open"  # 半开


@dataclass
class ServiceInstance:
    """服务实例"""
    instance_id: str
    host: str
    port: int
    healthy: bool = True
    load: float = 0.0
    last_health_check: float = field(default_factory=time.time)


class CircuitBreaker:
    """熔断器"""
    
    def __init__(
        self,
        failure_threshold: int = 5,
        timeout: int = 60,
        half_open_attempts: int = 3
    ):
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.failure_threshold = failure_threshold
        self.timeout = timeout
        self.half_open_attempts = half_open_attempts
        self.half_open_success = 0
        self.last_failure_time = 0
    
    def record_success(self):
        """记录成功"""
        if self.state == CircuitState.HALF_OPEN:
            self.half_open_success += 1
            if self.half_open_success >= self.half_open_attempts:
                self.state = CircuitState.CLOSED
                self.failure_count = 0
                self.half_open_success = 0
    
    def record_failure(self):
        """记录失败"""
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
    
    def can_execute(self) -> bool:
        """是否可以执行"""
        if self.state == CircuitState.CLOSED:
            return True
        
        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time >= self.timeout:
                self.state = CircuitState.HALF_OPEN
                return True
            return False
        
        return True


class LoadBalancer:
    """负载均衡器"""
    
    def __init__(self, strategy: str = "least_connection"):
        self.strategy = strategy
        self.instances: Dict[str, ServiceInstance] = {}
    
    def add_instance(self, instance: ServiceInstance):
        """添加实例"""
        self.instances[instance.instance_id] = instance
    
    def remove_instance(self, instance_id: str):
        """移除实例"""
        self.instances.pop(instance_id, None)
    
    def select_instance(self) -> Optional[ServiceInstance]:
        """选择实例"""
        healthy = [i for i in self.instances.values() if i.healthy]
        
        if not healthy:
            return None
        
        if self.strategy == "round_robin":
            return healthy[0]
        
        if self.strategy == "least_connection":
            return min(healthy, key=lambda x: x.load)
        
        return healthy[0]
    
    def update_load(self, instance_id: str, load: float):
        """更新负载"""
        if instance_id in self.instances:
            self.instances[instance_id].load = load


class AgentService:
    """Agent服务平台"""
    
    def __init__(self):
        self.load_balancer = LoadBalancer()
        self.circuit_breaker = CircuitBreaker()
        self.metrics: Dict[str, List[float]] = {
            "latency": deque(maxlen=1000),
            "error_rate": deque(maxlen=1000)
        }
    
    async def call(self, request: dict) -> dict:
        """调用服务"""
        if not self.circuit_breaker.can_execute():
            raise Exception("服务熔断中")
        
        instance = self.load_balancer.select_instance()
        if not instance:
            raise Exception("无可用服务实例")
        
        start = time.time()
        try:
            result = await self._forward_request(instance, request)
            
            self.circuit_breaker.record_success()
            self.load_balancer.update_load(
                instance.instance_id,
                instance.load - 1
            )
            
            self.metrics["latency"].append(time.time() - start)
            self.metrics["error_rate"].append(0)
            
            return result
        except Exception as e:
            self.circuit_breaker.record_failure()
            self.metrics["error_rate"].append(1)
            raise
    
    async def _forward_request(
        self,
        instance: ServiceInstance,
        request: dict
    ) -> dict:
        """转发请求"""
        self.load_balancer.update_load(
            instance.instance_id,
            instance.load + 1
        )
        await asyncio.sleep(0.1)  # 模拟网络请求
        return {"status": "success", "instance": instance.instance_id}
    
    def get_metrics(self) -> dict:
        """获取指标"""
        latencies = list(self.metrics["latency"])
        errors = list(self.metrics["error_rate"])
        
        return {
            "avg_latency": sum(latencies) / len(latencies) if latencies else 0,
            "p99_latency": sorted(latencies)[int(len(latencies) * 0.99)] if latencies else 0,
            "error_rate": sum(errors) / len(errors) if errors else 0,
            "circuit_state": self.circuit_breaker.state.value
        }
```

#### 4. 性能优化

- **异步处理**：非关键路径异步化，事件驱动架构
- **批处理**：小请求合并批处理，提升吞吐量

#### 5. 监控运维
- **全链路追踪**：OpenTelemetry，请求链路可视化
- **指标监控**：Prometheus + Grafana，自定义业务指标
- **日志聚合**：ELK栈，结构化日志和快速检索
