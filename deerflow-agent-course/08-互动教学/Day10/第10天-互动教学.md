@dataclass
class SubagentConfig:
    """子代理配置"""
    name: str
    role: str
    capabilities: List[str]
    max_execution_time: float = 60.0
    max_retries: int = 3
    timeout: float = 30.0

@dataclass
class SubagentResult:
    """子代理结果"""
    subagent_name: str
    success: bool
    output: Any
    error: Optional[str] = None
    execution_time: float = 0.0
    metadata: Dict = field(default_factory=dict)

### 子代理基类定义

```python
from abc import ABC, abstractmethod

class BaseSubagent(ABC):
    """子代理基类"""
    
    @property
    @abstractmethod
    def config(self) -> SubagentConfig:
        """返回子代理配置"""
        pass
    
    @abstractmethod
    async def execute(self, task: Dict) -> SubagentResult:
        """执行任务"""
        pass
    
    async def validate_task(self, task: Dict) -> tuple[bool, str]:
        """验证任务"""
        required_fields = ["description"]
        for field in required_fields:
            if field not in task:
                return False, f"缺少必需字段: {field}"
        return True, "验证通过"

class CodeExpertSubagent(BaseSubagent):
    """代码专家子代理"""
    
    @property
    def config(self) -> SubagentConfig:
        return SubagentConfig(
            name="code_expert",
            role="代码专家",
            capabilities=["代码生成", "代码审查", "代码优化"],
            max_execution_time=120.0
        )
    
    async def execute(self, task: Dict) -> SubagentResult:
        """执行代码任务"""
        import time
        start = time.time()
        
        # 验证任务
        is_valid, msg = await self.validate_task(task)
        if not is_valid:
            return SubagentResult(
                self.config.name, False, None, msg, time.time() - start
            )
        
        # 执行任务
        description = task["description"]
        language = task.get("language", "python")
        
        if "生成" in description or "写" in description:
            output = f"生成的{language}代码..."
        elif "审查" in description or "检查" in description:
            output = "代码审查完成，无重大问题"
        else:
            output = f"执行{language}任务完成"
        
        return SubagentResult(
            self.config.name, True, output, execution_time=time.time() - start
        )

# 测试
async def test_subagent():
    subagent = CodeExpertSubagent()
    result = await subagent.execute({

    print(f"子代理: {result.subagent_name}")
    print(f"成功: {result.success}")
    print(f"输出: {result.output}")
    print(f"耗时: {result.execution_time:.3f}s")

### 执行引擎实现

```python
class SubagentExecutor:
    """子代理执行引擎"""
    
    def __init__(self):
        self.subagents: Dict[str, BaseSubagent] = {}
        self.execution_history = []
    
    def register(self, subagent: BaseSubagent):
        """注册子代理"""
        self.subagents[subagent.config.name] = subagent
        print(f"🤖 已注册子代理: {subagent.config.name}")
    
    async def execute_single(
        self,
        subagent_name: str,
        task: Dict,
        timeout: float = None
    ) -> SubagentResult:
        """执行单个子代理任务"""
        import time
        
        subagent = self.subagents.get(subagent_name)
        if not subagent:
            return SubagentResult(
                subagent_name, False, None,
                error=f"子代理不存在: {subagent_name}"
            )
        
        timeout = timeout or subagent.config.timeout
        start = time.time()
        
        try:
            result = await asyncio.wait_for(
                subagent.execute(task),
                timeout=timeout
            )
            result.execution_time = time.time() - start
            return result
        
        except asyncio.TimeoutError:
            return SubagentResult(
                subagent_name, False, None,
                error=f"执行超时 ({timeout}s)",
                execution_time=timeout
            )
        
        except Exception as e:
            return SubagentResult(
                subagent_name, False, None,
                error=str(e),
                execution_time=time.time() - start
            )
    
    async def execute_parallel(
        self,
        tasks: List[Dict],
        max_concurrent: int = 5
    ) -> List[SubagentResult]:
        """并行执行多个子代理任务"""
        
        async def execute_task(task: Dict) -> SubagentResult:
            return await self.execute_single(
                task["subagent"],
                task["task"],
                task.get("timeout")
            )
        
        # 使用信号量限制并发
        semaphore = asyncio.Semaphore(max_concurrent)
        
        async def bounded_execute(task: Dict) -> SubagentResult:
            async with semaphore:
                return await execute_task(task)
        
        results = await asyncio.gather(*[
            bounded_execute(task) for task in tasks
        ])
        
        return list(results)

# 创建执行引擎并注册子代理
executor = SubagentExecutor()
executor.register(CodeExpertSubagent())

### 任务分解策略

```python
class TaskDecomposer:
    """任务分解器"""
    
    def decompose(self, task_description: str) -> List[Dict]:
        """分解任务"""
        tasks = []
        
        # 简单关键词匹配
        if any(word in task_description for word in ["搜索", "查找"]):
            tasks.append({
                "subagent": "web_search",
                "task": {"description": task_description}
            })
        
        if any(word in task_description for word in ["分析", "统计"]):
            tasks.append({
                "subagent": "data_analyst",
                "task": {"description": task_description}
            })
        
        if any(word in task_description for word in ["代码", "编程", "生成"]):
            tasks.append({
                "subagent": "code_expert",
                "task": {"description": task_description}
            })
        
        if any(word in task_description for word in ["写作", "撰写", "报告"]):
            tasks.append({
                "subagent": "writer",
                "task": {"description": task_description}
            })
        
        # 如果没有匹配，创建通用任务
        if not tasks:
            tasks.append({
                "subagent": "code_expert",
                "task": {"description": task_description}
            })
        
        return tasks

### 结果聚合策略

```python
@dataclass
class AggregatedResult:
    """聚合结果"""
    success: bool
    results: List[SubagentResult]
    summary: str
    conflicts: List[Dict] = field(default_factory=list)

class ResultAggregator:
    """结果聚合器"""
    
    def aggregate(self, results: List[SubagentResult]) -> AggregatedResult:
        """聚合多个子代理的结果"""
        
        # 统计成功和失败
        successes = [r for r in results if r.success]
        failures = [r for r in results if not r.success]
        
        # 生成摘要
        summary_parts = []
        for r in successes:
            summary_parts.append(f"[{r.subagent_name}]: {r.output}")
        
        summary = "\n".join(summary_parts) if summary_parts else "无成功结果"
        
        return AggregatedResult(
            success=len(successes) > 0,
            results=results,
            summary=summary,
            conflicts=[]
        )
    
    def resolve_conflicts(self, results: List[SubagentResult]) -> Any:
        """解决结果冲突"""
        # 策略1：使用最新的结果
        return results[-1].output if results else None

# 测试

        SubagentResult("data_analyst", True, "分析完成", execution_time=2.0),
        SubagentResult("writer", True, "报告生成完成", execution_time=1.0)
    ]
    
    aggregator = ResultAggregator()
    aggregated = aggregator.aggregate(results)
    
    print(f"聚合结果:")
    print(f"  成功: {aggregated.success}")
    print(f"  子结果数: {len(aggregated.results)}")
    print(f"  摘要:\n{aggregated.summary}")

### 综合实战项目

```python
class MultiAgentSystem:
    """多代理系统"""
    
    def __init__(self):
        self.executor = SubagentExecutor()
        self.decomposer = TaskDecomposer()
        self.aggregator = ResultAggregator()
    
    async def process_task(self, task_description: str) -> AggregatedResult:
        """处理复杂任务"""
        
        # 1. 分解任务
        subtasks = self.decomposer.decompose(task_description)
        print(f"📋 任务分解: {len(subtasks)} 个子任务")
        
        # 2. 并行执行
        results = await self.executor.execute_parallel(subtasks)
        
        # 3. 聚合结果
        aggregated = self.aggregator.aggregate(results)
        
        return aggregated
    
    def generate_report(self, result: AggregatedResult) -> str:
        """生成执行报告"""
        report = f"""
╔══════════════════════════════════════════════╗
║           任务执行报告                        ║
╠══════════════════════════════════════════════╣
║ 状态: {'✅ 成功' if result.success else '❌ 失败'}
║ 子任务数: {len(result.results)}
║ 成功数: {sum(1 for r in result.results if r.success)}
║ 失败数: {sum(1 for r in result.results if not r.success)}
╠══════════════════════════════════════════════╣
║ 执行详情:                                    ║
"""
        for r in result.results:
            status = "✅" if r.success else "❌"
            report += f"║ {status} {r.subagent_name}: {r.output[:30]}...\n"
        
        report += "╚══════════════════════════════════════════════╝"
        return report

# 测试
async def test_multi_agent():
    system = MultiAgentSystem()
    system.executor.register(CodeExpertSubagent())
    
    result = await system.process_task(
        "分析销售数据，生成Python代码进行可视化，并撰写报告"
    )
    
    print(system.generate_report(result))

asyncio.run(test_multi_agent())
