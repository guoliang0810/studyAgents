### 配置系统实现

```python
from typing import Dict, Any
import yaml

@dataclass
class SubagentSettings:
    """子代理设置"""
    enabled: bool = True
    max_concurrent: int = 3
    default_timeout: float = 60.0
    max_retries: int = 3
    priority: int = 1

class SubagentRegistry:
    """子代理注册表"""
    
    def __init__(self):
        self._subagents: Dict[str, BaseSubagent] = {}
        self._settings: Dict[str, SubagentSettings] = {}
        self._configs: Dict[str, Dict] = {}
    
    def register(
        self,
        subagent: BaseSubagent,
        settings: SubagentSettings = None
    ):
        """注册子代理"""
        name = subagent.config.name
        self._subagents[name] = subagent
        self._settings[name] = settings or SubagentSettings()
        print(f"✅ 注册子代理: {name}")
    
    def unregister(self, name: str):
        """注销子代理"""
        if name in self._subagents:
            del self._subagents[name]
            del self._settings[name]
            print(f"🗑️ 注销子代理: {name}")
    
    def get(self, name: str) -> Optional[BaseSubagent]:
        """获取子代理"""
        return self._subagents.get(name)
    
    def list_all(self) -> List[str]:
        """列出所有子代理"""
        return list(self._subagents.keys())
    
    def get_settings(self, name: str) -> SubagentSettings:
        """获取子代理设置"""
        return self._settings.get(name, SubagentSettings())
    
    def update_settings(self, name: str, settings: SubagentSettings):
        """更新设置"""
        self._settings[name] = settings
        print(f"⚙️ 更新设置: {name}")
    
    def load_from_yaml(self, config_path: str):
        """从YAML加载配置"""
        with open(config_path, 'r') as f:
            config = yaml.safe_load(f)
        
        for name, settings_dict in config.get('subagents', {}).items():
            self._configs[name] = settings_dict
            self._settings[name] = SubagentSettings(**settings_dict)
        
        print(f"📥 从YAML加载配置: {len(self._configs)} 个子代理配置")

# 测试
registry = SubagentRegistry()
registry.register(CodeExpertSubagent(), SubagentSettings(priority=2))
print(f"已注册: {registry.list_all()}")
```

### 动态加载机制

```python
import importlib
import inspect
from pathlib import Path

class DynamicSubagentLoader:
    """动态子代理加载器"""
    
    def __init__(self, registry: SubagentRegistry):
        self.registry = registry
    
    def discover_subagents(self, module_path: str) -> List[str]:
        """发现子代理"""
        discovered = []
        
        for file_path in Path(module_path).glob("*.py"):
            if file_path.name.startswith("_"):
                continue
            
            module_name = file_path.stem
            try:
                module = importlib.import_module(module_name)
                
                # 查找以Subagent结尾的类
                for name, obj in inspect.getmembers(module):
                    if (isinstance(obj, type) and 
                        name.endswith("Subagent") and 
                        issubclass(obj, BaseSubagent) and
                        obj != BaseSubagent):
                        
                        # 实例化并注册
                        subagent = obj()
                        self.registry.register(subagent)
                        discovered.append(name)
            
            except Exception as e:
                print(f"⚠️ 加载失败 {module_name}: {e}")
        
        return discovered
```

### 生命周期管理

```python
from enum import Enum

class SubagentState(Enum):
    IDLE = "idle"
    RUNNING = "running"
    PAUSED = "paused"
    STOPPED = "stopped"
    ERROR = "error"

class SubagentLifecycleManager:
    """子代理生命周期管理器"""
    
    def __init__(self):
        self.states: Dict[str, SubagentState] = {}
        self.managers: Dict[str, asyncio.Task] = {}
    
    def set_state(self, name: str, state: SubagentState):
        """设置状态"""
        self.states[name] = state
        print(f"🔄 {name}: {state.value}")
    
    def get_state(self, name: str) -> SubagentState:
        """获取状态"""
        return self.states.get(name, SubagentState.IDLE)
    
    async def start(self, name: str, executor: SubagentExecutor):
        """启动子代理"""
        self.set_state(name, SubagentState.RUNNING)
    
    async def stop(self, name: str):
        """停止子代理"""
        self.set_state(name, SubagentState.STOPPED)
        
        if name in self.managers:
            self.managers[name].cancel()
            del self.managers[name]
    
    async def pause(self, name: str):
        """暂停子代理"""
        self.set_state(name, SubagentState.PAUSED)
    
    async def resume(self, name: str):
        """恢复子代理"""
        self.set_state(name, SubagentState.RUNNING)
```

class SubagentManagementSystem:
    """子代理管理系统"""
    
    def __init__(self):
        self.registry = SubagentRegistry()
        self.executor = SubagentExecutor()
        self.lifecycle = SubagentLifecycleManager()
    
    def setup_default_subagents(self):
        """设置默认子代理"""
        # 注册内置子代理
        self.registry.register(CodeExpertSubagent())
        
        # 从配置文件加载
        # self.registry.load_from_yaml("config/subagents.yaml")
    
    async def execute_task(self, task: str) -> Dict:
        """执行任务"""
        # 获取所有启用的子代理
        enabled = [
            name for name in self.registry.list_all()
            if self.registry.get_settings(name).enabled
        ]
        
        # 注册到执行器
        for name in enabled:
            subagent = self.registry.get(name)
            self.executor.register(subagent)
        
        # 执行
        result = await self.executor.execute_single("code_expert", {"description": task})
        
        return {
            "success": result.success,
            "output": result.output,
            "error": result.error
        }

# 测试

    system.setup_default_subagents()
    
    result = await system.execute_task("生成快速排序代码")
