│ • 转换│ │ • 需要规划     │
└───────┘ └────────────────┘
""")

class TaskComplexityAnalyzer:
    """任务复杂度分析器"""
    
    COMPLEXITY_INDICATORS = {
        "high": ["分析", "设计", "优化", "比较", "评估"],
        "medium": ["生成", "创建", "查找", "获取"],
        "low": ["计算", "格式化", "转换"]
    }
    
    def analyze(self, task: str) -> str:
        """分析任务复杂度"""
        task_lower = task.lower()
        
        for indicator in self.COMPLEXITY_INDICATORS["high"]:
            if indicator in task_lower:
                return "high"
        
        for indicator in self.COMPLEXITY_INDICATORS["medium"]:
            if indicator in task_lower:
                return "medium"
        
        return "low"

# 测试

print(analyzer.analyze("搜索相关信息"))
print(analyzer.analyze("计算总和"))

## 🏫 第46节：智能委派策略（10:00-10:45）

### 委派策略实现

```python
class DelegationStrategy(Enum):
    """委派策略"""
    TOOL = "tool"              # 使用工具
    SUBAGENT = "subagent"      # 使用子代理
    MULTI_SUBAGENT = "multi"  # 多子代理并行
    HYBRID = "hybrid"          # 混合模式

class SmartDelegator:
    """智能任务委派器"""
    
    def __init__(self, executor: SubagentExecutor, registry: SubagentRegistry):
        self.executor = executor
        self.registry = registry
        self.analyzer = TaskComplexityAnalyzer()
    
    async def delegate(self, task: str) -> Dict:
        """委派任务"""
        complexity = self.analyzer.analyze(task)
        
        if complexity == "low":
            return await self._delegate_to_tool(task)
        elif complexity == "medium":
            return await self._delegate_to_subagent(task)
        else:
            return await self._delegate_to_multi(task)
    
    async def _delegate_to_tool(self, task: str) -> Dict:
        """委派给工具"""
        print(f"🔧 委派给工具: {task}")
        return {"strategy": "tool", "result": "工具执行完成"}
    
    async def _delegate_to_subagent(self, task: str) -> Dict:
        """委派给子代理"""
        print(f"🤖 委派给子代理: {task}")
        
        # 选择合适的子代理
        subagent = self._select_subagent(task)
        if subagent:
            result = await self.executor.execute_single(
                subagent.config.name,
                {"description": task}
            )
            return {"strategy": "subagent", "result": result.output}
        
        return {"strategy": "subagent", "result": "无合适的子代理"}
    
    async def _delegate_to_multi(self, task: str) -> Dict:
        """委派给多个子代理"""
        print(f"🚀 委派给多个子代理: {task}")
        return {"strategy": "multi", "result": "多代理协作完成"}
    
    def _select_subagent(self, task: str) -> Optional[BaseSubagent]:
        """选择合适的子代理"""
        task_lower = task.lower()
        
        for name in self.registry.list_all():
            subagent = self.registry.get(name)
            for cap in subagent.config.capabilities:
                if cap in task_lower:
                    return subagent
        
        return None
```

## 🏫 第47节：任务队列管理（11:00-11:45）

### 任务队列实现

```python
from collections import deque
import asyncio

@dataclass
class QueuedTask:
    """队列任务"""
    id: str
    task: str
    priority: int
    created_at: datetime
    status: str = "pending"

class TaskQueue:
    """任务队列"""
    
    def __init__(self, max_size: int = 100):
        self.max_size = max_size
        self.queue = deque()
        self.processing = set()
    
    def enqueue(self, task: QueuedTask) -> bool:
        """入队"""
        if len(self.queue) >= self.max_size:
            return False
        
        # 按优先级排序
        self.queue.appendleft(task)
        self.queue = deque(sorted(self.queue, key=lambda t: t.priority, reverse=True))
        return True
    
    def dequeue(self) -> Optional[QueuedTask]:
        """出队"""
        if not self.queue:
            return None
        return self.queue.popleft()
    
    def get_status(self) -> Dict:
        """获取队列状态"""
        return {
            "pending": len(self.queue),
            "processing": len(self.processing),
            "total": len(self.queue) + len(self.processing)
        }
```

```python
class IntelligentTaskSystem:
    """智能任务分发系统"""
    
    def __init__(self):
        self.registry = SubagentRegistry()
        self.executor = SubagentExecutor()
        self.delegator = SmartDelegator(self.executor, self.registry)
        self.queue = TaskQueue()
    
    async def submit_task(self, task: str, priority: int = 1) -> str:
        """提交任务"""
        task_id = f"task_{uuid.uuid4().hex[:8]}"
        
        queued_task = QueuedTask(
            id=task_id,
            task=task,
            priority=priority,
            created_at=datetime.now()
        )
        
        self.queue.enqueue(queued_task)
        return task_id
    
    async def process_next(self) -> Dict:
        """处理下一个任务"""
        task = self.queue.dequeue()
        if not task:
            return {"status": "empty", "message": "队列为空"}
        
        result = await self.delegator.delegate(task.task)
        return {
            "task_id": task.id,
            "result": result
        }
