@dataclass
class MemoryItem:
    """记忆条目"""
    id: str
    content: str
    memory_type: str  # short_term, long_term, working
    created_at: datetime
    access_count: int = 0
    last_accessed: datetime = None
    metadata: Dict = field(default_factory=dict)

class MemoryManager:
    """记忆管理器"""
    
    def __init__(self):
        self.short_term: List[MemoryItem] = []  # 短期记忆
        self.long_term: Dict[str, MemoryItem] = {}  # 长期记忆
        self.working: Dict[str, Any] = {}  # 工作记忆
    
    def add_short_term(self, content: str, metadata: Dict = None) -> MemoryItem:
        """添加短期记忆"""
        item = MemoryItem(
            id=f"st_{len(self.short_term)}",
            content=content,
            memory_type="short_term",
            created_at=datetime.now(),
            metadata=metadata or {}
        )
        self.short_term.append(item)
        
        # 保持最多10条短期记忆
        if len(self.short_term) > 10:
            self.short_term.pop(0)
        
        return item
    
    def add_long_term(self, content: str, key: str, metadata: Dict = None) -> MemoryItem:
        """添加长期记忆"""
        item = MemoryItem(
            id=f"lt_{key}",
            content=content,
            memory_type="long_term",
            created_at=datetime.now(),
            metadata=metadata or {}
        )
        self.long_term[key] = item
        return item
    
    def get_short_term(self, limit: int = 5) -> List[str]:
        """获取短期记忆"""
        return [item.content for item in self.short_term[-limit:]]
    
    def get_long_term(self, key: str) -> Optional[str]:
        """获取长期记忆"""
        item = self.long_term.get(key)
        if item:
            item.access_count += 1
            item.last_accessed = datetime.now()
            return item.content
        return None
    
    def set_working(self, key: str, value: Any):
        """设置工作记忆"""
        self.working[key] = value
    
    def get_working(self, key: str) -> Any:
        """获取工作记忆"""
        return self.working.get(key)
    
    def clear_working(self):
        """清空工作记忆"""
        self.working.clear()
```

### 短期记忆管理

```python
class ShortTermMemory:
    """短期记忆管理"""
    
    def __init__(self, max_items: int = 20):
        self.max_items = max_items
        self.items: List[MemoryItem] = []
    
    def add(self, role: str, content: str) -> MemoryItem:
        """添加对话记忆"""
        item = MemoryItem(
            id=f"conv_{len(self.items)}",
            content=f"{role}: {content}",
            memory_type="conversation",
            created_at=datetime.now(),
            metadata={"role": role}
        )
        self.items.append(item)
        
        # 自动清理
        if len(self.items) > self.max_items:
            self.items.pop(0)
        
        return item
    
    def get_context(self, max_turns: int = 10) -> str:
        """获取对话上下文"""
        recent = self.items[-max_turns:] if self.items else []
        return "\n".join([item.content for item in recent])
    
    def clear(self):
        """清空记忆"""
        self.items.clear()
    
    def summarize(self) -> str:
        """总结对话要点"""
        if not self.items:
            return "无对话记录"
        
        return f"对话共{len(self.items)}轮，涉及: {self.items[0].content[:50]}..."
```

### 向量存储基础

```python
import hashlib

class VectorStore:
    """简单的向量存储（实际项目用FAISS/Milvus）"""
    
    def __init__(self):
        self.vectors: Dict[str, List[float]] = {}
        self.contents: Dict[str, str] = {}
        self.dimension = 128
    
    def add(self, content: str, metadata: Dict = None) -> str:
        """添加记忆"""
        content_hash = hashlib.md5(content.encode()).hexdigest()
        
        # 简单向量（实际用embedding模型）
        vector = self._simple_embedding(content)
        
        self.vectors[content_hash] = vector
        self.contents[content_hash] = content
        
        return content_hash
    
    def _simple_embedding(self, text: str) -> List[float]:
        """简单文本向量化"""
        import random
        random.seed(hash(text) % (2**32))
        return [random.random() for _ in range(self.dimension)]
    
    def search(self, query: str, top_k: int = 3) -> List[tuple]:
        """相似度搜索（简化版）"""
        query_vector = self._simple_embedding(query)
        
        results = []
        for content_hash, vector in self.vectors.items():
            # 计算余弦相似度（简化）
            similarity = sum(q * v for q, v in zip(query_vector, vector)) / len(vector)
            results.append((content_hash, similarity, self.contents[content_hash]))
        
        results.sort(key=lambda x: x[1], reverse=True)
        return results[:top_k]

```python
class CompleteMemorySystem:
    """完整内存系统"""
    
    def __init__(self):
        self.short_term = ShortTermMemory()
        self.long_term = VectorStore()
        self.working = {}
    
    def remember_conversation(self, role: str, content: str):
        """记忆对话"""
        self.short_term.add(role, content)
    
    def save_important(self, key: str, content: str):
        """保存重要信息"""
        self.long_term.add(content, {"key": key})
        self.working[key] = content
    
    def recall(self, query: str) -> str:
        """召回相关记忆"""
        results = self.long_term.search(query)
        if results:
            return results[0][2]
        return None
    
    def get_context_for_llm(self) -> str:
        """获取LLM上下文"""
        context_parts = []
        
        # 添加短期记忆
        short_context = self.short_term.get_context()
        if short_context:
            context_parts.append(f"最近对话:\n{short_context}")
        
        return "\n\n".join(context_parts)
