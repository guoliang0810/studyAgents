### 注入策略

```python
class MemoryInjector:
    """记忆注入器"""
    
    def __init__(self, memory_system: CompleteMemorySystem):
        self.memory = memory_system
    
    def build_system_prompt(self, base_prompt: str) -> str:
        """构建带记忆的系统提示"""
        context = self.memory.get_context_for_llm()
        
        if context:
            return f"""{base_prompt}

## 对话上下文
{context}

## 重要信息
{self._get_important_info()}
"""
        return base_prompt
    
    def _get_important_info(self) -> str:
        """获取重要信息"""
        info_parts = []
        
        # 从工作记忆获取
        for key, value in self.memory.working.items():
            if key.startswith("important_"):
                info_parts.append(f"- {key}: {value}")
        
        return "\n".join(info_parts) if info_parts else "无"
    
    def inject_to_prompt(self, prompt: str, include_history: bool = True) -> str:
        """注入记忆到提示"""
        if include_history:
            history = self.memory.get_context_for_llm()
            if history:
                return f"""{history}

## 当前请求
{prompt}"""
        
        return prompt
```

### 智能记忆选择

```python
class MemorySelector:
    """记忆选择器"""
    
    def __init__(self, memory_system: CompleteMemorySystem):
        self.memory = memory_system
    
    def select_relevant(
        self,
        query: str,
        max_memories: int = 5
    ) -> List[MemoryItem]:
        """选择相关记忆"""
        # 从长期记忆搜索
        results = self.memory.long_term.search(query, top_k=max_memories)
        
        memories = []
        for content_hash, similarity, content in results:
            if similarity > 0.3:  # 相似度阈值
                memories.append({
                    "content": content,
                    "similarity": similarity
                })
        
        return memories
    
    def select_by_recency(self, max_items: int = 5) -> List[MemoryItem]:
        """按时间选择"""
        recent = self.memory.short_term.items[-max_items:]
        return [{"content": item.content, "recency": True} for item in recent]
    
    def select_by_importance(self) -> List[MemoryItem]:
        """按重要性选择"""
        important = []
        
        for key, value in self.memory.working.items():
            if key.startswith("important_"):
                important.append({"content": value, "importance": True})
        
        return important
```

```python
import json

class MemoryPersister:
    """记忆持久化"""
    
    def __init__(self, storage_path: str = "memory.json"):
        self.storage_path = storage_path
    
    async def save(self, memory_system: CompleteMemorySystem):
        """保存记忆"""
        data = {
            "short_term": [
                {"content": item.content, "created": item.created_at.isoformat()}
                for item in memory_system.short_term.items
            ],
            "long_term_keys": list(memory_system.long_term.contents.keys()),
            "working": memory_system.working
        }
        
        with open(self.storage_path, 'w') as f:
            json.dump(data, f, indent=2)
        
        print(f"💾 记忆已保存到 {self.storage_path}")
```
