```python
import os
from typing import Optional

class ConfigLoader:
    """配置加载器"""
    
    def __init__(self):
        self.config: Dict[str, Any] = {}
        self.watchers: List[Callable] = []
    
    def load_yaml(self, path: str) -> Dict:
        """加载YAML配置"""
        with open(path, 'r') as f:
            return yaml.safe_load(f)
    
    def load_env(self, prefix: str = "DEERFLOW_") -> Dict:
        """加载环境变量"""
        env_config = {}
        
        for key, value in os.environ.items():
            if key.startswith(prefix):
                config_key = key[len(prefix):].lower()
                env_config[config_key] = value
        
        return env_config
    
    def merge_configs(self, *configs: Dict) -> Dict:
        """合并多个配置"""
        merged = {}
        
        for config in configs:
            self._deep_merge(merged, config)
        
        return merged
    
    def _deep_merge(self, base: Dict, update: Dict):
        """深度合并"""
        for key, value in update.items():
            if key in base and isinstance(base[key], dict) and isinstance(value, dict):
                self._deep_merge(base[key], value)
            else:
                base[key] = value
    
    def load(self, config_path: str = None) -> AppConfig:
        """加载完整配置"""
        # 加载YAML
        yaml_config = self.load_yaml(config_path) if config_path else {}
        
        # 加载环境变量
        env_config = self.load_env()
        
        # 合并
        merged = self.merge_configs(self._get_defaults(), yaml_config, env_config)
        
        return self._to_app_config(merged)
    
    def _get_defaults(self) -> Dict:
        """获取默认配置"""
        return {
            "debug": False,
            "log_level": "INFO",
            "max_concurrent_requests": 10,
            "enable_memory": True,
            "enable_subagents": True
        }
    
    def _to_app_config(self, config: Dict) -> AppConfig:
        """转换为AppConfig"""
        models = []
        for model_data in config.get("models", []):
            models.append(ModelConfig(**model_data))
        
        return AppConfig(
            debug=config.get("debug", False),
            log_level=config.get("log_level", "INFO"),
            models=models,
            max_concurrent_requests=config.get("max_concurrent_requests", 10),
            enable_memory=config.get("enable_memory", True),
            enable_subagents=config.get("enable_subagents", True)
        )
```

### 配置验证

```python
class ConfigValidator:
    """配置验证器"""
    
    def validate(self, config: AppConfig) -> tuple[bool, List[str]]:
        """验证配置"""
        errors = []
        
        # 验证模型配置
        if not config.models:
            errors.append("至少需要配置一个模型")
        else:
            for model in config.models:
                if not model.name:
                    errors.append("模型名称不能为空")
                if model.temperature < 0 or model.temperature > 2:
                    errors.append(f"模型 {model.name} 的 temperature 必须在 0-2 之间")
        
        # 验证数值范围
        if config.max_concurrent_requests < 1:
            errors.append("max_concurrent_requests 必须大于 0")
        
        return len(errors) == 0, errors

class HotReloadConfig:
    """热更新配置"""
    
    def __init__(self, loader: ConfigLoader):
        self.loader = loader
        self.current_config = None
    
    def watch(self, path: str, callback: Callable):
        """监视配置变化"""
        import time
        
        last_modified = os.path.getmtime(path)
        
        while True:
            try:
                current_modified = os.path.getmtime(path)
                if current_modified > last_modified:
                    print(f"🔄 检测到配置变化，重新加载...")
                    new_config = self.loader.load(path)
                    callback(new_config)
                    last_modified = current_modified
            except Exception as e:
                print(f"⚠️ 监视错误: {e}")
            
            time.sleep(1)
```

```python
class EnvironmentConfig:
    """环境配置管理"""
    
    ENVIRONMENTS = ["development", "staging", "production"]
    
    def __init__(self):
        self.current_env = os.getenv("ENV", "development")
        self.configs: Dict[str, AppConfig] = {}
    
    def load_all(self, config_dir: str):
        """加载所有环境配置"""
        for env in self.ENVIRONMENTS:
            config_path = f"{config_dir}/{env}.yaml"
            if os.path.exists(config_path):
                loader = ConfigLoader()
                self.configs[env] = loader.load(config_path)
                print(f"✅ 加载 {env} 配置")
    
    def get_current(self) -> AppConfig:
        """获取当前环境配置"""
        return self.configs.get(self.current_env)
    
    def switch_env(self, env: str):
        """切换环境"""
        if env not in self.ENVIRONMENTS:
            raise ValueError(f"未知环境: {env}")
        
        self.current_env = env
        os.environ["ENV"] = env
        print(f"🔄 已切换到 {env} 环境")
```

class ConfigurationSystem:
    """配置管理系统"""
    
    def __init__(self):
        self.loader = ConfigLoader()
        self.validator = ConfigValidator()
        self.env_config = EnvironmentConfig()
        self.current_config: Optional[AppConfig] = None
    
    def initialize(self, config_path: str = "config/app.yaml"):
        """初始化配置系统"""
        # 加载配置
        self.current_config = self.loader.load(config_path)
        
        # 验证
        is_valid, errors = self.validator.validate(self.current_config)
        if not is_valid:
            raise ValueError(f"配置验证失败: {errors}")
        
        print(f"✅ 配置系统初始化完成")
        print(f"   模型数: {len(self.current_config.models)}")
        print(f"   内存: {'启用' if self.current_config.enable_memory else '禁用'}")
    
    def update(self, updates: Dict):
        """更新配置"""
        if not self.current_config:
            raise RuntimeError("配置系统未初始化")
        
        # 应用更新
        for key, value in updates.items():
            if hasattr(self.current_config, key):
                setattr(self.current_config, key, value)
        
        # 重新验证
        is_valid, errors = self.validator.validate(self.current_config)
        if not is_valid:
            raise ValueError(f"配置更新失败: {errors}")
        
        print(f"✅ 配置已更新")
