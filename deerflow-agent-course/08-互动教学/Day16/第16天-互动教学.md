```python
@dataclass
class PluginConfig:
    """插件配置"""
    name: str
    enabled: bool = True
    settings: Dict[str, Any] = field(default_factory=dict)

class PluginConfigRegistry:
    """插件配置注册表"""
    
    def __init__(self):
        self.plugins: Dict[str, PluginConfig] = {}
    
    def register(self, plugin: PluginConfig):
        """注册插件配置"""
        self.plugins[plugin.name] = plugin
        print(f"🔌 注册插件: {plugin.name}")
    
    def get(self, name: str) -> Optional[PluginConfig]:
        """获取插件配置"""
        return self.plugins.get(name)
    
    def enable(self, name: str):
        """启用插件"""
        if name in self.plugins:
            self.plugins[name].enabled = True
            print(f"✅ 启用插件: {name}")
    
    def disable(self, name: str):
        """禁用插件"""
        if name in self.plugins:
            self.plugins[name].enabled = False
            print(f"❌ 禁用插件: {name}")
    
    def list_enabled(self) -> List[str]:
        """列出已启用的插件"""
        return [name for name, cfg in self.plugins.items() if cfg.enabled]
```

### 版本控制

```python
import json
from datetime import datetime

@dataclass
class ConfigVersion:
    """配置版本"""
    version: str
    config: Dict
    created_at: datetime
    created_by: str
    comment: str = ""

class ConfigVersionManager:
    """配置版本管理器"""
    
    def __init__(self, storage_path: str = "config_versions"):
        self.storage_path = storage_path
        self.versions: List[ConfigVersion] = []
    
    def save_version(
        self,
        version: str,
        config: Dict,
        created_by: str,
        comment: str = ""
    ):
        """保存版本"""
        v = ConfigVersion(
            version=version,
            config=config,
            created_at=datetime.now(),
            created_by=created_by,
            comment=comment
        )
        self.versions.append(v)
        self._persist(v)
        print(f"💾 保存配置版本: {version}")
    
    def get_version(self, version: str) -> Optional[Dict]:
        """获取指定版本"""
        for v in self.versions:
            if v.version == version:
                return v.config
        return None
    
    def list_versions(self) -> List[str]:
        """列出所有版本"""
        return [v.version for v in self.versions]
    
    def rollback(self, version: str) -> Dict:
        """回滚到指定版本"""
        config = self.get_version(version)
        if config:
            print(f"🔄 回滚到版本: {version}")
            return config
        raise ValueError(f"版本不存在: {version}")

            }, f, indent=2)
```

```python
class ConfigTemplate:
    """配置模板"""
    
    TEMPLATES = {
        "minimal": {
            "debug": True,
            "enable_memory": True,
            "enable_subagents": False
        },
        "standard": {
            "debug": False,
            "enable_memory": True,
            "enable_subagents": True,
            "max_concurrent_requests": 10
        },
        "production": {
            "debug": False,
            "enable_memory": True,
            "enable_subagents": True,
            "max_concurrent_requests": 100,
            "log_level": "WARNING"
        }
    }
    
    @classmethod
    def get(cls, name: str) -> Dict:
        """获取模板"""
        if name not in cls.TEMPLATES:
            raise ValueError(f"未知模板: {name}")
        return cls.TEMPLATES[name].copy()
    
    @classmethod
    def apply_to_config(cls, name: str, config: Dict) -> Dict:
        """应用模板到配置"""
        template = cls.get(name)
        return {**template, **config}
```
