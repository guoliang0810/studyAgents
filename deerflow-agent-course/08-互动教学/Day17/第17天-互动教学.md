### 模型工厂实现

```python
from abc import ABC, abstractmethod

@dataclass
class ModelCapabilities:
    """模型能力"""
    supports_function_calling: bool = False
    supports_vision: bool = False
    supports_streaming: bool = True
    max_tokens: int = 4096
    supports_json_mode: bool = False

class BaseModel(ABC):
    """模型基类"""
    
    @property
    @abstractmethod
    def name(self) -> str:
        pass
    
    @property
    @abstractmethod
    def capabilities(self) -> ModelCapabilities:
        pass
    
    @abstractmethod
    async def generate(self, prompt: str, **kwargs) -> str:
        pass

class OpenAIModel(BaseModel):
    """OpenAI模型"""
    
    def __init__(self, model_name: str = "gpt-4", api_key: str = None):
        self._name = model_name
        self._api_key = api_key
    
    @property
    def name(self) -> str:
        return self._name
    
    @property
    def capabilities(self) -> ModelCapabilities:
        return ModelCapabilities(
            supports_function_calling=True,
            supports_vision=False,
            supports_streaming=True,
            max_tokens=8192,
            supports_json_mode=True
        )
    
    async def generate(self, prompt: str, **kwargs) -> str:
        print(f"🤖 [{self.name}] 生成响应...")
        return f"[模拟响应] {prompt[:50]}..."

class AnthropicModel(BaseModel):
    """Anthropic模型"""
    
    @property
    def name(self) -> str:
        return "claude-3-opus"
    
    @property
    def capabilities(self) -> ModelCapabilities:
        return ModelCapabilities(
            supports_function_calling=True,
            supports_vision=True,
            supports_streaming=True,
            max_tokens=200000,
            supports_json_mode=False
        )
    
    async def generate(self, prompt: str, **kwargs) -> str:
        print(f"🤖 [{self.name}] 生成响应...")
        return f"[模拟响应] {prompt[:50]}..."

class ModelFactory:
    """模型工厂"""
    
    _providers = {
        "openai": OpenAIModel,
        "anthropic": AnthropicModel,
    }
    
    @classmethod
    def create(cls, provider: str, **kwargs) -> BaseModel:
        """创建模型实例"""
        if provider not in cls._providers:
            raise ValueError(f"未知提供商: {provider}")
        
        return cls._providers[provider](**kwargs)
    
    @classmethod
    def register(cls, name: str, model_class: type):
        """注册新的模型"""
        cls._providers[name] = model_class
```

### 智能路由

```python
class ModelRouter:
    """模型路由器"""
    
    def __init__(self):
        self.models: Dict[str, BaseModel] = {}
        self.usage_stats: Dict[str, int] = {}
    
    def register(self, name: str, model: BaseModel):
        """注册模型"""
        self.models[name] = model
        self.usage_stats[name] = 0
    
    async def route(
        self,
        task: str,
        required_capabilities: ModelCapabilities = None
    ) -> BaseModel:
        """路由到合适的模型"""
        
        for name, model in self.models.items():
            if required_capabilities:
                # 检查能力
                caps = model.capabilities
                if (required_capabilities.supports_vision and not caps.supports_vision):
                    continue
                if (required_capabilities.supports_function_calling and 
                    not caps.supports_function_calling):
                    continue
            
            # 选择
            self.usage_stats[name] += 1
            return model
        
        raise ValueError("没有合适的模型")
    
    def get_least_used(self) -> BaseModel:
        """获取使用最少的模型"""
        if not self.usage_stats:
            raise ValueError("没有可用的模型")
        
        least_used = min(self.usage_stats.items(), key=lambda x: x[1])
        return self.models[least_used[0]]

```python
class MultiModelSystem:
    """多模型支持系统"""
    
    def __init__(self):
        self.factory = ModelFactory
        self.router = ModelRouter()
    
    def setup_default_models(self):
        """设置默认模型"""
        self.router.register("gpt4", self.factory.create("openai", model_name="gpt-4"))
        self.router.register("claude", self.factory.create("anthropic"))
        print("✅ 默认模型已设置")
    
    async def generate(
        self,
        prompt: str,
        preferred_model: str = None
    ) -> str:
        """生成响应"""
        if preferred_model and preferred_model in self.router.models:
            model = self.router.models[preferred_model]
        else:
            model = self.router.get_least_used()
        
        return await model.generate(prompt)
