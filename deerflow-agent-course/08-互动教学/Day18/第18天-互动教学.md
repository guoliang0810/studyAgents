```python
     │ 系统  │  │        │  │ 服务    │
     └───────┘  └────────┘  └─────────┘
""")

@dataclass
class MCPRequest:
    """MCP请求"""
    method: str
    params: Dict[str, Any]
    id: str = field(default_factory=lambda: uuid.uuid4().hex[:8])

@dataclass
class MCPResponse:
    """MCP响应"""
    result: Any
    error: Optional[str] = None
    id: str = ""
```

### MCP Client实现

```python
class MCPClient:
    """MCP客户端"""
    
    def __init__(self):
        self.tools: Dict[str, Callable] = {}
        self.resources: Dict[str, Any] = {}
    
    def register_tool(self, name: str, tool: Callable):
        """注册工具"""
        self.tools[name] = tool
        print(f"🔧 注册MCP工具: {name}")
    
    def register_resource(self, name: str, resource: Any):
        """注册资源"""
        self.resources[name] = resource
        print(f"📦 注册MCP资源: {name}")
    
    async def send_request(self, request: MCPRequest) -> MCPResponse:
        """发送请求"""
        method = request.method
        
        if method.startswith("tools/"):
            tool_name = method.replace("tools/", "")
            if tool_name in self.tools:
                try:
                    result = await self.tools[tool_name](**request.params)
                    return MCPResponse(result=result, id=request.id)
                except Exception as e:
                    return MCPResponse(error=str(e), id=request.id)
        
        if method.startswith("resources/"):
            resource_name = method.replace("resources/", "")
            if resource_name in self.resources:
                return MCPResponse(result=self.resources[resource_name], id=request.id)
        
        return MCPResponse(error=f"Unknown method: {method}", id=request.id)
    
    async def call_tool(self, name: str, params: Dict = None) -> Any:
        """调用工具"""
        request = MCPRequest(method=f"tools/{name}", params=params or {})
        response = await self.send_request(request)
        
        if response.error:
            raise ValueError(response.error)
        
        return response.result
```

## 🏫 第70节：MCP服务器实现（10:00-10:45）

### 服务器端

```python
class MCPServer:
    """MCP服务器"""
    
    def __init__(self):
        self.tools = {}
        self.resources = {}
    
    def tool(self, name: str = None):
        """工具装饰器"""
        def decorator(func: Callable) -> Callable:
            tool_name = name or func.__name__
            self.tools[tool_name] = func
            return func
        return decorator
    
    def resource(self, name: str):
        """资源装饰器"""
        def decorator(func: Callable) -> Callable:
            self.resources[name] = func
            return func
        return decorator
    
    async def handle_request(self, request: MCPRequest) -> MCPResponse:
        """处理请求"""
        method = request.method
        
        if method in self.tools:
            result = await self.tools[method](**request.params)
            return MCPResponse(result=result, id=request.id)
        
        if method in self.resources:
            result = self.resources[method]()
            return MCPResponse(result=result, id=request.id)
        
        return MCPResponse(error=f"Unknown: {method}", id=request.id)

# 使用示例
server = MCPServer()

@server.tool()
async def get_weather(city: str) -> str:
    return f"{city}的天气：晴，25°C"

@server.resource("database://users")
def get_users():
    return [{"id": 1, "name": "张三"}, {"id": 2, "name": "李四"}]
```
