### 检查点实现

```python
@dataclass
class Checkpoint:
    """检查点"""
    id: str
    timestamp: datetime
    state: Dict[str, Any]
    metadata: Dict[str, Any] = field(default_factory=dict)

class CheckpointManager:
    """检查点管理器"""
    
    def __init__(self, storage_dir: str = "checkpoints"):
        self.storage_dir = storage_dir
        self.checkpoints: List[Checkpoint] = []
    
    def create_checkpoint(self, state: Dict, metadata: Dict = None) -> str:
        """创建检查点"""
        checkpoint_id = f"cp_{len(self.checkpoints)}_{datetime.now().strftime('%Y%m%d%H%M%S')}"
        
        checkpoint = Checkpoint(
            id=checkpoint_id,
            timestamp=datetime.now(),
            state=state.copy(),
            metadata=metadata or {}
        )
        
        self.checkpoints.append(checkpoint)
        self._save_to_disk(checkpoint)
        
        print(f"💾 创建检查点: {checkpoint_id}")
        return checkpoint_id
    
    def restore_checkpoint(self, checkpoint_id: str) -> Optional[Dict]:
        """恢复检查点"""
        for cp in self.checkpoints:
            if cp.id == checkpoint_id:
                print(f"🔄 恢复检查点: {checkpoint_id}")
                return cp.state.copy()
        
        return None
    
    def list_checkpoints(self) -> List[str]:
        """列出检查点"""
        return [cp.id for cp in self.checkpoints]
    
    def _save_to_disk(self, checkpoint: Checkpoint):
        """保存到磁盘"""
        os.makedirs(self.storage_dir, exist_ok=True)
        path = f"{self.storage_dir}/{checkpoint.id}.json"
        
        with open(path, 'w') as f:
            json.dump({
                "id": checkpoint.id,
                "timestamp": checkpoint.timestamp.isoformat(),
                "state": checkpoint.state,
                "metadata": checkpoint.metadata
            }, f)
```

### 序列化策略

```python
import pickle
import json

class StateSerializer:
    """状态序列化器"""
    
    @staticmethod
    def to_json(state: Dict) -> str:
        """序列化为JSON"""
        def default_serializer(obj):
            if isinstance(obj, datetime):
                return obj.isoformat()
            if isinstance(obj, (set, frozenset)):
                return list(obj)
            raise TypeError(f"不可序列化: {type(obj)}")
        
        return json.dumps(state, default=default_serializer, indent=2)
    
    @staticmethod
    def from_json(json_str: str) -> Dict:
        """从JSON反序列化"""
        def date_hook(dct):
            for key, value in dct.items():
                if isinstance(value, str) and 'T' in value:
                    try:
                        dct[key] = datetime.fromisoformat(value)
                    except:
                        pass
            return dct
        
        return json.loads(json_str, object_hook=date_hook)
    
    @staticmethod
    def to_pickle(state: Dict, path: str):
        """序列化为pickle"""
        with open(path, 'wb') as f:
            pickle.dump(state, f)
    
    @staticmethod
    def from_pickle(path: str) -> Dict:
        """从pickle反序列化"""
        with open(path, 'rb') as f:
            return pickle.load(f)
```

## 🏫 第79节：增量检查点（11:00-11:45）

### 优化策略

```python
class IncrementalCheckpointManager(CheckpointManager):
    """增量检查点管理器"""
    
    def __init__(self, storage_dir: str = "checkpoints"):
        super().__init__(storage_dir)
        self.base_checkpoint: Optional[Dict] = None
        self.deltas: List[Dict] = []
    
    def create_checkpoint(self, state: Dict, metadata: Dict = None) -> str:
        """创建增量检查点"""
        if not self.base_checkpoint:
            # 首次，创建完整检查点
            self.base_checkpoint = state.copy()
            return super().create_checkpoint(state, metadata)
        
        # 计算差异
        delta = self._compute_delta(self.base_checkpoint, state)
        self.deltas.append(delta)
        
        # 只保存差异
        checkpoint_id = f"cp_{len(self.checkpoints)}"
        self.checkpoints.append(Checkpoint(
            id=checkpoint_id,
            timestamp=datetime.now(),
            state=delta,
            metadata={**metadata, "type": "incremental"}
        ))
        
        print(f"💾 创建增量检查点: {checkpoint_id} (差异大小: {len(str(delta))} bytes)")
        return checkpoint_id
    
    def restore_checkpoint(self, checkpoint_id: str) -> Optional[Dict]:
        """恢复增量检查点"""
        # 找到检查点
        target_index = -1
        for i, cp in enumerate(self.checkpoints):
            if cp.id == checkpoint_id:
                target_index = i
                break
        
        if target_index == -1:
            return None
        
        # 重建状态
        state = self.base_checkpoint.copy()
        for i in range(target_index + 1):
            if self.checkpoints[i].metadata.get("type") == "incremental":
                state = self._apply_delta(state, self.checkpoints[i].state)
            else:
                state = self.checkpoints[i].state.copy()
        
        return state
    
    def _compute_delta(self, base: Dict, current: Dict) -> Dict:
        """计算差异"""
        delta = {}
        for key in current:
            if key not in base or base[key] != current[key]:
                delta[key] = current[key]
        return delta
    
    def _apply_delta(self, base: Dict, delta: Dict) -> Dict:
        """应用差异"""
        result = base.copy()
        result.update(delta)
        return result
```

class RecoverableAgent:
    """可恢复的Agent"""
    
    def __init__(self):
        self.state = {}
        self.checkpoint_manager = CheckpointManager()
    
    def update_state(self, key: str, value: Any):
        """更新状态"""
        self.state[key] = value
    
    def save_checkpoint(self, metadata: Dict = None) -> str:
        """保存检查点"""
        return self.checkpoint_manager.create_checkpoint(self.state, metadata)
    
    def restore(self, checkpoint_id: str) -> bool:
        """恢复状态"""
        state = self.checkpoint_manager.restore_checkpoint(checkpoint_id)
        if state:
            self.state = state
            return True
        return False
    
    def get_latest_checkpoint(self) -> Optional[str]:
        """获取最新检查点"""
        checkpoints = self.checkpoint_manager.list_checkpoints()
        return checkpoints[-1] if checkpoints else None
