import pytest

class TestMemorySystem:
    """内存系统单元测试"""
    
    def test_short_term_memory(self):
        """测试短期记忆"""
        memory = CompleteMemorySystem()
        memory.remember_conversation("user", "Hello")
        
        context = memory.get_context_for_llm()
        assert "Hello" in context
    
    def test_long_term_memory(self):
        """测试长期记忆"""
        memory = CompleteMemorySystem()
        memory.save_important("key", "value")
        
        value = memory.working.get("key")
        assert value == "value"

class AgentTestSuite:
    """Agent测试套件"""
    
    def run_all_tests(self):
        """运行所有测试"""
        print("🧪 运行Agent测试套件...")
        
        tests = [
            ("内存系统", self.test_memory),
            ("子代理", self.test_subagent),
            ("工具系统", self.test_tools),
            ("中间件", self.test_middleware)
        ]
        
        passed = 0
        for name, test_func in tests:
            try:
                test_func()
                print(f"  ✅ {name}")
                passed += 1
            except Exception as e:
                print(f"  ❌ {name}: {e}")
        
        print(f"\n结果: {passed}/{len(tests)} 通过")
    
    def test_memory(self):
        memory = CompleteMemorySystem()
        assert memory is not None
    
    def test_subagent(self):
        agent = CodeExpertSubagent()
        assert agent.config.name == "code_expert"
    
    def test_tools(self):
        registry = ToolRegistry()
        registry.register(CalculatorTool())
        assert "calculator" in registry.list_tools()
    
    def test_middleware(self):
        middleware = AuthenticationMiddleware()
        assert middleware.name == "authentication"
