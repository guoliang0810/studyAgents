### 输入验证

```python
class InputValidator:
    """输入验证器"""
    
    MAX_PROMPT_LENGTH = 100000
    DANGEROUS_PATTERNS = [
        "ignore previous instructions",
        "disregard all rules",
        "sudo",
        "rm -rf"
    ]
    
    def validate(self, prompt: str) -> tuple[bool, str]:
        """验证输入"""
        # 长度检查
        if len(prompt) > self.MAX_PROMPT_LENGTH:
            return False, f"输入过长，最大 {self.MAX_PROMPT_LENGTH} 字符"
        
        # 危险模式检查
        prompt_lower = prompt.lower()
        for pattern in self.DANGEROUS_PATTERNS:
            if pattern in prompt_lower:
                return False, f"检测到危险内容: {pattern}"
        
        return True, "验证通过"
    
    def sanitize(self, prompt: str) -> str:
        """清理输入"""
        # 移除多余空白
        prompt = ' '.join(prompt.split())
        return prompt

# 测试
```

### 权限控制

```python
from enum import Enum

class Permission(Enum):
    READ = "read"
    WRITE = "write"
    EXECUTE_TOOL = "execute_tool"
    ACCESS_ADMIN = "access_admin"

@dataclass
class User:
    user_id: str
    role: str
    permissions: Set[Permission]

class AuthorizationManager:
    """授权管理器"""
    
    ROLE_PERMISSIONS = {
        "user": {Permission.READ, Permission.EXECUTE_TOOL},
        "premium": {Permission.READ, Permission.EXECUTE_TOOL, Permission.WRITE},
        "admin": {Permission.READ, Permission.WRITE, Permission.EXECUTE_TOOL, Permission.ACCESS_ADMIN}
    }
    
    def check_permission(self, user: User, permission: Permission) -> bool:
        """检查权限"""
        return permission in self.ROLE_PERMISSIONS.get(user.role, set())
    
    def authorize_tool(self, user: User, tool_name: str) -> bool:
        """授权工具使用"""
        if not self.check_permission(user, Permission.EXECUTE_TOOL):
            return False
        
        # 管理员工具检查
        admin_tools = {"delete_all", "system_config"}
        if tool_name in admin_tools:
            return self.check_permission(user, Permission.ACCESS_ADMIN)
        
        return True
```

### 加密存储

```python
import hashlib
import secrets

class DataSecurity:
    """数据安全"""
    
    @staticmethod
    def hash_password(password: str) -> str:
        """密码哈希"""
        salt = secrets.token_hex(16)
        hash_value = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode(),
            salt.encode(),
            100000
        )
        return f"{salt}${hash_value.hex()}"
    
    @staticmethod
    def verify_password(password: str, stored: str) -> bool:
        """验证密码"""
        salt, hash_value = stored.split('$')
        new_hash = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode(),
            salt.encode(),
            100000
        )
        return new_hash.hex() == hash_value
    
    @staticmethod
    def mask_sensitive(data: str, visible_chars: int = 4) -> str:
        """脱敏处理"""
        if len(data) <= visible_chars:
            return '*' * len(data)
        return data[:visible_chars] + '*' * (len(data) - visible_chars)
```
