```python
        # 合并变量

        return self.template.format(**merged)
```

### 上下文注入实战

**张老师**："让我们实现一个简单的上下文注入器："

```python
class SimpleContextInjector:
    """简单的上下文注入器"""
    
    def __init__(self, memory_size: int = 5):
        self.memory_size = memory_size
        self.conversation_history = []
    
    def add_message(self, role: str, content: str):
        """添加对话记录"""
        self.conversation_history.append({"role": role, "content": content})
        
        # 保持最近N条记录
        if len(self.conversation_history) > self.memory_size:
            self.conversation_history.pop(0)
    
    def inject_context(self, prompt: str) -> str:
        """注入上下文到提示中"""
        if not self.conversation_history:
            return prompt
        
        # 构建上下文
        context = "## 最近对话记录：\n"
        for msg in self.conversation_history:
            role = msg["role"]
            content = msg["content"][:100]  # 截断前100字符
            context += f"- {role}: {content}...\n"
        
        return f"{context}\n\n{prompt}"

# 使用示例
injector = SimpleContextInjector(memory_size=3)

# 模拟对话

injector.add_message("assistant", "好的！Python是非常好的编程语言。你想学哪个方面？")
injector.add_message("user", "我想学数据分析")

# 注入上下文
base_prompt = "请继续回答用户的问题："
enhanced_prompt = injector.inject_context(base_prompt)

print("增强后的提示：")
print(enhanced_prompt)
print("-" * 50)
print(f"对话历史: {injector.conversation_history}")
```

### 记忆检索策略

**张老师**："当对话历史很多时，我们需要智能检索相关记忆："

```python
import re
from typing import List, Dict

class SmartMemoryRetriever:
    """智能记忆检索器"""
    
    def __init__(self):
        self.all_memories = []
    
    def add_memory(self, memory: Dict):
        """添加记忆"""
        self.all_memories.append(memory)
    
    def retrieve_relevant(self, query: str, top_k: int = 3) -> List[Dict]:
        """检索相关记忆"""
        # 简单实现：基于关键词匹配
        query_keywords = set(query.lower().split())
        
        scored_memories = []
        for memory in self.all_memories:
            score = 0
            content = memory.get("content", "").lower()
            
            # 关键词匹配得分
            for keyword in query_keywords:
                if keyword in content:
                    score += 1
            
            # 时间权重（越新的记忆权重越高）
            time_weight = memory.get("recency", 1.0)
            score *= time_weight
            
            if score > 0:
                scored_memories.append((score, memory))
        
        # 按分数排序
        scored_memories.sort(key=lambda x: x[0], reverse=True)
        
        # 返回top_k个
        return [memory for _, memory in scored_memories[:top_k]]

# 使用示例
retriever = SmartMemoryRetriever()

# 添加一些记忆
retriever.add_memory({"content": "用户喜欢Python编程", "recency": 1.0})
retriever.add_memory({"content": "用户对数据分析感兴趣", "recency": 0.9})
retriever.add_memory({"content": "用户曾经问过Java问题", "recency": 0.5})
retriever.add_memory({"content": "用户是初学者", "recency": 0.8})

# 检索相关记忆
query = "Python数据分析"
relevant = retriever.retrieve_relevant(query, top_k=2)

print(f"查询: {query}")
print("相关记忆:")
for mem in relevant:
    print(f"  - {mem['content']} (权重: {mem['recency']})")
```

```python
class LearningMemorySystem:
    """学习进度记忆系统"""
    
    def __init__(self):
        self.learning_records = []
    
    def add_record(self, topic: str, duration: int, mastery: float):
        """添加学习记录"""
        record = {
            "topic": topic,
            "duration": duration,  # 分钟
            "mastery": mastery,    # 0-100
            "timestamp": "2024-03-27"  # 简化
        }
        self.learning_records.append(record)
    
    def get_progress_summary(self) -> str:
        """获取进度总结"""
        if not self.learning_records:
            return "暂无学习记录"
        
        total_duration = sum(r["duration"] for r in self.learning_records)
        avg_mastery = sum(r["mastery"] for r in self.learning_records) / len(self.learning_records)
        topics = [r["topic"] for r in self.learning_records]
        
        return f"""
学习进度总结：
- 总学习时长: {total_duration}分钟
- 平均掌握程度: {avg_mastery:.1f}%
- 学习主题: {', '.join(set(topics))}
- 总学习次数: {len(self.learning_records)}
"""
    
    def retrieve_relevant_topic(self, new_topic: str) -> List[dict]:
        """检索相关主题的学习记录"""
        relevant = []
        for record in self.learning_records:
            if new_topic.lower() in record["topic"].lower():
                relevant.append(record)
        return relevant
```

```python
from typing import Protocol, Dict, Any, Optional
from abc import ABC, abstractmethod

class Skill(ABC):
    """技能基类（抽象类）"""
    
    @property
    @abstractmethod
    def name(self) -> str:
        """技能名称"""
        pass
    
    @property  
    @abstractmethod
    def description(self) -> str:
        """技能描述"""
        pass
    
    @property
    def version(self) -> str:
        """技能版本"""
        return "1.0.0"
    
    @abstractmethod
    async def execute(self, context: Dict[str, Any]) -> Dict[str, Any]:
        """执行技能"""
        pass
    
    async def validate(self, input_data: Dict[str, Any]) -> bool:
        """验证输入"""
        return True

class SkillRegistry:
    """技能注册表"""
    
    def __init__(self):
        self._skills: Dict[str, Skill] = {}
    
    def register(self, skill: Skill):
        """注册技能"""
        self._skills[skill.name] = skill
        print(f"✓ 已注册技能: {skill.name} - {skill.description}")
    
    def get_skill(self, name: str) -> Optional[Skill]:
        """获取技能"""
        return self._skills.get(name)
    
    def list_skills(self) -> Dict[str, str]:
        """列出所有技能"""
        return {name: skill.description for name, skill in self._skills.items()}
```

```python
        """执行数据分析"""
        data = context.get("data", [])
        analysis_type = context.get("analysis_type", "summary")
        
        print(f"开始数据分析: {analysis_type}")
        print(f"数据量: {len(data)}条记录")
        
        # 模拟分析过程

                "count": len(data),

        else:
            result = {"error": f"未知分析类型: {analysis_type}"}
```

```python
import importlib
from pathlib import Path

class SkillLoader:
    """技能加载器"""
    
    def __init__(self, skill_dir: str):

    def discover_skills(self) -> list:
```

```python
class FileProcessingSkill(Skill):
    """文件处理技能"""
    
    @property
    def name(self) -> str:
        return "file_processing"
    
    @property
    def description(self) -> str:
        return "处理文件操作：读取、写入、转换格式"
    
    async def execute(self, context: Dict[str, Any]) -> Dict[str, Any]:
        """执行文件处理"""
        operation = context.get("operation", "read")
        file_path = context.get("file_path", "")
        content = context.get("content", "")
        
        print(f"执行文件操作: {operation}, 文件: {file_path}")
        
        if operation == "read":
            # 模拟读取
            return {"success": True, "content": f"文件{file_path}的内容..."}
        elif operation == "write":
            # 模拟写入
            return {"success": True, "message": f"已写入{len(content)}字符到{file_path}"}
        elif operation == "convert":
            # 模拟格式转换
            target_format = context.get("target_format", "txt")
            return {"success": True, "converted": f"转换为{target_format}格式"}
        else:
            return {"success": False, "error": f"未知操作: {operation}"}
    
    async def validate(self, input_data: Dict[str, Any]) -> bool:
        """验证输入"""
        required_fields = {"operation", "file_path"}
        if not required_fields.issubset(input_data.keys()):
            print(f"❌ 缺少必要字段: {required_fields - set(input_data.keys())}")
            return False
        
        operation = input_data["operation"]
        valid_operations = {"read", "write", "convert"}
        if operation not in valid_operations:
```

```python
import asyncio
from datetime import datetime, timedelta

class WeatherQuerySkill(Skill):
    """天气查询技能"""
    
    def __init__(self):
        self.cache = {}  # 缓存天气数据
        self.cache_duration = timedelta(hours=1)  # 缓存1小时
    
    @property
    def name(self) -> str:
        return "weather_query"
    
    @property
    def description(self) -> str:
        return "查询城市天气信息，提供温度和穿衣建议"
    
    async def _fetch_weather(self, city: str) -> dict:
        """模拟获取天气数据（实际项目会调用API）"""
        print(f"🌤️  查询{city}的天气...")
        await asyncio.sleep(0.5)  # 模拟网络延迟
        
        # 模拟不同城市的天气
        weather_data = {
            "北京": {"temp": 18, "condition": "晴朗", "humidity": 40},
            "上海": {"temp": 22, "condition": "多云", "humidity": 65},
            "广州": {"temp": 28, "condition": "小雨", "humidity": 85},
            "成都": {"temp": 20, "condition": "阴天", "humidity": 70}
        }
        
        if city in weather_data:
            return weather_data[city]
        else:
            raise ValueError(f"城市{city}不存在")
    
    def _get_clothing_advice(self, temp: int) -> str:
        """根据温度提供穿衣建议"""
        if temp < 0:
            return "极寒天气，建议穿羽绒服、帽子、手套"
        elif temp < 10:
            return "寒冷天气，建议穿厚外套、毛衣"
        elif temp < 20:
            return "凉爽天气，建议穿长袖、薄外套"
        elif temp < 30:
            return "温暖天气，建议穿短袖、薄长裤"
        else:
            return "炎热天气，建议穿短袖短裤，注意防晒"
    
    async def execute(self, context: Dict[str, Any]) -> Dict[str, Any]:
        """执行天气查询"""
        city = context.get("city", "")
        
        if not city:
            return {"success": False, "error": "请提供城市名称"}
        
        # 检查缓存
        cache_key = city
        if cache_key in self.cache:
            cached_data, cached_time = self.cache[cache_key]
            if datetime.now() - cached_time < self.cache_duration:
                print(f"📦 使用缓存数据: {city}")
                weather = cached_data
            else:
                # 缓存过期
                del self.cache[cache_key]
                weather = await self._fetch_weather(city)
        else:
            weather = await self._fetch_weather(city)
        
        # 更新缓存
        self.cache[cache_key] = (weather, datetime.now())
        
        # 生成建议
        temp = weather.get("temp", 0)
        advice = self._get_clothing_advice(temp)
        
        return {
            "success": True,
            "city": city,
            "temperature": f"{temp}°C",
            "condition": weather.get("condition", "未知"),
            "humidity": f"{weather.get('humidity', 0)}%",
            "advice": advice,
            "cached": cache_key in self.cache
        }
    
    async def validate(self, input_data: Dict[str, Any]) -> bool:
        """验证输入"""
        if "city" not in input_data:
            print("❌ 缺少城市名称")
            return False
        
        city = input_data["city"]
        if not isinstance(city, str) or not city.strip():
            print("❌ 城市名称无效")
            return False
        
        # 简单验证城市名称（实际会更复杂）
        if len(city) < 2:
            print("❌ 城市名称太短")
            return False
        
        return True
```

### 技能编排器

**张老师**："现在让我们创建一个技能编排器，可以组合多个技能："

```python
class SkillOrchestrator:
    """技能编排器"""
    
    def __init__(self, registry: SkillRegistry):
        self.registry = registry
    
    async def execute_skill_chain(self, skill_names: list, initial_context: dict) -> dict:
        """执行技能链"""
        context = initial_context.copy()
        results = []
        
        for skill_name in skill_names:
            skill = self.registry.get_skill(skill_name)
            if not skill:
                return {
                    "success": False,
                    "error": f"技能'{skill_name}'未找到",
                    "completed_skills": results
                }
            
            # 验证输入
            if not await skill.validate(context):
                return {
                    "success": False,
                    "error": f"技能'{skill_name}'输入验证失败",
                    "completed_skills": results
                }
            
            # 执行技能
            try:
                result = await skill.execute(context)
                results.append({
                    "skill": skill_name,
                    "result": result
                })
                
                # 将结果合并到上下文中（供下一个技能使用）
                if result.get("success"):
                    context.update(result)
                else:
                    return {
                        "success": False,
                        "error": f"技能'{skill_name}'执行失败",
                        "completed_skills": results
                    }
                    
            except Exception as e:
                return {
                    "success": False,
                    "error": f"技能'{skill_name}'执行异常: {e}",
                    "completed_skills": results
                }
        
        return {
            "success": True,
            "results": results,
            "final_context": context
        }

# 测试技能编排
```

```python
class TravelPlanningSkill(Skill):
    """旅行规划技能（组合技能）"""
    
    def __init__(self, orchestrator: SkillOrchestrator):
        self.orchestrator = orchestrator
    
    @property
    def name(self) -> str:
        return "travel_planning"
    
    @property
    def description(self) -> str:
        return "规划旅行，包括天气查询、行程安排、预算估算"
    
    async def execute(self, context: Dict[str, Any]) -> Dict[str, Any]:
        """执行旅行规划"""
        city = context.get("city", "")
        days = context.get("days", 3)
        
        print(f"✈️  开始规划{city}{days}天旅行...")

        weather_result = await self.orchestrator.execute_skill_chain(
            skill_names=["weather_query"],
            initial_context={"city": city}
        )
        
        if not weather_result["success"]:
            return weather_result
        
        weather_data = weather_result["results"][0]["result"]
        
        # 步骤2：根据天气规划活动
        temp = int(weather_data["temperature"].replace("°C", ""))
        condition = weather_data["condition"]
        
        activities = []
        if "雨" in condition:
            activities = ["参观博物馆", "室内购物", "咖啡馆休闲"]
        elif temp > 25:
            activities = ["海滩游玩", "水上活动", "夜市逛吃"]
        else:
            activities = ["城市观光", "公园散步", "当地美食体验"]
        
        # 步骤3：生成旅行报告
        report = f"""
{city}{days}天旅行规划报告
====================
📅 行程天数: {days}天
🌤️  天气状况: {condition}, {weather_data['temperature']}
💧 湿度: {weather_data['humidity']}
👕 穿衣建议: {weather_data['advice']}

🗺️ 推荐活动:
{chr(10).join(f'  • {act}' for act in activities)}

💰 预算估算:
  - 住宿: {days * 300}元 ({days}天 × 300元/天)
  - 餐饮: {days * 200}元 ({days}天 × 200元/天)
  - 交通: 约500元
  - 总计: {days * 500 + 500}元

🎯 温馨提示:
  1. 根据天气调整行程
  2. 准备合适的衣物
  3. 提前预订住宿
  4. 注意旅行安全
"""
        
        # 步骤4：保存报告
        file_result = await self.orchestrator.execute_skill_chain(
            skill_names=["file_processing"],
            initial_context={
                "operation": "write",
                "file_path": f"{city}_travel_plan.txt",
                "content": report
            }
        )
        
        return {
            "success": True,
            "city": city,
            "days": days,
            "weather": weather_data,
            "activities": activities,
            "report": report,
            "file_saved": file_result["success"],
            "filename": f"{city}_travel_plan.txt"
        }
    
    async def validate(self, input_data: Dict[str, Any]) -> bool:
        """验证输入"""
        if "city" not in input_data:
            return False
        
        days = input_data.get("days", 3)
        if not isinstance(days, int) or days < 1 or days > 30:
            return False
        
        return True
```
