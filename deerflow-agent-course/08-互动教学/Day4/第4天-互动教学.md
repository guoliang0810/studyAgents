### 中间件核心概念

**张老师**："DeerFlow中间件遵循**责任链模式**，每个中间件可以：
1. **预处理请求**：修改输入、验证权限、记录日志
2. **处理响应**：修改输出、格式化数据、错误处理
3. **传递控制**：决定是否继续执行链中的下一个中间件"

### 中间件接口定义

**张老师**："让我们看DeerFlow中真实的中间件接口："

```python
from typing import Protocol, Optional, Dict, Any
from dataclasses import dataclass

@dataclass
class AgentRequest:
    """Agent请求"""
    user_id: str
    message: str
    thread_id: Optional[str] = None
    metadata: Dict[str, Any] = None

@dataclass  
class AgentResponse:
    """Agent响应"""
    content: str
    success: bool = True
    error: Optional[str] = None
    metadata: Dict[str, Any] = None

class AgentMiddleware(Protocol):
    """Agent中间件协议"""
    
    @property
    def name(self) -> str:
        """中间件名称"""
        ...
    
    async def before_agent(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None
    ) -> AgentRequest:
        """Agent执行前调用 - 预处理请求"""
        ...
    
    async def after_agent(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """Agent执行后调用 - 处理响应"""
        ...
```

```python
        self.valid_tokens = {"user_token_123": "user_001", "admin_token_456": "admin_001"}
    
    async def before_agent(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None

            print(f"❌ [{self.name}] token无效，设置为匿名用户")

        return request
    
    async def after_agent(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """响应后处理（认证中间件通常不需要）"""
        return response
```

### 中间件链实现

**张老师**："让我们实现一个完整的中间件链："

```python
class MiddlewareChain:
    """中间件链"""
    
    def __init__(self, middlewares: list):
        self.middlewares = middlewares
        print(f"🔗 初始化中间件链，共 {len(middlewares)} 个中间件")
    
    async def execute_before(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None
    ) -> AgentRequest:
        """执行前置中间件链"""
        current_request = request
        
        for middleware in self.middlewares:
            print(f"  → 执行前置: {middleware.name}")
            current_request = await middleware.before_agent(current_request, config)
        
        return current_request
    
    async def execute_after(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """执行后置中间件链（逆序）"""
        current_response = response
        
        for middleware in reversed(self.middlewares):
            print(f"  ← 执行后置: {middleware.name}")
            current_response = await middleware.after_agent(current_response, config)
        
        return current_response

# 创建中间件链
async def test_middleware_chain():
    # 创建中间件
    auth = AuthenticationMiddleware()
    logger = LoggingMiddleware()
    
    # 创建链（注意顺序）
    chain = MiddlewareChain([auth, logger])
    
    # 创建请求
    request = AgentRequest(
        user_id="",
        message="测试请求",
        metadata={"auth_token": "user_token_123"}
    )
    
    print("\n执行前置链:")
    processed_request = await chain.execute_before(request)
    
    print("\n模拟Agent处理...")
    response = AgentResponse(content="Agent响应内容", success=True)
    
    print("\n执行后置链:")
    final_response = await chain.execute_after(response)

asyncio.run(test_middleware_chain())
```

### 限流中间件

**小王**："限流中间件应该记录请求次数，超过限制就拒绝..."

```python
import time
from collections import defaultdict

class RateLimitMiddleware:
    """限流中间件"""
    
    def __init__(self, requests_per_minute: int = 60):
        self.name = "rate_limit"
        self.requests_per_minute = requests_per_minute
        self.user_requests = defaultdict(list)  # 用户请求时间记录
    
    async def before_agent(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None
    ) -> AgentRequest:
        """检查限流"""
        current_time = time.time()
        user_id = request.user_id
        
        # 清理超过1分钟的记录
        self.user_requests[user_id] = [
            t for t in self.user_requests[user_id]
            if current_time - t < 60
        ]
        
        # 检查请求次数
        request_count = len(self.user_requests[user_id])
        
        if request_count >= self.requests_per_minute:
            print(f"⚠️ [{self.name}] 用户 {user_id} 请求过于频繁")
            # 添加标记，实际限流在Agent层处理
            if not request.metadata:
                request.metadata = {}
            request.metadata["rate_limited"] = True
        else:
            self.user_requests[user_id].append(current_time)
            print(f"✅ [{self.name}] 用户 {user_id} 请求通过 ({request_count + 1}/{self.requests_per_minute})")
        
        return request
    
    async def after_agent(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """后置处理"""
        return response

# 测试限流中间件
async def test_rate_limit():
    middleware = RateLimitMiddleware(requests_per_minute=5)
    
    for i in range(7):
        request = AgentRequest(user_id="test_user", message=f"请求{i}")
        await middleware.before_agent(request)
        await asyncio.sleep(0.1)  # 模拟请求间隔

asyncio.run(test_rate_limit())
```

### 线程数据中间件实现

**张老师**："让我们实现一个简化版的ThreadDataMiddleware："

```python
import uuid
from typing import Dict, Optional

class ThreadState:
    """简化的线程状态"""
    def __init__(self):
        self.thread_id: str = ""
        self.messages: list = []
        self.metadata: Dict = {}
        self.created_at: str = ""
    
    def to_dict(self) -> Dict:
        return {
            "thread_id": self.thread_id,
            "messages": self.messages,
            "metadata": self.metadata,
            "created_at": self.created_at
        }

class ThreadStorage:
    """线程存储（内存实现）"""
    def __init__(self):
        self.storage: Dict[str, Dict] = {}
    
    async def save(self, thread_id: str, state: Dict):
        self.storage[thread_id] = state
        print(f"💾 [{self.__class__.__name__}] 保存线程 {thread_id}")
    
    async def load(self, thread_id: str) -> Optional[Dict]:
        state = self.storage.get(thread_id)
        if state:
            print(f"📂 [{self.__class__.__name__}] 加载线程 {thread_id}")
        return state
    
    async def delete(self, thread_id: str):
        if thread_id in self.storage:
            del self.storage[thread_id]
            print(f"🗑️ [{self.__class__.__name__}] 删除线程 {thread_id}")

class ThreadDataMiddleware:
    """线程数据中间件"""
    
    def __init__(self, storage: ThreadStorage):
        self.name = "thread_data"
        self.storage = storage
    
    async def before_agent(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None
    ) -> AgentRequest:
        """Agent执行前：加载或创建线程数据"""
        
        thread_id = request.thread_id
        
        if not thread_id:
            # 创建新线程
            thread_id = self._generate_thread_id()
            request.thread_id = thread_id
            print(f"🆕 [{self.name}] 创建新线程: {thread_id}")
            
            # 创建初始状态
            initial_state = self._create_initial_state(thread_id)
            if not request.metadata:
                request.metadata = {}
            request.metadata["thread_state"] = initial_state
        else:
            # 加载现有线程
            print(f"📥 [{self.name}] 加载线程: {thread_id}")
            state = await self.storage.load(thread_id)
            if state:
                if not request.metadata:
                    request.metadata = {}
                request.metadata["thread_state"] = state
            else:
                print(f"⚠️ [{self.name}] 线程不存在，创建新状态")
                initial_state = self._create_initial_state(thread_id)
                if not request.metadata:
                    request.metadata = {}
                request.metadata["thread_state"] = initial_state
        
        return request
    
    async def after_agent(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """Agent执行后：保存线程数据"""
        
        thread_id = response.metadata.get("thread_id") if response.metadata else None
        thread_state = response.metadata.get("thread_state") if response.metadata else None
        
        if thread_id and thread_state:
            await self.storage.save(thread_id, thread_state)
            print(f"💾 [{self.name}] 线程数据已保存")
        
        return response
    
    def _generate_thread_id(self) -> str:
        """生成线程ID"""
        return f"thread_{uuid.uuid4().hex[:8]}"
    
    def _create_initial_state(self, thread_id: str) -> Dict:
        """创建初始线程状态"""
        return {
            "thread_id": thread_id,
            "messages": [],
            "metadata": {},
            "created_at": datetime.now().isoformat()
        }

# 测试线程数据中间件
async def test_thread_middleware():
    storage = ThreadStorage()
    middleware = ThreadDataMiddleware(storage)
    
    # 测试新线程
    print("\n=== 测试新线程创建 ===")
    request1 = AgentRequest(user_id="user1", message="你好", thread_id=None)
    result1 = await middleware.before_agent(request1)
    print(f"新线程ID: {result1.thread_id}")
    print(f"线程状态: {result1.metadata['thread_state']}")
    
    # 测试加载现有线程
    print("\n=== 测试线程加载 ===")
    request2 = AgentRequest(user_id="user1", message="继续对话", thread_id=result1.thread_id)
    result2 = await middleware.before_agent(request2)
    print(f"加载线程ID: {result2.thread_id}")
    
    # 模拟Agent更新状态并保存
    print("\n=== 模拟Agent保存状态 ===")
    response = AgentResponse(content="你好！有什么可以帮助你的？", success=True)
    response.metadata = {
        "thread_id": result1.thread_id,
        "thread_state": {
            "thread_id": result1.thread_id,
            "messages": [("user", "你好"), ("assistant", "你好！有什么可以帮助你的？")],
            "created_at": datetime.now().isoformat()
        }
    }
    await middleware.after_agent(response)

asyncio.run(test_thread_middleware())
```

### 状态合并策略

```python
def merge_messages(existing: list, new: list) -> list:
    """合并消息列表"""
    if existing is None:
        return new or []
    if new is None:
        return existing
    # 去重合并
    seen = set()
    merged = []
    for msg in existing + new:
        msg_id = str(msg)
        if msg_id not in seen:
            seen.add(msg_id)
            merged.append(msg)
    return merged

def merge_metadata(existing: dict, new: dict) -> dict:
    """合并元数据"""
    if existing is None:
        return new or {}
    if new is None:
        return existing
    # 深度合并
    merged = existing.copy()
    merged.update(new)
    return merged

# 测试状态合并
existing_state = {
    "messages": [("user", "你好")],
    "metadata": {"theme": "dark", "language": "zh"}
}

new_state = {
    "messages": [("assistant", "你好！")],
    "metadata": {"theme": "light"}
}

print("现有状态:", existing_state)
print("新状态:", new_state)
print("合并后:")
print("  消息:", merge_messages(existing_state["messages"], new_state["messages"]))
print("  元数据:", merge_metadata(existing_state["metadata"], new_state["metadata"]))
```

```python
import time
from typing import List, Dict, Any
from dataclasses import dataclass, field

@dataclass
class MiddlewareExecution:
    """中间件执行记录"""
    middleware_name: str
    phase: str  # "before" or "after"
    start_time: float
    end_time: float
    duration_ms: float = field(init=False)
    success: bool = True
    error: str = ""
    
    def __post_init__(self):
        self.duration_ms = (self.end_time - self.start_time) * 1000

class MiddlewareTracer:
    """中间件执行追踪器"""
    
    def __init__(self):
        self.executions: List[MiddlewareExecution] = []
        self.request_start: float = 0
        self.request_end: float = 0
    
    def start_request(self):
        """开始追踪请求"""
        self.request_start = time.time()
        print(f"⏱️  [{self.__class__.__name__}] 开始追踪请求")
    
    def record_execution(
        self,
        middleware_name: str,
        phase: str,
        start_time: float,
        end_time: float,
        success: bool = True,
        error: str = ""
    ):
        """记录中间件执行"""
        execution = MiddlewareExecution(
            middleware_name=middleware_name,
            phase=phase,
            start_time=start_time,
            end_time=end_time,
            success=success,
            error=error
        )
        self.executions.append(execution)
        status = "✅" if success else "❌"
        print(f"  {status} {middleware_name}.{phase}: {execution.duration_ms:.2f}ms")
    
    def end_request(self):
        """结束追踪请求"""
        self.request_end = time.time()
        total_duration = (self.request_end - self.request_start) * 1000
        print(f"⏱️  [{self.__class__.__name__}] 请求完成: {total_duration:.2f}ms")
    
    def generate_report(self) -> str:
        """生成执行报告"""
        if not self.executions:
            return "无执行记录"
        
        total_duration = (self.request_end - self.request_start) * 1000
        before_executions = [e for e in self.executions if e.phase == "before"]
        after_executions = [e for e in self.executions if e.phase == "after"]
        
        # 找出最慢的中间件
        slowest = max(self.executions, key=lambda e: e.duration_ms)
        
        report = f"""
═══════════════════════════════════════════
        中间件执行追踪报告
═══════════════════════════════════════════

📊 总体统计:
   • 总执行时间: {total_duration:.2f}ms
   • 前置中间件数: {len(before_executions)}
   • 后置中间件数: {len(after_executions)}
   • 成功执行: {sum(1 for e in self.executions if e.success)}
   • 失败执行: {sum(1 for e in self.executions if not e.success)}

⏱️  执行时间明细:

   【前置处理】
{chr(10).join(f"   • {e.middleware_name}: {e.duration_ms:.2f}ms" for e in before_executions)}

   【后置处理】
{chr(10).join(f"   • {e.middleware_name}: {e.duration_ms:.2f}ms" for e in after_executions)}

🐌 最慢中间件:
   • {slowest.middleware_name} ({slowest.phase}): {slowest.duration_ms:.2f}ms

⚠️  错误列表:
{chr(10).join(f"   • {e.middleware_name} ({e.phase}): {e.error}" for e in self.executions if not e.success) or "   • 无错误"}

═══════════════════════════════════════════
"""
        return report

# 创建追踪包装器
class TracedMiddleware:
    """带追踪功能的中间件包装器"""
    
    def __init__(self, middleware: AgentMiddleware, tracer: MiddlewareTracer):
        self.middleware = middleware
        self.tracer = tracer
    
    async def before_agent(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None
    ) -> AgentRequest:
        """追踪前置执行"""
        start = time.time()
        try:
            result = await self.middleware.before_agent(request, config)
            self.tracer.record_execution(
                self.middleware.name, "before", start, time.time(), success=True
            )
            return result
        except Exception as e:
            self.tracer.record_execution(
                self.middleware.name, "before", start, time.time(), success=False, error=str(e)
            )
            raise
    
    async def after_agent(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """追踪后置执行"""
        start = time.time()
        try:
            result = await self.middleware.after_agent(response, config)
            self.tracer.record_execution(
                self.middleware.name, "after", start, time.time(), success=True
            )
            return result
        except Exception as e:
            self.tracer.record_execution(
                self.middleware.name, "after", start, time.time(), success=False, error=str(e)
            )
            raise
```
