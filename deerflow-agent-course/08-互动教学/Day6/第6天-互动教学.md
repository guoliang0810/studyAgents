print("""
图片处理流程：

1. 【接收图片】
   • 用户上传图片或提供图片URL
   • 验证图片格式和大小

2. 【预处理】
   • 格式转换（PNG/JPEG/WebP）
   • 尺寸调整（缩放到合适大小）
   • 质量压缩（减少token消耗）

3. 【理解分析】
   • 调用多模态模型分析图片
   • 提取图片内容描述
   • 识别关键信息

4. 【结果处理】
   • 保存分析结果
   • 更新线程状态
   • 返回给用户
""")

@dataclass
class ImageData:
    """图片数据"""
    url: str
    content_type: str
    size: int
    dimensions: tuple  # (width, height)
    base64_data: Optional[str] = None

class ImageProcessor:
    """图片处理器"""
    
    SUPPORTED_FORMATS = {"png", "jpeg", "jpg", "webp", "gif"}
    MAX_SIZE = 10 * 1024 * 1024  # 10MB
    MAX_DIMENSION = 4096
    
    def validate_image(self, image: ImageData) -> tuple[bool, str]:
        """验证图片"""
        # 检查格式
        if image.content_type not in self.SUPPORTED_FORMATS:
            return False, f"不支持的图片格式: {image.content_type}"
        
        # 检查大小
        if image.size > self.MAX_SIZE:
            return False, f"图片太大: {image.size} bytes"
        
        # 检查尺寸
        if max(image.dimensions) > self.MAX_DIMENSION:
            return False, f"图片尺寸过大: {image.dimensions}"
        
        return True, "验证通过"
    
    async def preprocess(self, image: ImageData) -> ImageData:
        """预处理图片"""
        print(f"📷 预处理图片: {image.url}")
        print(f"   原始尺寸: {image.dimensions}")
        
        # 模拟缩放
        if max(image.dimensions) > 2048:
            scale = 2048 / max(image.dimensions)
            new_width = int(image.dimensions[0] * scale)
            new_height = int(image.dimensions[1] * scale)
            image.dimensions = (new_width, new_height)
            print(f"   缩放后: {image.dimensions}")
        
        return image
    
    async def analyze(self, image: ImageData, prompt: str = "描述这张图片") -> str:
        """分析图片（模拟多模态模型）"""
        print(f"🔍 分析图片内容...")
        
        # 模拟返回分析结果
        analysis = f"""图片分析结果：
- 尺寸: {image.dimensions[0]}x{image.dimensions[1]}
- 格式: {image.content_type}
- 大小: {image.size / 1024:.1f} KB
- 内容描述: 这是一张包含多种元素的图片，包含文字和图形信息。
"""
        
        return analysis

### ViewImageMiddleware实现

**张老师**："现在让我们实现ViewImageMiddleware："

```python
class ViewImageMiddleware:
    """图片查看中间件"""
    
    def __init__(self):
        self.name = "view_image"
        self.processor = ImageProcessor()
        self.viewed_images = {}  # 记录已查看的图片
    
    async def before_agent(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None
    ) -> AgentRequest:
        """检查请求中是否有图片"""
        
        # 从消息中提取图片
        images = self.extract_images(request.message)
        
        if images:
            print(f"🖼️ [{self.name}] 检测到 {len(images)} 张图片")
            
            # 处理每张图片
            processed_images = []
            for img_data in images:
                # 验证
                is_valid, msg = self.processor.validate_image(img_data)
                if is_valid:
                    # 预处理
                    processed = await self.processor.preprocess(img_data)
                    processed_images.append(processed)
                else:
                    print(f"   ⚠️ {msg}")
            
            # 保存到请求
            if not request.metadata:
                request.metadata = {}
            request.metadata["images"] = processed_images
            request.metadata["image_count"] = len(processed_images)
        
        return request
    
    async def after_agent(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """保存图片查看记录"""
        
        if response.metadata and "images" in response.metadata:
            images = response.metadata["images"]
            for img in images:
                self.viewed_images[img.url] = {
                    "dimensions": img.dimensions,
                    "viewed_at": datetime.now().isoformat()
                }
        
        return response
    
    def extract_images(self, message: str) -> List[ImageData]:
        """从消息中提取图片信息（简化实现）"""
        images = []
        
        # 简化：检测URL模式
        import re
        url_pattern = r'https?://[^\s]+\.(?:png|jpg|jpeg|gif|webp)'
        urls = re.findall(url_pattern, message, re.IGNORECASE)
        
        for url in urls:
            # 简化创建ImageData
            ext = url.split('.')[-1]
            images.append(ImageData(
                url=url,
                content_type=ext,
                size=1024 * 100,  # 模拟大小
                dimensions=(800, 600)  # 模拟尺寸
            ))
        
        return images

# 测试
async def test_view_image_middleware():
    middleware = ViewImageMiddleware()
    
    # 测试带图片的请求
    request = AgentRequest(
        user_id="user1",
        message="请分析这张图片 https://example.com/diagram.png"
    )
    
    result = await middleware.before_agent(request)
    
    if result.metadata and "images" in result.metadata:
        print(f"\n检测到 {len(result.metadata['images'])} 张图片")
        for img in result.metadata['images']:
            print(f"  - {img.url}")

asyncio.run(test_view_image_middleware())

### 限制策略实现

```python
class SubagentLimitMiddleware:
    """子代理限制中间件"""
    
    def __init__(
        self,
        max_subagents: int = 5,
        max_depth: int = 3,
        timeout_per_subagent: float = 60.0
    ):
        self.name = "subagent_limit"
        self.max_subagents = max_subagents
        self.max_depth = max_depth
        self.timeout_per_subagent = timeout_per_subagent
        self.current_subagents = 0
        self.subagent_calls = []
    
    async def before_agent(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None
    ) -> AgentRequest:
        """检查子代理限制"""
        
        # 获取当前深度
        depth = request.metadata.get("subagent_depth", 0) if request.metadata else 0
        
        # 检查深度限制
        if depth >= self.max_depth:
            print(f"🛑 [{self.name}] 超过最大深度限制 ({self.max_depth})")
            if not request.metadata:
                request.metadata = {}
            request.metadata["subagent_blocked"] = True
            request.metadata["block_reason"] = "max_depth_exceeded"
            return request
        
        # 检查子代理数量
        if self.current_subagents >= self.max_subagents:
            print(f"🛑 [{self.name}] 子代理数量已达上限 ({self.max_subagents})")
            if not request.metadata:
                request.metadata = {}
            request.metadata["subagent_blocked"] = True
            request.metadata["block_reason"] = "max_subagents_exceeded"
            return request
        
        # 更新计数
        self.current_subagents += 1
        self.subagent_calls.append({
            "depth": depth,
            "timestamp": time.time()
        })
        
        print(f"✅ [{self.name}] 允许子代理执行 (深度:{depth}, 数量:{self.current_subagents})")
        
        return request
    
    async def after_agent(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """子代理执行完成"""
        self.current_subagents = max(0, self.current_subagents - 1)
        return response

# 测试
async def test_subagent_limit():
    middleware = SubagentLimitMiddleware(max_subagents=2, max_depth=2)
    
    # 测试深度限制
    print("\n=== 测试深度限制 ===")
    for depth in range(4):
        request = AgentRequest(
            user_id="user1",
            message="执行任务",
            metadata={"subagent_depth": depth}
        )
        result = await middleware.before_agent(request)
        blocked = result.metadata.get("subagent_blocked", False) if result.metadata else False
        print(f"  深度 {depth}: {'阻塞' if blocked else '允许'}")

asyncio.run(test_subagent_limit())
```

### 任务管理中间件

**张老师**："欢迎回来！现在我们来学习TodoMiddleware。想象一个项目经理，跟踪每个任务的进度。"

```python
class TodoItem:
    """待办事项"""
    def __init__(self, id: str, content: str, status: str = "pending"):
        self.id = id
        self.content = content
        self.status = status  # pending, in_progress, completed
        self.created_at = datetime.now().isoformat()

class TodoMiddleware:
    """待办事项中间件"""
    
    def __init__(self):
        self.name = "todo"
        self.todos = []
    
    def extract_todos(self, message: str) -> List[str]:
        """从消息中提取待办事项"""
        todos = []
        
        # 检测待办模式
        import re
        patterns = [
            r'待办[：:]\s*(.+)',
            r'TODO[：:]\s*(.+)',
            r'任务[：:]\s*(.+)',
            r'需要做[：:]\s*(.+)'
        ]
        
        for pattern in patterns:
            matches = re.findall(pattern, message)
            todos.extend(matches)
        
        return todos
    
    async def before_agent(
        self,
        request: AgentRequest,
        config: Optional[Dict] = None
    ) -> AgentRequest:
        """提取和跟踪待办"""
        
        todos = self.extract_todos(request.message)
        
        if todos:
            for content in todos:
                todo = TodoItem(
                    id=f"todo_{len(self.todos) + 1}",
                    content=content
                )
                self.todos.append(todo)
                print(f"📋 [{self.name}] 新增待办: {content}")
            
            if not request.metadata:
                request.metadata = {}
            request.metadata["todos"] = self.todos
        
        return request
    
    async def after_agent(
        self,
        response: AgentResponse,
        config: Optional[Dict] = None
    ) -> AgentResponse:
        """返回待办列表"""
        
        if self.todos:
            print(f"📋 [{self.name}] 当前待办 ({len(self.todos)} 项):")
            for todo in self.todos:
                print(f"   [{todo.status}] {todo.content}")
        
        return response

# 测试
async def test_todo_middleware():
    middleware = TodoMiddleware()
    
    request = AgentRequest(
        user_id="user1",
        message="待办：1. 搜索资料 2. 整理结果 3. 生成报告"
    )
    
    await middleware.before_agent(request)
    response = AgentResponse(content="已完成")
    await middleware.after_agent(response)

asyncio.run(test_todo_middleware())
```

```python
class ImageTaskAgent:
    """图片任务Agent"""
    
    def __init__(self):
        self.image_middleware = ViewImageMiddleware()
        self.todo_middleware = TodoMiddleware()
        self.subagent_limit = SubagentLimitMiddleware()
    
    async def process(self, request: AgentRequest) -> AgentResponse:
        """处理请求"""
        # 1. 提取待办
        request = await self.todo_middleware.before_agent(request)
        
        # 2. 处理图片
        request = await self.image_middleware.before_agent(request)
        
        # 3. 检查子代理限制
        request = await self.subagent_limit.before_agent(request)
        
        # 4. 模拟处理
        content = "任务处理完成"
        if request.metadata:
            if "images" in request.metadata:
                content += f"\n处理了 {len(request.metadata['images'])} 张图片"
            if "todos" in request.metadata:
                content += f"\n跟踪了 {len(request.metadata['todos'])} 个待办"
        
        response = AgentResponse(content=content, metadata=request.metadata)
        
        # 5. 后置处理
        await self.subagent_limit.after_agent(response)
        await self.image_middleware.after_agent(response)
        await self.todo_middleware.after_agent(response)
        
        return response

# 测试完整系统
async def test_image_task_agent():
    agent = ImageTaskAgent()
    
    request = AgentRequest(
        user_id="user1",
        message="请分析这些图片 https://example.com/a.png https://example.com/b.png 并生成报告"
    )
    
    response = await agent.process(request)
    print(f"\n响应内容:\n{response.content}")

asyncio.run(test_image_task_agent())
