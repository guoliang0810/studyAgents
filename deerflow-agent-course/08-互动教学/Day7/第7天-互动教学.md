### 工具基类定义

**张老师**："让我们定义工具的基类："

```python
from abc import ABC, abstractmethod
from typing import Any, Dict, Optional
from dataclasses import dataclass

@dataclass
class ToolDefinition:
    """工具定义"""
    name: str
    description: str
    parameters: Dict[str, Any]
    returns: Dict[str, Any]
    category: str = "general"
    tags: list = None

class BaseTool(ABC):
    """工具基类"""
    
    @property
    @abstractmethod
    def definition(self) -> ToolDefinition:
        """返回工具定义"""
        pass
    
    @abstractmethod
    async def execute(self, **kwargs) -> Any:
        """执行工具"""
        pass
    
    def validate_parameters(self, params: Dict[str, Any]) -> tuple[bool, str]:
        """验证参数"""
        return True, "验证通过"
    
    def get_schema(self) -> Dict[str, Any]:
        """获取JSON Schema"""
        return {
            "name": self.definition.name,
            "description": self.definition.description,
            "parameters": self.definition.parameters
        }

class ToolRegistry:
    """工具注册表"""
    
    def __init__(self):
        self._tools: Dict[str, BaseTool] = {}
    
    def register(self, tool: BaseTool):
        """注册工具"""
        self._tools[tool.definition.name] = tool
        print(f"🔧 已注册工具: {tool.definition.name}")
    
    def get(self, name: str) -> Optional[BaseTool]:
        """获取工具"""
        return self._tools.get(name)
    
    def list_tools(self) -> list:
        """列出所有工具"""
        return [tool.definition for tool in self._tools.values()]
    
    def get_by_category(self, category: str) -> list:
        """按类别获取工具"""
        return [
            tool for tool in self._tools.values()
            if tool.definition.category == category
        ]

# 创建全局注册表
registry = ToolRegistry()
```

```python
                    "required": True
                }
            },

                "type": "number",
                "description": "计算结果"
```

### 沙箱工具实现

```python
import subprocess
import tempfile
import os
from pathlib import Path

class SandboxConfig:
    """沙箱配置"""
    def __init__(
        self,
        timeout: float = 30.0,
        memory_limit: str = "256MB",
        allowed_paths: list = None
    ):
        self.timeout = timeout
        self.memory_limit = memory_limit
        self.allowed_paths = allowed_paths or []

class SandboxTool(BaseTool):
    """沙箱执行工具"""
    
    def __init__(self, config: SandboxConfig = None):
        self.config = config or SandboxConfig()
    
    @property
    def definition(self) -> ToolDefinition:
        return ToolDefinition(
            name="sandbox_execute",
            description="在安全沙箱中执行代码",
            parameters={
                "code": {
                    "type": "string",
                    "description": "要执行的代码",
                    "required": True
                },
                "language": {
                    "type": "string",
                    "description": "编程语言 (python, javascript)",
                    "required": True
                }
            },
            returns={
                "type": "object",
                "description": "执行结果，包含输出和错误"
            },
            category="execution",
            tags=["sandbox", "code", "execute"]
        )
    
    async def execute(self, code: str, language: str = "python") -> Dict:
        """执行代码"""
        print(f"🔒 沙箱执行: {language} 代码 ({len(code)} 字符)")
        
        # 创建临时文件
        with tempfile.NamedTemporaryFile(
            mode='w',
            suffix=f'.{language}',
            delete=False
        ) as f:
            f.write(code)
            temp_file = f.name
        
        try:
            # 执行代码
            if language == "python":
                result = subprocess.run(
                    ["python", temp_file],
                    capture_output=True,
                    text=True,
                    timeout=self.config.timeout
                )
            elif language == "javascript":
                result = subprocess.run(
                    ["node", temp_file],
                    capture_output=True,
                    text=True,
                    timeout=self.config.timeout
                )
            else:
                return {"success": False, "error": f"不支持的语言: {language}"}
            
            return {
                "success": result.returncode == 0,
                "stdout": result.stdout,
                "stderr": result.stderr,
                "returncode": result.returncode
            }
        
        except subprocess.TimeoutExpired:
            return {"success": False, "error": "执行超时"}
        except Exception as e:
            return {"success": False, "error": str(e)}
        finally:
            # 清理临时文件
            os.unlink(temp_file)

# 测试沙箱
async def test_sandbox():
    sandbox = SandboxTool()
    
    # 执行Python代码
    result = await sandbox.execute(
        code="print('Hello from sandbox!')\nprint(2+2)",
        language="python"
    )
    
    print(f"成功: {result['success']}")
    print(f"输出: {result.get('stdout', '')}")

asyncio.run(test_sandbox())
```

### 工具执行器实现

**张老师**："现在让我们实现工具执行器："

```python
class ToolExecutor:
    """工具执行器"""
    
    def __init__(self, registry: ToolRegistry):
        self.registry = registry
        self.execution_history = []
    
    async def execute_tool(
        self,
        tool_name: str,
        parameters: Dict[str, Any]
    ) -> Dict[str, Any]:
        """执行工具"""
        
        # 查找工具
        tool = self.registry.get(tool_name)
        if not tool:
            return {"success": False, "error": f"工具不存在: {tool_name}"}
        
        # 验证参数
        is_valid, message = tool.validate_parameters(parameters)
        if not is_valid:
            return {"success": False, "error": f"参数验证失败: {message}"}
        
        # 记录开始
        start_time = time.time()
        
        try:
            # 执行工具
            result = await tool.execute(**parameters)
            
            # 记录成功
            self.execution_history.append({
                "tool": tool_name,
                "parameters": parameters,
                "result": result,
                "duration": time.time() - start_time,
                "success": True
            })
            
            return {"success": True, "result": result}
        
        except Exception as e:
            # 记录失败
            self.execution_history.append({
                "tool": tool_name,
                "parameters": parameters,
                "error": str(e),
                "duration": time.time() - start_time,
                "success": False
            })
            
            return {"success": False, "error": str(e)}

# 测试执行器
async def test_executor():
    executor = ToolExecutor(registry)
    
    result = await executor.execute_tool(
        "calculator",
        {"expression": "10 + 20 * 3"}
    )
    
    print(f"执行结果: {result}")

asyncio.run(test_executor())
```

                "type": "object",
                "description": "天气信息"
            },
            category="information",
            tags=["weather", "information"]
        )

            return {
                "city": city,
                "temperature": f"{data['temp']}°C",
                "condition": data['condition'],
                "humidity": f"{data['humidity']}%"
            }
        else:
            return {"city": city, "error": "城市未找到"}
