```python
@dataclass
class SandboxConfig:
    """沙箱配置"""
    timeout: float = 30.0          # 超时时间（秒）
    memory_limit: str = "256MB"     # 内存限制
    cpu_limit: float = 1.0          # CPU限制
    network_enabled: bool = False    # 是否启用网络
    disk_limit: str = "100MB"       # 磁盘限制
    allowed_paths: List[str] = []   # 允许访问的路径

@dataclass
class ExecutionResult:
    """执行结果"""
    success: bool
    stdout: str
    stderr: str
    exit_code: int
    duration: float
    error: Optional[str] = None

class Sandbox(ABC):
    """沙箱抽象基类"""
    
    @abstractmethod
    async def execute(
        self,
        code: str,
        language: str,
        config: SandboxConfig
    ) -> ExecutionResult:
        """执行代码"""
        pass
    
    @abstractmethod
    async def validate_code(self, code: str) -> tuple[bool, str]:
        """验证代码安全性"""
        pass
    
    @abstractmethod
    async def cleanup(self):
        """清理资源"""
        pass
```

### 沙箱接口定义

**张老师**："让我们定义完整的沙箱接口："

```python
from abc import ABC, abstractmethod
from typing import Optional, List
import asyncio

class Sandbox(ABC):
    """沙箱抽象基类"""
    
    @property
    @abstractmethod
    def name(self) -> str:
        """沙箱名称"""
        pass
    
    @abstractmethod
    async def execute(
        self,
        code: str,
        language: str,
        config: SandboxConfig
    ) -> ExecutionResult:
        """执行代码"""
        pass
    
    @abstractmethod
    async def validate_code(self, code: str) -> tuple[bool, str]:
        """验证代码安全性"""
        pass
    
    @abstractmethod
    async def cleanup(self):
        """清理资源"""
        pass

class SandboxFactory:
    """沙箱工厂"""
    
    _sandboxes = {
        "docker": "DockerSandbox",
        "process": "ProcessSandbox",
        "web": "WebAssemblySandbox"
    }
    
    @classmethod
    def create(cls, sandbox_type: str, config: SandboxConfig = None) -> Sandbox:
        """创建沙箱实例"""
        config = config or SandboxConfig()
        
        if sandbox_type == "docker":
            return DockerSandbox(config)
        elif sandbox_type == "process":
            return ProcessSandbox(config)
        elif sandbox_type == "web":
            return WebAssemblySandbox(config)
        else:
            raise ValueError(f"未知的沙箱类型: {sandbox_type}")
```

### ProcessSandbox实现

```python
import subprocess
import tempfile
import os

class ProcessSandbox(Sandbox):
    """进程级沙箱"""
    
    def __init__(self, config: SandboxConfig):
        self.config = config
        self.processes = []
    
    @property
    def name(self) -> str:
        return "ProcessSandbox"
    
    async def validate_code(self, code: str) -> tuple[bool, str]:
        """验证代码安全性"""
        # 危险模式检测
        dangerous_patterns = [
            "os.system", "subprocess.", "import os", 
            "__import__", "eval(", "exec(",
            "open(", "file(", "shutil"
        ]
        
        for pattern in dangerous_patterns:
            if pattern in code:
                # 检查是否在注释中
                if f"# {pattern}" not in code and f'"{pattern}"' not in code:
                    return False, f"检测到潜在危险代码: {pattern}"
        
        return True, "代码验证通过"
    
    async def execute(
        self,
        code: str,
        language: str,
        config: SandboxConfig
    ) -> ExecutionResult:
        """执行代码"""
        import time
        start_time = time.time()
        
        # 验证代码
        is_valid, message = await self.validate_code(code)
        if not is_valid:
            return ExecutionResult(
                success=False,
                stdout="",
                stderr=message,
                exit_code=-1,
                duration=0,
                error=message
            )
        
        # 创建临时文件
        suffix = ".py" if language == "python" else ".js"
        with tempfile.NamedTemporaryFile(
            mode='w', suffix=suffix, delete=False
        ) as f:
            f.write(code)
            temp_file = f.name
        
        try:
            # 确定执行命令
            cmd = ["python", temp_file] if language == "python" else ["node", temp_file]
            
            # 执行
            result = subprocess.run(
                cmd,
                capture_output=True,
                text=True,
                timeout=config.timeout
            )
            
            return ExecutionResult(
                success=result.returncode == 0,
                stdout=result.stdout,
                stderr=result.stderr,
                exit_code=result.returncode,
                duration=time.time() - start_time
            )
        
        except subprocess.TimeoutExpired:
            return ExecutionResult(
                success=False,
                stdout="",
                stderr="执行超时",
                exit_code=-1,
                duration=config.timeout,
                error="Execution timeout"
            )
        
        except Exception as e:
            return ExecutionResult(
                success=False,
                stdout="",
                stderr=str(e),
                exit_code=-1,
                duration=time.time() - start_time,
                error=str(e)
            )
        
        finally:
            os.unlink(temp_file)
    
    async def cleanup(self):
        """清理进程"""
        self.processes.clear()

# 测试
async def test_process_sandbox():
    sandbox = ProcessSandbox(SandboxConfig(timeout=5))
    
    # 测试安全代码
    result = await sandbox.execute(
        code="print('Hello from sandbox!')\nprint(sum([1,2,3]))",
        language="python",
        config=SandboxConfig()
    )
    
    print(f"执行成功: {result.success}")
    print(f"输出: {result.stdout}")
    print(f"耗时: {result.duration:.3f}s")

asyncio.run(test_process_sandbox())
```

### Docker沙箱架构

**张老师**："欢迎回来！现在让我们实现Docker沙箱，这是生产环境常用的方案。"

```python
class DockerSandbox(Sandbox):
    """Docker容器沙箱"""
    
    def __init__(self, config: SandboxConfig):
        self.config = config
        self.container_id = None
    
    @property
    def name(self) -> str:
        return "DockerSandbox"
    
    async def validate_code(self, code: str) -> tuple[bool, str]:
        """验证代码"""
        # 更严格的验证
        dangerous_patterns = [
            "rm -rf", "drop database", "format c:",
            "sudo", "chmod 777"
        ]
        
        for pattern in dangerous_patterns:
            if pattern in code.lower():
                return False, f"禁止的命令: {pattern}"
        
        return True, "验证通过"
    
    async def execute(
        self,
        code: str,
        language: str,
        config: SandboxConfig
    ) -> ExecutionResult:
        """在Docker容器中执行"""
        import time
        start_time = time.time()
        
        # 验证
        is_valid, msg = await self.validate_code(code)
        if not is_valid:
            return ExecutionResult(False, "", msg, -1, 0, msg)
        
        # 创建临时文件
        with tempfile.NamedTemporaryFile(
            mode='w', suffix=f'.{language}', delete=False
        ) as f:
            f.write(code)
            code_file = f.name
        
        try:
            # 确定Docker镜像
            image = "python:3.11-slim" if language == "python" else "node:18-slim"
            
            # Docker命令
            cmd = [
                "docker", "run", "--rm",
                "--network=none",
                "--memory=" + config.memory_limit,
                f"--cpus={config.cpu_limit}",
                "-v", f"{code_file}:/code/{os.path.basename(code_file)}",
                image,
                "python" if language == "python" else "node",
                f"/code/{os.path.basename(code_file)}"
            ]
            
            result = subprocess.run(
                cmd,
                capture_output=True,
                text=True,
                timeout=config.timeout
            )
            
            return ExecutionResult(
                success=result.returncode == 0,
                stdout=result.stdout,
                stderr=result.stderr,
                exit_code=result.returncode,
                duration=time.time() - start_time
            )
        
        except Exception as e:
            return ExecutionResult(False, "", str(e), -1, 0, str(e))
        
        finally:
            os.unlink(code_file)
    
    async def cleanup(self):
        """清理容器"""
        if self.container_id:
            subprocess.run(["docker", "rm", "-f", self.container_id])
            self.container_id = None
```

### 虚拟文件系统

**张老师**："现在我们来学习虚拟路径映射，这是沙箱安全的关键。"

```python
class VirtualPathMapper:
    """虚拟路径映射器"""
    
    def __init__(self, allowed_base_paths: List[str]):
        self.allowed_base_paths = allowed_base_paths
        self.virtual_mappings = {}
    
    def map_virtual_path(self, virtual_path: str, real_path: str) -> bool:
        """映射虚拟路径到真实路径"""
        # 验证真实路径在允许范围内
        real_path = os.path.abspath(real_path)
        
        for base in self.allowed_base_paths:
            if real_path.startswith(os.path.abspath(base)):
                self.virtual_mappings[virtual_path] = real_path
                return True
        
        return False
    
    def resolve_virtual_path(self, virtual_path: str) -> Optional[str]:
        """解析虚拟路径"""
        return self.virtual_mappings.get(virtual_path)
    
    def validate_path(self, path: str) -> tuple[bool, str]:
        """验证路径是否安全"""
        for virtual, real in self.virtual_mappings.items():
            if path.startswith(virtual):
                # 转换为真实路径
                actual_path = path.replace(virtual, real, 1)
                if os.path.abspath(actual_path).startswith(real):
                    return True, actual_path
                else:
                    return False, "路径遍历攻击检测"
        
        return False, "路径不在允许范围内"
```

class SecureSandbox:
    """安全沙箱系统"""
    
    def __init__(self, sandbox_type: str = "process"):
        self.config = SandboxConfig()
        self.sandbox = SandboxFactory.create(sandbox_type, self.config)
        self.path_mapper = VirtualPathMapper(["/tmp/sandbox"])
    
    async def execute_secure(
        self,
        code: str,
        language: str,
        virtual_working_dir: str = "/app"
    ) -> ExecutionResult:
        """安全执行代码"""
        
        # 1. 代码安全验证
        is_valid, msg = await self.sandbox.validate_code(code)
        if not is_valid:
            return ExecutionResult(False, "", msg, -1, 0, msg)
        
        # 2. 执行
        result = await self.sandbox.execute(code, language, self.config)
        
        # 3. 结果验证
        if result.success:
            # 检查输出是否包含敏感信息
            sensitive_patterns = ["password", "secret", "api_key"]
            for pattern in sensitive_patterns:
                if pattern.lower() in result.stdout.lower():
                    result.stdout = "[输出已脱敏]"
        
        return result
    
    async def cleanup(self):
        """清理资源"""
        await self.sandbox.cleanup()

# 测试

        language="python"
    )
