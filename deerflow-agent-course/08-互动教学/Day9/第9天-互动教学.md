```python
@dataclass
class VirtualPath:
    """虚拟路径定义"""
    virtual: str           # 虚拟路径，如 /app
    real: str             # 真实路径，如 /tmp/sandbox/app
    permissions: list      # 权限列表
    readonly: bool = False # 是否只读

class VirtualFileSystem:
    """虚拟文件系统"""
    
    def __init__(self):
        self.paths: Dict[str, VirtualPath] = {}
        self.access_log = []
    
    def mount(self, virtual_path: str, real_path: str, readonly: bool = False):
        """挂载虚拟路径"""
        vp = VirtualPath(
            virtual=virtual_path,
            real=real_path,
            permissions=["read"],
            readonly=readonly
        )
        if not readonly:
            vp.permissions.append("write")
        
        self.paths[virtual_path] = vp
        print(f"📁 挂载: {virtual_path} -> {real_path}")
    
    def resolve(self, virtual_path: str) -> Optional[str]:
        """解析虚拟路径为真实路径"""
        # 查找最长匹配的挂载点
        for vp_path, vp in sorted(self.paths.items(), key=lambda x: len(x[0]), reverse=True):
            if virtual_path.startswith(vp_path):
                return virtual_path.replace(vp_path, vp.real, 1)
        return None
    
    def validate_access(
        self,
        virtual_path: str,
        operation: str
    ) -> tuple[bool, str]:
        """验证访问权限"""
        for vp_path, vp in self.paths.items():
            if virtual_path.startswith(vp_path):
                if operation not in vp.permissions:
                    return False, f"无{operation}权限"
                if operation == "write" and vp.readonly:
                    return False, "路径只读"
                return True, "允许访问"
        
        return False, "路径未挂载"
    
    def log_access(self, path: str, operation: str, success: bool):
        """记录访问日志"""
        self.access_log.append({
            "path": path,
            "operation": operation,
            "success": success,
            "timestamp": datetime.now().isoformat()
        })

# 测试
vfs = VirtualFileSystem()
vfs.mount("/app", "/tmp/sandbox/app")
vfs.mount("/data", "/tmp/sandbox/data", readonly=True)

print(vfs.resolve("/app/file.txt"))
print(vfs.validate_access("/app/file.txt", "write"))
print(vfs.validate_access("/data/file.txt", "write"))
```

### 路径遍历攻击防护

**张老师**："现在让我们实现路径遍历攻击的防护："

```python
import os
import re

class PathTraversalDetector:
    """路径遍历检测器"""
    
    DANGEROUS_PATTERNS = [
        r'\.\./',           # ../
        r'\.\.%',           # ..%
        r'%2e%2e/',         # URL编码的 ../
        r'\.\.\\',          # ..\
        r'%c0%af',          # Unicode bypass
    ]
    
    def __init__(self):
        self.patterns = [re.compile(p, re.IGNORECASE) for p in self.DANGEROUS_PATTERNS]
    
    def detect(self, path: str) -> tuple[bool, str]:
        """检测路径遍历攻击"""
        for pattern in self.patterns:
            if pattern.search(path):
                return True, f"检测到危险模式: {pattern.pattern}"
        
        # 检查绝对路径
        if path.startswith('/') and not path.startswith('/app/'):
            return True, "禁止访问绝对路径"
        
        return False, "安全"
    
    def sanitize_path(self, path: str) -> str:
        """清理路径"""
        # 移除多余的斜杠
        path = re.sub(r'/+', '/', path)
        
        # 移除反斜杠
        path = path.replace('\\', '/')
        
        # 移除 ./
        path = re.sub(r'/\./', '/', path)
        
        # 处理 ../
        parts = []
        for part in path.split('/'):
            if part == '..':
                if parts:
                    parts.pop()
            elif part and part != '.':
                parts.append(part)
        
        return '/' + '/'.join(parts) if parts else '/'

# 测试
detector = PathTraversalDetector()
print(detector.detect("/app/../../../etc/passwd"))
print(detector.sanitize_path("/app/./data/../file.txt"))
```

## 🏫 第34节：权限控制模型（10:00-10:45）

### 细粒度权限控制

```python
from enum import Enum

class Permission(Enum):
    READ = "read"
    WRITE = "write"
    EXECUTE = "execute"
    DELETE = "delete"

@dataclass
class AccessControlEntry:
    """访问控制条目"""
    path: str
    permissions: Set[Permission]
    allowed_operations: List[str]
    denied_operations: List[str] = field(default_factory=list)

class AccessControlList:
    """访问控制列表"""
    
    def __init__(self):
        self.entries: List[AccessControlEntry] = []
    
    def add_entry(
        self,
        path: str,
        permissions: List[str],
        operations: List[str] = None
    ):
        """添加访问控制条目"""
        perms = {Permission(p) for p in permissions}
        ops = operations or []
        
        entry = AccessControlEntry(
            path=path,
            permissions=perms,
            allowed_operations=ops
        )
        self.entries.append(entry)
    
    def check_permission(
        self,
        path: str,
        operation: str,
        required_permission: Permission
    ) -> tuple[bool, str]:
        """检查权限"""
        for entry in self.entries:
            if path.startswith(entry.path):
                # 检查显式拒绝
                if operation in entry.denied_operations:
                    return False, f"操作{operation}被拒绝"
                
                # 检查权限
                if required_permission in entry.permissions:
                    return True, "允许"
                
                return False, f"缺少权限: {required_permission.value}"
        
        return False, "无访问控制条目"

# 测试
acl = AccessControlList()
acl.add_entry("/app", ["read", "write"], ["file:read", "file:write"])
print(acl.check_permission("/app/file.txt", "file:write", Permission.READ))
```

### 审计系统实现

```python
from typing import Optional
import json

@dataclass
class AuditLog:
    """审计日志"""
    timestamp: str
    user_id: str
    operation: str
    resource: str
    success: bool
    details: Optional[Dict] = None
    ip_address: Optional[str] = None

class AuditLogger:
    """审计日志记录器"""
    
    def __init__(self, log_file: str = "audit.log"):
        self.log_file = log_file
    
    def log(
        self,
        user_id: str,
        operation: str,
        resource: str,
        success: bool,
        details: Dict = None
    ):
        """记录审计日志"""
        entry = AuditLog(
            timestamp=datetime.now().isoformat(),
            user_id=user_id,
            operation=operation,
            resource=resource,
            success=success,
            details=details
        )
        
        with open(self.log_file, 'a') as f:
            f.write(json.dumps(asdict(entry)) + '\n')
    
    def query(
        self,
        user_id: str = None,
        operation: str = None,
        start_time: str = None,
        end_time: str = None
    ) -> List[AuditLog]:
        """查询审计日志"""
        logs = []
        
        if not os.path.exists(self.log_file):
            return logs
        
        with open(self.log_file, 'r') as f:
            for line in f:
                entry = json.loads(line)
                
                # 应用过滤条件
                if user_id and entry['user_id'] != user_id:
                    continue
                if operation and entry['operation'] != operation:
                    continue
                
                logs.append(entry)
        
        return logs

# 测试
logger = AuditLogger()
logger.log("user1", "file:read", "/app/data.txt", True)
logger.log("user1", "file:write", "/app/data.txt", False, {"error": "Permission denied"})
print(logger.query(user_id="user1"))
```

class SecureFileAccess:
    """安全文件访问系统"""
    
    def __init__(self):
        self.vfs = VirtualFileSystem()
        self.acl = AccessControlList()
        self.audit = AuditLogger()
        self.detector = PathTraversalDetector()
    
    async def read_file(
        self,
        user_id: str,
        virtual_path: str
    ) -> tuple[bool, str]:
        """安全读取文件"""
        
        # 1. 路径遍历检测
        is_dangerous, msg = self.detector.detect(virtual_path)
        if is_dangerous:
            self.audit.log(user_id, "file:read", virtual_path, False, {"reason": msg})
            return False, f"访问被拒绝: {msg}"
        
        # 2. 权限检查
        can_access, msg = self.acl.check_permission(
            virtual_path, "file:read", Permission.READ
        )
        if not can_access:
            self.audit.log(user_id, "file:read", virtual_path, False, {"reason": msg})
            return False, msg
        
        # 3. 路径解析
        real_path = self.vfs.resolve(virtual_path)
        if not real_path:
            self.audit.log(user_id, "file:read", virtual_path, False, {"reason": "路径未挂载"})
            return False, "路径未挂载"
        
        # 4. 执行读取
        try:
            with open(real_path, 'r') as f:
                content = f.read()
            
            self.audit.log(user_id, "file:read", virtual_path, True)
            return True, content
        
        except Exception as e:
            self.audit.log(user_id, "file:read", virtual_path, False, {"error": str(e)})
            return False, str(e)

# 测试
async def test_secure_file_access():
    system = SecureFileAccess()
    system.vfs.mount("/app", "/tmp/sandbox/app")
    system.acl.add_entry("/app", ["read"], ["file:read"])
    
    # 尝试路径遍历
    success, msg = await system.read_file("user1", "/app/../../../etc/passwd")
    print(f"路径遍历: success={success}, msg={msg[:50] if msg else 'None'}")

asyncio.run(test_secure_file_access())
